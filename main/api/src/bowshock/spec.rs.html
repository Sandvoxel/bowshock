6/// Longest accepted body, in nose diameters.
7const MAX_LENGTH_DIAMETERS: f64 = 40.0;
8/// Steepest accepted nose cone / transition half-angle.
9const MAX_HALF_ANGLE: f64 = 30.0 * std::f64::consts::PI / 180.0;

11/// Nose profile family.
12#[derive(Clone, Copy, Debug, PartialEq)]
13#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
14pub enum NoseShape {
15    /// Cone (half-angle at most 30 deg).
16    Cone,
17    /// Circular-arc tangent ogive.
18    TangentOgive,
19    /// Power series `r/R = (x/L)^n`, `n` in `[0.5, 1]`.
20    Power {
21        /// Exponent.
22        n: f64,
23    },
24    /// Parabolic series with parameter `k` in `[0, 1]` (`0` cone, `1` tangent parabola).
25    Parabolic {
26        /// Series parameter.
27        k: f64,
28    },
29    /// Haack series with parameter `c` in `[0, 1/3]` (`0` Von Karman, `1/3` LV-Haack).
30    Haack {
31        /// Series parameter.
32        c: f64,
33    },
34}

76/// Conversion between the internal units and the dimensional body.

86fn invalid<T>(field: &str, why: &str) -> Result<T, Error> {
87    Err(Error::InvalidInput(format!("{field}: {why}")))
88}

90fn positive(field: &str, v: f64) -> Result<(), Error> {
91    if v.is_finite() && v > 0.0 {
92        Ok(())
93    } else {
94        invalid(field, &format!("must be finite and > 0, got {v}"))
95    }
96}

98/// Edge profile of the fins.
99#[derive(Clone, Debug, PartialEq)]
100#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
101pub enum FinProfile {
102    /// Flat leading and trailing edges.
103    Square,
104    /// Rounded leading edge, flat trailing edge.
105    Rounded,
106    /// Rounded leading edge, tapered trailing edge.
107    Airfoil,
108    /// Sharp edges, double-wedge section (supersonic wave drag).
109    DoubleWedge,
110    /// User-defined symmetric section: upper-surface coordinates `[x/c, z]` from the leading edge (`x = 0`) to the trailing edge
111    /// (`x = 1`). Only the shape matters: `z` is normalised by its maximum and the maximum thickness is [`FinSpec::thickness`], so
112    /// the scale of `z` is irrelevant (a NACA `.dat` upper surface can be pasted as is).
113    Custom {
114        /// Upper-surface points `[x/c, z]`: at least 3, `x` strictly increasing from 0 to 1, `z >= 0` with a positive maximum.
115        section: Vec<[f64; 2]>,
116    },
117}

119/// One set of identical trapezoidal fins, in metres. `sweep` is the axial distance from the root leading edge to the tip leading edge (positive aft).
120#[derive(Clone, Debug, PartialEq)]
121#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
122pub struct FinSpec {
123    /// Number of fins, equally spaced around the body (3 to 8).
124    pub count: u32,
125    /// Chord at the body surface \[m\].
126    pub root_chord: f64,
127    /// Chord at the tip \[m\] (0 for a triangular fin).
128    pub tip_chord: f64,
129    /// Distance from the body surface to the tip \[m\].
130    pub span: f64,
131    /// Axial distance from the root leading edge to the tip leading edge \[m\], positive aft.
132    pub sweep: f64,
133    /// Maximum thickness \[m\].
134    pub thickness: f64,
135    /// Edge profile.
136    pub profile: FinProfile,
137}
138
139impl FinSpec {
140    /// Check the fin set; the error names the offending field as `fins.<field>`.
141    pub fn validate(&self) -> Result<(), Error> {
142        if !(3..=8).contains(&self.count) {
143            return invalid("fins.count", &format!("must be in 3..=8, got {}", self.count));
144        }
145        positive("fins.root_chord", self.root_chord)?;
146        positive("fins.span", self.span)?;
147        positive("fins.thickness", self.thickness)?;
148        if !(self.tip_chord.is_finite() && self.tip_chord >= 0.0) {
149            return invalid("fins.tip_chord", &format!("must be finite and >= 0, got {}", self.tip_chord));
150        }
151        if !(self.sweep.is_finite() && (self.sweep / self.span).atan().abs() <= 75f64.to_radians()) {
152            return invalid("fins.sweep", &format!("must be finite with a leading-edge sweep angle of at most 75 degrees, got {}", self.sweep));
153        }
154        if let FinProfile::Custom { section } = &self.profile {
155            validate_section(section)?;
156        }
157        let mac = crate::fins::mean_aerodynamic_chord(self.root_chord, self.tip_chord);
158        if self.thickness / mac > 0.2 {
159            return invalid("fins.thickness", &format!("thickness/mean chord must be <= 0.2, got {}", self.thickness / mac));
160        }
161        Ok(())
162    }
163}
164
165fn validate_section(s: &[[f64; 2]]) -> Result<(), Error> {
166    const F: &str = "fins.profile.section";
167    if s.len() < 3 {
168        return invalid(F, &format!("needs at least 3 points, got {}", s.len()));
169    }
170    if s.iter().any(|p| !(p[0].is_finite() && p[1].is_finite())) {
171        return invalid(F, "all coordinates must be finite");
172    }
173    if s[0][0].abs() > 1e-9 {
174        return invalid(F, &format!("first x must be 0, got {}", s[0][0]));
175    }
176    if (s[s.len() - 1][0] - 1.0).abs() > 1e-9 {
177        return invalid(F, &format!("last x must be 1, got {}", s[s.len() - 1][0]));
178    }
179    if s.windows(2).any(|w| w[1][0] <= w[0][0]) {
180        return invalid(F, "x must be strictly increasing");
181    }
182    if s.iter().any(|p| p[1] < 0.0) {
183        return invalid(F, "z must be >= 0");
184    }
185    if s.iter().all(|p| p[1] <= 0.0) {
186        return invalid(F, "the maximum z must be > 0");
187    }
188    Ok(())
189}

191/// Rocket motor exhausting through the centre of the base, modelled as an air jet (gamma = 1.4) at the nozzle exit.
192#[derive(Clone, Copy, Debug, PartialEq)]
193#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
194pub struct MotorSpec {
195    /// Diameter of the nozzle exit \[m\]; at most 0.95 of the base diameter.
196    pub exit_diameter: f64,
197    /// Mach number of the jet at the exit [-], in `[1, 5]`.
198    pub exit_mach: f64,
199    /// Static pressure at the exit \[Pa\].
200    pub exit_pressure_pa: f64,
201    /// Total temperature of the jet \[K\], in `[200, 4000]`.
202    pub total_temperature_k: f64,
203}
204
205impl MotorSpec {
206    /// Check the motor against the base diameter \[m\] of the body; `InvalidInput` names the offending field.
207    pub fn validate(&self, base_diameter: f64) -> Result<(), Error> {
208        positive("motor.exit_diameter", self.exit_diameter)?;
209        if self.exit_diameter > 0.95 * base_diameter {
210            return invalid("motor.exit_diameter", &format!("must be <= 0.95 of the base diameter {base_diameter}, got {}", self.exit_diameter));
211        }
212        if !(1.0..=5.0).contains(&self.exit_mach) {
213            return invalid("motor.exit_mach", &format!("must be in [1, 5], got {}", self.exit_mach));
214        }
215        positive("motor.exit_pressure_pa", self.exit_pressure_pa)?;
216        if !(200.0..=4000.0).contains(&self.total_temperature_k) {
217            return invalid("motor.total_temperature_k", &format!("must be in [200, 4000] K, got {}", self.total_temperature_k));
218        }
219        Ok(())
220    }
221}

223impl BodySpec {
224    /// Largest diameter of the body \[m\].
225    pub fn reference_diameter(&self) -> f64 {
226        self.aft.iter().fold(self.nose.diameter, |d, a| match *a {
227            AftSpec::Transition { aft_diameter, .. } => d.max(aft_diameter),
228            AftSpec::Tube { .. } => d,
229        })
230    }
231
232    /// Diameter of the flat base \[m\]: the diameter after the last aft section (the nose diameter when only tubes follow the nose).
233    pub fn base_diameter(&self) -> f64 {
234        self.aft.iter().fold(self.nose.diameter, |d, a| match *a {
235            AftSpec::Transition { aft_diameter, .. } => aft_diameter,
236            AftSpec::Tube { .. } => d,
237        })
238    }
239
240    /// Validate the body and convert it to the internal body (lengths divided by the nose diameter) and its [`Scale`].
241    #[doc(hidden)]
242    pub fn to_internal(&self) -> Result<(BaseBody, Scale), Error> {
243        let n = &self.nose;
244        positive("nose.length", n.length)?;
245        positive("nose.diameter", n.diameter)?;
246        let fineness = n.length / n.diameter;
247        if !(0.5 * (1.0 - 1e-9)..=12.0 * (1.0 + 1e-9)).contains(&fineness) {
248            return invalid("nose.length", &format!("fineness length / diameter = {fineness:.3} must be in [0.5, 12]"));
249        }
250        let nose = match n.shape {
251            NoseShape::Cone => {
252                let half_angle = (0.5 * n.diameter / n.length).atan();
253                if half_angle > MAX_HALF_ANGLE {
254                    return invalid("nose.shape", &format!("cone half-angle {:.1} deg exceeds 30 deg", half_angle.to_degrees()));
255                }
256                Nose::cone_fineness(fineness)
257            }
258            NoseShape::TangentOgive => Nose::TangentOgive { fineness },
259            NoseShape::Power { n } => {
260                if !(0.5..=1.0).contains(&n) {
261                    return invalid("nose.shape.n", &format!("must be in [0.5, 1], got {n}"));
262                }
263                Nose::Power { fineness, n }
264            }
265            NoseShape::Parabolic { k } => {
266                if !(0.0..=1.0).contains(&k) {
267                    return invalid("nose.shape.k", &format!("must be in [0, 1], got {k}"));
268                }
269                Nose::Parabolic { fineness, k }
270            }
271            NoseShape::Haack { c } => {
272                if !(0.0..=1.0 / 3.0).contains(&c) {
273                    return invalid("nose.shape.c", &format!("must be in [0, 1/3], got {c}"));
274                }
275                Nose::Haack { fineness, c }
276            }
277        };
278        if self.aft.is_empty() {
279            return invalid("aft", "needs at least one section");
280        }
281        let d = n.diameter;
282        let mut segments = Vec::with_capacity(self.aft.len());
283        let mut radius = 0.5; // internal radius at the start of the section
284        let mut total = n.length;
285        for (k, a) in self.aft.iter().enumerate() {
286            match *a {
287                AftSpec::Tube { length } => {
288                    positive(&format!("aft[{k}].length"), length)?;
289                    total += length;
290                    segments.push(Segment { length: length / d, slope: 0.0 });
291                }
292                AftSpec::Transition { length, aft_diameter } => {
293                    positive(&format!("aft[{k}].length"), length)?;
294                    positive(&format!("aft[{k}].aft_diameter"), aft_diameter)?;
295                    let (len, r_end) = (length / d, 0.5 * aft_diameter / d);
296                    let slope = (r_end - radius) / len;
297                    if slope.atan().abs() > MAX_HALF_ANGLE {
298                        return invalid(&format!("aft[{k}]"), &format!("transition half-angle {:.1} deg exceeds 30 deg", slope.atan().abs().to_degrees()));
299                    }
300                    total += length;
301                    radius = r_end;
302                    segments.push(Segment { length: len, slope });
303                }
304            }
305        }
306        if total > MAX_LENGTH_DIAMETERS * d {
307            return invalid("aft", &format!("total body length {:.1} nose diameters exceeds {MAX_LENGTH_DIAMETERS}", total / d));
308        }
309        let d_ref = self.reference_diameter();
310        let body = BaseBody { body: Body { nose }, segments, sting_radius: 0.0 };
311        Ok((body, Scale { d_nose: d, area_ratio: (d / d_ref).powi(2) }))
312    }
313}
