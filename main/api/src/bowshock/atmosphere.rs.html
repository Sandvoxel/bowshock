1//! International Standard Atmosphere (0-20 km) and explicit flight conditions: static state of the air for viscous analyses.
2
3use crate::error::Error;
4
5/// Specific gas constant of dry air [J/(kg K)].
6const GAS_CONSTANT: f64 = 287.05287;
7/// Ratio of specific heats of air [-].
8const GAMMA: f64 = 1.4;
9/// Sea-level standard temperature [K].
10const T0: f64 = 288.15;
11/// Sea-level standard pressure [Pa].
12const P0: f64 = 101_325.0;
13/// Temperature lapse rate of the troposphere [K/m].
14const LAPSE: f64 = 0.0065;
15/// Top of the troposphere, the lower bound of the isothermal layer [m].
16const TROPOPAUSE_M: f64 = 11_000.0;
17/// Top of the supported altitude range [m].
18const CEILING_M: f64 = 20_000.0;
19/// Isothermal temperature of the lower stratosphere [K].
20const T_STRATOSPHERE: f64 = 216.65;
21/// Pressure at the tropopause, the reference of the stratosphere exponential [Pa].
22const P_TROPOPAUSE: f64 = 22_632.06;
23/// Scale height of the lower stratosphere [m].
24const SCALE_HEIGHT_M: f64 = 6_341.62;
25/// Exponent of the troposphere pressure law [-].
26const PRESSURE_EXPONENT: f64 = 5.255_877;
27/// Sutherland reference viscosity [Pa s].
28const MU_REF: f64 = 1.716e-5;
29/// Sutherland reference temperature [K].
30const T_REF: f64 = 273.15;
31/// Sutherland constant of air [K].
32pub(crate) const SUTHERLAND_S: f64 = 110.4;
33/// Lowest explicit temperature accepted [K].
34const T_MIN: f64 = 100.0;
35/// Highest explicit temperature accepted [K].
36const T_MAX: f64 = 400.0;
37
38/// Source of the static state of the air.
39#[derive(Clone, Copy, Debug, PartialEq)]
40#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
41pub enum Atmosphere {
42    /// International Standard Atmosphere at a geometric altitude in metres (0 to 20 000 m).
43    Isa {
44        /// Geometric altitude \[m\].
45        altitude_m: f64,
46    },
47    /// Static temperature and pressure given directly.
48    Explicit {
49        /// Static temperature \[K\], within 100 to 400 K.
50        temperature_k: f64,
51        /// Static pressure \[Pa\], finite and positive.
52        pressure_pa: f64,
53    },
54}
55
56/// Static state of the air.
57#[derive(Clone, Copy, Debug, PartialEq)]
58pub struct AirState {
59    /// Static temperature \[K\].
60    pub temperature_k: f64,
61    /// Static pressure \[Pa\].
62    pub pressure_pa: f64,
63    /// Density [kg/m^3].
64    pub density: f64,
65    /// Speed of sound [m/s].
66    pub sound_speed: f64,
67    /// Dynamic viscosity from Sutherland's law [Pa s].
68    pub viscosity: f64,
69}
70
71/// Flight conditions of a viscous analysis.
72#[derive(Clone, Copy, Debug, PartialEq)]
73#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
74pub struct FlightConditions {
75    /// Static state of the air the vehicle flies through.
76    pub atmosphere: Atmosphere,
77}
78
79impl Atmosphere {
80    /// Static state of the air for this atmosphere.
81    ///
82    /// Returns `Error::InvalidInput` when the altitude is outside 0 to 20 000 m or the explicit temperature or pressure is
83    /// out of range.
84    pub fn state(&self) -> Result<AirState, Error> {
85        let (temperature_k, pressure_pa) = match *self {
86            Atmosphere::Isa { altitude_m: h } => {
87                if (0.0..=TROPOPAUSE_M).contains(&h) {
88                    let t = T0 - LAPSE * h;
89                    (t, P0 * (t / T0).powf(PRESSURE_EXPONENT))
90                } else if h > TROPOPAUSE_M && h <= CEILING_M {
91                    (T_STRATOSPHERE, P_TROPOPAUSE * (-(h - TROPOPAUSE_M) / SCALE_HEIGHT_M).exp())
92                } else {
93                    return Err(Error::InvalidInput(format!("altitude_m must be within 0 to {CEILING_M} m, got {h}")));
94                }
95            }
96            Atmosphere::Explicit { temperature_k, pressure_pa } => {
97                if !(T_MIN..=T_MAX).contains(&temperature_k) {
98                    return Err(Error::InvalidInput(format!("temperature_k must be within {T_MIN} to {T_MAX} K, got {temperature_k}")));
99                }
100                if !(pressure_pa.is_finite() && pressure_pa > 0.0) {
101                    return Err(Error::InvalidInput(format!("pressure_pa must be finite and positive, got {pressure_pa}")));
102                }
103                (temperature_k, pressure_pa)
104            }
105        };
106        let t = temperature_k;
107        Ok(AirState {
108            temperature_k: t,
109            pressure_pa,
110            density: pressure_pa / (GAS_CONSTANT * t),
111            sound_speed: (GAMMA * GAS_CONSTANT * t).sqrt(),
112            viscosity: MU_REF * (t / T_REF).powf(1.5) * (T_REF + SUTHERLAND_S) / (t + SUTHERLAND_S),
113        })
114    }
115}

117#[cfg(test)]
118mod tests {
119    use super::*;
120
121    #[test]
122    fn sea_level_matches_isa() {
123        let s = Atmosphere::Isa { altitude_m: 0.0 }.state().expect("sea level is valid");
124        assert_eq!(s.temperature_k, 288.15);
125        assert!((s.pressure_pa - 101_325.0).abs() < 1e-6);
126        assert!((s.density - 1.2250).abs() < 1e-4, "density {}", s.density);
127        assert!((s.sound_speed - 340.29).abs() < 0.01, "sound speed {}", s.sound_speed);
128        let rel = (s.viscosity - 1.789e-5).abs() / 1.789e-5;
129        assert!(rel < 0.005, "viscosity {}", s.viscosity);
130    }
131
132    #[test]
133    fn eleven_km_pressure() {
134        let s = Atmosphere::Isa { altitude_m: 11_000.0 }.state().expect("11 km is valid");
135        assert!((s.temperature_k - 216.65).abs() < 1e-9);
136        assert!((s.pressure_pa - 22_632.0).abs() < 5.0, "pressure {}", s.pressure_pa);
137    }
138
139    #[test]
140    fn isa_out_of_range_is_invalid() {
141        assert!(matches!(Atmosphere::Isa { altitude_m: -1.0 }.state(), Err(Error::InvalidInput(_))));
142        assert!(matches!(Atmosphere::Isa { altitude_m: 25_000.0 }.state(), Err(Error::InvalidInput(_))));
143        assert!(matches!(Atmosphere::Isa { altitude_m: f64::NAN }.state(), Err(Error::InvalidInput(_))));
144    }
145
146    #[test]
147    fn explicit_state_and_validation() {
148        assert!(matches!(Atmosphere::Explicit { temperature_k: 99.0, pressure_pa: 1e5 }.state(), Err(Error::InvalidInput(_))));
149        assert!(matches!(Atmosphere::Explicit { temperature_k: 300.0, pressure_pa: 0.0 }.state(), Err(Error::InvalidInput(_))));
150        let s = Atmosphere::Explicit { temperature_k: 300.0, pressure_pa: 101_325.0 }.state().expect("300 K is valid");
151        assert!((s.density - 1.1766).abs() < 1e-3, "density {}", s.density);
152    }
153}
