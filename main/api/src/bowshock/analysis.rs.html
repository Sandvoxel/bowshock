90/// Overall trust in a point, derived from how its run stopped.
91#[derive(Clone, Copy, Debug, PartialEq, Eq)]
92#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
93pub enum Quality {
94    /// The residual dropped by the requested number of orders.
95    Converged,
96    /// The residual did not drop enough, but the monitored quantity is stationary.
97    Stationary,
98    /// The iteration cap was reached without a stop criterion.

101    /// non-finite drag.
102    Failed,

105/// A condition under which a point is less reliable than its [`Quality`] suggests.
106#[derive(Clone, Copy, Debug, PartialEq)]
107#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
108pub enum Warning {
109    /// Transonic Mach number and a convex profile corner sharper than 2 degrees (shock position and drag are sensitive).
110    SharpCornerTransonic,
111    /// The base pressure still oscillates by `cp_base_range` over the monitor window although the residual did not drop.
112    UnsteadyWake {
113        /// Range of the area-mean base Cp over the monitor window.
114        cp_base_range: f64,
115    },
116    /// The pressure at the outer boundaries deviates from the free stream by `dp` (relative): the domain may be too small.
117    FarFieldDisturbed {
118        /// Relative pressure deviation.
119        dp: f64,
120    },
121    /// The first cell at the wall is too thick for a wall-resolved boundary layer: `y+` of the first cell reaches `y_plus_max`.
122    WallResolution {
123        /// Largest `y+` of the first wall cell.
124        y_plus_max: f64,
125    },
126    /// The total drag oscillated by `cd_range` (on the reference area) over the last two averaging windows, a bounded limit cycle
127    /// with a stationary mean; the point is reported at the crossing of that mean.
128    OscillatingDrag {
129        /// Max - min of the total drag over the averaging windows.
130        cd_range: f64,
131    },
132}

357    /// Free-stream Reynolds number, Sutherland constant and highest Mach number of a viscous sweep; `None` for the Euler equations.
358    fn rans(&self, machs: &[f64]) -> Result<Option<Rans>, Error> {
359        let Some(flight) = self.flight else {
360            return Ok(None);
361        };
362        if !self.dimensional {
363            return Err(Error::InvalidInput("viscous analysis needs a dimensional BodySpec".into()));
364        }
365        let air = flight.atmosphere.state()?;
366        let re_c = air.density * air.sound_speed * self.d_nose / air.viscosity;
367        let sutherland = SUTHERLAND_S / air.temperature_k;
368        // jet at the nozzle exit in solver units (rho = 1, c = 1, p = 1/gamma, T = 1 of the free stream)
369        let jet = self.motor.map(|m| {
370            let t = m.total_temperature_k / (1.0 + 0.2 * m.exit_mach * m.exit_mach) / air.temperature_k;
371            let p = m.exit_pressure_pa / air.pressure_pa;
372            let rho = p / t;
373            let u = m.exit_mach * t.sqrt();
374            let k = 1.5 * (0.01 * u).powi(2);
375            let mu = Viscous { re_c, sutherland, turbulence: Turbulence::Sst, wall_i_start: 0 }.mu(t);
376            Jet { r_exit: 0.5 * m.exit_diameter / self.d_nose, state: [rho, u, 0.0, p / GAMMA], turb: [k, rho * k / (10.0 * mu)] }
377        });
378        Ok(Some(Rans { re_c, sutherland, mach_max: machs.iter().copied().fold(f64::NEG_INFINITY, f64::max), jet }))
379    }

381    /// The solver cases of a sweep in ascending Mach order, after the input checks of [`Analysis::run`]; with the Mach numbers' original
382    /// positions and the Rans parameters.
383    fn prepare(&self, machs: &[f64]) -> Result<(Vec<usize>, Option<Rans>, Vec<Case>), Error> {
384        if machs.is_empty() {
385            return Err(Error::InvalidInput("no Mach numbers requested".into()));
386        }
387        for &m in machs {
388            if !(m.is_finite() && (MACH_RANGE.0..=MACH_RANGE.1).contains(&m)) {
389                return Err(Error::InvalidInput(format!("Mach number {m} is outside [{}, {}]", MACH_RANGE.0, MACH_RANGE.1)));
390            }
391        }
392        if !(self.d_ref.is_finite() && self.d_ref > 0.0) {
393            return Err(Error::InvalidInput(format!("reference diameter {} must be finite and > 0", self.d_ref)));
394        }
395
396        if let Some(f) = &self.fins {
397            f.validate()?;
398            if self.flight.is_none() {
399                return Err(Error::InvalidInput("fins need flight conditions: call Analysis::viscous".into()));
400            }
401            if !self.dimensional {
402                return Err(Error::InvalidInput("fins need a dimensional BodySpec".into()));
403            }
404        }
405
406        if let Some(m) = &self.motor {
407            if self.flight.is_none() {
408                return Err(Error::InvalidInput("motor-on analysis needs Analysis::viscous".into()));
409            }
410            if self.wake == Some(Wake::Sting) {
411                return Err(Error::InvalidInput("motor-on analysis needs the base wake".into()));
412            }
413            if !self.dimensional {
414                return Err(Error::InvalidInput("motor-on analysis needs a dimensional BodySpec".into()));
415            }
416            m.validate(self.body.base_radius() * 2.0 * self.d_nose)?;
417        }
418        // cases in ascending Mach order (stable), results restored to input order by the caller
419        let mut order: Vec<usize> = (0..machs.len()).collect();
420        order.sort_by(|&a, &b| machs[a].total_cmp(&machs[b]));
421        let rans = self.rans(machs)?;
422        let level = self.resolution.level();
423        let cases: Vec<Case> = order.iter().map(|&k| self.case(machs[k], level, rans.as_ref())).collect();
424        // the grids do not depend on the Mach number: check the first one
425        check_grid(&cases[0])?;
426        Ok((order, rans, cases))
427    }

435    /// Zero-lift drag at each Mach number of `machs` (each in `[0.1, 3.0]`).
436    pub fn run(&self, machs: &[f64]) -> Result<Polar, Error> {
437        let (order, rans, cases) = self.prepare(machs)?;
438        let level = self.resolution.level();
439        let cmp_cases: Option<Vec<Case>> = self.estimate_uncertainty.then(|| {
440            let cmp_level = if level >= 1 { level - 1 } else { level + 1 };
441            order.iter().map(|&k| self.case(machs[k], cmp_level, rans.as_ref())).collect()
442        });
443        if let Some(c) = &cmp_cases {
444            check_grid(&c[0])?;
445        }
446
447        let (results, backend, mut fallback) = self.sweep_on(&cases, self.backend)?;
448        let cmp_results = match &cmp_cases {
449            Some(c) => {
450                let b = match backend {
451                    BackendUsed::Cpu => Backend::Cpu,
452                    BackendUsed::Gpu => self.backend,
453                };
454                let (r, _, note) = self.sweep_on(c, b)?;
455                if let Some(note) = note {
456                    fallback = Some(format!("uncertainty sweep: {note}"));
457                }
458                Some(r)
459            }
460            None => None,
461        };
462
463        let mut cmp_iter = cmp_results.map(Vec::into_iter);
464        let mut slots: Vec<Option<PolarPoint>> = (0..machs.len()).map(|_| None).collect();
465        for (&k, detail) in order.iter().zip(results) {
466            let other = cmp_iter.as_mut().and_then(Iterator::next);
467            slots[k] = Some(self.point(machs[k], detail, other.as_ref()));
468        }
469        let points = slots.into_iter().map(|p| p.expect("every Mach number has a result")).collect();
470        Ok(Polar { points, backend, fallback, motor_on: self.motor.is_some() })
471    }

473    /// Wake treatment in use: the one set with [`Analysis::wake`], else a base wake for a viscous analysis and a sting otherwise.
474    fn wake_used(&self) -> Wake {
475        self.wake.unwrap_or(if self.flight.is_some() { Wake::Base } else { Wake::Sting })
476    }

478    fn case(&self, mach: f64, level: u32, rans: Option<&Rans>) -> Case {
479        let wake = self.wake_used();
480        let mut case = match rans {
481            Some(r) => Case::viscous(mach, self.body.clone(), level, wake, r.re_c, r.sutherland, r.mach_max),
482            None => Case::new(mach, self.body.clone(), level, wake, Numerics::auto(mach, level, wake)),
483        };
484        if let Some(n) = self.numerics {
485            case.numerics = n;
486        }
487        if let Some(cap) = self.max_iterations {
488            case.numerics.max_iter = case.numerics.max_iter.min(cap);
489        }
490        match rans.and_then(|r| r.jet) {
491            Some(jet) => case.with_motor(jet),
492            None => case,
493        }
494    }

496    /// Drag of one solver result, referenced to the reference area. A viscous analysis adds the skin friction and, with a base
497    /// wake, takes the base drag from the solution instead of the correlation.
498    fn drag(&self, mach: f64, detail: &PointResult) -> Drag {
499        let q = (self.d_nose / self.d_ref).powi(2);
500        let forebody_pressure = detail.cd_aft.iter().fold(detail.cd_nose, |a, c| a + c);
501        let viscous = self.flight.is_some();
502        let friction = viscous.then_some(detail.cd_friction);
503        let base_modelled = !(viscous && self.wake_used() == Wake::Base);
504        let base = if base_modelled { base_drag_coefficient(mach) * (self.body.base_radius() / self.body.body.radius()).powi(2) } else { detail.cd_base };
505        let mut total = (forebody_pressure + friction.unwrap_or(0.0) + base) * q;
506        let fins = self.fins.as_ref().zip(self.flight.and_then(|f| f.atmosphere.state().ok())).map(|(f, air)| crate::fins::fin_drag(f, mach, &air, self.d_ref));
507        if let Some(fd) = fins {
508            total += fd.friction + fd.pressure;
509        }
510        Drag { forebody_pressure: forebody_pressure * q, friction: friction.map(|f| f * q), base: base * q, base_modelled, fins, total }
511    }
512
513    fn point(&self, mach: f64, detail: PointResult, other: Option<&PointResult>) -> PolarPoint {
514        let drag = self.drag(mach, &detail);
515        let quality = quality(&detail, &drag);
516        let warnings = self.warnings(mach, &detail);
517        let cd_uncertainty = other.and_then(|o| {
518            let od = self.drag(mach, o);
519            (quality != Quality::Failed && self::quality(o, &od) != Quality::Failed).then(|| (drag.total - od.total).abs())
520        });
521        let detail = self.detail(&detail);
522        PolarPoint { mach, alpha: 0.0, drag, quality, warnings, cd_uncertainty, detail }
523    }

525    /// Public summary of a solver result: coefficients on the reference area, wall points in metres.
526    fn detail(&self, d: &PointResult) -> PointDetail {
527        let q = (self.d_nose / self.d_ref).powi(2);
528        let base = self.wake_used() == Wake::Base;
529        let x_base = self.body.x_base();
530        PointDetail {
531            iterations: d.iters,
532            residual_drop_orders: d.drop_orders,
533            stop_reason: d.stop_reason,
534            cd_nose: d.cd_nose * q,
535            cd_aft: d.cd_aft.iter().map(|c| c * q).collect(),
536            cd_friction: d.cd_friction * q,
537            cd_base_solver: if base { d.cd_base * q } else { f64::NAN },
538            cp_base: if base { d.cp_base } else { f64::NAN },
539            wall: d.wall.iter().filter(|w| w.x < x_base).map(|w| WallCp { x: w.x * self.d_nose, r: w.r * self.d_nose, cp: w.cp }).collect(),
540            y_plus_max: d.y_plus_max,
541            farfield_dp: d.farfield_dp,
542        }
543    }
544
545    fn warnings(&self, mach: f64, detail: &PointResult) -> Vec<Warning> {
546        let mut w = Vec::new();
547        if (0.7..1.05).contains(&mach) && self.has_sharp_corner() {
548            w.push(Warning::SharpCornerTransonic);
549        }
550        if self.wake_used() == Wake::Base && detail.cp_base_range.is_finite() && detail.cp_base_range >= 1e-3 && detail.stop_reason != StopReason::ResidualDrop
551        {
552            w.push(Warning::UnsteadyWake { cp_base_range: detail.cp_base_range });
553        }
554        if detail.farfield_dp > 0.02 {
555            w.push(Warning::FarFieldDisturbed { dp: detail.farfield_dp });
556        }
557        if detail.y_plus_max > 5.0 {
558            w.push(Warning::WallResolution { y_plus_max: detail.y_plus_max });
559        }
560        if detail.stop_reason == StopReason::MeanStationary {
561            let window = self.numerics.and_then(|n| n.mean_window).or(Numerics::viscous(mach, 0).mean_window).unwrap_or(0);
562            let k = (window / crate::engine::numerics::MONITOR_EVERY).max(1);
563            let h = &detail.monitor_history;
564            let (lo, hi) = h[h.len().saturating_sub(2 * k)..].iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
565            w.push(Warning::OscillatingDrag { cd_range: (hi - lo) * (self.d_nose / self.d_ref).powi(2) });
566        }
567        w
568    }

570    /// A convex corner sharper than 2 degrees at the nose end or between two aft segments.
571    fn has_sharp_corner(&self) -> bool {
572        let limit = 2f64.to_radians();
573        let segs = &self.body.segments;
574        let Some(first) = segs.first() else { return false };
575        let before = self.body.slope(self.body.body.nose_length() * (1.0 - 1e-9));
576        before.atan() - first.slope.atan() > limit || segs.windows(2).any(|w| w[0].slope.atan() - w[1].slope.atan() > limit)
577    }

579    /// Sweeps `cases` (ascending Mach) on `backend`; returns the backend used and, under `Auto`, why the GPU was not.
580    fn sweep_on(&self, cases: &[Case], backend: Backend) -> Result<(Vec<PointResult>, BackendUsed, Option<String>), Error> {
581        match backend {
582            Backend::Cpu => Ok((guard(|| self.sweep_cpu(cases))?, BackendUsed::Cpu, None)),
583            Backend::Gpu => Ok((guard(|| self.sweep_gpu_required(cases))?, BackendUsed::Gpu, None)),
584            Backend::Auto => self.sweep_auto(cases),
585        }
586    }
587
588    /// The sweep logic shared by the backends: first point cold; later points warm (when enabled) against the residual peak
589    /// of the most recent cold start; a warm point that ends `Failed` or `Stalled` is re-run cold on a fresh solver, and a point
590    /// after a `Failed` point starts cold on a fresh solver.
591    fn sweep<R: Runner>(&self, runner: &R, cases: &[Case]) -> Result<Vec<PointResult>, R::Error> {
592        let quality_of = |case: &Case, d: &PointResult| quality(d, &self.drag(case.mach, d));
593        let mut out: Vec<PointResult> = Vec::with_capacity(cases.len());
594        let mut state: Option<R::State> = None;
595        let mut r_ref = 0.0;
596        let mut previous_failed = false;
597        for case in cases {
598            let warm = match (&mut state, previous_failed) {
599                (Some(s), false) => {
600                    let r = runner.warm(s, case, r_ref)?;
601                    matches!(quality_of(case, &r), Quality::Converged | Quality::Stationary).then_some(r)
602                }
603                _ => None,
604            };
605            let result = match warm {
606                Some(r) => r,
607                None => {
608                    let (r, s) = runner.cold(case)?;
609                    r_ref = r.residual_peak;
610                    state = self.warm_start.then_some(s);
611                    r
612                }
613            };
614            previous_failed = quality_of(case, &result) == Quality::Failed;
615            out.push(result);
616        }
617        Ok(out)
618    }
619
620    fn sweep_cpu(&self, cases: &[Case]) -> Result<Vec<PointResult>, Error> {
621        self.sweep(&CpuBackend, cases)
622    }
623
624    #[cfg(feature = "gpu")]
625    fn sweep_gpu_required(&self, cases: &[Case]) -> Result<Vec<PointResult>, Error> {
626        self.sweep(&GpuBackend, cases)
627    }
628
629    #[cfg(not(feature = "gpu"))]
630    fn sweep_gpu_required(&self, _cases: &[Case]) -> Result<Vec<PointResult>, Error> {
631        Err(Error::Unsupported("the `gpu` feature is not enabled".into()))
632    }
633
634    #[cfg(feature = "gpu")]
635    fn sweep_auto(&self, cases: &[Case]) -> Result<(Vec<PointResult>, BackendUsed, Option<String>), Error> {
636        let why = match guard(|| self.sweep_gpu_required(cases)) {
637            Ok(results) => return Ok((results, BackendUsed::Gpu, None)),
638            Err(Error::Internal(m)) => m.replacen("solver panicked", "GPU solver panicked", 1),
639            Err(e) => format!("GPU unavailable: {e}"),
640        };
641        Ok((guard(|| self.sweep_cpu(cases))?, BackendUsed::Cpu, Some(why)))
642    }
643
644    #[cfg(not(feature = "gpu"))]
645    fn sweep_auto(&self, cases: &[Case]) -> Result<(Vec<PointResult>, BackendUsed, Option<String>), Error> {
646        Ok((guard(|| self.sweep_cpu(cases))?, BackendUsed::Cpu, None))
647    }
648}

650/// Runs `f`, turning a panic into `Error::Internal("solver panicked: ...")`.
651fn guard<T>(f: impl FnOnce() -> Result<T, Error>) -> Result<T, Error> {
652    std::panic::catch_unwind(std::panic::AssertUnwindSafe(f)).unwrap_or_else(|payload| {
653        let msg = payload.downcast_ref::<&str>().map(|s| (*s).to_string()).or_else(|| payload.downcast_ref::<String>().cloned());
654        Err(Error::Internal(format!("solver panicked: {}", msg.as_deref().unwrap_or("unknown"))))
655    })
656}

658/// Builds the grids of `case` and checks their quality (the wall-resolved grids of a viscous case against the tighter limit in `j`).
659fn check_grid(case: &Case) -> Result<(), Error> {
660    let probe = case.try_build_solver()?;
661    let max_j = if case.viscous.is_some() { MAX_STRETCHING_VISCOUS_J } else { MAX_STRETCHING };
662    check_quality(&probe.grid, probe.block(), MAX_STRETCHING, max_j)
663}
664
665/// Relative deviation of the outer-boundary pressure from the free stream above which a state is not a meaningful solution.
666const FARFIELD_DP_FAILED: f64 = 1.0;
667
668/// Quality of a solver result, first match wins. A point is [`Quality::Failed`] when the run stopped on a non-finite or blown-up
669/// state, when the density or pressure of a cell is not positive, when the total drag is not finite, or when the pressure at the
670/// outer boundaries deviates from the free stream by more than 100 % (`farfield_dp > 1`): the far field of a body in a flow of
671/// this size is undisturbed at that distance, so such a state is garbage even if it is finite and the run ended `Stalled`.
672fn quality(d: &PointResult, drag: &Drag) -> Quality {
673    let positive = |x: f64| x > 0.0; // false for NaN
674    if d.stop_reason == StopReason::NonFinite || !positive(d.min_rho) || !positive(d.min_p) || !drag.total.is_finite() || d.farfield_dp > FARFIELD_DP_FAILED {
675        return Quality::Failed;
676    }
677    match d.stop_reason {
678        StopReason::ResidualDrop => Quality::Converged,
679        StopReason::CdStationary | StopReason::MeanStationary | StopReason::BasePressureStationary | StopReason::ResidualPlateau => Quality::Stationary,
680        StopReason::MaxIterations => Quality::Stalled,
681        StopReason::NonFinite => Quality::Failed,
682    }
683}
684
685/// A solver backend of the sweep: cold runs on a fresh solver, warm runs on the solver kept from an earlier point.
686trait Runner {
687    type State;
688    type Error;
689    fn cold(&self, case: &Case) -> Result<(PointResult, Self::State), Self::Error>;
690    fn warm(&self, state: &mut Self::State, case: &Case, r_ref: f64) -> Result<PointResult, Self::Error>;
691}
692
693struct CpuBackend;
694
695impl Runner for CpuBackend {
696    type State = Solver;
697    type Error = Error;
698
699    fn cold(&self, case: &Case) -> Result<(PointResult, Solver), Error> {
700        let mut s = case.try_build_solver()?;
701        let r = run_on(case, &mut s, false, None);
702        Ok((r, s))
703    }
704
705    fn warm(&self, s: &mut Solver, case: &Case, r_ref: f64) -> Result<PointResult, Error> {
706        Ok(run_on(case, s, true, Some(r_ref)))
707    }
708}

726    }
