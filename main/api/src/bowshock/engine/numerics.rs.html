1//! Solver settings of a run: scheme, limiter, iteration limits and stop-rule tolerances.
2
3use crate::engine::case::Wake;
4use bowshock_cfd::flux::Limiter;
5use bowshock_cfd::implicit::{DtMode, ImplicitParams};
6use bowshock_cfd::solver::{Bc, Scheme};
7
8/// Sampling interval \[iterations\] of the monitor history.
9pub const MONITOR_EVERY: usize = 50;
10/// Mach number below which [`Numerics::auto`] uses the subsonic settings.
11pub const REGIME_SPLIT_MACH: f64 = 0.85;
12
13/// Pseudo-time parameters of the wall-resolved RANS runs of the facade: the summed-direction time step with a high CFL limit. The
14/// weaker-direction step of [`ImplicitParams::VISCOUS`] (which the flat-plate gates of the solver use) diverged within a few hundred
15/// iterations on the wall-resolved body grid of an ogive-cylinder at Mach 0.5 (cfl_max 10, also with a start CFL of 0.1).
16const RANS_IMPLICIT: ImplicitParams = ImplicitParams { dt_mode: DtMode::Sum, cfl_max: 50.0, sub_iters: 3, ..ImplicitParams::DEFAULT };
17
18/// Mach number from which a wall-resolved run uses [`RANS_IMPLICIT_SUPERSONIC`].
19const SUPERSONIC_MACH: f64 = 1.0;
20
21/// [`RANS_IMPLICIT`] with the CFL limit lowered to 20 for supersonic runs. At `cfl_max` 50 the implicit iteration can diverge in the
22/// supersonic wake behind the base once the CFL ramp (`growth` 1.02 per iteration) passes about 40, at iteration 195-240 (during the
23/// first-order phase): a short-wave oscillation of density and pressure on the wake rows next to the axis grows by about 30 % per
24/// iteration and the pressure turns negative within 5 iterations. The inviscid implicit scheme on the same grid does the same, so no
25/// viscous or turbulence term is involved, and it needs the ramp, not the second-order switch. Measured on the GPU solver at Medium:
26/// the polar body at Mach 1.5, 2, 2.5 and 3, the TP-2005 body at Mach 2 and the Arcas body at Mach 2 and 2.5 diverge at 50 (at 40 or 35
27/// the polar body at Mach 3, the TP-2005 body at Mach 3 and the Arcas body at Mach 2; the polar body at Mach 3 even at 30); at 20 every
28/// one of those, at Mach 1.5, 2, 2.5 and 3, reaches a stationary drag.
29const RANS_IMPLICIT_SUPERSONIC: ImplicitParams = ImplicitParams { cfl_max: 20.0, ..RANS_IMPLICIT };
30
31/// Numerical settings of one run.
32#[derive(Clone, Copy, Debug, PartialEq)]
33pub struct Numerics {
34    pub cfl: f64,
35    pub limiter: Limiter,
36    pub low_mach: bool,
37    pub scheme: Scheme,
38    /// First-order start-up iterations.
39    pub first_order_iters: usize,
40    pub max_iter: usize,
41    /// Stop rules other than a non-finite residual do not fire before this many iterations.
42    pub min_iter: usize,
43    /// Stop at this many orders of density-residual drop below the reference.
44    pub drop_orders: f64,
45    /// Boundary condition of the outflow plane.
46    pub outflow: Bc,
47    /// Length of the statistics window of the monitor \[iterations\].
48    pub monitor_window: usize,
49    /// Monitor range below which a run counts as stationary; `None`: no "stationary" stop rule.
50    pub monitor_tol: Option<f64>,
51    /// Averaging window \[iterations\] of the mean-stationary stop rule (`None`: no such rule; only [`Numerics::viscous`] sets it). A
52    /// viscous run whose total drag settles into a bounded limit cycle never satisfies `monitor_tol`; the rule stops it where the
53    /// drag crosses the stationary mean of the cycle. The window should span at least one period of the cycle.
54    pub mean_window: Option<usize>,
55}
56
57impl Numerics {
58    /// Line-implicit scheme, van Albada limiter, low-Mach correction (subsonic regime).
59    pub fn subsonic(level: u32, wake: Wake) -> Numerics {
60        Numerics {
61            cfl: 0.5,
62            limiter: Limiter::VanAlbada,
63            low_mach: true,
64            scheme: Scheme::Implicit(ImplicitParams::DEFAULT),
65            first_order_iters: 100,
66            max_iter: 10_000 << level,
67            min_iter: 0,
68            drop_orders: 6.0,
69            outflow: Bc::Characteristic,
70            monitor_window: 2000,
71            monitor_tol: match wake {
72                Wake::Base => Some(1e-3),
73                Wake::Sting => None,
74            },
75            mean_window: None,
76        }
77    }
78
79    /// Explicit SSP-RK3, minmod limiter (transonic regime).
80    pub fn transonic() -> Numerics {
81        Numerics {
82            cfl: 0.5,
83            limiter: Limiter::Minmod,
84            low_mach: false,
85            scheme: Scheme::Explicit,
86            first_order_iters: 300,
87            max_iter: 60_000,
88            min_iter: 0,
89            drop_orders: 6.0,
90            outflow: Bc::Characteristic,
91            monitor_window: 2000,
92            monitor_tol: Some(2e-4),
93            mean_window: None,
94        }
95    }
96
97    /// Subsonic settings below [`REGIME_SPLIT_MACH`], transonic settings otherwise; the stationary tolerance follows the wake.
98    pub fn auto(mach: f64, level: u32, wake: Wake) -> Numerics {
99        let mut n = if mach < REGIME_SPLIT_MACH { Numerics::subsonic(level, wake) } else { Numerics::transonic() };
100        n.monitor_tol = Some(match wake {
101            Wake::Base => 1e-3,
102            Wake::Sting => 2e-4,
103        });
104        n
105    }
106
107    /// Wall-resolved RANS (viscous terms and k-omega SST): line-implicit with the viscous parameters (CFL limit 20 from Mach 1, 50
108    /// below), van Albada limiter and the
109    /// low-Mach correction below [`REGIME_SPLIT_MACH`], minmod without correction otherwise; stationary tolerance 1e-4 on the
110    /// total CD, and the mean-stationary rule with a window of 6000 iterations (one period of the limit cycle of the subsonic runs).
111    pub fn viscous(mach: f64, level: u32) -> Numerics {
112        let subsonic = mach < REGIME_SPLIT_MACH;
113        Numerics {
114            cfl: 0.5,
115            limiter: if subsonic { Limiter::VanAlbada } else { Limiter::Minmod },
116            low_mach: subsonic,
117            scheme: Scheme::Implicit(if mach < SUPERSONIC_MACH { RANS_IMPLICIT } else { RANS_IMPLICIT_SUPERSONIC }),
118            first_order_iters: 200,
119            max_iter: 80_000 << level,
120            min_iter: 0,
121            drop_orders: 5.0,
122            outflow: Bc::Characteristic,
123            monitor_window: 2000,
124            monitor_tol: Some(1e-4),
125            mean_window: Some(6000),
126        }
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133
134    #[test]
135    fn auto_regime_split_and_tolerance() {
136        let lo = Numerics::auto(0.5, 1, Wake::Sting);
137        assert_eq!(lo.scheme, Scheme::Implicit(ImplicitParams::DEFAULT));
138        assert_eq!(lo.max_iter, 20_000);
139        assert_eq!(lo.monitor_tol, Some(2e-4));
140        let hi = Numerics::auto(REGIME_SPLIT_MACH, 1, Wake::Base);
141        assert_eq!(hi.scheme, Scheme::Explicit);
142        assert_eq!(hi.max_iter, 60_000);
143        assert_eq!(hi.monitor_tol, Some(1e-3));
144        assert_eq!(Numerics::subsonic(0, Wake::Sting).monitor_tol, None);
145    }
146
147    #[test]
148    fn viscous_cfl_limit_is_lower_from_mach_one() {
149        let cfl_max = |mach: f64| match Numerics::viscous(mach, 1).scheme {
150            Scheme::Implicit(p) => p.cfl_max,
151            s => panic!("{s:?}"),
152        };
153        assert_eq!([cfl_max(0.5), cfl_max(0.99), cfl_max(1.0), cfl_max(3.0)], [50.0, 50.0, 20.0, 20.0]);
154    }
