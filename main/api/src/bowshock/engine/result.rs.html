3use std::fmt;

5/// Why an iteration loop stopped.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7#[cfg_attr(feature = "serde", derive(serde::Serialize, serde::Deserialize))]
8pub enum StopReason {
9    /// The residual dropped by the requested number of orders.
10    ResidualDrop,
11    /// The forebody or total drag is stationary over the monitor window.
12    CdStationary,
13    /// The base pressure is stationary over the monitor window.
14    BasePressureStationary,
15    /// The monitored total drag oscillates in a bounded limit cycle whose mean is stationary; the run stopped where the drag
16    /// crosses that mean.
17    MeanStationary,
18    /// The residual stopped decreasing.
19    ResidualPlateau,
20    /// The iteration cap was reached.
21    MaxIterations,
22    /// The state became non-finite or non-physical.
23    NonFinite,
24}
25
26impl fmt::Display for StopReason {
27    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
28        f.pad(match self {
29            StopReason::ResidualDrop => "residual drop reached",
30            StopReason::CdStationary => "CD stationary",
31            StopReason::BasePressureStationary => "base pressure stationary",
32            StopReason::MeanStationary => "mean stationary",
33            StopReason::ResidualPlateau => "residual plateau",
34            StopReason::MaxIterations => "max iterations",
35            StopReason::NonFinite => "non-finite residual",
36        })
37    }
38}

40/// One wall face of the body.
41#[derive(Clone, Copy, Debug)]
42pub struct WallPoint {
43    /// Face centre (x/d, r/d).
44    pub x: f64,
45    pub r: f64,
46    /// `s = (r/R)^2` at the two face end nodes (for the base-area-referenced drag integral).
47    pub s0: f64,
48    pub s1: f64,
49    /// Cp from the wall value extrapolated linearly from cells j=0,1, and from the first cell.
50    pub cp: f64,
51    pub cp_cell: f64,
52}
53
54/// One base-face sample: radius range, Cp from the wall-extrapolated and the first-cell pressure.
55#[derive(Clone, Copy, Debug)]
56pub struct BasePoint {
57    pub r0: f64,
58    pub r1: f64,
59    pub cp: f64,
60    pub cp_cell: f64,
61}
62
63/// Linear interpolation of the wall `Cp` (extrapolated values) at axial position `x` (None outside the wall range).
64pub fn wall_cp_at(wall: &[WallPoint], x: f64) -> Option<f64> {
65    for w in wall.windows(2) {
66        if x >= w[0].x && x <= w[1].x {
67            let t = (x - w[0].x) / (w[1].x - w[0].x);
68            return Some(w[0].cp + t * (w[1].cp - w[0].cp));
69        }
70    }
71    None
72}

74/// Everything a run reports. Drag coefficients are referenced to the maximum cross-section `pi R^2`; the pressure terms of a
75/// viscous run add `cd_friction`.
76#[derive(Clone, Debug)]
77pub struct PointResult {
78    pub mach: f64,
79    /// Angle of attack \[rad\]; always 0.0 (the solver is axisymmetric).
80    pub alpha: f64,
81    pub iters: usize,
82    pub residual: Vec<f64>,
83    /// Largest residual after iteration 20 (cold start) / of the run (warm start).
84    pub residual_peak: f64,
85    /// Orders dropped from `residual_peak` to the last residual.
86    pub drop_orders: f64,
87    pub stop_reason: StopReason,
88    /// Wall faces of the nose, cylinder and boattail (and sting extension) in axial order.
89    pub wall: Vec<WallPoint>,
90    /// Base-face samples (empty without base), `r0 < r1`.
91    pub base_wall: Vec<BasePoint>,
92    /// Forebody CD from the extrapolated wall Cp and from the first-cell Cp.
93    pub cd_nose: f64,
94    pub cd_nose_cell: f64,
95    /// Pressure drag of each aft segment (`body.segments` order), from the extrapolated wall Cp.
96    pub cd_aft: Vec<f64>,
97    pub cd_base: f64,
98    /// Skin-friction drag of the body surface (`0.0` for the Euler equations).
99    pub cd_friction: f64,
100    /// `cd_nose + cd_aft[0] + cd_aft[1] + .. + cd_friction + cd_base` (`cd_friction` is 0 for the Euler equations).
101    pub cd_total: f64,
102    /// Base drag of the empirical free-flight correlation scaled to `pi R^2`; 0 without base, NaN for a sting-mounted base.
103    pub cd_base_model: f64,
104    /// `cd_nose + cd_aft[0] + cd_aft[1] + .. + cd_friction + cd_base_model`.
105    pub cd_total_model: f64,
106    /// Area-averaged base Cp (extrapolated wall values) and the first-cell variant; NaN without base.
107    pub cp_base: f64,
108    pub cp_base_cell: f64,
109    /// Mean and range of the area-mean base Cp over the monitor window (NaN range without enough history or without base, and
110    /// for a viscous run, whose monitor is the total CD).
111    pub cp_base_mean: f64,
112    pub cp_base_range: f64,
113    /// Max - min of the monitor over the monitor window when it is the forebody CD (sting wake) or the total CD (viscous run);
114    /// NaN otherwise or without enough history.
115    pub cd_range: f64,
116    /// Monitor samples `(iteration, value)` every `MONITOR_EVERY` iterations.
117    pub monitor_history: Vec<(usize, f64)>,
118    pub cp_min: f64,
119    pub x_cp_min: f64,
120    pub m_max: f64,
121    /// Maximum wall-cell Mach number.
122    pub m_wall_max: f64,
123    /// Max `|p/p_inf - 1|` over the boundary cells of the inflow, outer and outflow boundaries.
124    pub farfield_dp: f64,
125    pub min_rho: f64,
126    pub min_p: f64,
127    /// Largest wall distance `y+` of the first cell over the body surface (NaN for the Euler equations).
128    pub y_plus_max: f64,
129}
