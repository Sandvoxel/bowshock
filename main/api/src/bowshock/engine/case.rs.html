10/// Treatment of the region behind the body.
11#[derive(Clone, Copy, Debug, PartialEq, Eq)]
12pub enum Wake {
13    /// The cylinder continues as a sting to the outflow plane (`large_domain_grid`), no base.
14    Sting,
15    /// Finite body with a flat base and a wake region (`base_domain_grid`).
16    Base,
17}
18
19/// Motor jet at the nozzle exit, internal units.
20#[derive(Clone, Copy, Debug, PartialEq)]
21pub struct Jet {
22    /// Exit radius.
23    pub r_exit: f64,
24    /// Primitive state at the exit (supersonic, axial).
25    pub state: Prim,
26    /// `[k, omega]` of the jet.
27    pub turb: [f64; 2],
28}
29
30/// One Mach number on one body.
31#[derive(Clone, Debug)]
32pub struct Case {
33    pub mach: f64,
34    pub body: BaseBody,
35    pub wake: Wake,
36    pub grid_base: BaseGridParams,
37    pub grid_sting: LargeGridParams,
38    pub numerics: Numerics,
39    /// Viscous flow parameters (RANS with the SST model); `None`: Euler equations.
40    pub viscous: Option<Viscous>,
41    /// Motor jet from the base; `None`: power-off.
42    pub motor: Option<Jet>,
43}
44
45impl Case {
46    /// Domain `10 L` upstream and radially, `5 L` downstream (`L` = total body length), grid level 0/1/2 (cells per diameter
47    /// doubling each level).
48    pub fn new(mach: f64, body: BaseBody, level: u32, wake: Wake, numerics: Numerics) -> Case {
49        let f = 1usize << level;
50        let dx_tip = 0.03 / f as f64;
51        let l_body = body.x_base();
52        let n_sting = if body.sting_radius > 0.0 { 4 * f } else { 0 };
53        let nseg = body.segments.len();
54        // cells per segment: tubes are finer than cones, never finer than the end-cell size `dx_tip` allows
55        let count = |per: usize, length: f64| per.min(even_ceil(length / dx_tip)).max(4);
56        let n_seg_base: Vec<usize> = body.segments.iter().map(|s| count(if s.slope == 0.0 { 40 * f } else { 24 * f }, s.length)).collect();
57        let n_seg_sting: Vec<usize> = body
58            .segments
59            .iter()
60            .enumerate()
61            .map(|(k, s)| {
62                count(
63                    if s.slope != 0.0 {
64                        24 * f
65                    } else if k + 1 == nseg {
66                        36 * f
67                    } else {
68                        40 * f
69                    },
70                    s.length,
71                )
72            })
73            .collect();
74        let grid_base = BaseGridParams {
75            body: body.clone(),
76            x_up: 10.0 * l_body,
77            x_down: 5.0 * l_body,
78            r_far: 10.0 * l_body,
79            n_up: 28 * f,
80            n_nose: 48 * f,
81            n_seg: n_seg_base,
82            n_wake: 40 * f,
83            n_sting,
84            n_gap: 12 * f,
85            nj_out: 64 * f,
86            dx_tip,
87            dr_wall: 0.03 / f as f64,
88            gap_one_sided: false,
89            nozzle: None,
90        };
91        let grid_sting = LargeGridParams {
92            body: body.clone(),
93            x_up: 10.0 * l_body,
94            x_down: 5.0 * l_body,
95            r_far: 10.0 * l_body,
96            n_up: 30 * f,
97            n_nose: 48 * f,
98            n_seg: n_seg_sting,
99            n_ext: 30 * f,
100            nj: 70 * f,
101            dx_tip,
102            dr_wall: 0.03 / f as f64,
103        };
104        Case { mach, body, wake, grid_base, grid_sting, numerics, viscous: None, motor: None }
105    }
106
107    /// Wall-resolved RANS case: [`Case::new`] with [`Numerics::viscous`], first cell height at the wall for `y+ ~ 1` at the highest
108    /// Mach number of the sweep `mach_max` (flat-plate friction estimate over the body length), radial cell counts for a growth
109    /// ratio of 1.2, the gap behind the base clustered at the base corner only, and the SST model with the no-slip surface starting at
110    /// the nose tip. `re_c = rho c d_nose / mu` of the free stream, `sutherland = 110.4 K / T_inf`.
111    pub fn viscous(mach: f64, body: BaseBody, level: u32, wake: Wake, re_c: f64, sutherland: f64, mach_max: f64) -> Case {
112        let mut c = Case::new(mach, body, level, wake, Numerics::viscous(mach, level));
113        let cf = 0.027 * (re_c * mach_max * c.body.x_base()).powf(-1.0 / 7.0);
114        let y1 = 1.0 / (re_c * mach_max * (0.5 * cf).sqrt());
115        let count = |h: f64| layer_count(h, y1);
116        let (re, rs) = (c.body.base_radius(), c.body.sting_radius);
117        c.grid_base.dr_wall = y1;
118        c.grid_base.nj_out = count(c.grid_base.r_far - re);
119        c.grid_base.n_gap = count(re - rs);
120        c.grid_base.gap_one_sided = true;
121        c.grid_sting.dr_wall = y1;
122        c.grid_sting.nj = count(c.grid_sting.r_far);
123        let wall_i_start = match wake {
124            Wake::Base => c.grid_base.n_up,
125            Wake::Sting => c.grid_sting.n_up,
126        };
127        c.viscous = Some(Viscous { re_c, sutherland, turbulence: Turbulence::Sst, wall_i_start });
128        c
129    }
130
131    /// Add a motor jet from the centre of the base to a case built by [`Case::viscous`] (panics otherwise, or without the
132    /// base wake): splits the gap behind the base into the nozzle exit (rows clustered at the lip, `n_jet` rows) and the annulus
133    /// around it (`n_ann` rows, at least 4), `n_gap = n_jet + n_ann`.
134    pub fn with_motor(mut self, jet: Jet) -> Case {
135        assert!(self.viscous.is_some() && self.wake == Wake::Base, "a motor needs a viscous case with the base wake");
136        let y1 = self.grid_base.dr_wall;
137        let re = self.body.base_radius();
138        let n_jet = layer_count(jet.r_exit, y1);
139        let n_ann = (2 * layer_count(0.5 * (re - jet.r_exit), y1)).max(4);
140        self.grid_base.n_gap = n_jet + n_ann;
141        self.grid_base.nozzle = Some(NozzleRows { r_exit: jet.r_exit, n_jet });
142        self.motor = Some(jet);
143        self
144    }
145
146    /// Scale the domain: inflow plane / outer boundary at `up` and the outflow plane at `down` total body lengths.
147    pub fn with_domain(mut self, up: f64, down: f64) -> Case {
148        let l = self.body.x_base();
149        self.grid_base.x_up = up * l;
150        self.grid_base.r_far = up * l;
151        self.grid_base.x_down = down * l;
152        self.grid_sting.x_up = up * l;
153        self.grid_sting.r_far = up * l;
154        self.grid_sting.x_down = down * l;
155        self
156    }
157
158    /// Cell counts `(ni, nj)` of the grid selected by `wake`.
159    pub fn cells(&self) -> (usize, usize) {
160        let g = &self.grid_sting;
161        let b = &self.grid_base;
162        match self.wake {
163            Wake::Base => (b.n_up + b.n_nose + b.n_seg.iter().sum::<usize>() + b.n_wake, b.n_sting + b.n_gap + b.nj_out),
164            Wake::Sting => (g.n_up + g.n_nose + g.n_seg.iter().sum::<usize>() + g.n_ext, g.nj),
165        }
166    }

168    /// Solver on the grid selected by `wake`, configured from `numerics`; panics on invalid grid parameters.
169    pub fn build_solver(&self) -> Solver {
170        self.try_build_solver().expect("valid grid parameters")
171    }
172
173    /// Like [`Case::build_solver`], returning the grid error instead of panicking.
174    pub fn try_build_solver(&self) -> Result<Solver, Error> {
175        let n = &self.numerics;
176        let cfg = Config {
177            cfl: n.cfl,
178            limiter: n.limiter,
179            freestream: freestream(self.mach),
180            bc: [Bc::Characteristic, n.outflow, Bc::SlipWall, Bc::Characteristic],
181            first_order_iters: n.first_order_iters,
182        };
183        let mut s;
184        match self.wake {
185            Wake::Base => {
186                let (grid, block) = base_domain_grid(&self.grid_base)?;
187                s = Solver::new(grid, cfg);
188                s.set_block(Some(block));
189                if let Some(jet) = &self.motor {
190                    let n_jet = self.grid_base.nozzle.map_or(0, |z| z.n_jet);
191                    s.set_nozzle(Some(Nozzle { j_end: n_jet, state: jet.state, turb: jet.turb }));
192                }
193            }
194            Wake::Sting => {
195                s = Solver::new(large_domain_grid(&self.grid_sting)?, cfg);
196            }
197        }
198        s.low_mach = n.low_mach;
199        s.set_scheme(n.scheme);
200        s.set_viscous(self.viscous);
201        Ok(s)
202    }
203}

205/// Smallest even integer `>= x` (`x >= 0`).
206fn even_ceil(x: f64) -> usize {
207    let n = x.ceil() as usize;
208    n + n % 2
209}

211/// Number of rows of a wall-clustered layer of height `h` with first cell height `y1` and growth ratio 1.2 (even).
212fn layer_count(h: f64, y1: f64) -> usize {
213    even_ceil((1.0 + 0.2 * h / y1).ln() / 1.2f64.ln())
