1//! Incompressible potential flow past a closed body of revolution at zero incidence: an independent reference for the
2//! low-Mach Euler results (no CFD code involved).
3//!
4//! Method (method of fundamental solutions / axial source distribution): the flow is `U e_x` plus the field of `n_s` point
5//! sources of unknown strength `q_k` placed on the axis inside the body,
6//!
7//! ```text
8//!   V(P) = U e_x + sum_k q_k (P - P_k) / (4 pi |P - P_k|^3),
9//! ```
10//!
11//! and the strengths solve the no-penetration condition `V.n = 0` at `n_c > n_s` collocation points on the surface (regularised
12//! least squares through the normal equations; the net source `sum q_k = 0` is added as a soft constraint). The surface speed gives
13//! `Cp = 1 - |V|^2 / U^2`. Axisymmetric potential flows are represented by axial singularity distributions, so the solution
14//! converges to the exact one as the point sets are refined; it is verified against the exact sphere solution
15//! (`Cp = 1 - 9/4 sin^2 theta`) in the unit tests.
16//!
17//! Compressibility (linearised, thin body): Goethert's rule for bodies of revolution, `Cp_M[R](x) = Cp_inc[beta R](x) / beta^2`,
18//! `beta = sqrt(1 - M^2)` (the Prandtl-Glauert transformation of the axisymmetric small-perturbation equation
19//! `beta^2 phi_xx + phi_rr + phi_r / r = 0`), i.e. the incompressible solution for the body with radius scaled by `beta`.
20
21use std::f64::consts::PI;

23/// A closed body of revolution given by its meridian polyline `(x, r)` from the nose tip (`r = 0`) to the tail tip (`r = 0`).
24#[derive(Clone, Debug)]
25pub struct Meridian {
26    pub pts: Vec<(f64, f64)>,
27}
28
29/// Solution on the surface: panel mid-points and their `Cp`.
30#[derive(Clone, Debug)]
31pub struct PotentialSolution {
32    pub x: Vec<f64>,
33    pub r: Vec<f64>,
34    pub cp: Vec<f64>,
35    /// Largest `|sum_k q_k| / max |q_k|` (closure check) and the RMS no-penetration residual `V.n / U` at the collocation points.
36    pub closure: f64,
37    pub residual: f64,
38}
39
40impl PotentialSolution {
41    /// `Cp` at axial position `x` on the upper half of the body (linear interpolation of the panel values).
42    pub fn cp_at(&self, x: f64) -> Option<f64> {
43        for k in 0..self.x.len() - 1 {
44            let (a, b) = (self.x[k], self.x[k + 1]);
45            if (x >= a && x <= b) || (x >= b && x <= a) {
46                if (b - a).abs() < 1e-14 {
47                    return Some(self.cp[k]);
48                }
49                let t = (x - a) / (b - a);
50                return Some(self.cp[k] + t * (self.cp[k + 1] - self.cp[k]));
51            }
52        }
53        None
54    }
55}

57fn solve_dense(a: &mut [f64], b: &mut [f64], n: usize) {
58    // Gaussian elimination with partial pivoting (a: n x n row-major)
59    for c in 0..n {
60        let mut piv = c;
61        for r in c + 1..n {
62            if a[r * n + c].abs() > a[piv * n + c].abs() {
63                piv = r;
64            }
65        }
66        if piv != c {
67            for k in 0..n {
68                a.swap(c * n + k, piv * n + k);
69            }
70            b.swap(c, piv);
71        }
72        let d = a[c * n + c];
73        for r in c + 1..n {
74            let f = a[r * n + c] / d;
75            if f != 0.0 {
76                for k in c..n {
77                    a[r * n + k] -= f * a[c * n + k];
78                }
79                b[r] -= f * b[c];
80            }
81        }
82    }
83    for c in (0..n).rev() {
84        let mut s = b[c];
85        for k in c + 1..n {
86            s -= a[c * n + k] * b[k];
87        }
88        b[c] = s / a[c * n + c];
89    }
90}

92/// Solve the potential flow past `body` with `n_src` axial sources (Chebyshev-clustered over the interior of the body's axial
93/// extent, `margin` = fraction of the length kept free at both ends) and one collocation point per polyline segment (refine the
94/// polyline for more). Free stream `U = 1` along `+x`.
95pub fn solve(body: &Meridian, n_src: usize, margin: f64) -> PotentialSolution {
96    let n_c = body.pts.len() - 1;
97    let (x0, x1) = (body.pts[0].0, body.pts.last().unwrap().0);
98    let len = x1 - x0;
99    // Chebyshev-Lobatto-like source positions
100    let xs: Vec<f64> = (0..n_src)
101        .map(|k| {
102            let t = 0.5 * (1.0 - (PI * (k as f64 + 0.5) / n_src as f64).cos());
103            x0 + margin * len + (1.0 - 2.0 * margin) * len * t
104        })
105        .collect();
106    // collocation points, normals (pointing out of the body; the meridian runs tip -> tail on the upper half, so the outward
107    // normal of a segment with tangent (tx, tr) is (tr, -tx)).
108    let mut col = Vec::with_capacity(n_c);
109    for k in 0..n_c {
110        let (a, b) = (body.pts[k], body.pts[k + 1]);
111        let (tx, tr) = (b.0 - a.0, b.1 - a.1);
112        let l = tx.hypot(tr);
113        col.push((0.5 * (a.0 + b.0), 0.5 * (a.1 + b.1), tr / l, -tx / l));
114    }
115    // velocity influence of a unit source at (xk, 0) on point (x, r): (dx, dr) / (4 pi d^3)
116    let infl = |x: f64, r: f64, xk: f64| -> (f64, f64) {
117        let (dx, d2) = (x - xk, (x - xk) * (x - xk) + r * r);
118        let f = 1.0 / (4.0 * PI * d2 * d2.sqrt());
119        (dx * f, r * f)
120    };
121    // normal equations for min sum (V.n)^2 + soft closure
122    let mut a = vec![0.0; n_src * n_src];
123    let mut rhs = vec![0.0; n_src];
124    let mut rows: Vec<Vec<f64>> = Vec::with_capacity(n_c);
125    let mut b0 = Vec::with_capacity(n_c);
126    for &(x, r, nx, nr) in &col {
127        let row: Vec<f64> = xs
128            .iter()
129            .map(|&xk| {
130                let (vx, vr) = infl(x, r, xk);
131                vx * nx + vr * nr
132            })
133            .collect();
134        b0.push(-nx); // U e_x . n = nx, moved to the right-hand side
135        rows.push(row);
136    }
137    for (row, &b) in rows.iter().zip(b0.iter()) {
138        for i in 0..n_src {
139            rhs[i] += row[i] * b;
140            for j in 0..n_src {
141                a[i * n_src + j] += row[i] * row[j];
142            }
143        }
144    }
145    let scale = (0..n_src).map(|i| a[i * n_src + i]).fold(0.0, f64::max);
146    for i in 0..n_src {
147        a[i * n_src + i] += 1e-13 * scale;
148    }
149    let w_closure = 1.0 * n_c as f64;
150    for i in 0..n_src {
151        for j in 0..n_src {
152            a[i * n_src + j] += w_closure * 1e-6;
153        }
154    }
155    let mut q = rhs;
156    solve_dense(&mut a, &mut q, n_src);
157    let qmax = q.iter().fold(0.0f64, |m, v| m.max(v.abs()));
158    let closure = q.iter().sum::<f64>().abs() / qmax.max(1e-300);
159    let (mut cx, mut cr, mut cp) = (Vec::new(), Vec::new(), Vec::new());
160    let mut res2 = 0.0;
161    for &(x, r, nx, nr) in &col {
162        let (mut vx, mut vr) = (1.0, 0.0);
163        for (k, &xk) in xs.iter().enumerate() {
164            let (ux, ur) = infl(x, r, xk);
165            vx += q[k] * ux;
166            vr += q[k] * ur;
167        }
168        res2 += (vx * nx + vr * nr).powi(2);
169        cx.push(x);
170        cr.push(r);
171        cp.push(1.0 - (vx * vx + vr * vr));
172    }
173    PotentialSolution { x: cx, r: cr, cp, closure, residual: (res2 / n_c as f64).sqrt() }
174}

176/// Closed body: nose profile `r_nose(x)` on `[0, l_nose]`, cylinder of radius `r_cyl` up to `x = l_nose + l_cyl`, mirrored tail
177/// (the nose profile reversed), `n_nose` / `n_cyl` segments (cosine clustering at the ends of each part).
178pub fn nose_cylinder_body(r_nose: &dyn Fn(f64) -> f64, l_nose: f64, l_cyl: f64, r_cyl: f64, n_nose: usize, n_cyl: usize) -> Meridian {
179    let cl = |k: usize, n: usize| 0.5 * (1.0 - (PI * k as f64 / n as f64).cos());
180    let mut pts = Vec::new();
181    for k in 0..=n_nose {
182        let x = l_nose * cl(k, n_nose);
183        pts.push((x, if k == 0 { 0.0 } else { r_nose(x) }));
184    }
185    for k in 1..n_cyl {
186        pts.push((l_nose + l_cyl * cl(k, n_cyl), r_cyl));
187    }
188    let x_tail = l_nose + l_cyl;
189    for k in 0..=n_nose {
190        let xt = l_nose * (1.0 - cl(n_nose - k, n_nose));
191        // tail point at x_tail + xt has the radius of the nose at l_nose - xt
192        let xn = l_nose - xt;
193        pts.push((x_tail + xt, if k == n_nose { 0.0 } else { r_nose(xn.max(0.0)) }));
194    }
195    Meridian { pts }
196}

198/// Goethert-rule compressible `Cp` of the same body at Mach `m` (incompressible solution for the radius scaled by `beta`,
199/// divided by `beta^2`).
200pub fn solve_compressible(body: &Meridian, mach: f64, n_src: usize, margin: f64) -> PotentialSolution {
201    let beta = (1.0 - mach * mach).sqrt();
202    let scaled = Meridian { pts: body.pts.iter().map(|&(x, r)| (x, beta * r)).collect() };
203    let mut sol = solve(&scaled, n_src, margin);
204    for c in sol.cp.iter_mut() {
205        *c /= beta * beta;
206    }
207    for r in sol.r.iter_mut() {
208        *r /= beta;
209    }
210    sol
211}

213#[cfg(test)]
214mod tests {
215    use super::*;
216
217    #[test]
218    fn sphere_matches_the_exact_solution() {
219        // sphere of radius 0.5 centred at x = 0.5, polyline from the nose to the tail (angle from the nose)
220        let n = 160;
221        let pts: Vec<(f64, f64)> = (0..=n)
222            .map(|k| {
223                let th = PI * k as f64 / n as f64;
224                (0.5 - 0.5 * th.cos(), 0.5 * th.sin())
225            })
226            .collect();
227        let sol = solve(&Meridian { pts }, 60, 0.02);
228        let mut worst: f64 = 0.0;
229        for k in 0..sol.x.len() {
230            let th = ((0.5 - sol.x[k]) / 0.5).clamp(-1.0, 1.0).acos(); // angle from the nose
231            let exact = 1.0 - 2.25 * th.sin().powi(2);
232            worst = worst.max((sol.cp[k] - exact).abs());
233        }
234        assert!(worst < 5e-3, "sphere Cp error {worst}");
235        assert!(sol.closure < 1e-3, "net source {}", sol.closure);
236    }
237
238    #[test]
239    fn slender_body_cp_scales_like_the_thickness_squared_and_goethert_is_nearly_mach_independent() {
240        // slender ellipsoid-like nose-cylinder: compressibility has (almost) no effect on a slender axisymmetric body
241        let nose = |x: f64| 0.5 * (1.0 - ((3.0 - x) / 3.0).powi(2)).max(0.0).sqrt();
242        let body = nose_cylinder_body(&nose, 3.0, 8.0, 0.5, 80, 120);
243        let inc = solve(&body, 200, 0.003);
244        let m5 = solve_compressible(&body, 0.5, 200, 0.003);
245        let cp_inc = inc.cp_at(1.5).unwrap();
246        let cp_m5 = m5.cp_at(1.5).unwrap();
247        assert!(cp_inc < 0.0 && (cp_m5 - cp_inc).abs() < 0.2 * cp_inc.abs(), "{cp_inc} vs {cp_m5}");
248        assert!(inc.residual < 1e-3, "no-penetration residual {}", inc.residual);
249    }
