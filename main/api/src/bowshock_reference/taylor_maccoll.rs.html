1//! Taylor-Maccoll solution for supersonic flow over a sharp circular cone at zero incidence.
2//!
3//! In spherical coordinates (R, theta) with velocities `Vr`, `Vt` made non-dimensional by
4//! `V_max = sqrt(2 h0)` the conical flow obeys (Anderson, *Modern Compressible Flow*, eq. 10.15)
5//!
6//! ```text
7//!   Vr'  = Vt
8//!   Vt'  = ( Vt^2 Vr - k (2 Vr + Vt cot(theta)) ) / ( k - Vt^2 ),   k = (g-1)/2 (1 - Vr^2 - Vt^2)
9//! ```
10//!
11//! Starting just behind an oblique shock of angle `beta` (velocity components from the oblique
12//! shock relations) the ODE is integrated with RK4 towards smaller `theta` until `Vt = 0`, the
13//! cone surface. `beta` is found by bisection such that that angle equals the cone half-angle.
14
15use bowshock_cfd::gas::GAMMA;
16
17#[derive(Clone, Copy, Debug)]
18pub struct ConeSolution {
19    pub beta: f64,
20    /// Cone-surface Mach number and `p / p_inf`.
21    pub mach_surface: f64,
22    pub p_ratio_surface: f64,
23    pub cp_surface: f64,
24    /// Post-shock Mach number and `p / p_inf`.
25    pub mach_post_shock: f64,
26    pub p_ratio_post_shock: f64,
27}
28
29/// Flow deflection angle `delta` of an oblique shock with wave angle `beta`.
30pub fn oblique_deflection(m1: f64, beta: f64) -> f64 {
31    let g = GAMMA;
32    let num = 2.0 / beta.tan() * (m1 * m1 * beta.sin().powi(2) - 1.0);
33    let den = m1 * m1 * (g + (2.0 * beta).cos()) + 2.0;
34    (num / den).atan()
35}
36
37/// Post-shock `(M2, p2/p1)` of an oblique shock.
38pub fn oblique_post_shock(m1: f64, beta: f64) -> (f64, f64) {
39    let g = GAMMA;
40    let mn1 = m1 * beta.sin();
41    let delta = oblique_deflection(m1, beta);
42    let mn2sq = (1.0 + 0.5 * (g - 1.0) * mn1 * mn1) / (g * mn1 * mn1 - 0.5 * (g - 1.0));
43    let m2 = mn2sq.sqrt() / (beta - delta).sin();
44    let p21 = 1.0 + 2.0 * g / (g + 1.0) * (mn1 * mn1 - 1.0);
45    (m2, p21)
46}
47
48#[inline]
49fn rhs(th: f64, y: [f64; 2]) -> [f64; 2] {
50    let k = 0.5 * (GAMMA - 1.0) * (1.0 - y[0] * y[0] - y[1] * y[1]);
51    let vt = y[1];
52    let dvt = (vt * vt * y[0] - k * (2.0 * y[0] + vt / th.tan())) / (k - vt * vt);
53    [vt, dvt]
54}
55
56fn rk4(th: f64, y: [f64; 2], h: f64) -> [f64; 2] {
57    let k1 = rhs(th, y);
58    let k2 = rhs(th + 0.5 * h, [y[0] + 0.5 * h * k1[0], y[1] + 0.5 * h * k1[1]]);
59    let k3 = rhs(th + 0.5 * h, [y[0] + 0.5 * h * k2[0], y[1] + 0.5 * h * k2[1]]);
60    let k4 = rhs(th + h, [y[0] + h * k3[0], y[1] + h * k3[1]]);
61    [y[0] + h / 6.0 * (k1[0] + 2.0 * k2[0] + 2.0 * k3[0] + k4[0]), y[1] + h / 6.0 * (k1[1] + 2.0 * k2[1] + 2.0 * k3[1] + k4[1])]
62}
63
64/// Integrate from the shock; returns `(theta_cone, Vr_surface)` (the angle where `Vt` crosses zero).
65fn integrate_from_shock(m1: f64, beta: f64) -> (f64, f64) {
66    let (m2, _) = oblique_post_shock(m1, beta);
67    let delta = oblique_deflection(m1, beta);
68    // V/Vmax from M: V^2/Vmax^2 = (g-1)/2 M^2 / (1 + (g-1)/2 M^2)
69    let a = 0.5 * (GAMMA - 1.0) * m2 * m2;
70    let v = (a / (1.0 + a)).sqrt();
71    let mut y = [v * (beta - delta).cos(), -v * (beta - delta).sin()];
72    let mut th = beta;
73    let h = -1e-5;
74    while th > 1e-3 {
75        let yn = rk4(th, y, h);
76        if yn[1] >= 0.0 {
77            // linear interpolation of the zero crossing
78            let t = y[1] / (y[1] - yn[1]);
79            return (th + t * h, y[0] + t * (yn[0] - y[0]));
80        }
81        y = yn;
82        th += h;
83    }
84    (0.0, y[0])
85}

87/// Solve the weak-shock Taylor-Maccoll cone flow for Mach `m1` and cone half-angle `theta_c` \[rad\].
88/// Returns `None` if no attached conical shock exists.
89pub fn solve_cone(m1: f64, theta_c: f64) -> Option<ConeSolution> {
90    let mu = (1.0 / m1).asin();
91    // scan for the first bracket where theta_cone(beta) exceeds the target (weak branch)
92    let step = 2e-3;
93    let mut lo = mu + 1e-6;
94    let mut found = None;
95    let mut b = lo + step;
96    while b < 0.5 * std::f64::consts::PI {
97        if integrate_from_shock(m1, b).0 >= theta_c {
98            found = Some((lo, b));
99            break;
100        }
101        lo = b;
102        b += step;
103    }
104    let (mut lo, mut hi) = found?;
105    for _ in 0..50 {
106        let mid = 0.5 * (lo + hi);
107        if integrate_from_shock(m1, mid).0 >= theta_c {
108            hi = mid;
109        } else {
110            lo = mid;
111        }
112    }
113    let beta = 0.5 * (lo + hi);
114    let (_, vr) = integrate_from_shock(m1, beta);
115    let (m2, p21) = oblique_post_shock(m1, beta);
116    let g = GAMMA;
117    // total pressure behind the shock, then isentropic change to the cone surface
118    let p02_p1 = p21 * (1.0 + 0.5 * (g - 1.0) * m2 * m2).powf(g / (g - 1.0));
119    let t_t0 = 1.0 - vr * vr; // T/T0 with V in units of Vmax
120    let p_ratio_surface = p02_p1 * t_t0.powf(g / (g - 1.0));
121    // c^2/Vmax^2 = (g-1)/2 (1 - V^2)  =>  M^2 = V^2 / ((g-1)/2 (1-V^2))
122    let mach_surface = (vr * vr / (0.5 * (g - 1.0) * t_t0)).sqrt();
123    Some(ConeSolution {
124        beta,
125        mach_surface,
126        p_ratio_surface,
127        cp_surface: 2.0 * (p_ratio_surface - 1.0) / (g * m1 * m1),
128        mach_post_shock: m2,
129        p_ratio_post_shock: p21,
130    })
131}

133#[cfg(test)]
134mod tests {
135    use super::*;
136
137    const D: f64 = std::f64::consts::PI / 180.0;
138
139    #[test]
140    fn oblique_shock_table_value() {
141        // NACA 1135 / Anderson: M=2, delta=10 deg -> weak beta = 39.31 deg.
142        assert!((oblique_deflection(2.0, 39.3139 * D) / D - 10.0).abs() < 2e-3);
143    }
144
145    #[test]
146    fn cone_mach_2p35_10deg_nasa_wind_validation_case() {
147        // NASA NPARC Alliance validation archive "10 Degree Cone at Mach 2.35"
148        // (https://www.grc.nasa.gov/WWW/wind/valid/cone10/cone10.html). Its Euler CFD results
149        // (WIND/NPARC/Wind-US, Table 4) give cone-surface M3 = 2.1467..2.1469 and
150        // p3/p1 = 1.3740..1.3741, and agree with this solver. (The page's "theory" row,
151        // p3/p1 = 1.4234 and beta = 27.1843 deg, is not consistent with its own CFD nor with the
152        // M=2/15 deg reference below, so it is deliberately not used.)
153        let s = solve_cone(2.35, 10.0 * D).unwrap();
154        assert!((s.mach_surface - 2.1468).abs() < 3e-4, "M3 {}", s.mach_surface);
155        assert!((s.p_ratio_surface - 1.3740).abs() < 3e-4, "p3/p1 {}", s.p_ratio_surface);
156    }
157
158    #[test]
159    fn cone_mach_2_15deg_mallard_tutorial() {
160        // MatthewBonanni/mallard docs/tutorials/cone.md (taylor_maccoll.py):
161        // M=2, 15 deg: shock 33.9147 deg, p/pinf 1.56629, Cp 0.20225, surface Mach 1.7069.
162        let s = solve_cone(2.0, 15.0 * D).unwrap();
163        assert!((s.beta / D - 33.9147).abs() < 2e-3, "beta {}", s.beta / D);
164        assert!((s.p_ratio_surface - 1.56629).abs() < 2e-4, "p {}", s.p_ratio_surface);
165        assert!((s.cp_surface - 0.20225).abs() < 1e-4, "cp {}", s.cp_surface);
166        assert!((s.mach_surface - 1.7069).abs() < 5e-4, "M {}", s.mach_surface);
167    }
168
169    #[test]
170    fn cone_mach_2_10deg() {
171        // M=2, 10 deg: independent cross-check through the oblique-shock relations - the
172        // computed beta must lie between the Mach angle (30 deg) and the wedge angle for the same
173        // deflection (39.31 deg), and the conical shock is weaker than the 10 deg wedge shock.
174        let s = solve_cone(2.0, 10.0 * D).unwrap();
175        let b = s.beta / D;
176        assert!(b > 30.0 && b < 39.31, "beta {b}");
177        assert!((b - 31.206).abs() < 0.01, "beta {b}"); // self-consistent value, matches the ~31.2 deg in tables
178        assert!(s.p_ratio_surface > 1.0 && s.p_ratio_surface < 1.5);
179    }
