1//! Independent surface-pressure theory for slender bodies of revolution (no CFD involved):
2//!
3//! * **tangent cone**: local Cp of the Taylor-Maccoll cone whose half angle equals the local surface slope;
4//! * **shock-expansion**: Taylor-Maccoll cone flow at the (conical) nose tip, then Prandtl-Meyer expansion
5//!   along the surface through the turning angle `theta_tip - theta_local` (isentropic, surface streamline
6//!   only; shock curvature/vorticity and the axisymmetric relief are ignored).
7//!
8//! Both are classical first estimates; neither is exact on a curved ogive.
9
10use crate::taylor_maccoll::{solve_cone, ConeSolution};
11use bowshock_cfd::gas::GAMMA;
12
13/// Prandtl-Meyer function `nu(M)` \[rad\].
14pub fn prandtl_meyer(m: f64) -> f64 {
15    let g = GAMMA;
16    let k = ((g + 1.0) / (g - 1.0)).sqrt();
17    let b = (m * m - 1.0).max(0.0).sqrt();
18    k * (b / k).atan() - b.atan()
19}
20
21/// Inverse of [`prandtl_meyer`] by bisection.
22pub fn mach_from_pm(nu: f64) -> f64 {
23    let (mut lo, mut hi) = (1.0f64, 200.0f64);
24    for _ in 0..200 {
25        let mid = 0.5 * (lo + hi);
26        if prandtl_meyer(mid) < nu {
27            lo = mid;
28        } else {
29            hi = mid;
30        }
31    }
32    0.5 * (lo + hi)
33}
34
35/// `p/p_inf` -> `Cp` at free-stream Mach `m`.
36pub fn cp_from_ratio(p_ratio: f64, m: f64) -> f64 {
37    2.0 * (p_ratio - 1.0) / (GAMMA * m * m)
38}

40/// Tangent-ogive geometry. Lengths are in units of the body diameter `d = 1` unless noted.
41#[derive(Clone, Copy, Debug)]
42pub struct Ogive {
43    /// Nose length / diameter.
44    pub fineness: f64,
45}
46
47impl Ogive {
48    pub fn radius(&self) -> f64 {
49        0.5
50    }
51    pub fn nose_length(&self) -> f64 {
52        self.fineness
53    }
54    /// Ogive arc radius `rho = (L^2 + R^2) / (2R)`.
55    pub fn rho(&self) -> f64 {
56        let (l, r) = (self.fineness, self.radius());
57        (l * l + r * r) / (2.0 * r)
58    }
59    /// Body radius at `x` (cylinder for `x >= L`).
60    pub fn r(&self, x: f64) -> f64 {
61        let l = self.fineness;
62        if x >= l {
63            return self.radius();
64        }
65        let x = x.max(0.0);
66        let rho = self.rho();
67        (rho * rho - (l - x).powi(2)).max(0.0).sqrt() - (rho - self.radius())
68    }
69    /// Surface slope `dr/dx`.
70    pub fn slope(&self, x: f64) -> f64 {
71        let l = self.fineness;
72        if x >= l {
73            return 0.0;
74        }
75        let x = x.max(0.0);
76        let rho = self.rho();
77        (l - x) / (rho * rho - (l - x).powi(2)).max(1e-300).sqrt()
78    }
79    /// Half angle of the cone tangent to the nose tip \[rad\].
80    pub fn tip_half_angle(&self) -> f64 {
81        self.slope(0.0).atan()
82    }
83}

85/// Theory evaluator at one Mach number.
86pub struct SurfaceTheory {
87    pub mach: f64,
88    pub body: Ogive,
89    /// Taylor-Maccoll solution of the tip cone (None if the shock would detach).
90    pub tip: Option<ConeSolution>,
91}
92
93impl SurfaceTheory {
94    pub fn new(mach: f64, body: Ogive) -> Self {
95        SurfaceTheory { mach, body, tip: solve_cone(mach, body.tip_half_angle()) }
96    }
97
98    /// Tangent-cone Cp at `x`. Below 1 deg of local slope the cone Cp is scaled linearly with the slope
99    /// (`Cp -> 0` as `theta -> 0`) because the cone solver does not resolve vanishing cone angles.
100    pub fn cp_tangent_cone(&self, x: f64) -> f64 {
101        let theta = self.body.slope(x).atan();
102        let floor = 1f64.to_radians();
103        if theta >= floor {
104            solve_cone(self.mach, theta).map_or(f64::NAN, |s| s.cp_surface)
105        } else {
106            solve_cone(self.mach, floor).map_or(f64::NAN, |s| s.cp_surface) * theta / floor
107        }
108    }
109
110    /// Shock-expansion Cp at `x`.
111    pub fn cp_shock_expansion(&self, x: f64) -> f64 {
112        let Some(tip) = &self.tip else { return f64::NAN };
113        let theta0 = self.body.tip_half_angle();
114        let theta = self.body.slope(x).atan();
115        let nu = prandtl_meyer(tip.mach_surface) + (theta0 - theta);
116        let m2 = mach_from_pm(nu);
117        let t = |m: f64| 1.0 + 0.5 * (GAMMA - 1.0) * m * m;
118        let p_ratio = tip.p_ratio_surface * (t(tip.mach_surface) / t(m2)).powf(GAMMA / (GAMMA - 1.0));
119        cp_from_ratio(p_ratio, self.mach)
120    }
121
122    /// Forebody pressure-drag coefficient referenced to the base area `pi R^2`:
123    /// `CD = (1/R^2) * integral Cp d(r^2)`, integrated over the nose with `n` midpoint panels
124    /// uniform in `x`.
125    pub fn forebody_cd(&self, cp: impl Fn(&Self, f64) -> f64, n: usize) -> f64 {
126        let l = self.body.nose_length();
127        let r2 = self.body.radius().powi(2);
128        let mut s = 0.0;
129        for k in 0..n {
130            let (x0, x1) = (l * k as f64 / n as f64, l * (k + 1) as f64 / n as f64);
131            let xm = 0.5 * (x0 + x1);
132            s += cp(self, xm) * (self.body.r(x1).powi(2) - self.body.r(x0).powi(2));
133        }
134        s / r2
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    #[test]
143    fn pm_roundtrip() {
144        for m in [1.2, 2.0, 3.7] {
145            assert!((mach_from_pm(prandtl_meyer(m)) - m).abs() < 1e-9);
146        }
147        // known value: nu(2) = 26.38 deg
148        assert!((prandtl_meyer(2.0).to_degrees() - 26.3798).abs() < 1e-3);
149    }
150
151    #[test]
152    fn ogive_geometry() {
153        let o = Ogive { fineness: 3.0 };
154        assert!((o.rho() - 9.25).abs() < 1e-12);
155        assert!(o.r(0.0).abs() < 1e-12);
156        assert!((o.r(3.0) - 0.5).abs() < 1e-12);
157        assert!((o.tip_half_angle().to_degrees() - (3.0f64 / 8.75).atan().to_degrees()).abs() < 1e-12);
158    }
159
160    #[test]
161    fn shock_expansion_starts_at_cone_value() {
162        let th = SurfaceTheory::new(2.0, Ogive { fineness: 3.0 });
163        let tip = th.tip.unwrap();
164        assert!((th.cp_shock_expansion(0.0) - tip.cp_surface).abs() < 1e-9);
165        assert!((th.cp_tangent_cone(0.0) - tip.cp_surface).abs() < 1e-6);
166    }
