1//! Progress of a running solve: a callback that sees where each run is and can stop it.
2//!
3//! The solvers iterate for seconds to minutes per Mach number. [`Analysis::on_progress`](crate::Analysis::on_progress) registers a
4//! callback that is called from the thread that runs the analysis, once per [`SAMPLE_INTERVAL`] solver iterations (the interval at
5//! which the stop rules sample their monitor), with a [`Progress`] snapshot. Reporting reads numbers the solver already has: it does
6//! not change the iteration, the stop rules or the result. The callback decides whether to go on: returning
7//! [`ControlFlow::Break`] ends the run at once and the analysis returns [`Error::Cancelled`](crate::Error::Cancelled).
8//!
9//! The callback runs on the solver's thread between iterations: keep it short (hand the snapshot to a channel, throttle there).
10
11use std::fmt;
12use std::ops::ControlFlow;
13use std::sync::Arc;
14
15/// Solver iterations between two [`Progress`] reports: the sampling interval of the stop rules' monitor.
16pub const SAMPLE_INTERVAL: usize = crate::engine::numerics::MONITOR_EVERY;
17
18/// Where one run (one Mach number, one angle of attack) is.
19#[derive(Clone, Copy, Debug, PartialEq)]
20pub struct Progress {
21    /// Mach number of the run.
22    pub mach: f64,
23    /// Angle of attack \[rad\]: 0 for the zero-lift drag solver, the angle of the 3D stability solve.
24    pub alpha: f64,
25    /// Iterations done in this run.
26    pub iteration: usize,
27    /// The iteration cap of this run.
28    pub max_iterations: usize,
29    /// Orders of magnitude the density residual has dropped from its peak (0 until it has a peak).
30    pub residual_drop_orders: f64,
31    /// The quantity the stop rule watches at the latest sample: the total drag coefficient of a viscous drag run (on the analysis's
32    /// reference area), the normal force coefficient of a 3D run; NaN before the first sample. Other runs report their own monitor
33    /// (the nose drag of an Euler body with a sting, the base pressure of an Euler base wake) in the solver's internal units.
34    pub monitor: f64,
35    /// Range (max − min) of the monitor over the last stop-rule window relative to the magnitude of its mean; `None` until the
36    /// window is full. A run stops on its stationary rule when this drops below [`Progress::tolerance`] (and the residual condition
37    /// of the rule holds).
38    pub spread: Option<f64>,
39    /// The relative tolerance of the stationary rule, if the run has one.
40    pub tolerance: Option<f64>,
41}
42
43/// The callback of [`Analysis::on_progress`](crate::Analysis::on_progress), cheap to clone.
44#[derive(Clone)]
45pub struct ProgressSink(Arc<dyn Fn(&Progress) -> ControlFlow<()> + Send + Sync>);
46
47impl ProgressSink {
48    /// A sink calling `f`.
49    pub fn new(f: impl Fn(&Progress) -> ControlFlow<()> + Send + Sync + 'static) -> ProgressSink {
50        ProgressSink(Arc::new(f))
51    }
52
53    /// Reports `progress`; `Break` asks the run to stop.
54    pub(crate) fn report(&self, progress: &Progress) -> ControlFlow<()> {
55        (self.0)(progress)
56    }
57
58    /// A sink that multiplies the monitor by `k` (the coefficient scaling of the reference area) before it reports.
59    pub(crate) fn scaled(&self, k: f64) -> ProgressSink {
60        let inner = self.clone();
61        ProgressSink::new(move |p| inner.report(&Progress { monitor: p.monitor * k, ..*p }))
62    }
63}
64
65impl fmt::Debug for ProgressSink {
66    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
67        f.write_str("ProgressSink")
68    }
69}

71/// Range over the magnitude of the mean of the last `k` samples of `hist` (`(iteration, monitor)`); `None` with fewer than `k`
72/// samples, `k = 0`, or a mean of zero.
73pub(crate) fn spread(hist: &[(usize, f64)], k: usize) -> Option<f64> {
74    if k == 0 || hist.len() < k {
75        return None;
76    }
77    let w = &hist[hist.len() - k..];
78    let (lo, hi, sum) = w.iter().fold((f64::INFINITY, f64::NEG_INFINITY, 0.0), |(l, h, s), c| (l.min(c.1), h.max(c.1), s + c.1));
79    let mean = (sum / k as f64).abs();
80    (mean > 0.0 && mean.is_finite()).then_some((hi - lo) / mean)
81}
82
83/// Orders of drop of a residual `r` below its reference `peak` (0 without a peak).
84pub(crate) fn drop_orders(peak: f64, r: f64) -> f64 {
85    if peak > 0.0 && r.is_finite() {
86        (peak / r.max(1e-300)).log10()
87    } else {
88        0.0
89    }
90}

92#[cfg(test)]
93mod tests {
94    use super::*;
95
96    #[test]
97    fn the_spread_is_the_range_over_the_mean_of_the_last_window() {
98        let hist: Vec<(usize, f64)> = [9.0, 1.0, 1.01, 0.99, 1.0].iter().enumerate().map(|(i, m)| ((i + 1) * 50, *m)).collect();
99        // the last four samples: range 0.02, mean 1.0
100        assert!((spread(&hist, 4).unwrap() - 0.02).abs() < 1e-12);
101        // the early outlier is outside the window of 4 but inside a window of 5
102        assert!(spread(&hist, 5).unwrap() > 1.0);
103        assert_eq!(spread(&hist, 6), None, "the window is not full");
104        assert_eq!(spread(&hist, 0), None);
105        assert_eq!(spread(&[(50, 0.0), (100, 0.0)], 2), None, "a zero mean has no relative spread");
106    }
107
108    #[test]
109    fn the_drop_is_in_orders_below_the_peak() {
110        assert!((drop_orders(1.0, 1e-3) - 3.0).abs() < 1e-12);
111        assert_eq!(drop_orders(0.0, 1e-3), 0.0);
112        assert_eq!(drop_orders(1.0, f64::NAN), 0.0);
113    }
114
115    #[test]
116    fn a_scaled_sink_scales_the_monitor_and_passes_the_verdict_through() {
117        let seen = Arc::new(std::sync::Mutex::new(Vec::new()));
118        let log = seen.clone();
119        let sink = ProgressSink::new(move |p| {
120            log.lock().unwrap().push(p.monitor);
121            if p.iteration >= 100 {
122                ControlFlow::Break(())
123            } else {
124                ControlFlow::Continue(())
125            }
126        });
127        let scaled = sink.scaled(2.0);
128        let p =
129            Progress { mach: 0.8, alpha: 0.0, iteration: 50, max_iterations: 1000, residual_drop_orders: 0.0, monitor: 0.25, spread: None, tolerance: None };
130        assert_eq!(scaled.report(&p), ControlFlow::Continue(()));
131        assert_eq!(scaled.report(&Progress { iteration: 100, ..p }), ControlFlow::Break(()));
132        assert_eq!(*seen.lock().unwrap(), vec![0.5, 0.5]);
133    }
