1//! 3D body-fitted grid of the half domain (`theta` in `[0, pi]`, pitch-plane symmetry) around a sting-mounted body with thick fins.
2//!
3//! The axisymmetric large-domain grid ([`super::body_grid`]) is revolved into `nk` azimuthal cells. Fins are one-cell-thick solid
4//! layers in `k` (see `Grid3::solid`): the layer edges follow the fin thickness over the planform and relax to the nominal cell width
5//! away from it, so the fin faces are flat planes through the body axis. Lengths are in nose diameters.
6//!
7//! * `Grid3` nodes are `(x, r sin(theta), r cos(theta))`: the pitch plane is `y = 0`, `theta = 0` points to `+z` (the lee side
8//!   for `alpha > 0`) and `theta = pi` to `-z`.
9//! * Fins sit at `theta_f = pi/n + 2 pi m / n`; those in `[0, pi]` are kept. For odd `n` the fin at `theta = pi` lies in the
10//!   symmetry plane and is a half fin owning the last cell `k = nk - 1`.
11//! * With fins, the root chord must lie on one tube segment. That segment is rebuilt as upstream piece, fin piece and downstream piece
12//!   (`32 << level` cells over the root chord, clustered at both edges); the sweep and taper are applied as a piecewise-linear
13//!   remap of the segment's `x` nodes per row (the cells outside the fin stretch or shrink instead of folding), blended away over
14//!   four rows above the fin tip. The radial fractions are remapped so that a node row lies exactly on the fin tip.
15

24/// Trapezoidal fins in nose diameters.
25#[derive(Clone, Debug)]
26pub struct Fins3 {
27    pub count: u32,
28    pub root_chord: f64,
29    pub tip_chord: f64,
30    /// Radial extent from the body surface to the tip.
31    pub span: f64,
32    /// Axial distance root leading edge to tip leading edge, positive aft.
33    pub sweep: f64,
34    /// Maximum thickness.
35    pub thickness: f64,
36    /// Distance by which the root trailing edge lies ahead of the body base (0 = flush).
37    pub te_to_base: f64,
38    pub profile: FinProfile,
39}

41/// Grid parameters: the sting grid of the 2D case plus the fins and the azimuthal resolution.
42#[derive(Clone, Debug)]
43pub struct Grid3Params {
44    pub sting: LargeGridParams,
45    pub fins: Option<Fins3>,
46    /// Azimuthal cells over `[0, pi]` (the plan default is `24 << level`, see [`Grid3Params::default_nk`]).
47    pub nk: usize,
48    /// Grid level: sets the cells over the fin root chord (`32 << level`) unless `fin_chord_cells` overrides it.
49    pub level: u32,
50    /// Cells over the fin root chord (`None`: `32 << level`; the ANF study found 16 too coarse for the fin loading at M >= 2).
51    pub fin_chord_cells: Option<usize>,
52    /// Radial cells from the body wall to the fin tip (`None`: whatever the sting grid's radial spacing puts there, with the tip
53    /// row snapped to the nearest node). With `Some(n)` the radial fractions are regraded for the whole grid: `n` cells in
54    /// `[R, R + span]` starting at the wall spacing (uniform when that is already finer than `span / n`), then a geometric fan of
55    /// at least the sting grid's outer cell count with growth `<= 1.2` to the far boundary. `nj` grows accordingly.
56    pub span_cells: Option<usize>,
57}

60    /// Default azimuthal cell count of a grid level.

62        24usize << level
63    }

66/// Half thickness of the fin section at the chordwise fraction `xi` (chord length `chord`, maximum thickness `t`), floored at
67/// 25 % of `t/2` so that edges are blunted.
68pub(crate) fn half_thickness(profile: &FinProfile, t: f64, chord: f64, xi: f64) -> f64 {
69    let xi = xi.clamp(0.0, 1.0);
70    let r = 0.5 * t;
71    let h = match profile {
72        FinProfile::Square => r,
73        FinProfile::Rounded => {
74            let s = (xi * chord).min((1.0 - xi) * chord);
75            if s >= r {
76                r
77            } else {
78                (r * r - (r - s) * (r - s)).max(0.0).sqrt()
79            }
80        }
81        FinProfile::Airfoil => 5.0 * t * (0.2969 * xi.sqrt() - 0.1260 * xi - 0.3516 * xi * xi + 0.2843 * xi.powi(3) - 0.1015 * xi.powi(4)).max(0.0),
82        FinProfile::DoubleWedge => t * xi.min(1.0 - xi),
83        FinProfile::Custom { section } => {
84            let z: Vec<[f64; 2]> = normalised(section).collect();
85            let mut v = z[z.len() - 1][1];
86            for w in z.windows(2) {
87                if xi <= w[1][0] {
88                    let d = w[1][0] - w[0][0];
89                    let f = if d > 0.0 { ((xi - w[0][0]) / d).clamp(0.0, 1.0) } else { 1.0 };
90                    v = w[0][1] + f * (w[1][1] - w[0][1]);
91                    break;
92                }
93            }
94            r * v
95        }
96    };
97    h.max(0.25 * r)
98}

117/// Azimuthal layer structure: fin angles and the number of cells in each gap.
118struct Layers {
119    theta_f: Vec<f64>,
120    half_last: bool,
121    /// Cells of the gap before each layer, plus one trailing gap behind the last layer unless that is a half fin.
122    gaps: Vec<usize>,
123    /// First `k` of the cell owned by each layer.
124    k_f: Vec<usize>,
125}
126
127fn layers(count: u32, nk: usize) -> Result<Layers, Error> {
128    let n = count as f64;
129    let mut theta_f = Vec::new();
130    let mut half_last = false;
131    for m in 0..count {
132        let th = PI / n + 2.0 * PI * m as f64 / n;
133        if th <= PI + 1e-9 {
134            if (PI - th).abs() < 1e-9 {
135                half_last = true;
136                theta_f.push(PI);
137            } else {
138                theta_f.push(th);
139            }
140        }
141    }
142    let nl = theta_f.len();
143    let nh = PI / (2.0 * nk as f64);
144    let mut widths = Vec::new();
145    let mut prev = 0.0;
146    for (l, &th) in theta_f.iter().enumerate() {
147        let half = half_last && l + 1 == nl;
148        widths.push((if half { PI - nh } else { th - nh }) - prev);
149        prev = if half { PI } else { th + nh };
150    }
151    if !half_last {
152        widths.push(PI - prev);
153    }
154    let too_close = || Error::Grid("fins too close for nk cells".to_string());
155    if nk < nl || widths.iter().any(|&w| w <= 0.0) || nk - nl < 2 * widths.len() {
156        return Err(too_close());
157    }
158    let remaining = nk - nl;
159    let total: f64 = widths.iter().sum();
160    let q: Vec<f64> = widths.iter().map(|w| remaining as f64 * w / total).collect();
161    let mut gaps: Vec<usize> = q.iter().map(|&x| (x.round() as usize).max(2)).collect();
162    while gaps.iter().sum::<usize>() != remaining {
163        if gaps.iter().sum::<usize>() > remaining {
164            let i = (0..gaps.len()).filter(|&i| gaps[i] > 2).max_by(|&a, &b| (gaps[a] as f64 - q[a]).total_cmp(&(gaps[b] as f64 - q[b])));
165            gaps[i.ok_or_else(too_close)?] -= 1;
166        } else {
167            let i = (0..gaps.len()).max_by(|&a, &b| (q[a] - gaps[a] as f64).total_cmp(&(q[b] - gaps[b] as f64))).unwrap();
168            gaps[i] += 1;
169        }
170    }
171    let mut k_f = Vec::new();
172    let mut k = 0;
173    for l in 0..nl {
174        k += gaps[l];
175        k_f.push(k);
176        k += 1;
177    }
178    Ok(Layers { theta_f, half_last, gaps, k_f })
179}

181/// Azimuth of the nodes `k = 0..=nk` of one `(i, j)` column for the layer half-angle `alpha`.
182fn fill_theta(theta: &mut [f64], lay: &Layers, alpha: f64) {
183    let nk = theta.len() - 1;
184    theta[0] = 0.0;
185    let (mut k, mut prev) = (0usize, 0.0f64);
186    let nl = lay.theta_f.len();
187    for (l, &th) in lay.theta_f.iter().enumerate() {
188        let half = lay.half_last && l + 1 == nl;
189        let lo = if half { PI - alpha } else { th - alpha };
190        let hi = if half { PI } else { th + alpha };
191        let ng = lay.gaps[l];
192        for m in 1..=ng {
193            theta[k + m] = prev + (lo - prev) * m as f64 / ng as f64;
194        }
195        k += ng;
196        theta[k + 1] = hi;
197        k += 1;
198        prev = hi;
199    }
200    if lay.gaps.len() > nl {
201        let ng = lay.gaps[nl];
202        for m in 1..=ng {
203            theta[k + m] = prev + (PI - prev) * m as f64 / ng as f64;
204        }
205    }
206    theta[nk] = PI;
207}

209fn smoothstep(u: f64) -> f64 {
210    let u = u.clamp(0.0, 1.0);
211    u * u * (3.0 - 2.0 * u)
212}

214/// Choose the first candidate `(node, distance, outer cell width)` that a geometric fan of `m` cells starting at `dx` with growth
215/// `q <= 1.2` reaches with a last cell within 1.4 of the outer width; returns `(node, m, q)` (the best compromise if none does).
216fn fan(dx: f64, cands: &[(usize, f64, f64)]) -> Option<(usize, usize, f64)> {
217    let growth = |m: usize, d: f64| -> f64 {
218        // dx (q^m - 1)/(q - 1) = d
219        let (mut lo, mut hi) = (0.2f64, 3.0f64);
220        for _ in 0..80 {
221            let q = 0.5 * (lo + hi);
222            let sum = if (q - 1.0).abs() < 1e-12 { dx * m as f64 } else { dx * (q.powi(m as i32) - 1.0) / (q - 1.0) };
223            if sum < d {
224                lo = q;
225            } else {
226                hi = q;
227            }
228        }
229        0.5 * (lo + hi)
230    };
231    let mut best: Option<(f64, usize, usize, f64)> = None;
232    for &(o, d, w) in cands {
233        for m in 1..=200usize {
234            let q = growth(m, d);
235            let last = dx * q.powi(m as i32 - 1);
236            let score = (q.max(1.0 / q) / 1.2).max((last / w).max(w / last) / 1.4);
237            if score <= 1.0 {
238                return Some((o, m, q));
239            }
240            if best.is_none_or(|b| score < b.0) {
241                best = Some((score, o, m, q));
242            }
243        }
244    }
245    best.map(|b| (b.1, b.2, b.3))
246}

248/// Rebuild the tube segment that carries the fins in `xs` and compute the radial remap.
249fn plan_fins(p: &Grid3Params, f: &Fins3, xs: &mut Vec<f64>) -> Result<FinPlan, Error> {
250    let sp = &p.sting;
251    let bad = |m: &str| Err(Error::InvalidInput(m.to_string()));
252    let ok = |v: f64| v.is_finite();
253    if !(f.count >= 1 && ok(f.root_chord) && f.root_chord > 0.0 && ok(f.tip_chord) && f.tip_chord >= 0.0 && ok(f.span) && f.span > 0.0)
254        || !(ok(f.sweep) && ok(f.thickness) && f.thickness > 0.0 && ok(f.te_to_base) && f.te_to_base >= 0.0)
255    {
256        return bad("3D fins need count >= 1, positive root chord, span and thickness, tip chord >= 0 and te_to_base >= 0");
257    }
258    let bb = &sp.body;
259    let x_te_root = bb.x_base() - f.te_to_base;
260    let x_le_root = x_te_root - f.root_chord;
261    let nseg = bb.segments.len();
262    let unsupported = || Error::Unsupported("3D fins must sit on a tube".to_string());
263    let k = (0..nseg)
264        .find(|&k| {
265            let x0 = bb.station(k).0;
266            bb.segments[k].slope == 0.0 && x_le_root >= x0 - 1e-9 && x_te_root <= x0 + bb.segments[k].length + 1e-9
267        })
268        .ok_or_else(unsupported)?;
269    let a = sp.n_up + sp.n_nose + sp.n_seg[..k].iter().sum::<usize>();
270    let b_old = a + sp.n_seg[k];
271    let (x_a, x_b) = (xs[a], xs[b_old]);
272    let x_le_root = if x_le_root - x_a > 1e-9 { x_le_root } else { x_a };
273    let x_te_root = if x_b - x_te_root > 1e-9 { x_te_root } else { x_b };
274    let flush = x_te_root >= x_b;
275    let nf = p.fin_chord_cells.unwrap_or(32usize << p.level);
276    if nf < 2 {
277        return Err(Error::InvalidInput("3D fins need at least 2 cells over the root chord".to_string()));
278    }
279    let mut fin_offs = two_sided_nodes(f.root_chord, nf, 0.3 * f.root_chord / nf as f64);
280    fin_offs[nf] = x_te_root - x_le_root;
281    let dx_f = fin_offs[1] - fin_offs[0];
282    // Regrade: the fin piece replaces the original nodes over its chord. Geometric fans (growth <= 1.2) lead from its end cells
283    // to the first original node where the spacing has caught up with the original one (ratio <= 1.5 on both sides).
284    // Candidates: `(node, distance from the fin edge, width of the original cell beyond the node)`.
285    let up: Vec<(usize, f64, f64)> = (a..b_old).rev().filter(|&o| x_le_root - xs[o] > 1e-9).map(|o| (o, x_le_root - xs[o], xs[o] - xs[o - 1])).collect();
286    let o_max = if flush { (b_old + 60).min(xs.len() - 1) } else { b_old };
287    let down: Vec<(usize, f64, f64)> = (a..=o_max)
288        .filter(|&o| xs[o] - x_te_root > 1e-9)
289        .map(|o| (o, xs[o] - x_te_root, if o + 1 < xs.len() { xs[o + 1] - xs[o] } else { xs[o] - xs[o - 1] }))
290        .collect();
291    let mut new = xs[..=a].to_vec();
292    if let Some((o, m, q)) = fan(dx_f, &up) {
293        {
294            new = xs[..=o].to_vec();
295            let sizes: Vec<f64> = (0..m).rev().map(|i| dx_f * q.powi(i as i32)).collect();
296            let mut s = xs[o];
297            for w in &sizes[..m - 1] {
298                s += w;
299                new.push(s);
300            }
301            new.push(x_le_root);
302        }
303    }
304    let i_le = new.len() - 1;
305    new.extend((1..=nf).map(|i| x_le_root + fin_offs[i]));
306    let i_te = new.len() - 1;
307    let (o_down, m_down) = match fan(dx_f, &down) {
308        Some((o, m, q)) => {
309            let mut s = 0.0;
310            for i in 0..m - 1 {
311                s += dx_f * q.powi(i as i32);
312                new.push(x_te_root + s);
313            }
314            new.push(xs[o]);
315            (o, m)
316        }
317        None => (b_old, 0),
318    };
319    new.extend_from_slice(&xs[o_down + 1..]);
320    let i_seg1 = if flush { i_te } else { i_te + m_down + (b_old - o_down) };
321    *xs = new;
322
323    let r_t = bb.station(k).1;
324    if sp.r_far.is_nan() || sp.r_far <= r_t + f.span || sp.nj < 6 {
325        return Err(Error::Grid("the fin tip must lie inside the domain".to_string()));
326    }
327    let sj0 = radial_fractions(sp.dr_wall, sp.r_far, sp.nj);
328    let target = f.span / (sp.r_far - r_t);
329    let j_tip0 = (1..=sp.nj).min_by(|&a, &b| (sj0[a] - target).abs().total_cmp(&(sj0[b] - target).abs())).unwrap();
330    let (sj, s_remap, j_tip) = match p.span_cells {
331        None => {
332            let s_remap: Vec<f64> = (0..=sp.nj)
333                .map(|j| {
334                    if j == j_tip0 {
335                        target
336                    } else if j < j_tip0 {
337                        sj0[j] * target / sj0[j_tip0]
338                    } else {
339                        target + (sj0[j] - sj0[j_tip0]) * (1.0 - target) / (1.0 - sj0[j_tip0])
340                    }
341                })
342                .collect();
343            (sj0, s_remap, j_tip0)
344        }
345        Some(n) => {
346            if n < 2 {
347                return Err(Error::InvalidInput("span_cells >= 2 is required".to_string()));
348            }
349            let first = sj0[1];
350            let inner = if first * n as f64 >= target {
351                (0..=n).map(|j| target * j as f64 / n as f64).collect::<Vec<_>>()
352            } else {
353                geometric_offsets(first, target, n)
354            };
355            let c_last = inner[n] - inner[n - 1];
356            let outer = 1.0 - target;
357            // growth q of `m` cells c_last q, c_last q^2, .. summing to `outer`
358            let growth = |m: usize| -> f64 {
359                let (mut lo, mut hi) = (0.2f64, 4.0f64);
360                for _ in 0..200 {
361                    let q = 0.5 * (lo + hi);
362                    let (mut s, mut d) = (0.0, c_last);
363                    for _ in 0..m {
364                        d *= q;
365                        s += d;
366                    }
367                    if s < outer {
368                        lo = q;
369                    } else {
370                        hi = q;
371                    }
372                }
373                0.5 * (lo + hi)
374            };
375            let mut m = (sp.nj - j_tip0).max(4);
376            while growth(m) > 1.2 {
377                m += 1;
378                if m > 2000 {
379                    return Err(Error::Grid("span_cells: no outer fan with growth <= 1.2".to_string()));
380                }
381            }
382            let q = growth(m);
383            let mut s = inner;
384            let (mut acc, mut d) = (target, c_last);
385            for _ in 0..m {
386                d *= q;
387                acc += d;
388                s.push(acc);
389            }
390            *s.last_mut().unwrap() = 1.0;
391            (s.clone(), s, n)
392        }
393    };
394    let nj = s_remap.len() - 1;
395    if j_tip + 4 > nj {
396        return Err(Error::Grid("the fin tip is too close to the outer boundary".to_string()));
397    }
398    // the swept tip trailing edge must stay inside the rebuilt segment (flush: not behind the base)
399    let x_te_tip = x_le_root + f.sweep + f.tip_chord;
400    let room = if i_te == i_seg1 { 1e-9 } else { 0.5 * (x_b - x_te_root) };
401    if x_te_tip > x_te_root + room || x_le_root + f.sweep < x_a - 1e-9 {
402        return Err(Error::Unsupported("3D fin tip trailing edge lies behind the root trailing edge room".to_string()));
403    }
404    Ok(FinPlan { i_seg0: a, i_seg1, i_le, i_te, j_tip, x_le_root, x_te_root, r_t, sj, s_remap })
405}

407/// Build the 3D grid. With fins the grid has `sting.nj`, `nk` cells in `j`, `k` and the axial count of `axial_nodes` changed by
408/// the rebuilt fin segment.
409pub fn body_grid3(p: &Grid3Params) -> Result<Grid3, Error> {
410    let sp = &p.sting;
411    let nk = p.nk;
412    if nk < 2 {
413        return Err(Error::Grid("nk >= 2 is required".to_string()));
414    }
415    if !(sp.nj >= 1 && sp.dr_wall > 0.0 && sp.r_far > 0.0) {
416        return Err(Error::Grid("nj >= 1, dr_wall > 0 and r_far > 0 are required".to_string()));
417    }
418    let mut xs = axial_nodes(sp)?;
419    let plan = match &p.fins {
420        Some(f) => Some((plan_fins(p, f, &mut xs)?, f)),
421        None => None,
422    };
423    let lay = match &p.fins {
424        Some(f) => Some(layers(f.count, nk)?),
425        None => None,
426    };
427    let ni = xs.len() - 1;
428    let sj = match &plan {
429        Some((pl, _)) => pl.sj.clone(),
430        None => radial_fractions(sp.dr_wall, sp.r_far, sp.nj),
431    };
432    let nj = sj.len() - 1;
433    let n2 = (ni + 1) * (nj + 1);
434
435    // (x, r) of every (i, j) node
436    let mut xn = vec![0.0; n2];
437    let mut rn = vec![0.0; n2];
438    let r_in: Vec<f64> = xs.iter().map(|&x| sp.body.r(x)).collect();
439    for j in 0..=nj {
440        for i in 0..=ni {
441            xn[j * (ni + 1) + i] = xs[i];
442            rn[j * (ni + 1) + i] = r_in[i] + sj[j] * (sp.r_far - r_in[i]);
443        }
444    }
445    // per-row planform (frac along the span, LE position, chord) and the half-angle of the fin layer at each column
446    let mut alpha = Vec::new();
447    if let Some((pl, f)) = &plan {
448        let wcol = |i: usize| -> f64 {
449            let d = if i < pl.i_le { pl.i_le - i } else { i.saturating_sub(pl.i_te) };
450            if d <= 4 {
451                1.0
452            } else {
453                (1.0 - (d as f64 - 4.0) / 4.0).max(0.0)
454            }
455        };
456        for j in 0..=nj {
457            for i in 0..=ni {
458                let s = sj[j] + wcol(i) * (pl.s_remap[j] - sj[j]);
459                rn[j * (ni + 1) + i] = r_in[i] + s * (sp.r_far - r_in[i]);
460            }
461        }
462        let row = |j: usize| -> (f64, f64, f64) {
463            let r = pl.r_t + pl.s_remap[j.min(pl.j_tip)] * (sp.r_far - pl.r_t);
464            let frac = ((r - pl.r_t) / f.span).clamp(0.0, 1.0);
465            (frac, pl.x_le_root + f.sweep * frac, f.root_chord + (f.tip_chord - f.root_chord) * frac)
466        };
467        // sweep and taper: piecewise-linear remap of the segment nodes
468        let nodes0: Vec<f64> = xs.clone();
469        for j in 0..=(pl.j_tip + 3).min(nj) {
470            let vj = if j <= pl.j_tip { 1.0 } else { 1.0 - (j - pl.j_tip) as f64 / 4.0 };
471            let (_, x_le, chord) = row(j);
472            let mut knots = vec![(pl.i_seg0, nodes0[pl.i_seg0], nodes0[pl.i_seg0])];
473            if pl.i_le == pl.i_seg0 {
474                knots.clear();
475            }
476            knots.push((pl.i_le, pl.x_le_root, x_le));
477            knots.push((pl.i_te, pl.x_te_root, x_le + chord));
478            if pl.i_te != pl.i_seg1 {
479                knots.push((pl.i_seg1, nodes0[pl.i_seg1], nodes0[pl.i_seg1]));
480            }
481            for w in knots.windows(2) {
482                for i in w[0].0..=w[1].0 {
483                    let old = nodes0[i];
484                    let d = w[1].1 - w[0].1;
485                    let t = if d.abs() > 0.0 { (old - w[0].1) / d } else { 1.0 };
486                    let new = w[0].2 + t * (w[1].2 - w[0].2);
487                    xn[j * (ni + 1) + i] = old + vj * (new - old);
488                }
489            }
490        }
491        // layer half-angle at every (i, j) node
492        alpha = vec![0.0; n2];
493        let t = f.thickness;
494        let h_at = |i: usize, j: usize| -> f64 {
495            let (_, _, chord) = row(j);
496            let xi = (nodes0[i] - pl.x_le_root) / f.root_chord;
497            half_thickness(&f.profile, t, chord, xi)
498        };
499        for j in 0..=nj {
500            for i in 0..=ni {
501                let r = rn[j * (ni + 1) + i];
502                let di = if i < pl.i_le { pl.i_le - i } else { i.saturating_sub(pl.i_te) };
503                let dj = j.saturating_sub(pl.j_tip);
504                let d = di.max(dj);
505                let (ic, jc) = (i.clamp(pl.i_le, pl.i_te), j.min(pl.j_tip));
506                let h_f = h_at(ic, jc);
507                let nominal = 0.5 * r * PI / nk as f64;
508                let w = if d == 0 { h_f } else { h_f + (nominal - h_f) * smoothstep(d as f64 / 4.0) };
509                alpha[j * (ni + 1) + i] = if r < 1e-12 { (PI / (2.0 * nk as f64)).asin() } else { (w / r).min(0.99).asin() };
510            }
511        }
512    }
513
514    let nn = (ni + 1) * (nj + 1) * (nk + 1);
515    let (mut x, mut y, mut z) = (vec![0.0; nn], vec![0.0; nn], vec![0.0; nn]);
516    let mut theta = vec![0.0; nk + 1];
517    for j in 0..=nj {
518        for i in 0..=ni {
519            let c = j * (ni + 1) + i;
520            match &lay {
521                Some(l) => fill_theta(&mut theta, l, alpha[c]),
522                None => {
523                    for (k, t) in theta.iter_mut().enumerate() {
524                        *t = PI * k as f64 / nk as f64;
525                    }
526                }
527            }
528            let (xc, r) = (xn[c], rn[c]);
529            for k in 0..=nk {
530                let id = (k * (nj + 1) + j) * (ni + 1) + i;
531                x[id] = xc;
532                y[id] = r * theta[k].sin();
533                z[id] = r * theta[k].cos();
534            }
535        }
536    }
537    let mut g = Grid3::try_from_nodes(ni, nj, nk, x, y, z).map_err(Error::Grid)?;
538    g.axis_cells = sp.n_up;
539    if let (Some((pl, _)), Some(l)) = (&plan, &lay) {
540        g.solid = l.k_f.iter().map(|&kf| SolidBox { i: pl.i_le..pl.i_te, j: 0..pl.j_tip, k: kf..kf + 1 }).collect();
541        for k in 0..nk {
542            for j in 0..nj {
543                for i in 0..ni {
544                    if !g.is_solid(i, j, k) && g.vol[g.cell(i, j, k)] <= 0.0 {
545                        return Err(Error::Grid(format!("non-positive fluid cell volume at ({i},{j},{k})")));
546                    }
547                }
548            }
549        }
550    } else {
551        for k in 0..nk {
552            for j in 0..nj {
553                for i in 0..ni {
554                    if g.vol[g.cell(i, j, k)] <= 0.0 && i >= sp.n_up {
555                        return Err(Error::Grid(format!("non-positive fluid cell volume at ({i},{j},{k})")));
556                    }
557                }
558            }
559        }
560    }
561    Ok(g)
562}

590    /// Distance of node `(i, j, k)` from the fin mid-plane at angle `th`.
591    fn dist(g: &Grid3, i: usize, j: usize, k: usize, th: f64) -> f64 {
592        let id = (k * (g.nj + 1) + j) * (g.ni + 1) + i;
593        g.y[id] * th.cos() - g.z[id] * th.sin()
594    }

597        let id = (k * (g.nj + 1) + j) * (g.ni + 1) + i;
598        [g.x[id], g.y[id], g.z[id]]
599    }

717    /// `(max ratio, index of the larger cell's left node)` of adjacent cell widths.
718    fn max_ratio(xs: &[f64], from: usize) -> (f64, usize) {
719        let mut m = (1.0, 0);
720        for i in from..xs.len() - 2 {
721            let (a, b) = (xs[i + 1] - xs[i], xs[i + 2] - xs[i + 1]);
722            let q = (a / b).max(b / a);
723            if q > m.0 {
724                m = (q, i);
725            }
726        }
727        m
728    }
