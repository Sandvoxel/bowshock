1//! GPU runner of a [`Case3`]: the procedure and stop rule of [`crate::engine::run3::run3_on`] with the iterations executed by
2//! `bowshock_cfd::gpu::GpuSolver3` (extended precision).
3//!
4//! The GPU records [`SYNC`] iterations back to back and reads their residuals. CN is sampled once per full chunk from the downloaded
5//! state, so the stop rule sees the same samples as the CPU loop; the reported stop iteration is the CPU-equivalent one, the GPU may
6//! execute up to `SYNC - 1` further iterations and the final forces are those of the state after them.
7
8use crate::engine::case3::Case3;
9use crate::engine::run3::{forces_on, PointResult3, StopRule3, SAMPLE_EVERY};
10use crate::error::Error;
11use bowshock_cfd::gpu::{GpuSolver3, Precision};
12use bowshock_cfd::solver3::{Config3, Scheme3, Solver3};
13
14/// Iterations between two synchronisations (the CN sampling interval).
15pub const SYNC: usize = SAMPLE_EVERY;
16
17/// GPU solver plus the host solver (grid and config) of one 3D case.
18pub struct GpuRunner3 {
19    pub gpu: GpuSolver3,
20    pub host: Solver3,
21    /// Kernel build + geometry upload, seconds.
22    pub setup_seconds: f64,
23}
24
25impl GpuRunner3 {
26    /// Builds the host solver of `case` and uploads it to the default GPU context.
27    pub fn new(case: &Case3, prec: Precision) -> Result<GpuRunner3, Error> {
28        let host = case.build_solver()?;
29        let t0 = std::time::Instant::now();
30        let ctx = luma::default_context()?;
31        if ctx.is_cpu() {
32            return Err(Error::Gpu("no GPU: Luma opened a software (CPU) Vulkan device".into()));
33        }
34        let gpu = GpuSolver3::from_host(&ctx, &host, prec)?;
35        Ok(GpuRunner3 { gpu, host, setup_seconds: t0.elapsed().as_secs_f64() })
36    }
37
38    /// Iterate to a stop rule: as [`crate::engine::run3::run3_on`] (`warm`: keep the field, set the free stream of `case`, restart
39    /// the CFL ramp).
40    pub fn run_on(&mut self, case: &Case3, warm: bool) -> Result<PointResult3, Error> {
41        let n = &case.numerics;
42        let cfg = Config3 {
43            freestream: case.freestream(),
44            cfl: self.host.cfg.cfl,
45            limiter: n.limiter,
46            first_order_iters: n.first_order_iters,
47            low_mach: n.low_mach,
48            scheme: Scheme3::Implicit(n.implicit),
49        };
50        self.gpu.set_config(&cfg)?;
51        self.host.cfg = cfg;
52        if warm {
53            let it = self.gpu.iter().max(n.first_order_iters);
54            self.gpu.set_iter(it);
55            self.gpu.restart_ramp()?;
56        } else {
57            self.gpu.reset()?;
58        }
59        self.gpu.context().sync()?;
60
61        let x_base = case.body.x_base();
62        let p_inf = self.host.cfg.freestream[4];
63        let mut rule = StopRule3::new(case, warm);
64        let mut done = 0usize;
65        let mut stop: Option<usize> = None;
66        while done < n.max_iter && stop.is_none() {
67            let len = SYNC.min(n.max_iter - done);
68            let h = self.gpu.run(len)?;
69            let cn = if len == SYNC { Some(forces_on(case, &self.host.grid, &self.gpu.state_f64()?, p_inf, x_base).cn) } else { None };
70            for (k, r) in h.into_iter().enumerate() {
71                let it = done + k;
72                let sample = if it % SYNC == SYNC - 1 { cn } else { None };
73                if rule.check(it, r, sample).is_some() {
74                    stop = Some(it + 1);
75                    break;
76                }
77            }
78            done += len;
79            if stop.is_none() && len == SYNC && rule.report(case, done).is_break() {
80                rule.cancel();
81                stop = Some(done);
82            }
83        }
84        let f = forces_on(case, &self.host.grid, &self.gpu.state_f64()?, p_inf, x_base);
85        Ok(rule.finish(case, f, stop.unwrap_or(done)))
86    }
87}
