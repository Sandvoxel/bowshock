1//! Solver settings of a run: scheme, limiter, iteration limits and stop-rule tolerances.
2
3use crate::engine::case::Wake;
4use bowshock_cfd::flux::Limiter;
5use bowshock_cfd::implicit::{DtMode, ImplicitParams};
6use bowshock_cfd::solver::{Bc, Scheme};
7
8/// Sampling interval \[iterations\] of the monitor history.
9pub const MONITOR_EVERY: usize = 50;
10/// Window \[iterations\] of the relative drag rule ([`Numerics::drag_rel_tol`]); a multiple of [`MONITOR_EVERY`].
11pub const DRAG_REL_WINDOW: usize = 500;
12/// Mach number below which [`Numerics::auto`] uses the subsonic settings.
13pub const REGIME_SPLIT_MACH: f64 = 0.85;
14
15/// Pseudo-time parameters of the wall-resolved RANS runs of the facade: the summed-direction time step with a high CFL limit. The
16/// weaker-direction step of [`ImplicitParams::VISCOUS`] (which the flat-plate gates of the solver use) diverged within a few hundred
17/// iterations on the wall-resolved body grid of an ogive-cylinder at Mach 0.5 (cfl_max 10, also with a start CFL of 0.1).
18const RANS_IMPLICIT: ImplicitParams = ImplicitParams { dt_mode: DtMode::Sum, cfl_max: 50.0, sub_iters: 3, ..ImplicitParams::DEFAULT };
19
20/// Mach number from which a wall-resolved run uses [`RANS_IMPLICIT_SUPERSONIC`].
21const SUPERSONIC_MACH: f64 = 1.0;
22
23/// [`RANS_IMPLICIT`] with the CFL limit lowered to 20 for supersonic runs. At `cfl_max` 50 the implicit iteration can diverge in the
24/// supersonic wake behind the base once the CFL ramp (`growth` 1.02 per iteration) passes about 40, at iteration 195-240 (during the
25/// first-order phase): a short-wave oscillation of density and pressure on the wake rows next to the axis grows by about 30 % per
26/// iteration and the pressure turns negative within 5 iterations. The inviscid implicit scheme on the same grid does the same, so no
27/// viscous or turbulence term is involved, and it needs the ramp, not the second-order switch. Measured on the GPU solver at Medium:
28/// the polar body at Mach 1.5, 2, 2.5 and 3, the TP-2005 body at Mach 2 and the Arcas body at Mach 2 and 2.5 diverge at 50 (at 40 or 35
29/// the polar body at Mach 3, the TP-2005 body at Mach 3 and the Arcas body at Mach 2; the polar body at Mach 3 even at 30); at 20 every
30/// one of those, at Mach 1.5, 2, 2.5 and 3, reaches a stationary drag.
31const RANS_IMPLICIT_SUPERSONIC: ImplicitParams = ImplicitParams { cfl_max: 20.0, ..RANS_IMPLICIT };
32
33/// Numerical settings of one run.
34#[derive(Clone, Copy, Debug, PartialEq)]
35pub struct Numerics {
36    pub cfl: f64,
37    pub limiter: Limiter,
38    pub low_mach: bool,
39    pub scheme: Scheme,
40    /// First-order start-up iterations.
41    pub first_order_iters: usize,
42    pub max_iter: usize,
43    /// Stop rules other than a non-finite residual do not fire before this many iterations.
44    pub min_iter: usize,
45    /// Stop at this many orders of density-residual drop below the reference.
46    pub drop_orders: f64,
47    /// Boundary condition of the outflow plane.
48    pub outflow: Bc,
49    /// Length of the statistics window of the monitor \[iterations\].
50    pub monitor_window: usize,
51    /// Monitor range below which a run counts as stationary; `None`: no "stationary" stop rule.
52    pub monitor_tol: Option<f64>,
53    /// Averaging window \[iterations\] of the mean-stationary stop rule (`None`: no such rule; only [`Numerics::viscous`] sets it). A
54    /// viscous run whose total drag settles into a bounded limit cycle never satisfies `monitor_tol`; the rule stops it where the
55    /// drag crosses the stationary mean of the cycle. The window should span at least one period of the cycle.
56    pub mean_window: Option<usize>,
57    /// Relative stationary rule on the monitor: stop once `(max - min) / |mean|` of the last [`DRAG_REL_WINDOW`] iterations is below
58    /// this (after a residual drop of 2 orders). Only meaningful for a viscous case, whose monitor is the total CD; `None` by default.
59    pub drag_rel_tol: Option<f64>,
60}
61
62impl Numerics {
63    /// Line-implicit scheme, van Albada limiter, low-Mach correction (subsonic regime).
64    pub fn subsonic(level: u32, wake: Wake) -> Numerics {
65        Numerics {
66            cfl: 0.5,
67            limiter: Limiter::VanAlbada,
68            low_mach: true,
69            scheme: Scheme::Implicit(ImplicitParams::DEFAULT),
70            first_order_iters: 100,
71            max_iter: 10_000 << level,
72            min_iter: 0,
73            drop_orders: 6.0,
74            outflow: Bc::Characteristic,
75            monitor_window: 2000,
76            monitor_tol: match wake {
77                Wake::Base => Some(1e-3),
78                Wake::Sting => None,
79            },
80            mean_window: None,
81            drag_rel_tol: None,
82        }
83    }
84
85    /// Explicit SSP-RK3, minmod limiter (transonic regime).
86    pub fn transonic() -> Numerics {
87        Numerics {
88            cfl: 0.5,
89            limiter: Limiter::Minmod,
90            low_mach: false,
91            scheme: Scheme::Explicit,
92            first_order_iters: 300,
93            max_iter: 60_000,
94            min_iter: 0,
95            drop_orders: 6.0,
96            outflow: Bc::Characteristic,
97            monitor_window: 2000,
98            monitor_tol: Some(2e-4),
99            mean_window: None,
100            drag_rel_tol: None,
101        }
102    }
103
104    /// Subsonic settings below [`REGIME_SPLIT_MACH`], transonic settings otherwise; the stationary tolerance follows the wake.
105    pub fn auto(mach: f64, level: u32, wake: Wake) -> Numerics {
106        let mut n = if mach < REGIME_SPLIT_MACH { Numerics::subsonic(level, wake) } else { Numerics::transonic() };
107        n.monitor_tol = Some(match wake {
108            Wake::Base => 1e-3,
109            Wake::Sting => 2e-4,
110        });
111        n
112    }
113
114    /// Wall-resolved RANS (viscous terms and k-omega SST): line-implicit with the viscous parameters (CFL limit 20 from Mach 1, 50
115    /// below), van Albada limiter and the
116    /// low-Mach correction below [`REGIME_SPLIT_MACH`], minmod without correction otherwise; stationary tolerance 1e-4 on the
117    /// total CD, and the mean-stationary rule with a window of 6000 iterations (one period of the limit cycle of the subsonic runs).
118    pub fn viscous(mach: f64, level: u32) -> Numerics {
119        let subsonic = mach < REGIME_SPLIT_MACH;
120        Numerics {
121            cfl: 0.5,
122            limiter: if subsonic { Limiter::VanAlbada } else { Limiter::Minmod },
123            low_mach: subsonic,
124            scheme: Scheme::Implicit(if mach < SUPERSONIC_MACH { RANS_IMPLICIT } else { RANS_IMPLICIT_SUPERSONIC }),
125            first_order_iters: 200,
126            max_iter: 80_000 << level,
127            min_iter: 0,
128            drop_orders: 5.0,
129            outflow: Bc::Characteristic,
130            monitor_window: 2000,
131            monitor_tol: Some(1e-4),
132            mean_window: Some(6000),
133            drag_rel_tol: None,
134        }
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    #[test]
143    fn auto_regime_split_and_tolerance() {
144        let lo = Numerics::auto(0.5, 1, Wake::Sting);
145        assert_eq!(lo.scheme, Scheme::Implicit(ImplicitParams::DEFAULT));
146        assert_eq!(lo.max_iter, 20_000);
147        assert_eq!(lo.monitor_tol, Some(2e-4));
148        let hi = Numerics::auto(REGIME_SPLIT_MACH, 1, Wake::Base);
149        assert_eq!(hi.scheme, Scheme::Explicit);
150        assert_eq!(hi.max_iter, 60_000);
151        assert_eq!(hi.monitor_tol, Some(1e-3));
152        assert_eq!(Numerics::subsonic(0, Wake::Sting).monitor_tol, None);
153    }
154
155    #[test]
156    fn viscous_cfl_limit_is_lower_from_mach_one() {
157        let cfl_max = |mach: f64| match Numerics::viscous(mach, 1).scheme {
158            Scheme::Implicit(p) => p.cfl_max,
159            s => panic!("{s:?}"),
160        };
161        assert_eq!([cfl_max(0.5), cfl_max(0.99), cfl_max(1.0), cfl_max(3.0)], [50.0, 50.0, 20.0, 20.0]);
162    }
