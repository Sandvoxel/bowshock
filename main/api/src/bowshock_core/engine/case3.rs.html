15/// Numerical settings and stop-rule tolerances of a 3D run.
16#[derive(Clone, Copy, Debug, PartialEq)]
17pub struct Numerics3 {
18    pub limiter: Limiter,
19    pub low_mach: bool,
20    pub implicit: ImplicitParams,
21    /// First-order start-up iterations.
22    pub first_order_iters: usize,
23    pub max_iter: usize,
24    /// Stationary rule: relative CN range over the last `window` iterations.
25    pub rel_tol: f64,
26    /// Window \[iterations\] of the stationary rule (a multiple of [`crate::engine::run3::SAMPLE_EVERY`]).
27    pub window: usize,
28}
29
30impl Numerics3 {
31    /// Van Albada with the low-Mach correction below Mach 0.85, minmod above; the implicit defaults with `cfl_max` 10 for every Mach
32    /// number (the three-factor scheme diverged on the coarse ANF grid after the ramp passed about 15, at 20 and 30; 10 and 5 converge).
33    pub fn auto(mach: f64, level: u32) -> Numerics3 {
34        let subsonic = mach < 0.85;
35        Numerics3 {
36            limiter: if subsonic { Limiter::VanAlbada } else { Limiter::Minmod },
37            low_mach: subsonic,
38            implicit: ImplicitParams { cfl_max: 10.0, ..ImplicitParams::DEFAULT },
39            first_order_iters: 100,
40            max_iter: 8000 << level,
41            rel_tol: 0.002,
42            window: 500,
43        }
44    }
45}

47/// One Mach number and incidence on one body.
48#[derive(Clone, Debug)]
49pub struct Case3 {
50    pub mach: f64,
51    /// Incidence \[rad\]; the air moves towards `+z`.
52    pub alpha: f64,
53    pub body: BaseBody,
54    pub grid: Grid3Params,
55    pub numerics: Numerics3,
56    /// Where the run reports its progress and can be stopped; `None`: silent.
57    pub progress: Option<ProgressSink>,
58}
59
60impl Case3 {
61    /// Sting grid of the 2D case ([`crate::engine::case::Case::new`] with `Wake::Sting`), `24 << level` azimuthal cells and
62    /// [`Numerics3::auto`].
63    pub fn new(mach: f64, alpha: f64, body: BaseBody, fins: Option<Fins3>, level: u32) -> Case3 {
64        let sting = sting_grid(&body, level);
65        let grid = Grid3Params { sting, fins, nk: Grid3Params::default_nk(level), level, fin_chord_cells: None, span_cells: None };
66        Case3 { mach, alpha, body, grid, numerics: Numerics3::auto(mach, level), progress: None }
67    }
68
69    /// Free-stream state at the case's Mach number and incidence.
70    pub fn freestream(&self) -> bowshock_cfd::gas3::Prim3 {
71        freestream3(self.mach, self.alpha)
72    }
73
74    /// Cartesian grid, then the implicit CPU solver with the free stream everywhere.
75    pub fn build_solver(&self) -> Result<Solver3, Error> {
76        let grid = body_grid3(&self.grid)?;
77        let n = &self.numerics;
78        let cfg = Config3 {
79            freestream: self.freestream(),
80            cfl: 0.5,
81            limiter: n.limiter,
82            first_order_iters: n.first_order_iters,
83            low_mach: n.low_mach,
84            scheme: Scheme3::Implicit(n.implicit),
85        };
86        Ok(Solver3::new(grid, cfg))
87    }
88
89    /// Army-Navy Basic Finner body: 10 d long, 10 degree half-angle cone nose, cylindrical afterbody.
90    #[doc(hidden)]
91    pub fn anf_body() -> BaseBody {
92        let nose = Body { nose: Nose::Cone { half_angle: 10f64.to_radians() } };
93        let l_nose = nose.nose_length();
94        BaseBody::cylinder(nose, 10.0 - l_nose)
95    }
96
97    /// ANF fins: four fins of 1 d chord and 1 d exposed span, thickness [`bowshock_reference::anf::FIN_THICKNESS_CAL`], flush with the
98    /// base. Section: sharp leading edge and a blunt trailing edge of the full thickness (0.08 cal), as Abiza et al. (p. 3) describe
99    /// them. The source gives no bevel angle: the `FinProfile::Custom` section `[[0,0],[0.1,1],[1,1]]` (a wedge over the first
100    /// 0.1 chord, about 22 degrees per side, floored at 25 % of the half thickness) is **assumed**, and CN_alpha at M >= 2.35 is
101    /// sensitive to it.
102    #[doc(hidden)]
103    pub fn anf_fins() -> Fins3 {
104        let profile = FinProfile::Custom { section: vec![[0.0, 0.0], [0.1, 1.0], [1.0, 1.0]] };
105        Fins3 {
106            count: 4,
107            root_chord: 1.0,
108            tip_chord: 1.0,
109            span: 1.0,
110            sweep: 0.0,
111            thickness: bowshock_reference::anf::FIN_THICKNESS_CAL,
112            te_to_base: 0.0,
113            profile,
114        }
115    }
116
117    /// Coarse ANF-like case for tests: [`Case3::anf_body`] with [`Case3::anf_fins`], level 0, `nk = 8`, 16 cells over the fin chord and the sting grid counts
118    /// `n_up 10, n_nose 16, n_seg [12], n_ext 10, nj 24`. `max_iter` 3000.
119    #[doc(hidden)]
120    pub fn tiny_anf(mach: f64, alpha: f64) -> Case3 {
121        let mut c = Case3::new(mach, alpha, Case3::anf_body(), Some(Case3::anf_fins()), 0);
122        let s = &mut c.grid.sting;
123        s.n_up = 10;
124        s.n_nose = 16;
125        s.n_seg = vec![12];
126        s.n_ext = 10;
127        s.nj = 24;
128        c.grid.nk = 8;
129        c.grid.fin_chord_cells = Some(16);
130        c.numerics.max_iter = 3000;
131        c
132    }
