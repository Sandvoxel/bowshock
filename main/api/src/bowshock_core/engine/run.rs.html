1//! CPU runner: iterate a [`Solver`] built from a [`Case`] to convergence, the stopping rules and the post-processing of the
2//! final state (wall Cp, forebody / afterbody / base pressure drag, field indicators).
3//!
4//! Lengths are in body diameters (`d = 1`, `R = 0.5`). Drag coefficients are pressure drag only (inviscid), referenced to the
5//! maximum cross-section `pi R^2`; one formula covers every wall part, `CD = integral Cp d(r^2 / R^2)` along the surface from the
6//! nose tip downstream (the base face is traversed from its outer to its inner radius, so `CD_base = -Cp_b A_b / A_ref`).
7

19/// Reports the run's progress after `iteration` iterations (residual `r`, residual reference `reference`, monitor history `hist`) to
20/// the case's sink; `Break` when the sink asks to stop, and `Continue` without a sink.
21pub(crate) fn report(case: &Case, iteration: usize, r: f64, reference: f64, hist: &[(usize, f64)]) -> ControlFlow<()> {
22    let Some(sink) = &case.progress else { return ControlFlow::Continue(()) };
23    sink.report(&Progress {
24        mach: case.mach,
25        alpha: 0.0,
26        iteration,
27        max_iterations: case.numerics.max_iter,
28        residual_drop_orders: drop_orders(reference, r),
29        monitor: hist.last().map_or(f64::NAN, |h| h.1),
30        spread: spread(hist, DRAG_REL_WINDOW / MONITOR_EVERY),
31        tolerance: case.numerics.drag_rel_tol,
32    })
33}

35/// Cold-start run on a fresh solver.
36pub fn run_cold(case: &Case) -> PointResult {
37    let mut s = case.build_solver();
38    run_on(case, &mut s, false, None)
39}

41/// Iterate `s` (built from `case`) to convergence. `warm`: keep the field of the previous run, only the free stream changes
42/// (no first-order start-up; the implicit CFL ramp restarts). `r_ref`: residual reference of the drop criterion (`None` = peak
43/// of this run after iteration 20 / from the start when warm).
44pub fn run_on(case: &Case, s: &mut Solver, warm: bool, r_ref: Option<f64>) -> PointResult {
45    let n = case.numerics;
46    s.cfg.freestream = freestream(case.mach);
47    s.cfg.cfl = n.cfl;
48    s.cfg.limiter = n.limiter;
49    s.cfg.first_order_iters = n.first_order_iters;
50    s.low_mach = n.low_mach;
51    if s.scheme() != n.scheme {
52        s.set_scheme(n.scheme);
53    }
54    if !warm {
55        let q = prim_to_cons(&freestream(case.mach));
56        s.u.fill(q);
57        if s.viscous().is_some() {
58            // the turbulence state is not part of `u`: back to the free stream
59            s.set_viscous(s.viscous());
60        }
61    }
62    s.iter = if warm { n.first_order_iters } else { 0 };
63    s.restart_ramp();
64    let skip = if warm { 0 } else { 20 };
65    let sample = case.wake == Wake::Base || n.monitor_tol.is_some() || n.drag_rel_tol.is_some();
66    let mut residual = Vec::new();
67    let mut hist: Vec<(usize, f64)> = Vec::new();
68    let mut rmax: f64 = 0.0;
69    let mut reason = StopReason::MaxIterations;
70    for it in 0..n.max_iter {
71        let r = s.step();
72        residual.push(r);
73        if it >= skip {
74            rmax = rmax.max(r);
75        }
76        if sample && it % MONITOR_EVERY == MONITOR_EVERY - 1 {
77            hist.push((it + 1, monitor(case, s)));
78        }
79        if let Some(why) = stop_reason(case, it, r, r_ref.unwrap_or(rmax), skip, &hist) {
80            reason = why;
81            break;
82        }
83        if it % MONITOR_EVERY == MONITOR_EVERY - 1 && report(case, it + 1, r, r_ref.unwrap_or(rmax), &hist).is_break() {
84            reason = StopReason::Cancelled;
85            break;
86        }
87    }
88    finish(case, s, residual, rmax, hist, reason)
89}

91/// Largest magnitude of the convergence monitor (a drag or pressure coefficient) of a meaningful solution; a sample beyond it, or a
92/// non-finite one, means the state has blown up (the residual of such a state can still be finite).
93pub const MONITOR_LIMIT: f64 = 1e3;
94
95/// Stopping rule after iteration `it` with density residual `r`, in this order: non-finite residual, or a monitor sample taken at
96/// this iteration (the last of `hist`) that is non-finite or larger than [`MONITOR_LIMIT`] in magnitude (always); then, from
97/// `min_iter` on, `drop_orders` below `reference` ("residual drop reached"); or, with a `drag_rel_tol`, a viscous case at a monitor
98/// sample (`it + 1 >= DRAG_REL_WINDOW`, residual dropped 2 orders) whose total-CD range over the last `DRAG_REL_WINDOW` iterations
99/// is below `drag_rel_tol` times the magnitude of its mean (`CdStationary`); or, with a `monitor_tol`, every 500 iterations
100/// from `2 monitor_window` on and once the residual has dropped 2 orders, a monitor range (`hist`, one sample per `MONITOR_EVERY`
101/// iterations) below the tolerance over the last `monitor_window` iterations (`BasePressureStationary` for an inviscid base wake,
102/// `CdStationary` for a sting and for every viscous case, whose monitor is the total CD; the inviscid wake can be unsteady, the
103/// residual then stalls); then, for a viscous case with a `mean_window` and at every monitor sample, [`mean_stationary`]
104/// (`MeanStationary`). `skip` is 20 on a cold start, 0 warm.
105pub fn stop_reason(case: &Case, it: usize, r: f64, reference: f64, skip: usize, hist: &[(usize, f64)]) -> Option<StopReason> {
106    let n = &case.numerics;
107    if !r.is_finite() || hist.last().is_some_and(|&(last, m)| last == it + 1 && (!m.is_finite() || m.abs() > MONITOR_LIMIT)) {
108        return Some(StopReason::NonFinite);
109    }
110    if it + 1 < n.min_iter {
111        return None;
112    }
113    if it > skip && reference / r >= 10f64.powf(n.drop_orders) {
114        return Some(StopReason::ResidualDrop);
115    }
116    if let Some(tol) = n.drag_rel_tol {
117        let k = DRAG_REL_WINDOW / MONITOR_EVERY;
118        if case.viscous.is_some()
119            && hist.last().is_some_and(|&(last, _)| last == it + 1)
120            && it + 1 >= DRAG_REL_WINDOW
121            && reference / r >= 100.0
122            && hist.len() >= k
123        {
124            let w = &hist[hist.len() - k..];
125            let (lo, hi) = w.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
126            let mean = w.iter().map(|c| c.1).sum::<f64>() / k as f64;
127            if hi - lo < tol * mean.abs() {
128                return Some(StopReason::CdStationary);
129            }
130        }
131    }
132    if let Some(tol) = n.monitor_tol {
133        if it % 500 == 499 && it + 1 >= 2 * n.monitor_window && reference / r >= 100.0 {
134            let k = n.monitor_window / MONITOR_EVERY;
135            let (lo, hi) = hist[hist.len() - k..].iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
136            if hi - lo < tol {
137                return Some(match (case.viscous.is_some(), case.wake) {
138                    (false, Wake::Base) => StopReason::BasePressureStationary,
139                    (true, _) | (false, Wake::Sting) => StopReason::CdStationary,
140                });
141            }
142        }
143    }
144    if let (true, Some(window), Some(&(last, _))) = (case.viscous.is_some(), n.mean_window, hist.last()) {
145        if last == it + 1 && mean_stationary(hist, window / MONITOR_EVERY) {
146            return Some(StopReason::MeanStationary);
147        }
148    }
149    None
150}

152/// Mean-stationary test of a monitor history on the samples with iteration `> 2000` (the start-up transient is not judged), with
153/// `k` samples per averaging window: `m1` is the mean of the last `k` samples, `m0` the mean of the `k` before them and
154/// `m = (m0 + m1) / 2`. True when there are `2k` such samples, `|m1 - m0| <= 2e-3 |m|` (the mean of the oscillation is stationary),
155/// `max - min <= 0.05 |m|` over the last `2k` samples (the oscillation is bounded) and the last two samples lie on opposite sides
156/// of `m` (the state is at the mean of the cycle). `k = 0` never holds.
157pub fn mean_stationary(hist: &[(usize, f64)], k: usize) -> bool {
158    let from = hist.partition_point(|&(it, _)| it <= 2000);
159    let h = &hist[from..];
160    if k == 0 || h.len() < 2 * k {
161        return false;
162    }
163    let w = &h[h.len() - 2 * k..];
164    let mean = |s: &[(usize, f64)]| s.iter().map(|c| c.1).sum::<f64>() / k as f64;
165    let (m0, m1) = (mean(&w[..k]), mean(&w[k..]));
166    let m = 0.5 * (m0 + m1);
167    let (lo, hi) = w.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
168    let (a, b) = (w[2 * k - 2].1 - m, w[2 * k - 1].1 - m);
169    (m1 - m0).abs() <= 2e-3 * m.abs() && hi - lo <= 0.05 * m.abs() && a * b < 0.0
170}

172/// Convergence monitor of the current state: the total CD of a viscous case, else the nose CD (sting wake) or the area-mean base
173/// Cp (base wake).
174pub fn monitor(case: &Case, s: &Solver) -> f64 {
175    let bb = &case.body;
176    if case.viscous.is_some() {
177        let (wall, cd_nose, _) = wall_distribution(bb, s);
178        let fore = aft_drag(bb, &wall).iter().fold(cd_nose, |a, c| a + c);
179        return fore + friction(case, s).0 + base_state(s, bb.body.radius().powi(2)).1;
180    }
181    match case.wake {
182        Wake::Sting => wall_distribution(bb, s).1,
183        Wake::Base => base_state(s, bb.body.radius().powi(2)).2,
184    }
185}

187/// Pressure drag of each aft segment (`body.segments` order) from the extrapolated wall Cp of `wall`.
188fn aft_drag(bb: &BaseBody, wall: &[WallPoint]) -> Vec<f64> {
189    let mut cd_aft = vec![0.0; bb.segments.len()];
190    let stations: Vec<f64> = (0..=bb.segments.len()).map(|k| bb.station(k).0).collect();
191    for w in wall {
192        // a face belongs to segment k when stations[k] <= x < stations[k + 1]; the sting extension behind the base has no
193        // projected area and is not counted, nor is the nose
194        if let Some(k) = (0..cd_aft.len()).find(|&k| w.x >= stations[k] && w.x < stations[k + 1]) {
195            cd_aft[k] += w.cp * (w.s1 - w.s0);
196        }
197    }
198    cd_aft
199}

201/// Skin-friction drag and the largest `y+` of the first cell over the no-slip body-surface faces ahead of the base plane
202/// (the sting extension of a `Wake::Sting` grid and the surface of a sting behind the base are not part of the body): `(cd_friction, y_plus_max)`,
203/// `(0.0, NaN)` for the Euler equations. With the wall tangent `t = (nr, -nx)` pointing downstream, the first-cell tangential
204/// velocity `u_t`, its wall distance `d_n` and `tau_w = (mu(T) + mu_t,w) u_t / d_n` (`mu_t,w` the Aupoix wall eddy viscosity of a rough wall,
205/// `Viscous::roughness > 0`, else 0): `CD = sum 2 tau_w t_x r_m len / (q_inf R^2)`, `y+ = d_n sqrt(rho tau_w) / mu`. Works on the
206/// conserved state `s.u`.
207pub fn friction(case: &Case, s: &Solver) -> (f64, f64) {
208    let Some(v) = s.viscous() else {
209        return (0.0, f64::NAN);
210    };
211    let g = &s.grid;
212    let ni = g.ni;
213    let fs = s.cfg.freestream;
214    let q_inf = 0.5 * fs[0] * fs[1] * fs[1];
215    let r2 = case.body.body.radius().powi(2);
216    let x_base = case.body.x_base();
217    let rows = match s.block() {
218        None => vec![0],
219        Some(b) => vec![b.j_end, b.j_sting],
220    };
221    let (mut cd, mut y_plus_max) = (0.0, 0.0f64);
222    for j in rows {
223        for i in 0..ni {
224            if !s.noslip_jface(i, j) {
225                continue;
226            }
227            let (n0, n1) = (g.node(i, j), g.node(i + 1, j));
228            let xm = 0.5 * (n0.0 + n1.0);
229            if xm >= x_base {
230                continue;
231            }
232            let f = &g.jface[j * ni + i];
233            // the fluid cell above the face
234            let c = j * ni + i;
235            let w = cons_to_prim(&s.u[c]);
236            let (tx, tr) = (f.nr, -f.nx);
237            let u_t = w[1] * tx + w[2] * tr;
238            let d_n = ((g.xc[c] - xm) * f.nx + (g.rc[c] - f.rm) * f.nr).abs();
239            let mu = v.mu(temperature(&w));
240            // rough wall: the discrete wall shear is (mu + mu_t,w) u_t / d_n with the Aupoix wall eddy viscosity of the cell state (0 for a smooth wall)
241            let mu_tw = if v.rough() { w[0] * bowshock_cfd::turbulence::aupoix_wall(mu / w[0], w[1].hypot(w[2]), d_n, v.roughness)[2] } else { 0.0 };
242            let tau = (mu + mu_tw) * u_t / d_n;
243            cd += 2.0 * tau * tx * f.rm * f.len / (q_inf * r2);
244            y_plus_max = y_plus_max.max(d_n * (w[0] * tau.abs()).sqrt() / mu);
245        }
246    }
247    (cd, y_plus_max)
248}

250/// Wall points of the body (and of the sting extension) in axial order, and the nose drag from the extrapolated wall Cp and from
251/// the first-cell Cp: `(wall, cd_nose, cd_nose_cell)`.
252pub fn wall_distribution(body: &BaseBody, s: &Solver) -> (Vec<WallPoint>, f64, f64) {
253    let g = &s.grid;
254    let ni = g.ni;
255    let fs = s.cfg.freestream;
256    let (p_inf, q_inf) = (fs[3], 0.5 * fs[0] * fs[1] * fs[1]);
257    let cp = |p: f64| (p - p_inf) / q_inf;
258    let (ln, rb2) = (body.body.nose_length(), body.body.radius().powi(2));
259    let blk = s.block();
260    let mut wall = Vec::new();
261    let (mut cd, mut cd_cell) = (0.0, 0.0);
262    let i_wall_end = blk.map_or(ni, |b| b.i_end);
263    for i in 0..i_wall_end {
264        let j0 = blk.map_or(0, |b| b.col_start(i));
265        let (n0, n1) = (g.node(i, j0), g.node(i + 1, j0));
266        let (xm, rm) = (0.5 * (n0.0 + n1.0), 0.5 * (n0.1 + n1.1));
267        if xm <= 0.0 {
268            continue;
269        }
270        let (p0, p1) = (s.pressure(i, j0), s.pressure(i, j0 + 1));
271        let d0 = (g.xc[j0 * ni + i] - xm).hypot(g.rc[j0 * ni + i] - rm);
272        let d1 = (g.xc[(j0 + 1) * ni + i] - xm).hypot(g.rc[(j0 + 1) * ni + i] - rm);
273        let pw = p0 + (p0 - p1) * d0 / (d1 - d0);
274        let (s0, s1) = (n0.1 * n0.1 / rb2, n1.1 * n1.1 / rb2);
275        if n1.0 <= ln + 1e-12 {
276            cd += cp(pw) * (s1 - s0);
277            cd_cell += cp(p0) * (s1 - s0);
278        }
279        wall.push(WallPoint { x: xm, r: rm, s0, s1, cp: cp(pw), cp_cell: cp(p0) });
280    }
281    (wall, cd, cd_cell)
282}

284/// Base-face samples and integrals of the current state: `(samples, CD_base, area-mean Cp_b, area-mean first-cell Cp_b)`;
285/// NaN means / empty samples without a base. Wall pressure by linear extrapolation from the two first cells behind the base.
286pub fn base_state(s: &Solver, rb2: f64) -> (Vec<BasePoint>, f64, f64, f64) {
287    let Some(b) = s.block() else {
288        return (Vec::new(), 0.0, f64::NAN, f64::NAN);
289    };
290    let g = &s.grid;
291    let ni = g.ni;
292    let fs = s.cfg.freestream;
293    let (p_inf, q_inf) = (fs[3], 0.5 * fs[0] * fs[1] * fs[1]);
294    let cp = |p: f64| (p - p_inf) / q_inf;
295    let ie = b.i_end;
296    let (mut pts, mut cd, mut mean, mut mean_cell, mut s_sum) = (Vec::new(), 0.0, 0.0, 0.0, 0.0);
297    let j0 = s.nozzle().map_or(b.j_sting, |n| n.j_end);
298    for j in j0..b.j_end {
299        let (n0, n1) = (g.node(ie, j), g.node(ie, j + 1));
300        let (xm, rm) = (n0.0, 0.5 * (n0.1 + n1.1));
301        let (p0, p1) = (s.pressure(ie, j), s.pressure(ie + 1, j));
302        let d0 = (g.xc[j * ni + ie] - xm).hypot(g.rc[j * ni + ie] - rm);
303        let d1 = (g.xc[j * ni + ie + 1] - xm).hypot(g.rc[j * ni + ie + 1] - rm);
304        let pw = p0 + (p0 - p1) * d0 / (d1 - d0);
305        let ds = (n1.1 * n1.1 - n0.1 * n0.1) / rb2;
306        cd -= cp(pw) * ds;
307        mean += cp(pw) * ds;
308        mean_cell += cp(p0) * ds;
309        s_sum += ds;
310        pts.push(BasePoint { r0: n0.1, r1: n1.1, cp: cp(pw), cp_cell: cp(p0) });
311    }
312    (pts, cd, mean / s_sum, mean_cell / s_sum)
313}

315/// Post-process the state in `s` (works for a host `Solver` whose `u` was overwritten with another solver's state).
316pub fn finish(case: &Case, s: &Solver, residual: Vec<f64>, rmax: f64, hist: Vec<(usize, f64)>, reason: StopReason) -> PointResult {
317    let g = &s.grid;
318    let (ni, nj) = (g.ni, g.nj);
319    let fs = s.cfg.freestream;
320    let p_inf = fs[3];
321    let n = &case.numerics;
322    let bb = &case.body;
323    let rb2 = bb.body.radius().powi(2);
324    let blk = s.block();
325    let (wall, cd_nose, cd_nose_cell) = wall_distribution(bb, s);
326    let cd_aft = aft_drag(bb, &wall);
327    let (base_wall, cd_base, cp_base, cp_base_cell) = base_state(s, rb2);
328    let cd_base_model = match (case.wake, bb.sting_radius > 0.0) {
329        (Wake::Sting, _) => 0.0,
330        (Wake::Base, false) => base_drag_coefficient(case.mach) * bb.base_radius().powi(2) / rb2,
331        (Wake::Base, true) => f64::NAN,
332    };
333
334    let (cp_min, x_cp_min) = wall.iter().fold((f64::INFINITY, 0.0), |(m, xm), w| if w.cp < m { (w.cp, w.x) } else { (m, xm) });
335    let i_wall_end = blk.map_or(ni, |b| b.i_end);
336    let mut m_wall_max: f64 = 0.0;
337    for i in 0..i_wall_end {
338        let j0 = blk.map_or(0, |b| b.col_start(i));
339        if s.grid.xc[j0 * ni + i] > 0.0 {
340            let w = cons_to_prim(&s.u[j0 * ni + i]);
341            m_wall_max = m_wall_max.max(w[1].hypot(w[2]) / sound_speed(&w));
342        }
343    }
344    let dev = |i: usize, j: usize| (s.pressure(i, j) / p_inf - 1.0).abs();
345    let mut farfield_dp: f64 = 0.0;
346    for i in 0..ni {
347        farfield_dp = farfield_dp.max(dev(i, nj - 1));
348    }
349    for j in 0..nj {
350        farfield_dp = farfield_dp.max(dev(0, j)).max(dev(ni - 1, j));
351    }
352    let (mut min_rho, mut min_p, mut m_max) = (f64::INFINITY, f64::INFINITY, 0.0f64);
353    for (c, q) in s.u.iter().enumerate() {
354        if blk.is_some_and(|b| b.solid(c % ni, c / ni)) {
355            continue;
356        }
357        let w = cons_to_prim(q);
358        min_rho = min_rho.min(w[0]);
359        min_p = min_p.min(w[3]);
360        m_max = m_max.max(w[1].hypot(w[2]) / sound_speed(&w));
361    }
362    let nwin = (n.monitor_window / MONITOR_EVERY).min(hist.len());
363    let (mut cp_base_mean, mut cp_base_range, mut cd_range) = (cp_base, f64::NAN, f64::NAN);
364    match (case.viscous.is_some(), case.wake) {
365        (false, Wake::Base) => {
366            if nwin >= 2 {
367                let w = &hist[hist.len() - nwin..];
368                let (lo, hi) = w.iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
369                cp_base_mean = w.iter().map(|c| c.1).sum::<f64>() / nwin as f64;
370                cp_base_range = hi - lo;
371            }
372        }
373        // the monitor is the forebody CD (sting wake) or, for a viscous run, the total CD
374        (true, _) | (false, Wake::Sting) => {
375            if nwin >= 2 {
376                let (lo, hi) = hist[hist.len() - nwin..].iter().fold((f64::INFINITY, f64::NEG_INFINITY), |(l, h), c| (l.min(c.1), h.max(c.1)));
377                cd_range = hi - lo;
378            }
379        }
380    }
381    let fore = cd_aft.iter().fold(cd_nose, |a, c| a + c);
382    let (cd_friction, y_plus_max) = friction(case, s);
383    let (cd_total, cd_total_model) = (fore + cd_friction + cd_base, fore + cd_friction + cd_base_model);
384    let last = residual.last().copied().unwrap_or(rmax);
385    let drop_orders = if rmax > 0.0 && last > 0.0 { (rmax / last).log10() } else { 0.0 };
386    PointResult {
387        mach: case.mach,
388        alpha: 0.0,
389        iters: residual.len(),
390        residual,
391        residual_peak: rmax,
392        drop_orders,
393        stop_reason: reason,
394        wall,
395        base_wall,
396        cd_nose,
397        cd_nose_cell,
398        cd_aft,
399        cd_base,
400        cd_friction,
401        cd_total,

442    #[test]
443    fn stop_reason_order_and_gates() {
444        let flat: Vec<(usize, f64)> = (1..=40).map(|k| (k * 50, -0.1)).collect();
445        let (c, cs) = (case(Some(1e-3), Wake::Base), case(Some(1e-3), Wake::Sting));
446        // non-finite beats everything, even before min_iter
447        let mut g = case(Some(1e-3), Wake::Base);
448        g.numerics.min_iter = 10_000;
449        assert_eq!(stop_reason(&g, 5, f64::NAN, 1.0, 0, &[]), Some(StopReason::NonFinite));
450        assert_eq!(stop_reason(&g, 5, 1e-9, 1.0, 0, &[]), None);
451        // so does a blown-up monitor sampled at this iteration, with a finite residual
452        for bad in [f64::NAN, f64::INFINITY, 1.5e3, -1.5e3] {
453            assert_eq!(stop_reason(&g, 49, 1e-9, 1.0, 0, &[(50, bad)]), Some(StopReason::NonFinite), "{bad}");
454        }
455        assert_eq!(stop_reason(&g, 49, 1e-9, 1.0, 0, &[(50, 900.0)]), None);
456        assert_eq!(stop_reason(&g, 48, 1e-9, 1.0, 0, &[(50, 1.5e3)]), None);
457        // residual drop before the stationary rule
458        assert_eq!(stop_reason(&c, 4999, 1e-7, 1.0, 20, &flat), Some(StopReason::ResidualDrop));
459        // stationary rule: needs it % 500 == 499, 2 orders of drop and a flat history
460        assert_eq!(stop_reason(&c, 4999, 1e-3, 1.0, 20, &flat), Some(StopReason::BasePressureStationary));
461        assert_eq!(stop_reason(&cs, 4999, 1e-3, 1.0, 20, &flat), Some(StopReason::CdStationary));
462        assert_eq!(stop_reason(&c, 4998, 1e-3, 1.0, 20, &flat), None);
463        assert_eq!(stop_reason(&c, 4999, 0.5, 1.0, 20, &flat), None);
464        assert_eq!(stop_reason(&case(None, Wake::Base), 4999, 1e-3, 1.0, 20, &flat), None);
465        let mut moving = flat.clone();
466        moving[39].1 = 0.1;
467        assert_eq!(stop_reason(&c, 4999, 1e-3, 1.0, 20, &moving), None);
468        // a viscous case stops on a stationary total CD for either wake
469        let mut v = case(Some(1e-3), Wake::Base);
470        v.viscous = Some(Viscous { re_c: 1e6, sutherland: 0.38, turbulence: Turbulence::Sst, wall_i_start: 28, roughness: 0.0 });
471        assert_eq!(stop_reason(&v, 4999, 1e-3, 1.0, 20, &flat), Some(StopReason::CdStationary));
472    }

474    #[test]
475    fn relative_drag_rule_stops_a_flat_viscous_history() {
476        let hist = |rel: f64| -> Vec<(usize, f64)> { (1..=10).map(|k| (k * 50, 0.4 * (1.0 + if k % 2 == 0 { rel } else { 0.0 }))).collect() };
477        let mut v = case(None, Wake::Base);
478        v.viscous = Some(Viscous { re_c: 1e6, sutherland: 0.38, turbulence: Turbulence::Sst, wall_i_start: 28, roughness: 0.0 });
479        v.numerics.drag_rel_tol = Some(2e-3);
480        // (a) relative range 1e-3, 3 orders of drop: stops at the first sample at iteration 500, not before
481        assert_eq!(stop_reason(&v, 499, 1e-3, 1.0, 20, &hist(1e-3)), Some(StopReason::CdStationary));
482        assert_eq!(stop_reason(&v, 498, 1e-3, 1.0, 20, &hist(1e-3)), None);
483        assert_eq!(stop_reason(&v, 449, 1e-3, 1.0, 20, &hist(1e-3)[..9]), None);
484        // (b) only 1 order of drop
485        assert_eq!(stop_reason(&v, 499, 1e-1, 1.0, 20, &hist(1e-3)), None);
486        // (c) relative range 1e-2
487        assert_eq!(stop_reason(&v, 499, 1e-3, 1.0, 20, &hist(1e-2)), None);
488        // an inviscid case ignores the rule
489        let mut inv = v.clone();
490        inv.viscous = None;
491        assert_eq!(stop_reason(&inv, 499, 1e-3, 1.0, 20, &hist(1e-3)), None);
492    }

494    fn cycle(end: usize, mean: f64, amp: f64, drift_per_1000: f64) -> Vec<(usize, f64)> {
495        (1..=end / 50)
496            .map(|k| {
497                let it = k * 50;
498                (it, mean + amp * (std::f64::consts::TAU * it as f64 / 6000.0).sin() + drift_per_1000 * it as f64 / 1000.0)
499            })
500            .collect()
501    }

503    #[test]
504    fn mean_stationary_stops_at_a_mean_crossing_of_a_limit_cycle() {
505        let mut v = case(Some(1e-4), Wake::Base);
506        v.viscous = Some(Viscous { re_c: 1e6, sutherland: 0.38, turbulence: Turbulence::Sst, wall_i_start: 28, roughness: 0.0 });
507        v.numerics.mean_window = Some(6000);
508        let full = cycle(40_000, 0.1475, 0.0015, 0.0);
509        // first monitor sample that stops: not before 2 windows after the start-up, and with the last two samples across the mean
510        let stop = (1..=full.len()).find(|&n| stop_reason(&v, full[n - 1].0 - 1, 1e-3, 1.0, 20, &full[..n]).is_some()).expect("a stop");
511        let it = full[stop - 1].0;
512        assert!(it >= 2000 + 2 * 6000, "stopped at {it}");
513        assert_eq!(stop_reason(&v, it - 1, 1e-3, 1.0, 20, &full[..stop]), Some(StopReason::MeanStationary));
514        let (a, b) = (full[stop - 2].1 - 0.1475, full[stop - 1].1 - 0.1475);
515        assert!(a * b <= 0.0, "not at a mean crossing: {a} {b}");
516        // a call between two monitor samples never stops
517        assert_eq!(stop_reason(&v, it, 1e-3, 1.0, 20, &full[..stop]), None);
518        // Euler (no viscous model) and a case without a mean window never use the rule
519        let e = case(Some(1e-4), Wake::Base);
520        assert_eq!(stop_reason(&e, it - 1, 1e-3, 1.0, 20, &full[..stop]), None);
521        v.numerics.mean_window = None;
522        assert_eq!(stop_reason(&v, it - 1, 1e-3, 1.0, 20, &full[..stop]), None);
523    }

527        let any = |h: &[(usize, f64)]| (240..=h.len()).any(|n| mean_stationary(&h[..n], 120));
