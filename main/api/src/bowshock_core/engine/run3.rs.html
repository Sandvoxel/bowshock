1//! CPU runner of a [`Case3`]: iterate the [`Solver3`] to a converged or stationary normal force, and integrate the wall pressure into
2//! the force coefficients.
3//!
4//! Lengths are in nose diameters, the coefficients are referenced to `S_ref = pi/4` (nose diameter 1; the facade rescales to the
5//! reference diameter). Pressure forces only (Euler). The solver covers the half domain `theta in [0, pi]`, so forces are doubled.
6

15use std::ops::ControlFlow;

17/// Iterations between samples of the normal force.
18pub const SAMPLE_EVERY: usize = 50;
19/// Orders of density-residual drop that end a run as [`StopReason::ResidualDrop`].
20pub const DROP_ORDERS: f64 = 6.0;
21/// Iterations before the stationary rule may fire (it has no residual condition: the residual of a fin-body flow can plateau at
22/// 1 to 2 orders of drop with the CN flat to four digits).
23pub const STATIONARY_MIN_ITERS: usize = 1000;
24
25/// Pressure-force coefficients of the body plus fins.
26#[derive(Clone, Copy, Debug)]
27pub struct LiftForces {
28    /// Normal force coefficient (`+z`, towards the lee side at positive incidence).
29    pub cn: f64,
30    /// Axial (pressure) force coefficient, `+x` downstream.
31    pub ca: f64,
32    /// Centre of pressure of the normal force from the nose tip, nose diameters; NaN when `|sum Fz| < 1e-12`.
33    pub x_cp: f64,
34}
35
36/// Result of one 3D run.
37#[derive(Clone, Debug)]
38pub struct PointResult3 {
39    pub mach: f64,
40    /// Incidence \[rad\].
41    pub alpha: f64,
42    pub cn: f64,
43    pub ca: f64,
44    /// Nose diameters from the tip.
45    pub x_cp: f64,
46    pub iterations: usize,
47    /// `ResidualDrop` (converged), `CdStationary` (CN stationary), `MaxIterations` or `NonFinite`.
48    pub stop_reason: StopReason,
49    /// Orders of density-residual drop below the reference (peak) at the end of the run.
50    pub drop_orders: f64,
51    /// `(iteration, CN)` every [`SAMPLE_EVERY`] iterations.
52    pub cn_history: Vec<(usize, f64)>,
53}

55/// Wall forces of the current state of `s`: the `j = 0` faces on the body (`i >= axis_cells`, centroid `x <=` base station) and every
56/// face between a fluid and a solid cell. See [`lift_forces_upto`] for the wall pressure.
57pub fn lift_forces(case: &Case3, s: &Solver3) -> LiftForces {
58    lift_forces_upto(case, s, case.body.x_base())
59}
60
61/// [`lift_forces`] restricted to the wall faces with centroid `x <= x_max`. The wall pressure is extrapolated linearly from the two
62/// first fluid cells normal to the wall, as the 2D facade does (`engine::run::wall_distribution`): `pw = p0 + (p0 - p1) d0 / (d1 - d0)`,
63/// `d` the distance from the face centre to the cell centroid. A fin face whose second cell is solid or outside the grid uses the first-cell
64/// pressure. `dF = (pw - p_inf) area n_into_body`.
65pub fn lift_forces_upto(case: &Case3, s: &Solver3, x_max: f64) -> LiftForces {
66    forces_on(case, &s.grid, &s.u, s.cfg.freestream[4], x_max)
67}
68
69/// [`lift_forces_upto`] on a grid, a conserved state `u` (index `Grid3::cell`, e.g. downloaded from the GPU) and the free-stream pressure.
70pub fn forces_on(case: &Case3, g: &Grid3, u: &[Cons3], p_inf: f64, x_max: f64) -> LiftForces {
71    let acc = accumulate(g, u, p_inf, x_max, 0.0);
72    let (mut fx, mut fz, mut mz) = (0.0, 0.0, 0.0);
73    for a in &acc {
74        fx += a[0];
75        fz += a[1];
76        mz += a[2];
77    }
78    coefficients(case, [fx, fz, mz])
79}

81/// Number of force parts of [`LiftBreakdown`].
82const PARTS: usize = 5;
83
84/// Pressure-force coefficients of one part of the wetted surface; `x_cp` (nose diameters from the tip) is NaN when `|sum Fz| < 1e-12`.
85#[derive(Clone, Copy, Debug)]
86pub struct PartForces {
87    pub cn: f64,
88    pub ca: f64,
89    pub x_cp: f64,
90}
91
92/// [`LiftForces`] split by the surface the pressure acts on. The parts sum to [`forces_on`] (`x_max` = the base station).
93#[derive(Clone, Copy, Debug)]
94pub struct LiftBreakdown {
95    /// Body wall (`j = 0` faces) up to the end of the nose.
96    pub nose: PartForces,
97    /// Body wall behind the nose up to the base station.
98    pub afterbody: PartForces,
99    /// Fin side faces (faces between a fluid and a solid cell in `k`).
100    pub fin_sides: PartForces,
101    /// Fin leading and trailing edge faces (`i` faces of the fin boxes).
102    pub fin_edges: PartForces,
103    /// Fin tip faces (`j` faces of the fin boxes).
104    pub fin_tips: PartForces,
105}
106
107impl LiftBreakdown {
108    /// The whole fin: sides, edges and tip.
109    pub fn fins(&self) -> PartForces {
110        let parts = [self.fin_sides, self.fin_edges, self.fin_tips];
111        let cn: f64 = parts.iter().map(|p| p.cn).sum();
112        let ca: f64 = parts.iter().map(|p| p.ca).sum();
113        let m: f64 = parts.iter().filter(|p| p.x_cp.is_finite()).map(|p| p.cn * p.x_cp).sum();
114        PartForces { cn, ca, x_cp: if cn.abs() < 1e-12 { f64::NAN } else { m / cn } }
115    }
116}

118/// [`forces_on`] split into nose, afterbody and fin faces (the nose ends at the nose length of `case.body`, the body wall at its base
119/// station).
120pub fn breakdown_on(case: &Case3, g: &Grid3, u: &[Cons3], p_inf: f64) -> LiftBreakdown {
121    let acc = accumulate(g, u, p_inf, case.body.x_base(), case.body.body.nose_length());
122    let part = |a: [f64; 3]| {
123        let f = coefficients(case, a);
124        PartForces { cn: f.cn, ca: f.ca, x_cp: f.x_cp }
125    };
126    LiftBreakdown { nose: part(acc[0]), afterbody: part(acc[1]), fin_sides: part(acc[2]), fin_edges: part(acc[3]), fin_tips: part(acc[4]) }
127}

129fn coefficients(case: &Case3, [fx, fz, mz]: [f64; 3]) -> LiftForces {
130    let q = 0.5 * case.mach * case.mach;
131    let s_ref = std::f64::consts::FRAC_PI_4;
132    let norm = 2.0 / (q * s_ref);
133    LiftForces { cn: norm * fz, ca: norm * fx, x_cp: if fz.abs() < 1e-12 { f64::NAN } else { mz / fz } }
134}

136/// Dimensional force sums `[Fx, Fz, Mz]` (half domain, `Mz = sum x Fz`) of the five parts: body wall up to `x_nose`, body wall from
137/// `x_nose` to `x_max`, fin `k` faces, fin `i` faces, fin `j` faces.
138fn accumulate(g: &Grid3, u: &[Cons3], p_inf: f64, x_max: f64, x_nose: f64) -> [[f64; 3]; PARTS] {
139    let (ni, nj, nk) = (g.ni, g.nj, g.nk);
140    let p = |c: usize| cons_to_prim3(&u[c])[4];
141    let solid = |i: usize, j: usize, k: usize| g.is_solid(i, j, k);
142    let dist = |c: usize, f: &Face3| {
143        let x = g.xc[c];
144        ((x[0] - f.c[0]).powi(2) + (x[1] - f.c[1]).powi(2) + (x[2] - f.c[2]).powi(2)).sqrt()
145    };
146    let mut acc = [[0.0; 3]; PARTS];
147    // `a`: first fluid cell, `b`: the fluid cell behind it (if any); `sign`: +1 when the face normal points into the body
148    let mut add = |part: usize, a: usize, b: Option<usize>, f: &Face3, sign: f64| {
149        if f.c[0] > x_max {
150            return;
151        }
152        let part = if part == 0 && f.c[0] > x_nose { 1 } else { part };
153        let p0 = p(a);
154        let pw = match b {
155            Some(b) => {
156                let (d0, d1) = (dist(a, f), dist(b, f));
157                p0 + (p0 - p(b)) * d0 / (d1 - d0)
158            }
159            None => p0,
160        };
161        let d = (pw - p_inf) * f.area * sign;
162        acc[part][0] += d * f.n[0];
163        acc[part][1] += d * f.n[2];
164        acc[part][2] += d * f.n[2] * f.c[0];
165    };
166    for k in 0..nk {
167        for i in g.axis_cells..ni {
168            let f = &g.jface[k * (nj + 1) * ni + i];
169            // normal points into the fluid: into the body is its negative
170            if !solid(i, 0, k) {
171                add(0, g.cell(i, 0, k), (nj > 1 && !solid(i, 1, k)).then(|| g.cell(i, 1, k)), f, -1.0);
172            }
173        }
174    }
175    for k in 0..nk {
176        for j in 0..nj {
177            for i in 1..ni {
178                let f = &g.iface[(k * nj + j) * (ni + 1) + i];
179                match (solid(i - 1, j, k), solid(i, j, k)) {
180                    (false, true) => add(3, g.cell(i - 1, j, k), (i >= 2 && !solid(i - 2, j, k)).then(|| g.cell(i - 2, j, k)), f, 1.0),
181                    (true, false) => add(3, g.cell(i, j, k), (i + 1 < ni && !solid(i + 1, j, k)).then(|| g.cell(i + 1, j, k)), f, -1.0),
182                    _ => {}
183                }
184            }
185        }
186    }
187    for k in 0..nk {
188        for j in 1..nj {
189            for i in 0..ni {
190                let f = &g.jface[(k * (nj + 1) + j) * ni + i];
191                match (solid(i, j - 1, k), solid(i, j, k)) {
192                    (false, true) => add(4, g.cell(i, j - 1, k), (j >= 2 && !solid(i, j - 2, k)).then(|| g.cell(i, j - 2, k)), f, 1.0),
193                    (true, false) => add(4, g.cell(i, j, k), (j + 1 < nj && !solid(i, j + 1, k)).then(|| g.cell(i, j + 1, k)), f, -1.0),
194                    _ => {}
195                }
196            }
197        }
198    }
199    for k in 1..nk {
200        for j in 0..nj {
201            for i in 0..ni {
202                let f = &g.kface[(k * nj + j) * ni + i];
203                match (solid(i, j, k - 1), solid(i, j, k)) {
204                    (false, true) => add(2, g.cell(i, j, k - 1), (k >= 2 && !solid(i, j, k - 2)).then(|| g.cell(i, j, k - 2)), f, 1.0),
205                    (true, false) => add(2, g.cell(i, j, k), (k + 1 < nk && !solid(i, j, k + 1)).then(|| g.cell(i, j, k + 1)), f, -1.0),
206                    _ => {}
207                }
208            }
209        }
210    }
211    acc
212}

214/// Iterate `s` (built from `case`) until a stop rule fires, in this order at every iteration: a non-finite residual (or, at a sample,
215/// CN) gives `NonFinite`; a residual drop of [`DROP_ORDERS`] orders below the peak residual (from iteration 20 cold, from the start
216/// warm) gives `ResidualDrop`; at a sample with `it + 1 >= max(window, STATIONARY_MIN_ITERS)` and a range of the last
217/// `window / SAMPLE_EVERY` CN samples below `rel_tol |mean|` (whatever the residual did) gives `CdStationary`; the iteration cap
218/// gives `MaxIterations`.
219///
220/// `warm`: keep the field of the previous run, set the free stream of `case` and restart the CFL ramp (no first-order start-up).
221pub fn run3_on(case: &Case3, s: &mut Solver3, warm: bool) -> PointResult3 {
222    s.cfg.freestream = case.freestream();
223    s.restart_ramp();
224    if warm {
225        s.iter = s.iter.max(case.numerics.first_order_iters);
226    }
227    let mut rule = StopRule3::new(case, warm);
228    let mut it = 0;
229    while it < case.numerics.max_iter {
230        let r = s.step();
231        let cn = ((it + 1) % SAMPLE_EVERY == 0).then(|| lift_forces(case, s).cn);
232        it += 1;
233        if rule.check(it - 1, r, cn).is_some() {
234            break;
235        }
236        if cn.is_some() && rule.report(case, it).is_break() {
237            rule.cancel();
238            break;
239        }
240    }
241    rule.finish(case, lift_forces(case, s), it)
242}

244/// The stop rules of a 3D run, shared by the CPU and GPU runners. Feed it every iteration's residual (and, at iterations
245/// `it + 1 = 0 mod SAMPLE_EVERY`, the sampled CN); see [`run3_on`] for the order of the rules.
246pub struct StopRule3 {
247    skip: usize,
248    nwin: usize,
249    window: usize,
250    rel_tol: f64,
251    rmax: f64,
252    r: f64,
253    hist: Vec<(usize, f64)>,
254    reason: StopReason,
255}
256
257impl StopRule3 {
258    pub fn new(case: &Case3, warm: bool) -> StopRule3 {
259        let n = &case.numerics;
260        StopRule3 {
261            skip: if warm { 0 } else { 20 },
262            nwin: (n.window / SAMPLE_EVERY).max(1),
263            window: n.window,
264            rel_tol: n.rel_tol,
265            rmax: 0.0,
266            r: f64::NAN,
267            hist: Vec::new(),
268            reason: StopReason::MaxIterations,
269        }
270    }
271
272    /// Reports the run's progress after `iterations` iterations to the case's sink; `Break` when the sink asks to stop, and `Continue`
273    /// without a sink.
274    pub fn report(&self, case: &Case3, iterations: usize) -> ControlFlow<()> {
275        let Some(sink) = &case.progress else { return ControlFlow::Continue(()) };
276        sink.report(&Progress {
277            mach: case.mach,
278            alpha: case.alpha,
279            iteration: iterations,
280            max_iterations: case.numerics.max_iter,
281            residual_drop_orders: drop_orders(self.rmax, self.r),
282            monitor: self.hist.last().map_or(f64::NAN, |h| h.1),
283            spread: spread(&self.hist, self.nwin),
284            tolerance: Some(self.rel_tol),
285        })
286    }
287
288    /// Marks the run as ended by the sink: [`StopRule3::finish`] reports [`StopReason::Cancelled`].
289    pub fn cancel(&mut self) {
290        self.reason = StopReason::Cancelled;
291    }
292
293    /// Residual `r` of iteration `it` (0-based) and, at a sample iteration, the CN of the state after it. `Some(reason)` ends the run
294    /// with `it + 1` iterations.
295    pub fn check(&mut self, it: usize, r: f64, cn: Option<f64>) -> Option<StopReason> {
296        self.r = r;
297        let stop = |s: &mut Self, why| {
298            s.reason = why;
299            Some(why)
300        };
301        if !r.is_finite() {
302            return stop(self, StopReason::NonFinite);
303        }
304        if it >= self.skip {
305            self.rmax = self.rmax.max(r);
306        }
307        let drop = if self.rmax > 0.0 { (self.rmax / r.max(1e-300)).log10() } else { 0.0 };
308        if drop >= DROP_ORDERS {
309            return stop(self, StopReason::ResidualDrop);
310        }
311        if let Some(cn) = cn {
312            if !cn.is_finite() || cn.abs() > MONITOR_LIMIT {
313                return stop(self, StopReason::NonFinite);
314            }
315            self.hist.push((it + 1, cn));
316            if it + 1 >= self.window.max(STATIONARY_MIN_ITERS) && self.hist.len() >= self.nwin {
317                let w = &self.hist[self.hist.len() - self.nwin..];
318                // a state that no longer changes (every sample bit-identical) without a residual drop is a frozen run, not a converged one
319                if drop < 1.0 && w.iter().all(|&(_, c)| c.to_bits() == w[0].1.to_bits()) {
320                    return stop(self, StopReason::NonFinite);
321                }
322                let (lo, hi, sum) = w.iter().fold((f64::INFINITY, f64::NEG_INFINITY, 0.0), |a, &(_, c)| (a.0.min(c), a.1.max(c), a.2 + c));
323                if hi - lo < self.rel_tol * (sum / self.nwin as f64).abs() {
324                    return stop(self, StopReason::CdStationary);
325                }
326            }
327        }
328        None
329    }
330
331    /// Result after `iterations` iterations with the final forces `f`.
332    pub fn finish(self, case: &Case3, f: LiftForces, iterations: usize) -> PointResult3 {
333        let mut reason = self.reason;
334        if reason != StopReason::NonFinite && !(f.cn.is_finite() && f.ca.is_finite() && f.cn.abs() <= MONITOR_LIMIT) {
335            reason = StopReason::NonFinite;
336        }
337        let drop_orders = if self.rmax > 0.0 && self.r.is_finite() { (self.rmax / self.r.max(1e-300)).log10() } else { 0.0 };
338        PointResult3 {
339            mach: case.mach,
340            alpha: case.alpha,
341            cn: f.cn,
342            ca: f.ca,
343            x_cp: f.x_cp,
344            iterations,
345            stop_reason: reason,
346            drop_orders,
347            cn_history: self.hist,
348        }
349    }
350}

352/// Cold run on a fresh solver.
353pub fn run3_cold(case: &Case3) -> Result<PointResult3, crate::error::Error> {
354    let mut s = case.build_solver()?;
355    Ok(run3_on(case, &mut s, false))
356}

362    /// Feeds a synthetic run: residual `res(it)` every iteration, `cn(it + 1)` at the sample iterations. The stop, if any.
363    fn feed(res: impl Fn(usize) -> f64, cn: impl Fn(usize) -> f64) -> Option<(usize, StopReason)> {
364        let mut rule = StopRule3::new(&Case3::tiny_anf(2.0, 2f64.to_radians()), false);
365        for it in 0..3000 {
366            let sample = (it + 1) % SAMPLE_EVERY == 0;
367            if let Some(why) = rule.check(it, res(it), sample.then(|| cn(it + 1))) {
368                return Some((it + 1, why));
369            }
370        }
371        None
372    }
373
374    #[test]
375    fn a_frozen_state_without_a_residual_drop_is_a_failure() {
376        // every CN sample bit-identical, the residual flat at 0.5: frozen, not stationary
377        let stop = feed(|_| 0.5, |_| 0.316_928_495_035_045);
378        assert_eq!(stop, Some((1000, StopReason::NonFinite)));
379    }
380
381    #[test]
382    fn a_stationary_state_with_a_residual_drop_still_stops_stationary() {
383        // identical CN samples but the residual fell 2 orders (a converged limit): CdStationary as before
384        let stop = feed(|it| if it < 100 { 1.0 } else { 1e-2 }, |_| 0.3);
385        assert_eq!(stop, Some((1000, StopReason::CdStationary)));
386        // a CN that wiggles at round-off level without a drop is not frozen (not bit-identical): CdStationary
387        let stop = feed(|_| 0.5, |it| 0.3 + 1e-9 * (it % 3) as f64);
388        assert_eq!(stop, Some((1000, StopReason::CdStationary)));
389    }
