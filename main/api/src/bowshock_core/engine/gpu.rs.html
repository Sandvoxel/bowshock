1//! GPU runner of a [`Case`]: the same procedure, stopping rules and post-processing as the CPU [`crate::engine::run::run_on`], with the
2//! iterations executed by `bowshock_cfd::gpu::GpuSolver` (Luma array ops).
3//!
4//! The GPU records [`SYNC`] iterations back to back and reads the residual history. The monitor of the CPU loop (every
5//! [`MONITOR_EVERY`] iterations, which is the synchronisation interval) is sampled once per full chunk: the base-pressure
6//! history of an inviscid `Wake::Base` case by a GPU kernel that reads back only the two cells behind the base, the forebody CD
7//! of an inviscid `Wake::Sting` case and the total CD of a viscous case (friction included, which reads only the mean state) from
8//! the state downloaded into the host solver. The residual-based stopping rules run on the history
9//! iteration by iteration, so the reported stop iteration is the CPU-equivalent one; the GPU may execute up to `SYNC - 1`
10//! further iterations (`iters_exec`) and the final wall state is the one after those.
11

22type Result<T, E = Error> = std::result::Result<T, E>;

24/// Iterations between two synchronisations; equal to the monitor sampling interval of the CPU runner.
25pub const SYNC: usize = MONITOR_EVERY;
26
27/// GPU solver plus the host solver (grid, config, block, and a state buffer for post-processing) of one case.
28pub struct GpuRunner {
29    pub gpu: GpuSolver,
30    pub host: Solver,
31    /// Kernel build + geometry upload, seconds.
32    pub setup_seconds: f64,
33}
34
35/// Result of one GPU run plus its cost.
36pub struct GpuRun {
37    /// Same fields as the CPU result (post-processed with [`finish`] on the final GPU state);
38    /// `iters` / `residual` end at the CPU-equivalent stopping iteration.
39    pub result: PointResult,
40    /// Iterations after which the stopping rule fired / iterations actually executed (a multiple of [`SYNC`]).
41    pub iters_stop: usize,
42    pub iters_exec: usize,
43    /// Wall time of the iteration loop (kernels, residual-history and state read-back, monitor sampling).
44    pub loop_seconds: f64,
45}
46
47impl GpuRun {
48    /// Microseconds per executed iteration.
49    pub fn us_per_iter(&self) -> f64 {
50        1e6 * self.loop_seconds / self.iters_exec as f64
51    }
52}

54impl GpuRunner {
55    /// Builds the host solver of `case` and uploads it to the default GPU context.
56    pub fn new(case: &Case, prec: Precision) -> Result<GpuRunner> {
57        let host = case.try_build_solver()?;
58        let t0 = Instant::now();
59        let ctx = luma::default_context()?;
60        // The native CPU solver is faster than the GPU kernels on a software rasterizer, so `Backend::Auto` falls back to it.
61        if ctx.is_cpu() {
62            return Err(Error::Gpu("no GPU: Luma opened a software (CPU) Vulkan device".into()));
63        }
64        let gpu = GpuSolver::from_host(&ctx, &host, prec)?;
65        Ok(GpuRunner { gpu, host, setup_seconds: t0.elapsed().as_secs_f64() })
66    }
67
68    /// Copies the GPU state into the host solver's `u`.
69    pub fn download(&mut self) -> Result<()> {
70        self.host.u = self.gpu.state_f64()?;
71        Ok(())
72    }
73
74    /// Iterate to convergence on the GPU: the same procedure and arguments as [`crate::engine::run::run_on`]
75    /// (`warm`: keep the field, only change the free stream; `r_ref`: reference residual of the drop criterion).
76    pub fn run_on(&mut self, case: &Case, warm: bool, r_ref: Option<f64>) -> Result<GpuRun> {
77        let n = case.numerics;
78        let fs = freestream(case.mach);
79        let cfg = &mut self.host.cfg;
80        cfg.freestream = fs;
81        cfg.cfl = n.cfl;
82        cfg.limiter = n.limiter;
83        cfg.first_order_iters = n.first_order_iters;
84        self.gpu.set_config(&self.host.cfg)?;
85        self.host.low_mach = n.low_mach;
86        self.gpu.set_low_mach(n.low_mach);
87        if self.gpu.scheme() != n.scheme {
88            self.gpu.set_scheme(n.scheme)?;
89            self.host.set_scheme(n.scheme);
90        }
91        if warm {
92            self.gpu.set_iter(n.first_order_iters);
93            self.gpu.restart_ramp()?; // implicit scheme: CFL ramp restarts at every Mach number
94        } else {
95            self.gpu.reset()?; // free stream of `case.mach`, iteration 0, ramp restarted
96        }
97        self.gpu.context().sync()?;
98
99        let viscous = case.viscous.is_some();
100        let sampler = (case.wake == Wake::Base && !viscous).then(|| BaseSampler::new(&self.host, case.body.body.radius().powi(2)));
101        let sample = case.wake == Wake::Base || viscous || n.monitor_tol.is_some() || n.drag_rel_tol.is_some();
102        let mut residual = Vec::new();
103        let mut hist: Vec<(usize, f64)> = Vec::new();
104        let mut rmax: f64 = 0.0;
105        let mut reason = StopReason::MaxIterations;
106        let mut iters_stop = None;
107        let skip = if warm { 0 } else { 20 };
108        let mut done = 0usize;
109        let t0 = Instant::now();
110        while done < n.max_iter && iters_stop.is_none() {
111            let len = SYNC.min(n.max_iter - done);
112            let h = self.gpu.run(len)?;
113            // monitor sample of iteration `done + SYNC` (needs the state at exactly the end of a full chunk)
114            let value = if sample && len == SYNC {
115                Some(match &sampler {
116                    Some(sampler) => sampler.cp_base(&self.gpu.base_cells()?),
117                    None if viscous => {
118                        // total CD incl. friction: needs only the mean state (`u`) on the host, which `download` provides
119                        self.download()?;
120                        monitor(case, &self.host)
121                    }
122                    None => {
123                        self.download()?;
124                        wall_distribution(&case.body, &self.host).1
125                    }
126                })
127            } else {
128                None
129            };
130            let last_r = h.last().copied();
131            for (k, r) in h.into_iter().enumerate() {
132                let it = done + k;
133                residual.push(r);
134                if it >= skip {
135                    rmax = rmax.max(r);
136                }
137                if sample && it % SYNC == SYNC - 1 {
138                    hist.push((it + 1, value.expect("full chunk")));
139                }
140                if let Some(why) = stop_reason(case, it, r, r_ref.unwrap_or(rmax), skip, &hist) {
141                    reason = why;
142                    iters_stop = Some(it + 1);
143                    break;
144                }
145            }
146            done += len;
147            if iters_stop.is_none() && len == SYNC && report(case, done, last_r.unwrap_or(f64::NAN), r_ref.unwrap_or(rmax), &hist).is_break() {
148                reason = StopReason::Cancelled;
149                iters_stop = Some(done);
150            }
151        }
152        let loop_seconds = t0.elapsed().as_secs_f64();
153        self.download()?;
154        let result = finish(case, &self.host, residual, rmax, hist, reason);
155        Ok(GpuRun { iters_stop: iters_stop.unwrap_or(done), iters_exec: done, loop_seconds, result })
156    }
157}

159/// Host side of the GPU base-pressure sample: the geometric weights of [`crate::engine::run::base_state`] per base row (extrapolation
160/// factor `d0 / (d1 - d0)` and area weight), applied in f64 to the two cells per row of [`GpuSolver::base_cells`].
161pub struct BaseSampler {
162    rows: Vec<(f64, f64)>,
163    /// Leading entries of `base_cells()` that belong to the nozzle exit (not part of the base drag).
164    skip: usize,
165    p_inf: f64,
166    q_inf: f64,
167    s_sum: f64,
168}
169
170impl BaseSampler {
171    /// Weights of the base rows of `s` (which must have a solid block); `rb2` is the squared reference radius.
172    pub fn new(s: &Solver, rb2: f64) -> BaseSampler {
173        let b = s.block().expect("a case with a base has a solid block");
174        let (g, ni, fs) = (&s.grid, s.grid.ni, s.cfg.freestream);
175        let ie = b.i_end;
176        let j0 = s.nozzle().map_or(b.j_sting, |n| n.j_end);
177        let rows: Vec<(f64, f64)> = (j0..b.j_end)
178            .map(|j| {
179                let (n0, n1) = (g.node(ie, j), g.node(ie, j + 1));
180                let (xm, rm) = (n0.0, 0.5 * (n0.1 + n1.1));
181                let d0 = (g.xc[j * ni + ie] - xm).hypot(g.rc[j * ni + ie] - rm);
182                let d1 = (g.xc[j * ni + ie + 1] - xm).hypot(g.rc[j * ni + ie + 1] - rm);
183                (d0 / (d1 - d0), (n1.1 * n1.1 - n0.1 * n0.1) / rb2)
184            })
185            .collect();
186        let s_sum = rows.iter().map(|r| r.1).sum();
187        BaseSampler { rows, skip: j0 - b.j_sting, p_inf: fs[3], q_inf: 0.5 * fs[0] * fs[1] * fs[1], s_sum }
188    }
189
190    /// Area-mean Cp of the extrapolated base-wall pressure (`base_state(..).2`).
191    pub fn cp_base(&self, cells: &[[[f64; 4]; 2]]) -> f64 {
192        let mean: f64 = self
193            .rows
194            .iter()
195            .zip(cells.iter().skip(self.skip))
196            .map(|(&(k, ds), c)| {
197                let (p0, p1) = (cons_to_prim(&c[0])[3], cons_to_prim(&c[1])[3]);
198                (p0 + (p0 - p1) * k - self.p_inf) / self.q_inf * ds
199            })
200            .sum();
201        mean / self.s_sum
202    }
203}
