1//! `bowshock-core`: zero-lift drag of axisymmetric rockets versus Mach number (0.1 to 3) from CFD: the axisymmetric Euler or
2//! Reynolds-averaged Navier-Stokes equations (k-omega SST) on the GPU (Luma) or the CPU, plus empirical fin drag and a motor-on
3//! base flow with the exhaust jet.
4//!
5//! Describe the body with [`BodySpec`], optionally add [`FinSpec`] fins and a [`MotorSpec`] motor, give the flight conditions
6//! ([`FlightConditions`]) and run an [`Analysis`] over Mach numbers to get a [`Polar`]. The guide is at
7//! <https://sandvoxel.github.io/bowshock/>.
8//!
9//! ```no_run
10//! use bowshock_core::{AftSpec, Analysis, Atmosphere, BodySpec, FinProfile, FinSpec, FlightConditions, MotorSpec, NoseShape, NoseSpec};
11//!
12//! # fn main() -> Result<(), bowshock_core::Error> {
13//! let body = BodySpec {
14//!     nose: NoseSpec { shape: NoseShape::TangentOgive, length: 0.30, diameter: 0.10 },
15//!     aft: vec![AftSpec::Tube { length: 0.60 }, AftSpec::Transition { length: 0.10, aft_diameter: 0.08 }],
16//! };
17//! let fins = FinSpec { count: 4, root_chord: 0.12, tip_chord: 0.06, span: 0.08, sweep: 0.06, thickness: 0.003, roughness: 0.0, root_te_to_base: 0.0, profile: FinProfile::Rounded };
18//! let motor = MotorSpec { exit_diameter: 0.05, exit_mach: 2.5, exit_pressure_pa: 101_325.0, total_temperature_k: 1500.0 };
19//! let flight = FlightConditions { atmosphere: Atmosphere::Isa { altitude_m: 0.0 } };
20//!
21//! let polar = Analysis::new(&body)?.fins(fins).viscous(flight).motor(motor).run(&[0.5, 0.9, 1.5])?;
22//! for p in &polar.points {
23//!     println!("M {:.2}: CD {:.4} (base {:.4}), {:?}", p.mach, p.drag.total, p.drag.base, p.quality);
24//! }
25//! # Ok(())
26//! # }
27//! ```
