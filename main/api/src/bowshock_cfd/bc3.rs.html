1//! 3D characteristic far-field and static-pressure outflow ghost states, ported from [`crate::bc`] with the outward
2//! unit normal `n: [f64; 3]`. The Riemann-invariant logic acts on the normal velocity; the tangential velocity vector
3//! is taken from the upwind side (free stream on inflow, interior on outflow). The 2D [`Regime`] is reused.
4
5use crate::bc::Regime;
6use crate::gas3::{sound_speed3, Prim3, GAMMA};
7
8/// Normal Mach number `v.n / c` of `w` for the outward unit normal `n`.
9#[inline]
10pub fn normal_mach3(w: &Prim3, n: [f64; 3]) -> f64 {
11    (w[1] * n[0] + w[2] * n[1] + w[3] * n[2]) / sound_speed3(w)
12}
13
14/// Regime of a face from the interior state and the outward normal.
15#[inline]
16pub fn regime3(interior: &Prim3, n: [f64; 3]) -> Regime {
17    let m = normal_mach3(interior, n);
18    if m <= -1.0 {
19        Regime::SupersonicInflow
20    } else if m < 0.0 {
21        Regime::SubsonicInflow
22    } else if m < 1.0 {
23        Regime::SubsonicOutflow
24    } else {
25        Regime::SupersonicOutflow
26    }
27}
28
29/// Ghost state of the characteristic far-field BC. `fs` is the free stream, `interior` the first interior cell,
30/// `n` the outward unit normal. Returns the regime used as well.
31pub fn characteristic_ghost3(fs: &Prim3, interior: &Prim3, n: [f64; 3]) -> (Prim3, Regime) {
32    let reg = regime3(interior, n);
33    let g1 = GAMMA - 1.0;
34    let ghost = match reg {
35        Regime::SupersonicInflow => *fs,
36        Regime::SupersonicOutflow => *interior,
37        Regime::SubsonicInflow | Regime::SubsonicOutflow => {
38            let (ci, ce) = (sound_speed3(interior), sound_speed3(fs));
39            let uni = interior[1] * n[0] + interior[2] * n[1] + interior[3] * n[2];
40            let une = fs[1] * n[0] + fs[2] * n[1] + fs[3] * n[2];
41            let r_out = uni + 2.0 * ci / g1;
42            let r_in = une - 2.0 * ce / g1;
43            let un_b = 0.5 * (r_out + r_in);
44            let c_b = (0.25 * g1 * (r_out - r_in)).max(1e-8);
45            // reference state carrying entropy and tangential velocity
46            let (rf, un_ref) = if un_b >= 0.0 { (interior, uni) } else { (fs, une) };
47            let s = rf[4] / rf[0].powf(GAMMA);
48            let rho = (c_b * c_b / (GAMMA * s)).powf(1.0 / g1);
49            let p = rho * c_b * c_b / GAMMA;
50            let dn = un_b - un_ref;
51            [rho, rf[1] + dn * n[0], rf[2] + dn * n[1], rf[3] + dn * n[2], p]
52        }
53    };
54    (ghost, reg)
55}

57/// Fixed static-pressure outflow: subsonic faces (`M_n < 1`) take `rho, u, v, w` from the interior and
58/// `p = p_back`; supersonic faces extrapolate.
59pub fn pressure_outflow_ghost3(p_back: f64, interior: &Prim3, n: [f64; 3]) -> Prim3 {
60    if normal_mach3(interior, n) >= 1.0 {
61        *interior
62    } else {
63        [interior[0], interior[1], interior[2], interior[3], p_back]
64    }
65}

74    fn unit(rng: &mut Rng) -> [f64; 3] {
75        let n = [rng.range(-1.0, 1.0), rng.range(-1.0, 1.0), rng.range(-1.0, 1.0)];
76        let l = (n[0] * n[0] + n[1] * n[1] + n[2] * n[2]).sqrt();
77        [n[0] / l, n[1] / l, n[2] / l]
78    }
79
80    fn state(rng: &mut Rng, vmax: f64) -> Prim3 {
81        [rng.range(0.5, 2.0), rng.range(-vmax, vmax), rng.range(-vmax, vmax), rng.range(-vmax, vmax), rng.range(0.5, 2.0)]
82    }

84    #[test]
85    fn matches_2d_in_xz_plane() {
86        let mut rng = Rng::new(5);
87        let mut seen = [false; 4];
88        for it in 0..4000 {
89            let vmax = [0.3, 1.0, 3.0][it % 3];
90            let t = rng.range(0.0, std::f64::consts::TAU);
91            let (nx, nr) = (t.cos(), t.sin());
92            let n = [nx, 0.0, nr];
93            let mut w = state(&mut rng, vmax);
94            w[2] = 0.0;
95            let w2 = [w[0], w[1], w[3], w[4]];
96            let fs3 = freestream3(rng.range(0.3, 3.0), rng.range(-0.3, 0.3));
97            let fs2 = [fs3[0], fs3[1], fs3[3], fs3[4]];
98            let (g3, r3) = characteristic_ghost3(&fs3, &w, n);
99            let (g2, r2) = bc::characteristic_ghost(&fs2, &w2, nx, nr);
100            assert_eq!(r3, r2);
101            seen[r3 as usize] = true;
102            assert_eq!(g3[2], 0.0);
103            for (a, b) in [(g3[0], g2[0]), (g3[1], g2[1]), (g3[3], g2[2]), (g3[4], g2[3])] {
104                assert!((a - b).abs() < 1e-14 * (1.0 + b.abs()), "{a} vs {b}");
105            }
106            let pb = rng.range(0.3, 1.5);
107            let o3 = pressure_outflow_ghost3(pb, &w, n);
108            let o2 = bc::pressure_outflow_ghost(pb, &w2, nx, nr);
109            assert_eq!(o3[2], 0.0);
110            for (a, b) in [(o3[0], o2[0]), (o3[1], o2[1]), (o3[3], o2[2]), (o3[4], o2[3])] {
111                assert!((a - b).abs() < 1e-14);
112            }
113        }
114        assert!(seen.iter().all(|&s| s), "every regime exercised: {seen:?}");
115    }

117    #[test]
118    fn rotation_invariance() {
119        let mut rng = Rng::new(9);
120        for it in 0..4000 {
121            let vmax = [0.3, 1.0, 3.0][it % 3];
122            let r = random_rotation(&mut rng);
123            let n = unit(&mut rng);
124            let w = state(&mut rng, vmax);
125            let fs = freestream3(rng.range(0.3, 3.0), rng.range(-0.3, 0.3));
126            let (g, reg) = characteristic_ghost3(&fs, &w, n);
127            let (gr, regr) = characteristic_ghost3(&rot5(&r, &fs), &rot5(&r, &w), rot(&r, n));
128            assert_eq!(reg, regr);
129            let want = rot5(&r, &g);
130            for k in 0..5 {
131                assert!((want[k] - gr[k]).abs() < 1e-12 * (1.0 + want[k].abs()));
132            }
133            let pb = rng.range(0.3, 1.5);
134            let o = pressure_outflow_ghost3(pb, &w, n);
135            let or = pressure_outflow_ghost3(pb, &rot5(&r, &w), rot(&r, n));
136            let want = rot5(&r, &o);
137            for k in 0..5 {
138                assert!((want[k] - or[k]).abs() < 1e-12 * (1.0 + want[k].abs()));
139            }
140        }
141    }
