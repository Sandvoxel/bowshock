1//! Viscous terms of the axisymmetric compressible Navier-Stokes equations (laminar; the eddy-viscosity arrays are carried
2//! for the turbulence model but stay zero until it is attached).
3//!
4//! Units: `rho_inf = 1`, `c_inf = 1`, `p_inf = 1/gamma`, `T = gamma p / rho` (`T_inf = 1`), lengths in nose diameters.
5//!
6//! ```text
7//!   mu(T)  = (1 / re_c) T^1.5 (1 + S) / (T + S)            S = sutherland = 110.4 K / T_inf[K],  re_c = rho_inf c_inf d / mu_inf
8//!   kappa  = (mu / PRANDTL + mu_t / PRANDTL_T) / (gamma - 1)         q = -kappa grad T        (c_p = 1 / (gamma - 1))
9//! ```
10//!
11//! The residual of a cell uses the total face flux `F - Fv` in place of the Euler flux `F` and the radial-momentum source
12//! `(p - t_tt) A / V` in place of `p A / V` (`t_tt` = hoop stress):
13//!
14//! ```text
15//!   V dU/dt = - sum_faces w_f (F - Fv).n + (0, 0, (p - t_tt) A, 0),            w_f = len_f r_m
16//!   Fv.n    = [0, txx nx + txr nr, txr nx + trr nr, (u txx + v txr + kappa T_x) nx + (u txr + v trr + kappa T_r) nr]
17//!   div     = u_x + v_r + v / r;   txx = mue (2 u_x - 2/3 div);  trr = mue (2 v_r - 2/3 div);  txr = mue (u_r + v_x);  mue = mu + mu_t
18//!   t_tt    = (mu_c + mu_t,c) (2 v_c / r_c - 2/3 div_c)
19//! ```
20//!
21//! Three data-parallel passes, called from `Solver::eval_residual` after the inviscid face fluxes:
22//!
23//! 1. `cell_pass`: `mu(T)` and the Green-Gauss gradient `(u_x, u_r, v_x, v_r, T_x, T_r)` of every fluid cell in the meridian
24//!    plane, `g = (1/A) sum_faces phi_f n_f len_f` with `phi_f = (phi_c + phi_neighbour) / 2`;
25//! 2. `face_pass`: the face gradient `g_f = gbar + ((phi_R - phi_L) / |d| - gbar.e) e` (`d` = distance vector of the two
26//!    adjacent cell centres, `e = d / |d|`, `gbar` the mean of the two cell gradients: a compact normal derivative with the
27//!    averaged tangential part) and the viscous flux of every face into `fvi`, `fvj`; faces on the axis (`r_m = 0`) and inside
28//!    the solid block carry no flux;
29//! 3. the cell residual in `solver.rs` (with `hoop_stress` and the diffusive time-step limit `diffusion_rate`).
30//!
31//! A state across a boundary is the ghost cell of `Solver::w`; the solid cell across a wall of a body block is its ghost as
32//! well (block ghost in the j direction, [`wall_ghost`] across the base wall). A ghost sits at the mirror image of the centre
33//! of its cell in the face centre and carries the gradient, viscosity and eddy viscosity of that cell (the eddy viscosity with
34//! the sign flipped across a no-slip face, so it vanishes at the wall).
35//!
36//! No-slip walls (`Walls`): the face where the viscous flow starts (`Viscous::wall_i_start`) and the body surface, base wall
37//! and sting surface of a block. The wall ghost of a no-slip face is `(rho, -u, -v, p)` (adiabatic wall, zero normal pressure
38//! gradient), that of a slip face the reflection about the face normal.
39

45impl Viscous {
46    /// Dynamic viscosity (Sutherland's law) at the static temperature `t` (free-stream temperature = 1).
47    #[inline]
48    pub fn mu(&self, t: f64) -> f64 {
49        t.powf(1.5) * (1.0 + self.sutherland) / (t + self.sutherland) / self.re_c
50    }
51}

53/// Static temperature `gamma p / rho` of a primitive state.
54#[inline]
55pub fn temperature(w: &Prim) -> f64 {
56    GAMMA * w[3] / w[0]
57}

59/// Ghost state across a wall face with unit normal `(nx, nr)`: `(rho, -u, -v, p)` on a no-slip face, else the reflection.
60#[inline]
61pub fn wall_ghost(w: &Prim, f: &Face, noslip: bool) -> Prim {
62    if noslip {
63        [w[0], -w[1], -w[2], w[3]]
64    } else {
65        reflect(w, f.nx, f.nr)
66    }
67}

69/// Which wall faces are no-slip (only with viscous terms on). The single classification used by the ghost fills, the flux
70/// kernels and the implicit operator. A nozzle ([`Nozzle`]) turns the first base rows (`j < j_end`) into a fixed inflow
71/// state: those faces are never no-slip walls ([`Walls::jet_iface`]).
72///
73/// * Without a block: a `jmin` face with a slip-wall boundary condition and `i >= wall_i_start`.
74/// * With a block: the body-surface faces (`j = j_end`, `wall_i_start <= i < i_end`), every base-wall face (`i = i_end`,
75///   `j_sting <= j < j_end`) and every sting-surface face (`j = j_sting > 0`, `i >= i_end`); the axis stays a slip wall.
76#[derive(Clone, Copy, Debug)]
77pub(crate) struct Walls {
78    pub visc: Option<Viscous>,
79    pub jmin_slip: bool,
80    pub blk: Option<Block>,
81    pub nozzle: Option<Nozzle>,
82}
83
84impl Walls {
85    pub(crate) fn new(visc: Option<Viscous>, bc: &[Bc; 4], blk: Option<Block>, nozzle: Option<Nozzle>) -> Walls {
86        Walls { visc, jmin_slip: bc[2] == Bc::SlipWall, blk, nozzle }
87    }
88
89    /// Is the j-face `(i, j)` (between cells `j - 1` and `j` of column `i`) a no-slip wall?
90    #[inline]
91    pub(crate) fn noslip_jface(&self, i: usize, j: usize) -> bool {
92        let Some(v) = self.visc else { return false };
93        match self.blk {
94            None => j == 0 && self.jmin_slip && i >= v.wall_i_start,
95            Some(b) => (j == b.j_end && i >= v.wall_i_start && i < b.i_end) || (j == b.j_sting && b.j_sting > 0 && i >= b.i_end),
96        }
97    }
98
99    /// Is the i-face `(i, j)` (between cells `i - 1` and `i` of row `j`) the no-slip base wall?
100    #[inline]
101    pub(crate) fn noslip_iface(&self, i: usize, j: usize) -> bool {
102        self.visc.is_some() && self.blk.is_some_and(|b| i == b.i_end && j >= b.j_sting && j < b.j_end) && !self.nozzle.is_some_and(|n| j < n.j_end)
103    }
104
105    /// Is the i-face `(i, j)` a nozzle exit face (fixed inflow state)?
106    #[inline]
107    pub(crate) fn jet_iface(&self, i: usize, j: usize) -> bool {
108        self.blk.is_some_and(|b| i == b.i_end) && self.nozzle.is_some_and(|n| j < n.j_end)
109    }
110}

131/// Sides of i-face `(i, j)`; `None` inside the solid block. `w` is the primitive array with ghost layers.
132pub(crate) fn iface_sides(grid: &Grid, w: &[Prim], walls: &Walls, i: usize, j: usize) -> Option<FaceSides> {
133    let ni = grid.ni;
134    let (ii, jj) = (i as isize, j as isize);
135    if let Some(b) = walls.blk {
136        if j < b.j_end {
137            if j < b.j_sting || i < b.i_end {
138                return None;
139            }
140            if i == b.i_end {
141                let c = j * ni + i;
142                let wr = w[pidx(ni, ii, jj)];
143                if let Some(n) = walls.nozzle.filter(|n| j < n.j_end) {
144                    let fixed = walls.visc.map(|v| {
145                        let mu_t = if v.turbulence == Turbulence::Sst { n.state[0] * n.turb[0] / n.turb[1] } else { 0.0 };
146                        (v.mu(temperature(&n.state)), mu_t)
147                    });
148                    return Some(FaceSides {
149                        l: Side { w: n.state, cell: c, ghost: true, fixed },
150                        r: Side { w: wr, cell: c, ghost: false, fixed: None },
151                        noslip: false,
152                    });
153                }
154                let noslip = walls.noslip_iface(i, j);
155                let wl = wall_ghost(&wr, &grid.iface[j * (ni + 1) + i], noslip);
156                return Some(FaceSides { l: Side { w: wl, cell: c, ghost: true, fixed: None }, r: Side { w: wr, cell: c, ghost: false, fixed: None }, noslip });
157            }
158        }
159    }
160    let side = |w: Prim, cell: usize, ghost: bool| Side { w, cell, ghost, fixed: None };
161    Some(if i == 0 {
162        let c = j * ni;
163        FaceSides { l: side(w[pidx(ni, -1, jj)], c, true), r: side(w[pidx(ni, 0, jj)], c, false), noslip: false }
164    } else if i == ni {
165        let c = j * ni + ni - 1;
166        FaceSides { l: side(w[pidx(ni, ii - 1, jj)], c, false), r: side(w[pidx(ni, ii, jj)], c, true), noslip: false }
167    } else {
168        FaceSides { l: side(w[pidx(ni, ii - 1, jj)], j * ni + i - 1, false), r: side(w[pidx(ni, ii, jj)], j * ni + i, false), noslip: false }
169    })
170}

172/// Sides of j-face `(i, j)`; `None` inside the solid block.
173pub(crate) fn jface_sides(grid: &Grid, w: &[Prim], walls: &Walls, i: usize, j: usize) -> Option<FaceSides> {
174    let (ni, nj) = (grid.ni, grid.nj);
175    let (ii, jj) = (i as isize, j as isize);
176    let side = |w: Prim, cell: usize, ghost: bool| Side { w, cell, ghost, fixed: None };
177    if let Some(b) = walls.blk {
178        let j0 = b.col_start(i);
179        if j < j0 {
180            return None;
181        }
182        if j == j0 && j0 > 0 {
183            // body or sting surface: the solid cell below holds the block ghost
184            let c = j * ni + i;
185            return Some(FaceSides { l: side(w[pidx(ni, ii, jj - 1)], c, true), r: side(w[pidx(ni, ii, jj)], c, false), noslip: walls.noslip_jface(i, j) });
186        }
187    }
188    Some(if j == 0 {
189        let c = i;
190        FaceSides { l: side(w[pidx(ni, ii, -1)], c, true), r: side(w[pidx(ni, ii, 0)], c, false), noslip: walls.noslip_jface(i, 0) }
191    } else if j == nj {
192        let c = (nj - 1) * ni + i;
193        FaceSides { l: side(w[pidx(ni, ii, jj - 1)], c, false), r: side(w[pidx(ni, ii, jj)], c, true), noslip: false }
194    } else {
195        FaceSides { l: side(w[pidx(ni, ii, jj - 1)], (j - 1) * ni + i, false), r: side(w[pidx(ni, ii, jj)], j * ni + i, false), noslip: false }
196    })
197}

199/// Centre of i-face `(i, j)` (mean of its end nodes).
200#[inline]
201pub(crate) fn iface_centre(grid: &Grid, i: usize, j: usize) -> (f64, f64) {
202    let (a, b) = (grid.node(i, j), grid.node(i, j + 1));
203    (0.5 * (a.0 + b.0), 0.5 * (a.1 + b.1))
204}

206/// Centre of j-face `(i, j)`.
207#[inline]
208pub(crate) fn jface_centre(grid: &Grid, i: usize, j: usize) -> (f64, f64) {
209    let (a, b) = (grid.node(i, j), grid.node(i + 1, j));
210    (0.5 * (a.0 + b.0), 0.5 * (a.1 + b.1))
211}

213/// Position of a side: the centre of its cell, or the mirror image of that centre in the face centre for a ghost.
214#[inline]
215fn side_pos(grid: &Grid, s: &Side, xf: f64, rf: f64) -> (f64, f64) {
216    let (x, r) = (grid.xc[s.cell], grid.rc[s.cell]);
217    if s.ghost {
218        (2.0 * xf - x, 2.0 * rf - r)
219    } else {
220        (x, r)
221    }
222}

224/// `(|d|, e_x, e_r)`: distance and direction between the two sides.
225#[inline]
226pub(crate) fn stencil(grid: &Grid, sides: &FaceSides, xf: f64, rf: f64) -> (f64, f64, f64) {
227    let (xl, rl) = side_pos(grid, &sides.l, xf, rf);
228    let (xr, rr) = side_pos(grid, &sides.r, xf, rf);
229    let (dx, dr) = (xr - xl, rr - rl);
230    let dd = dx.hypot(dr);
231    (dd, dx / dd, dr / dd)
232}

234/// Face viscosity and eddy viscosity: mean of the two sides, a ghost taking its cell's values (eddy viscosity negated
235/// across a no-slip face; a jet ghost has the fixed jet values).
236#[inline]
237pub(crate) fn face_mu(mu: &[f64], mu_t: &[f64], sides: &FaceSides) -> (f64, f64) {
238    let mu_side = |s: &Side| s.fixed.map_or(mu[s.cell], |f| f.0);
239    let mut_side = |s: &Side| match s.fixed {
240        Some(f) => f.1,
241        None if s.ghost && sides.noslip => -mu_t[s.cell],
242        None => mu_t[s.cell],
243    };
244    (0.5 * (mu_side(&sides.l) + mu_side(&sides.r)), 0.5 * (mut_side(&sides.l) + mut_side(&sides.r)))
245}

247/// `(u, v, T)` of a primitive state.
248#[inline]
249fn phi(w: &Prim) -> [f64; 3] {
250    [w[1], w[2], temperature(w)]
251}

253/// Pass 1: viscosity and meridian-plane Green-Gauss gradient `(u_x, u_r, v_x, v_r, T_x, T_r)` of every fluid cell (zero in
254/// solid cells).
255pub(crate) fn cell_pass(v: &Viscous, grid: &Grid, w: &[Prim], walls: &Walls, grad: &mut [[f64; 6]], mu: &mut [f64]) {
256    let ni = grid.ni;
257    grad.par_chunks_mut(ni).zip(mu.par_chunks_mut(ni)).enumerate().for_each(|(j, (grow, mrow))| {
258        for i in 0..ni {
259            if walls.blk.is_some_and(|b| b.solid(i, j)) {
260                grow[i] = [0.0; 6];
261                mrow[i] = 0.0;
262                continue;
263            }
264            let wc = &w[pidx(ni, i as isize, j as isize)];
265            mrow[i] = v.mu(temperature(wc));
266            let pc = phi(wc);
267            let fl = &grid.iface[j * (ni + 1) + i];
268            let fr = &grid.iface[j * (ni + 1) + i + 1];
269            let fb = &grid.jface[j * ni + i];
270            let ft = &grid.jface[(j + 1) * ni + i];
271            let nb_l = iface_sides(grid, w, walls, i, j).expect("fluid cell face").l.w;
272            let nb_r = iface_sides(grid, w, walls, i + 1, j).expect("fluid cell face").r.w;
273            let nb_b = jface_sides(grid, w, walls, i, j).expect("fluid cell face").l.w;
274            let nb_t = jface_sides(grid, w, walls, i, j + 1).expect("fluid cell face").r.w;
275            let mut g = [0.0; 6];
276            for (f, s, nb) in [(fr, 1.0, &nb_r), (fl, -1.0, &nb_l), (ft, 1.0, &nb_t), (fb, -1.0, &nb_b)] {
277                let pn = phi(nb);
278                for m in 0..3 {
279                    let pf = 0.5 * (pc[m] + pn[m]);
280                    g[2 * m] += s * pf * f.nx * f.len;
281                    g[2 * m + 1] += s * pf * f.nr * f.len;
282                }
283            }
284            let inv_a = 1.0 / grid.area[j * ni + i];
285            for x in g.iter_mut() {
286                *x *= inv_a;
287            }
288            grow[i] = g;
289        }
290    });
291}

293/// Viscous flux `Fv.n` of one face (see the module documentation); `f` is the face, `(xf, rf)` its centre.
294fn viscous_flux(grid: &Grid, grad: &[[f64; 6]], mu: &[f64], mu_t: &[f64], sides: &FaceSides, f: &Face, xf: f64, rf: f64) -> Cons {
295    if f.rm == 0.0 {
296        return [0.0; 4];
297    }
298    let (dd, ex, er) = stencil(grid, sides, xf, rf);
299    let (gl, gr) = (&grad[sides.l.cell], &grad[sides.r.cell]);
300    let (pl, pr) = (phi(&sides.l.w), phi(&sides.r.w));
301    let mut gf = [0.0; 6];
302    for m in 0..3 {
303        let (gbx, gbr) = (0.5 * (gl[2 * m] + gr[2 * m]), 0.5 * (gl[2 * m + 1] + gr[2 * m + 1]));
304        let corr = (pr[m] - pl[m]) / dd - (gbx * ex + gbr * er);
305        gf[2 * m] = gbx + corr * ex;
306        gf[2 * m + 1] = gbr + corr * er;
307    }
308    let (mu_f, mut_f) = face_mu(mu, mu_t, sides);
309    let (uf, vf) = (0.5 * (sides.l.w[1] + sides.r.w[1]), 0.5 * (sides.l.w[2] + sides.r.w[2]));
310    let (u_x, u_r, v_x, v_r, t_x, t_r) = (gf[0], gf[1], gf[2], gf[3], gf[4], gf[5]);
311    let div = u_x + v_r + vf / f.rm;
312    let mue = mu_f + mut_f;
313    let txx = mue * (2.0 * u_x - 2.0 / 3.0 * div);
314    let trr = mue * (2.0 * v_r - 2.0 / 3.0 * div);
315    let txr = mue * (u_r + v_x);
316    let kappa = (mu_f / PRANDTL + mut_f / PRANDTL_T) / (GAMMA - 1.0);
317    [0.0, txx * f.nx + txr * f.nr, txr * f.nx + trr * f.nr, (uf * txx + vf * txr + kappa * t_x) * f.nx + (uf * txr + vf * trr + kappa * t_r) * f.nr]
318}

320/// Pass 2: viscous fluxes of every i-face (`fvi`, `(ni + 1) * nj`) and j-face (`fvj`, `ni * (nj + 1)`).
321pub(crate) fn face_pass(grid: &Grid, w: &[Prim], walls: &Walls, grad: &[[f64; 6]], mu: &[f64], mu_t: &[f64], fvi: &mut [Cons], fvj: &mut [Cons]) {
322    let ni = grid.ni;
323    fvi.par_chunks_mut(ni + 1).enumerate().for_each(|(j, row)| {
324        for (i, out) in row.iter_mut().enumerate() {
325            *out = match iface_sides(grid, w, walls, i, j) {
326                Some(s) => {
327                    let (xf, rf) = iface_centre(grid, i, j);
328                    viscous_flux(grid, grad, mu, mu_t, &s, &grid.iface[j * (ni + 1) + i], xf, rf)
329                }
330                None => [0.0; 4],
331            };
332        }
333    });
334    fvj.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
335        for (i, out) in row.iter_mut().enumerate() {
336            *out = match jface_sides(grid, w, walls, i, j) {
337                Some(s) => {
338                    let (xf, rf) = jface_centre(grid, i, j);
339                    viscous_flux(grid, grad, mu, mu_t, &s, &grid.jface[j * ni + i], xf, rf)
340                }
341                None => [0.0; 4],
342            };
343        }
344    });
345}

347/// Hoop stress `t_tt = (mu_c + mu_t,c)(2 v_c / r_c - 2/3 div_c)`, `div_c = u_x + v_r + v_c / r_c`, from the cell gradient `g`.
348#[inline]
349pub(crate) fn hoop_stress(mue: f64, g: &[f64; 6], v_c: f64, r_c: f64) -> f64 {
350    let div = g[0] + g[3] + v_c / r_c;
351    mue * (2.0 * v_c / r_c - 2.0 / 3.0 * div)
352}

354/// Diffusive contribution `2 lv len^2 / A` to the spectral radius `L_d` of one direction (`len`: mean length of the two faces of
355/// that direction), with `lv = gamma (mu / PRANDTL + mu_t / PRANDTL_T) / rho`.
356#[inline]
357pub(crate) fn diffusion_rate(mu_c: f64, mu_t_c: f64, rho_c: f64, len: f64, area: f64) -> f64 {
358    let lv = GAMMA * (mu_c / PRANDTL + mu_t_c / PRANDTL_T) / rho_c;
359    2.0 * lv * len * len / area
360}

362/// Diffusive spectral radius `lam = gamma (mu_f / PRANDTL + mu_t,f / PRANDTL_T) / (rho_f |d|)` of the faces `(lower, upper)`
363/// of the cell `(i, j)` in the line direction (`0.0` for a face inside the solid block): the implicit operator adds `2 lam I`
364/// to the dissipation matrix of the face.
365pub(crate) fn face_lambdas(grid: &Grid, w: &[Prim], walls: &Walls, mu: &[f64], mu_t: &[f64], dir_i: bool, i: usize, j: usize) -> (f64, f64) {
366    let lam = |sides: Option<FaceSides>, (xf, rf): (f64, f64)| match sides {
367        Some(s) => {
368            let (dd, _, _) = stencil(grid, &s, xf, rf);
369            let (mu_f, mut_f) = face_mu(mu, mu_t, &s);
370            let rho_f = 0.5 * (s.l.w[0] + s.r.w[0]);
371            GAMMA * (mu_f / PRANDTL + mut_f / PRANDTL_T) / (rho_f * dd)
372        }
373        None => 0.0,
374    };
375    if dir_i {
376        (lam(iface_sides(grid, w, walls, i, j), iface_centre(grid, i, j)), lam(iface_sides(grid, w, walls, i + 1, j), iface_centre(grid, i + 1, j)))
377    } else {
378        (lam(jface_sides(grid, w, walls, i, j), jface_centre(grid, i, j)), lam(jface_sides(grid, w, walls, i, j + 1), jface_centre(grid, i, j + 1)))
379    }
380}

409    /// Skin friction coefficient `cf = mu_w u_t / d_n / (0.5 M^2)` of the wall face (j = 0, i >= i0) nearest to `x_target`.
410    pub(crate) fn cf_at(s: &Solver, v: &Viscous, mach: f64, x_target: f64) -> (f64, f64) {
411        let g = &s.grid;
412        let (i, _) =
413            (v.wall_i_start..g.ni).map(|i| (i, (0.5 * (g.node(i, 0).0 + g.node(i + 1, 0).0) - x_target).abs())).min_by(|a, b| a.1.total_cmp(&b.1)).unwrap();
414        let f = &g.jface[i];
415        let (xf, rf) = (0.5 * (g.node(i, 0).0 + g.node(i + 1, 0).0), 0.5 * (g.node(i, 0).1 + g.node(i + 1, 0).1));
416        let w = s.prim(i, 0);
417        let (tx, tr) = (f.nr, -f.nx);
418        let ut = w[1] * tx + w[2] * tr;
419        let dn = ((g.xc[i] - xf) * f.nx + (g.rc[i] - rf) * f.nr).abs();
420        (xf, v.mu(temperature(&w)) * ut / dn / (0.5 * mach * mach))
421    }

423    /// G1: laminar flat plate (Blasius) in the annulus formulation, `Re_x = re_c M x`.
424    ///
425    /// The domain height is 0.6: with 0.3 the confinement of the displacement body raises the edge velocity to 1.004-1.007 U
426    /// and the measured `cf / Blasius - 1` is +1.4 % (x = 0.50) / +3.2 % (x = 0.89), independent of the wall-normal and
427    /// streamwise resolution (n_plate 120, nj 72, dr_wall 1e-4: +1.5 % / +3.5 %); with 0.6 it is +0.64 % / +1.63 %, with 1.2 it
428    /// is +0.26 % / +0.76 %. The density residual plateaus about 4.4 orders below its peak (limiter chatter) while `cf` is
429    /// stationary to three digits, so the run ends when both checked `cf` values changed by less than 1e-5 in 250 iterations
430    /// (about 1000 iterations, 4 s in release; the 6000-iteration limit took about 25 s).
431    #[test]
432    fn g1_laminar_flat_plate_matches_blasius() {
433        let (mach, re_c) = (0.2, 1e6);
434        let grid = plate_grid(&PlateGridParams { r0: 20.0, x_up: 0.5, length: 1.0, height: 0.6, n_up: 20, n_plate: 80, nj: 64, dx_le: 0.005, dr_wall: 2e-4 });
435        let cfg = Config {
436            cfl: 0.5,
437            limiter: Limiter::VanAlbada,
438            freestream: freestream(mach),
439            bc: [Bc::Characteristic, Bc::PressureOutflow, Bc::SlipWall, Bc::Characteristic],
440            first_order_iters: 50,
441        };
442        let mut s = Solver::new(grid, cfg);
443        s.low_mach = true;
444        s.set_scheme(Scheme::Implicit(ImplicitParams::VISCOUS));
445        let v = Viscous { re_c, sutherland: 0.368, turbulence: Turbulence::Laminar, wall_i_start: 20 };
446        s.set_viscous(Some(v));
447        let (mut peak, mut last, mut iters) = (0.0f64, f64::INFINITY, 0);
448        let mut cf_prev = [f64::NAN; 2];
449        for it in 0..6000 {
450            let r = s.step();
451            iters = it + 1;
452            last = r;
453            if it > 20 {
454                peak = peak.max(r);
455                if peak / r >= 1e5 {
456                    break;
457                }
458            }
459            // the residual plateaus; the quantity under test is stationary once both checked cf values change by < 1e-5 in 250 iterations
460            if it % 250 == 249 {
461                let cf = [cf_at(&s, &v, mach, 0.5).1, cf_at(&s, &v, mach, 0.9).1];
462                let settled = cf.iter().zip(&cf_prev).all(|(c, p)| (c / p - 1.0).abs() < 1e-5);
463                cf_prev = cf;
464                if settled {
465                    break;
466                }
467            }
468        }
469        println!("G1: {iters} iterations, density residual {last:.3e} (peak {peak:.3e}, drop {:.2} orders)", (peak / last).log10());
470        let mut worst = 0.0f64;
471        for x_target in [0.5, 0.9] {
472            let (x, cf) = cf_at(&s, &v, mach, x_target);
473            let blasius = 0.664 / (re_c * mach * x).sqrt();
474            let err = (cf / blasius - 1.0).abs();
475            println!("G1: x = {x:.4}  cf = {cf:.6e}  Blasius = {blasius:.6e}  error = {:+.2} %", 100.0 * (cf / blasius - 1.0));
476            worst = worst.max(err);
477        }
478        assert!(worst < 0.03, "G1: skin friction off by {:.2} % from Blasius", 100.0 * worst);
479    }
