1//! Viscous terms of the axisymmetric compressible Navier-Stokes equations (laminar; the eddy-viscosity arrays are carried
2//! for the turbulence model but stay zero until it is attached).
3//!
4//! Units: `rho_inf = 1`, `c_inf = 1`, `p_inf = 1/gamma`, `T = gamma p / rho` (`T_inf = 1`), lengths in nose diameters.
5//!
6//! ```text
7//!   mu(T)  = (1 / re_c) T^1.5 (1 + S) / (T + S)            S = sutherland = 110.4 K / T_inf[K],  re_c = rho_inf c_inf d / mu_inf
8//!   kappa  = (mu / PRANDTL + mu_t / PRANDTL_T) / (gamma - 1)         q = -kappa grad T        (c_p = 1 / (gamma - 1))
9//! ```
10//!
11//! The residual of a cell uses the total face flux `F - Fv` in place of the Euler flux `F` and the radial-momentum source
12//! `(p - t_tt) A / V` in place of `p A / V` (`t_tt` = hoop stress):
13//!
14//! ```text
15//!   V dU/dt = - sum_faces w_f (F - Fv).n + (0, 0, (p - t_tt) A, 0),            w_f = len_f r_m
16//!   Fv.n    = [0, txx nx + txr nr, txr nx + trr nr, (u txx + v txr + kappa T_x) nx + (u txr + v trr + kappa T_r) nr]
17//!   div     = u_x + v_r + v / r;   txx = mue (2 u_x - 2/3 div);  trr = mue (2 v_r - 2/3 div);  txr = mue (u_r + v_x);  mue = mu + mu_t
18//!   t_tt    = (mu_c + mu_t,c) (2 v_c / r_c - 2/3 div_c)
19//! ```
20//!
21//! Three data-parallel passes, called from `Solver::eval_residual` after the inviscid face fluxes:
22//!
23//! 1. `cell_pass`: `mu(T)` and the Green-Gauss gradient `(u_x, u_r, v_x, v_r, T_x, T_r)` of every fluid cell in the meridian
24//!    plane, `g = (1/A) sum_faces phi_f n_f len_f` with `phi_f = (phi_c + phi_neighbour) / 2`;
25//! 2. `face_pass`: the face gradient `g_f = gbar + ((phi_R - phi_L) / |d| - gbar.e) e` (`d` = distance vector of the two
26//!    adjacent cell centres, `e = d / |d|`, `gbar` the mean of the two cell gradients: a compact normal derivative with the
27//!    averaged tangential part) and the viscous flux of every face into `fvi`, `fvj`; faces on the axis (`r_m = 0`) and inside
28//!    the solid block carry no flux;
29//! 3. the cell residual in `solver.rs` (with `hoop_stress` and the diffusive time-step limit `diffusion_rate`).
30//!
31//! A state across a boundary is the ghost cell of `Solver::w`; the solid cell across a wall of a body block is its ghost as
32//! well (block ghost in the j direction, [`wall_ghost`] across the base wall). A ghost sits at the mirror image of the centre
33//! of its cell in the face centre and carries the gradient, viscosity and eddy viscosity of that cell (the eddy viscosity
34//! `2 mu_t,w - mu_t` across a no-slip face, `-mu_t` for a smooth wall, so it is the wall value `mu_t,w` there: 0 when smooth).
35//!
36//! No-slip walls (`Walls`): the face where the viscous flow starts (`Viscous::wall_i_start`) and the body surface, base wall
37//! and sting surface of a block. The wall ghost of a no-slip face is `(rho, -u, -v, p)` (adiabatic wall, zero normal pressure
38//! gradient), that of a slip face the reflection about the face normal.
39

45impl Viscous {
46    /// Dynamic viscosity (Sutherland's law) at the static temperature `t` (free-stream temperature = 1).
47    #[inline]
48    pub fn mu(&self, t: f64) -> f64 {
49        t.powf(1.5) * (1.0 + self.sutherland) / (t + self.sutherland) / self.re_c
50    }
51
52    /// Does the run use a rough wall (SST with `roughness > 0`)? Then `Solver::rough_wall` holds the Aupoix wall state of the wall cells.
53    #[inline]
54    pub fn rough(&self) -> bool {
55        self.turbulence == Turbulence::Sst && self.roughness > 0.0
56    }
57}

59/// Static temperature `gamma p / rho` of a primitive state.
60#[inline]
61pub fn temperature(w: &Prim) -> f64 {
62    GAMMA * w[3] / w[0]
63}

65/// Ghost state across a wall face with unit normal `(nx, nr)`: `(rho, -u, -v, p)` on a no-slip face, else the reflection.
66#[inline]
67pub fn wall_ghost(w: &Prim, f: &Face, noslip: bool) -> Prim {
68    if noslip {
69        [w[0], -w[1], -w[2], w[3]]
70    } else {
71        reflect(w, f.nx, f.nr)
72    }
73}

75/// Which wall faces are no-slip (only with viscous terms on). The single classification used by the ghost fills, the flux
76/// kernels and the implicit operator. A nozzle ([`Nozzle`]) turns the first base rows (`j < j_end`) into a fixed inflow
77/// state: those faces are never no-slip walls ([`Walls::jet_iface`]).
78///
79/// * Without a block: a `jmin` face with a slip-wall boundary condition and `i >= wall_i_start`.
80/// * With a block: the body-surface faces (`j = j_end`, `wall_i_start <= i < i_end`), every base-wall face (`i = i_end`,
81///   `j_sting <= j < j_end`) and every sting-surface face (`j = j_sting > 0`, `i >= i_end`); the axis stays a slip wall.
82#[derive(Clone, Copy, Debug)]
83pub(crate) struct Walls {
84    pub visc: Option<Viscous>,
85    pub jmin_slip: bool,
86    pub blk: Option<Block>,
87    pub nozzle: Option<Nozzle>,
88}
89
90impl Walls {
91    pub(crate) fn new(visc: Option<Viscous>, bc: &[Bc; 4], blk: Option<Block>, nozzle: Option<Nozzle>) -> Walls {
92        Walls { visc, jmin_slip: bc[2] == Bc::SlipWall, blk, nozzle }
93    }
94
95    /// Is the j-face `(i, j)` (between cells `j - 1` and `j` of column `i`) a no-slip wall?
96    #[inline]
97    pub(crate) fn noslip_jface(&self, i: usize, j: usize) -> bool {
98        let Some(v) = self.visc else { return false };
99        match self.blk {
100            None => j == 0 && self.jmin_slip && i >= v.wall_i_start,
101            Some(b) => (j == b.j_end && i >= v.wall_i_start && i < b.i_end) || (j == b.j_sting && b.j_sting > 0 && i >= b.i_end),
102        }
103    }
104
105    /// Is the i-face `(i, j)` (between cells `i - 1` and `i` of row `j`) the no-slip base wall?
106    #[inline]
107    pub(crate) fn noslip_iface(&self, i: usize, j: usize) -> bool {
108        self.visc.is_some() && self.blk.is_some_and(|b| i == b.i_end && j >= b.j_sting && j < b.j_end) && !self.nozzle.is_some_and(|n| j < n.j_end)
109    }
110
111    /// Is the j-face `(i, j)` a rough no-slip wall (`Viscous::roughness`)? The body surface (the domain `jmin` wall without a block): not the
112    /// base wall and not the sting, whose coarse cells are no place for a roughness wall state.
113    #[inline]
114    pub(crate) fn rough_jface(&self, i: usize, j: usize) -> bool {
115        let Some(v) = self.visc else { return false };
116        v.rough()
117            && match self.blk {
118                None => j == 0 && self.jmin_slip && i >= v.wall_i_start,
119                Some(b) => j == b.j_end && i >= v.wall_i_start && i < b.i_end,
120            }
121    }
122
123    /// Is the i-face `(i, j)` a nozzle exit face (fixed inflow state)?
124    #[inline]
125    pub(crate) fn jet_iface(&self, i: usize, j: usize) -> bool {
126        self.blk.is_some_and(|b| i == b.i_end) && self.nozzle.is_some_and(|n| j < n.j_end)
127    }
128}

149/// Sides of i-face `(i, j)`; `None` inside the solid block. `w` is the primitive array with ghost layers.
150pub(crate) fn iface_sides(grid: &Grid, w: &[Prim], walls: &Walls, i: usize, j: usize) -> Option<FaceSides> {
151    let ni = grid.ni;
152    let (ii, jj) = (i as isize, j as isize);
153    if let Some(b) = walls.blk {
154        if j < b.j_end {
155            if j < b.j_sting || i < b.i_end {
156                return None;
157            }
158            if i == b.i_end {
159                let c = j * ni + i;
160                let wr = w[pidx(ni, ii, jj)];
161                if let Some(n) = walls.nozzle.filter(|n| j < n.j_end) {
162                    let fixed = walls.visc.map(|v| {
163                        let mu_t = if v.turbulence == Turbulence::Sst { n.state[0] * n.turb[0] / n.turb[1] } else { 0.0 };
164                        (v.mu(temperature(&n.state)), mu_t)
165                    });
166                    return Some(FaceSides {
167                        l: Side { w: n.state, cell: c, ghost: true, fixed },
168                        r: Side { w: wr, cell: c, ghost: false, fixed: None },
169                        noslip: false,
170                    });
171                }
172                let noslip = walls.noslip_iface(i, j);
173                let wl = wall_ghost(&wr, &grid.iface[j * (ni + 1) + i], noslip);
174                return Some(FaceSides { l: Side { w: wl, cell: c, ghost: true, fixed: None }, r: Side { w: wr, cell: c, ghost: false, fixed: None }, noslip });
175            }
176        }
177    }
178    let side = |w: Prim, cell: usize, ghost: bool| Side { w, cell, ghost, fixed: None };
179    Some(if i == 0 {
180        let c = j * ni;
181        FaceSides { l: side(w[pidx(ni, -1, jj)], c, true), r: side(w[pidx(ni, 0, jj)], c, false), noslip: false }
182    } else if i == ni {
183        let c = j * ni + ni - 1;
184        FaceSides { l: side(w[pidx(ni, ii - 1, jj)], c, false), r: side(w[pidx(ni, ii, jj)], c, true), noslip: false }
185    } else {
186        FaceSides { l: side(w[pidx(ni, ii - 1, jj)], j * ni + i - 1, false), r: side(w[pidx(ni, ii, jj)], j * ni + i, false), noslip: false }
187    })
188}

190/// Sides of j-face `(i, j)`; `None` inside the solid block.
191pub(crate) fn jface_sides(grid: &Grid, w: &[Prim], walls: &Walls, i: usize, j: usize) -> Option<FaceSides> {
192    let (ni, nj) = (grid.ni, grid.nj);
193    let (ii, jj) = (i as isize, j as isize);
194    let side = |w: Prim, cell: usize, ghost: bool| Side { w, cell, ghost, fixed: None };
195    if let Some(b) = walls.blk {
196        let j0 = b.col_start(i);
197        if j < j0 {
198            return None;
199        }
200        if j == j0 && j0 > 0 {
201            // body or sting surface: the solid cell below holds the block ghost
202            let c = j * ni + i;
203            return Some(FaceSides { l: side(w[pidx(ni, ii, jj - 1)], c, true), r: side(w[pidx(ni, ii, jj)], c, false), noslip: walls.noslip_jface(i, j) });
204        }
205    }
206    Some(if j == 0 {
207        let c = i;
208        FaceSides { l: side(w[pidx(ni, ii, -1)], c, true), r: side(w[pidx(ni, ii, 0)], c, false), noslip: walls.noslip_jface(i, 0) }
209    } else if j == nj {
210        let c = (nj - 1) * ni + i;
211        FaceSides { l: side(w[pidx(ni, ii, jj - 1)], c, false), r: side(w[pidx(ni, ii, jj)], c, true), noslip: false }
212    } else {
213        FaceSides { l: side(w[pidx(ni, ii, jj - 1)], (j - 1) * ni + i, false), r: side(w[pidx(ni, ii, jj)], j * ni + i, false), noslip: false }
214    })
215}

217/// Centre of i-face `(i, j)` (mean of its end nodes).
218#[inline]
219pub(crate) fn iface_centre(grid: &Grid, i: usize, j: usize) -> (f64, f64) {
220    let (a, b) = (grid.node(i, j), grid.node(i, j + 1));
221    (0.5 * (a.0 + b.0), 0.5 * (a.1 + b.1))
222}

224/// Centre of j-face `(i, j)`.
225#[inline]
226pub(crate) fn jface_centre(grid: &Grid, i: usize, j: usize) -> (f64, f64) {
227    let (a, b) = (grid.node(i, j), grid.node(i + 1, j));
228    (0.5 * (a.0 + b.0), 0.5 * (a.1 + b.1))

231/// Position of a side: the centre of its cell, or the mirror image of that centre in the face centre for a ghost.
232#[inline]
233fn side_pos(grid: &Grid, s: &Side, xf: f64, rf: f64) -> (f64, f64) {
234    let (x, r) = (grid.xc[s.cell], grid.rc[s.cell]);
235    if s.ghost {
236        (2.0 * xf - x, 2.0 * rf - r)
237    } else {
238        (x, r)
239    }
240}

242/// `(|d|, e_x, e_r)`: distance and direction between the two sides.
243#[inline]
244pub(crate) fn stencil(grid: &Grid, sides: &FaceSides, xf: f64, rf: f64) -> (f64, f64, f64) {
245    let (xl, rl) = side_pos(grid, &sides.l, xf, rf);
246    let (xr, rr) = side_pos(grid, &sides.r, xf, rf);
247    let (dx, dr) = (xr - xl, rr - rl);
248    let dd = dx.hypot(dr);
249    (dd, dx / dd, dr / dd)
250}

252/// Face viscosity and eddy viscosity: mean of the two sides, a ghost taking its cell's values (eddy viscosity `2 mu_t,w - mu_t` across a
253/// no-slip face, `-mu_t` for a smooth wall (`rough` empty): the face value is the wall eddy viscosity `mu_t,w = rough[cell][2]`; a jet ghost
254/// has the fixed jet values).
255#[inline]
256pub(crate) fn face_mu(mu: &[f64], mu_t: &[f64], rough: &[[f64; 3]], sides: &FaceSides) -> (f64, f64) {
257    let mu_side = |s: &Side| s.fixed.map_or(mu[s.cell], |f| f.0);
258    let mut_side = |s: &Side| match s.fixed {
259        Some(f) => f.1,
260        None if s.ghost && sides.noslip => {
261            if rough.is_empty() {
262                -mu_t[s.cell]
263            } else {
264                2.0 * rough[s.cell][2] - mu_t[s.cell]
265            }
266        }
267        None => mu_t[s.cell],
268    };
269    (0.5 * (mu_side(&sides.l) + mu_side(&sides.r)), 0.5 * (mut_side(&sides.l) + mut_side(&sides.r)))
270}

272/// `(u, v, T)` of a primitive state.
273#[inline]
274fn phi(w: &Prim) -> [f64; 3] {
275    [w[1], w[2], temperature(w)]
276}

278/// Pass 1: viscosity and meridian-plane Green-Gauss gradient `(u_x, u_r, v_x, v_r, T_x, T_r)` of every fluid cell (zero in
279/// solid cells).
280pub(crate) fn cell_pass(v: &Viscous, grid: &Grid, w: &[Prim], walls: &Walls, grad: &mut [[f64; 6]], mu: &mut [f64]) {
281    let ni = grid.ni;
282    grad.par_chunks_mut(ni).zip(mu.par_chunks_mut(ni)).enumerate().for_each(|(j, (grow, mrow))| {
283        for i in 0..ni {
284            if walls.blk.is_some_and(|b| b.solid(i, j)) {
285                grow[i] = [0.0; 6];
286                mrow[i] = 0.0;
287                continue;
288            }
289            let wc = &w[pidx(ni, i as isize, j as isize)];
290            mrow[i] = v.mu(temperature(wc));
291            let pc = phi(wc);
292            let fl = &grid.iface[j * (ni + 1) + i];
293            let fr = &grid.iface[j * (ni + 1) + i + 1];
294            let fb = &grid.jface[j * ni + i];
295            let ft = &grid.jface[(j + 1) * ni + i];
296            let nb_l = iface_sides(grid, w, walls, i, j).expect("fluid cell face").l.w;
297            let nb_r = iface_sides(grid, w, walls, i + 1, j).expect("fluid cell face").r.w;
298            let nb_b = jface_sides(grid, w, walls, i, j).expect("fluid cell face").l.w;
299            let nb_t = jface_sides(grid, w, walls, i, j + 1).expect("fluid cell face").r.w;
300            let mut g = [0.0; 6];
301            for (f, s, nb) in [(fr, 1.0, &nb_r), (fl, -1.0, &nb_l), (ft, 1.0, &nb_t), (fb, -1.0, &nb_b)] {
302                let pn = phi(nb);
303                for m in 0..3 {
304                    let pf = 0.5 * (pc[m] + pn[m]);
305                    g[2 * m] += s * pf * f.nx * f.len;
306                    g[2 * m + 1] += s * pf * f.nr * f.len;
307                }
308            }
309            let inv_a = 1.0 / grid.area[j * ni + i];
310            for x in g.iter_mut() {
311                *x *= inv_a;
312            }
313            grow[i] = g;
314        }
315    });
316}

318/// Viscous flux `Fv.n` of one face (see the module documentation); `f` is the face, `(xf, rf)` its centre.
319fn viscous_flux(grid: &Grid, grad: &[[f64; 6]], mu: &[f64], mu_t: &[f64], rough: &[[f64; 3]], sides: &FaceSides, f: &Face, xf: f64, rf: f64) -> Cons {
320    if f.rm == 0.0 {
321        return [0.0; 4];
322    }
323    let (dd, ex, er) = stencil(grid, sides, xf, rf);
324    let (gl, gr) = (&grad[sides.l.cell], &grad[sides.r.cell]);
325    let (pl, pr) = (phi(&sides.l.w), phi(&sides.r.w));
326    let mut gf = [0.0; 6];
327    for m in 0..3 {
328        let (gbx, gbr) = (0.5 * (gl[2 * m] + gr[2 * m]), 0.5 * (gl[2 * m + 1] + gr[2 * m + 1]));
329        let corr = (pr[m] - pl[m]) / dd - (gbx * ex + gbr * er);
330        gf[2 * m] = gbx + corr * ex;
331        gf[2 * m + 1] = gbr + corr * er;
332    }
333    let (mu_f, mut_f) = face_mu(mu, mu_t, rough, sides);
334    let (uf, vf) = (0.5 * (sides.l.w[1] + sides.r.w[1]), 0.5 * (sides.l.w[2] + sides.r.w[2]));
335    let (u_x, u_r, v_x, v_r, t_x, t_r) = (gf[0], gf[1], gf[2], gf[3], gf[4], gf[5]);
336    let div = u_x + v_r + vf / f.rm;
337    let mue = mu_f + mut_f;
338    let txx = mue * (2.0 * u_x - 2.0 / 3.0 * div);
339    let trr = mue * (2.0 * v_r - 2.0 / 3.0 * div);
340    let txr = mue * (u_r + v_x);
341    let kappa = (mu_f / PRANDTL + mut_f / PRANDTL_T) / (GAMMA - 1.0);
342    [0.0, txx * f.nx + txr * f.nr, txr * f.nx + trr * f.nr, (uf * txx + vf * txr + kappa * t_x) * f.nx + (uf * txr + vf * trr + kappa * t_r) * f.nr]
343}

345/// Pass 2: viscous fluxes of every i-face (`fvi`, `(ni + 1) * nj`) and j-face (`fvj`, `ni * (nj + 1)`).
346#[allow(clippy::too_many_arguments)]
347pub(crate) fn face_pass(
348    grid: &Grid,
349    w: &[Prim],
350    walls: &Walls,
351    grad: &[[f64; 6]],
352    mu: &[f64],
353    mu_t: &[f64],
354    rough: &[[f64; 3]],
355    fvi: &mut [Cons],
356    fvj: &mut [Cons],
357) {
358    let ni = grid.ni;
359    fvi.par_chunks_mut(ni + 1).enumerate().for_each(|(j, row)| {
360        for (i, out) in row.iter_mut().enumerate() {
361            *out = match iface_sides(grid, w, walls, i, j) {
362                Some(s) => {
363                    let (xf, rf) = iface_centre(grid, i, j);
364                    viscous_flux(grid, grad, mu, mu_t, rough, &s, &grid.iface[j * (ni + 1) + i], xf, rf)
365                }
366                None => [0.0; 4],
367            };
368        }
369    });
370    fvj.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
371        for (i, out) in row.iter_mut().enumerate() {
372            *out = match jface_sides(grid, w, walls, i, j) {
373                Some(s) => {
374                    let (xf, rf) = jface_centre(grid, i, j);
375                    viscous_flux(grid, grad, mu, mu_t, rough, &s, &grid.jface[j * ni + i], xf, rf)
376                }
377                None => [0.0; 4],
378            };
379        }
380    });
381}

383/// Rough-wall state `[k_w, omega_w, mu_t,w]` of every fluid cell that has a no-slip face (the cell above / behind it), from the primitives `w`, the
384/// viscosity `mu` and the wall distance `dwall`; zero elsewhere. Cells of a rough wall (`Walls::rough_jface`) take `turbulence::aupoix_wall` with
385/// the roughness `ks` phased in over the first wall cells (`turbulence::rough_ramp`) and `mu_t,w = rho nu_t,w`; the other no-slip cells (base wall,
386/// sting) the smooth state `[0, 60 nu / (beta1 d^2), 0]`.
387pub(crate) fn rough_pass(grid: &Grid, w: &[Prim], walls: &Walls, dwall: &[f64], mu: &[f64], ks: f64, out: &mut [[f64; 3]]) {
388    let ni = grid.ni;
389    let wall_i_start = walls.visc.map_or(0, |v| v.wall_i_start);
390    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
391        for (i, o) in row.iter_mut().enumerate() {
392            *o = [0.0; 3];
393            let rough = walls.rough_jface(i, j);
394            if walls.blk.is_some_and(|b| b.solid(i, j)) || !(rough || walls.noslip_jface(i, j) || walls.noslip_iface(i, j)) {
395                continue;
396            }
397            let c = j * ni + i;
398            let wc = &w[pidx(ni, i as isize, j as isize)];
399            let nu = mu[c] / wc[0];
400            *o = if rough {
401                let ks_eff = ks * crate::turbulence::rough_ramp(i, wall_i_start);
402                let [k_w, om_w, nu_t] = crate::turbulence::aupoix_wall(nu, wc[1].hypot(wc[2]), dwall[c], ks_eff);
403                [k_w, om_w, wc[0] * nu_t]
404            } else {
405                [0.0, 60.0 * nu / (crate::turbulence::BETA1 * dwall[c] * dwall[c]), 0.0]
406            };
407        }
408    });
409}

411/// Hoop stress `t_tt = (mu_c + mu_t,c)(2 v_c / r_c - 2/3 div_c)`, `div_c = u_x + v_r + v_c / r_c`, from the cell gradient `g`.
412#[inline]
413pub(crate) fn hoop_stress(mue: f64, g: &[f64; 6], v_c: f64, r_c: f64) -> f64 {
414    let div = g[0] + g[3] + v_c / r_c;
415    mue * (2.0 * v_c / r_c - 2.0 / 3.0 * div)
416}
417
418/// Diffusive contribution `2 lv len^2 / A` to the spectral radius `L_d` of one direction (`len`: mean length of the two faces of
419/// that direction), with `lv = gamma (mu / PRANDTL + mu_t / PRANDTL_T) / rho`.
420#[inline]
421pub(crate) fn diffusion_rate(mu_c: f64, mu_t_c: f64, rho_c: f64, len: f64, area: f64) -> f64 {
422    let lv = GAMMA * (mu_c / PRANDTL + mu_t_c / PRANDTL_T) / rho_c;
423    2.0 * lv * len * len / area
424}
425
426/// Diffusive spectral radius `lam = gamma (mu_f / PRANDTL + mu_t,f / PRANDTL_T) / (rho_f |d|)` of the faces `(lower, upper)`
427/// of the cell `(i, j)` in the line direction (`0.0` for a face inside the solid block): the implicit operator adds `2 lam I`
428/// to the dissipation matrix of the face.
429pub(crate) fn face_lambdas(
430    grid: &Grid,
431    w: &[Prim],
432    walls: &Walls,
433    mu: &[f64],
434    mu_t: &[f64],
435    rough: &[[f64; 3]],
436    dir_i: bool,
437    i: usize,
438    j: usize,
439) -> (f64, f64) {
440    let lam = |sides: Option<FaceSides>, (xf, rf): (f64, f64)| match sides {
441        Some(s) => {
442            let (dd, _, _) = stencil(grid, &s, xf, rf);
443            let (mu_f, mut_f) = face_mu(mu, mu_t, rough, &s);
444            let rho_f = 0.5 * (s.l.w[0] + s.r.w[0]);
445            GAMMA * (mu_f / PRANDTL + mut_f / PRANDTL_T) / (rho_f * dd)
446        }
447        None => 0.0,
448    };
449    if dir_i {
450        (lam(iface_sides(grid, w, walls, i, j), iface_centre(grid, i, j)), lam(iface_sides(grid, w, walls, i + 1, j), iface_centre(grid, i + 1, j)))
451    } else {
452        (lam(jface_sides(grid, w, walls, i, j), jface_centre(grid, i, j)), lam(jface_sides(grid, w, walls, i, j + 1), jface_centre(grid, i, j + 1)))
453    }
454}

483    /// Skin friction coefficient `cf = mu_w u_t / d_n / (0.5 M^2)` of the wall face (j = 0, i >= i0) nearest to `x_target`.
484    pub(crate) fn cf_at(s: &Solver, v: &Viscous, mach: f64, x_target: f64) -> (f64, f64) {
485        let g = &s.grid;
486        let (i, _) =
487            (v.wall_i_start..g.ni).map(|i| (i, (0.5 * (g.node(i, 0).0 + g.node(i + 1, 0).0) - x_target).abs())).min_by(|a, b| a.1.total_cmp(&b.1)).unwrap();
488        let f = &g.jface[i];
489        let (xf, rf) = (0.5 * (g.node(i, 0).0 + g.node(i + 1, 0).0), 0.5 * (g.node(i, 0).1 + g.node(i + 1, 0).1));
490        let w = s.prim(i, 0);
491        let (tx, tr) = (f.nr, -f.nx);
492        let ut = w[1] * tx + w[2] * tr;
493        let dn = ((g.xc[i] - xf) * f.nx + (g.rc[i] - rf) * f.nr).abs();
494        let mu = v.mu(temperature(&w));
495        // rough wall: the wall shear is (mu + mu_t,w) u_t / d_n with the Aupoix wall eddy viscosity of the cell state
496        let mu_tw = if v.rough() { w[0] * crate::turbulence::aupoix_wall(mu / w[0], w[1].hypot(w[2]), dn, v.roughness)[2] } else { 0.0 };
497        (xf, (mu + mu_tw) * ut / dn / (0.5 * mach * mach))
498    }

500    /// G1: laminar flat plate (Blasius) in the annulus formulation, `Re_x = re_c M x`.
501    ///
502    /// The domain height is 0.6: with 0.3 the confinement of the displacement body raises the edge velocity to 1.004-1.007 U
503    /// and the measured `cf / Blasius - 1` is +1.4 % (x = 0.50) / +3.2 % (x = 0.89), independent of the wall-normal and
504    /// streamwise resolution (n_plate 120, nj 72, dr_wall 1e-4: +1.5 % / +3.5 %); with 0.6 it is +0.64 % / +1.63 %, with 1.2 it
505    /// is +0.26 % / +0.76 %. The density residual plateaus about 4.4 orders below its peak (limiter chatter) while `cf` is
506    /// stationary to three digits, so the run ends when both checked `cf` values changed by less than 1e-5 in 250 iterations
507    /// (about 1000 iterations, 4 s in release; the 6000-iteration limit took about 25 s).
508    #[test]
509    fn g1_laminar_flat_plate_matches_blasius() {
510        let (mach, re_c) = (0.2, 1e6);
511        let grid = plate_grid(&PlateGridParams { r0: 20.0, x_up: 0.5, length: 1.0, height: 0.6, n_up: 20, n_plate: 80, nj: 64, dx_le: 0.005, dr_wall: 2e-4 });
512        let cfg = Config {
513            cfl: 0.5,
514            limiter: Limiter::VanAlbada,
515            freestream: freestream(mach),
516            bc: [Bc::Characteristic, Bc::PressureOutflow, Bc::SlipWall, Bc::Characteristic],
517            first_order_iters: 50,
518        };
519        let mut s = Solver::new(grid, cfg);
520        s.low_mach = true;
521        s.set_scheme(Scheme::Implicit(ImplicitParams::VISCOUS));
522        let v = Viscous { re_c, sutherland: 0.368, turbulence: Turbulence::Laminar, wall_i_start: 20, roughness: 0.0 };
523        s.set_viscous(Some(v));
524        let (mut peak, mut last, mut iters) = (0.0f64, f64::INFINITY, 0);
525        let mut cf_prev = [f64::NAN; 2];
526        for it in 0..6000 {
527            let r = s.step();
528            iters = it + 1;
529            last = r;
530            if it > 20 {
531                peak = peak.max(r);
532                if peak / r >= 1e5 {
533                    break;
534                }
535            }
536            // the residual plateaus; the quantity under test is stationary once both checked cf values change by < 1e-5 in 250 iterations
537            if it % 250 == 249 {
538                let cf = [cf_at(&s, &v, mach, 0.5).1, cf_at(&s, &v, mach, 0.9).1];
539                let settled = cf.iter().zip(&cf_prev).all(|(c, p)| (c / p - 1.0).abs() < 1e-5);
540                cf_prev = cf;
541                if settled {
542                    break;
543                }
544            }
545        }
546        println!("G1: {iters} iterations, density residual {last:.3e} (peak {peak:.3e}, drop {:.2} orders)", (peak / last).log10());
547        let mut worst = 0.0f64;
548        for x_target in [0.5, 0.9] {
549            let (x, cf) = cf_at(&s, &v, mach, x_target);
550            let blasius = 0.664 / (re_c * mach * x).sqrt();
551            let err = (cf / blasius - 1.0).abs();
552            println!("G1: x = {x:.4}  cf = {cf:.6e}  Blasius = {blasius:.6e}  error = {:+.2} %", 100.0 * (cf / blasius - 1.0));
553            worst = worst.max(err);
554        }
555        assert!(worst < 0.03, "G1: skin friction off by {:.2} % from Blasius", 100.0 * worst);
556    }
