1// The explicit solver in array ops. This file is `include!`d twice by `gpu/mod.rs`, once per precision, into a module that
2// defines `S` (the element type: `f32` or `Df32`), `c(f64) -> S` (a constant rounded from f64), `up(..)` (host f64 data to
3// an `Array<S>`), `to_f64(..)` (an `Array<S>` to host f64 data) and the residual bookkeeping of the loops (`records`,
4// `read_history`, `rn_host`).
5//
6// Every function mirrors the CPU routine of the same name in the CPU modules of this crate expression by expression; the array code is
7// the equation, the doc comment states it.

16/// Array of one field (`rho`, `u`, ...) over a block of cells or faces.
17pub type A = Array<S>;
18/// Four fields: primitive `[rho, u, v, p]`, conservative `[rho, rho u, rho v, E]` or a flux. A record of arrays
19/// (`luma::Fields`): `slice`, `pad`, `assign` and `cache` apply to the four fields at once.
20pub type Rec = [A; 4];
21/// The loop state: the conserved record, the density residual of the step that produced it (one element), the CFL of the next
22/// iteration (one element; the ramp of the implicit scheme is loop state, the explicit scheme passes it through) and the turbulence
23/// state of a viscous run: `k`, `omega`, the eddy viscosity (`[ni, nj]` with the SST model, one dummy element otherwise) and the
24/// density residual norm of `k` of the last update (one element).
25pub type State = [A; 10];
26
27fn rec(f: impl FnMut(usize) -> A) -> Rec {
28    std::array::from_fn(f)
29}
30
31/// The conserved record of a loop state.
32fn unpack(s: &State) -> Rec {
33    rec(|k| s[k].clone())
34}

36fn pack(u: Rec, rn: A, cfl: A, t: [A; 4]) -> State {
37    let [u0, u1, u2, u3] = u;
38    let [k, om, mu_t, trn] = t;
39    [u0, u1, u2, u3, rn, cfl, k, om, mu_t, trn]
40}

42/// The loop state of host data: the conserved state `u` (cell-major, `[rho, rho u, rho v, E]`), a zero residual and the CFL `cfl`.
43fn upload_state(ctx: &Context, u: &[[f64; 4]], shape: [usize; 2], cfl: f64) -> Result<State> {
44    let comp = |k: usize| up(ctx, &u.iter().map(|q| q[k]).collect::<Vec<_>>(), shape);
45    let dummy = || up(ctx, &[0.0], 1);
46    Ok([comp(0)?, comp(1)?, comp(2)?, comp(3)?, up(ctx, &[0.0], 1)?, up(ctx, &[cfl], 1)?, dummy()?, dummy()?, dummy()?, dummy()?])
47}

49fn download_state(s: &State) -> Result<Vec<[f64; 4]>> {
50    let v: Vec<Vec<f64>> = (0..4).map(|k| to_f64(&s[k])).collect::<Result<_>>()?;
51    Ok((0..v[0].len()).map(|c| [v[0][c], v[1][c], v[2][c], v[3][c]]).collect())
52}

60/// `rho = q0, u = q1 / rho, v = q2 / rho, p = (g - 1) (E - rho (u^2 + v^2) / 2)`.
61pub fn cons_to_prim(q: Rec) -> Rec {
62    let u = &q[1] / &q[0];
63    let v = &q[2] / &q[0];
64    let p = c(G1) * (&q[3] - c(0.5) * &q[0] * (&u * &u + &v * &v));
65    let [rho, ..] = q;
66    [rho, u, v, p]
67}

69/// `F.n = [rho u_n, rho u u_n + p n_x, rho v u_n + p n_r, (E + p) u_n]` with `u_n = u n_x + v n_r`.
70pub fn physical_flux(w: &Rec, nx: &A, nr: &A) -> Rec {
71    let [rho, u, v, p] = w;
72    let un = u * nx + v * nr;
73    let e = p / c(G1) + c(0.5) * rho * (u * u + v * v);
74    flux_from(w, &un, &e, nx, nr)
75}
76
77/// [`physical_flux`] of a state whose normal velocity `u_n` and total energy `E` are known.
78fn flux_from(w: &Rec, un: &A, e: &A, nx: &A, nr: &A) -> Rec {
79    let [rho, u, v, p] = w;
80    [rho * un, rho * u * un + p * nx, rho * v * un + p * nr, (e + p) * un]
81}

83/// One side of a face with the quantities both Riemann solvers need, built once: normal velocity `u_n = u n_x + v n_r`,
84/// total energy `E = p / (g - 1) + rho (u^2 + v^2) / 2`, sound speed `c = sqrt(g p / rho)` and the flux `F.n`.
85struct Side<'a> {
86    w: &'a Rec,
87    un: A,
88    e: A,
89    c: A,
90    f: Rec,
91}
92
93impl<'a> Side<'a> {
94    fn new(w: &'a Rec, nx: &A, nr: &A) -> Side<'a> {
95        let [rho, u, v, p] = w;
96        let un = u * nx + v * nr;
97        let e = p / c(G1) + c(0.5) * rho * (u * u + v * v);
98        let f = flux_from(w, &un, &e, nx, nr);
99        Side { w, un, e, c: (c(GAMMA) * p / rho).sqrt(), f }
100    }
101}

103/// `w` reflected about a wall of unit normal `(n_x, n_r)`: `v - 2 (v.n) n`.
104fn reflect(w: &Rec, nx: &A, nr: &A) -> Rec {
105    let un = &w[1] * nx + &w[2] * nr;
106    [w[0].clone(), &w[1] - c(2.0) * &un * nx, &w[2] - c(2.0) * &un * nr, w[3].clone()]
107}

113/// Limited slope `phi(a, b)`, `a = q_i - q_{i-1}`, `b = q_{i+1} - q_i`; zero unless `a b > 0`.
114/// minmod: the smaller of `a`, `b` in modulus. van Albada: `(a (b^2 + e) + b (a^2 + e)) / (a^2 + b^2 + 2 e)`, `e = 1e-14`.
115fn limited_slope(a: &A, b: &A, lim: Limiter) -> A {
116    let same_sign = (a * b).gt(c(0.0));
117    match lim {
118        Limiter::FirstOrder => unreachable!("first order reconstruction has no slopes"),
119        Limiter::Minmod => select(&same_sign, select(&a.abs().lt(b.abs()), a, b), c(0.0)),
120        Limiter::VanAlbada => {
121            let eps = 1e-14;
122            let slope = (a * (b * b + c(eps)) + b * (a * a + c(eps))) / (a * a + b * b + c(2.0 * eps));
123            select(&same_sign, slope, c(0.0))
124        }
125        Limiter::Venkatakrishnan => {
126            let eps = 1e-14;
127            let slope = c(2.0) * b * (a * a + a * b + c(eps)) / (c(2.0) * a * a + a * b + b * b + c(2.0 * eps));
128            select(&same_sign, slope, c(0.0))
129        }
130    }
131}

133/// MUSCL reconstruction at the face between the cells `m1 | p0` (`m2`, `p1` the outer neighbours):
134/// `l = m1 + phi(m1 - m2, p0 - m1) / 2`, `r = p0 - phi(p0 - m1, p1 - p0) / 2`;
135/// a face whose reconstructed density or pressure is not positive falls back to the cell value.
136fn reconstruct(m2: &Rec, m1: &Rec, p0: &Rec, p1: &Rec, lim: Limiter) -> (Rec, Rec) {
137    if lim == Limiter::FirstOrder {
138        return (m1.clone(), p0.clone());
139    }
140    let l = rec(|k| &m1[k] + c(0.5) * limited_slope(&(&m1[k] - &m2[k]), &(&p0[k] - &m1[k]), lim));
141    let r = rec(|k| &p0[k] - c(0.5) * limited_slope(&(&p0[k] - &m1[k]), &(&p1[k] - &p0[k]), lim));
142    let ok_l = l[0].gt(c(0.0)) & l[3].gt(c(0.0));
143    let ok_r = r[0].gt(c(0.0)) & r[3].gt(c(0.0));
144    (rec(|k| select(&ok_l, &l[k], &m1[k])), rec(|k| select(&ok_r, &r[k], &p0[k])))
145}

147/// HLLC flux for the 2-D Euler equations projected on the unit normal `(n_x, n_r)` (Toro ch. 10), wave speeds
148/// `S_L = min(u_nL - c_L, u~_n - c~)`, `S_R = max(u_nR + c_R, u~_n + c~)` from Roe averages (`sqrt(rho)` weights):
149/// `F = F_L` for `S_L >= 0`, `F_R` for `S_R <= 0`, else `F_K + S_K (U*_K - U_K)` of the side `K` of the contact speed `S_M`.
150fn hllc_speeds(l: &Side, r: &Side, nx: &A, nr: &A) -> [A; 3] {
151    let [rl, ul, vl, pl] = l.w;
152    let [rr, ur, vr, pr] = r.w;
153    let (unl, unr, cl, cr, el, er) = (&l.un, &r.un, &l.c, &r.c, &l.e, &r.e);
154
155    // Roe averages
156    let sql = rl.sqrt();
157    let sqr = rr.sqrt();
158    let inv = c(1.0) / (&sql + &sqr);
159    let uh = (&sql * ul + &sqr * ur) * &inv;
160    let vh = (&sql * vl + &sqr * vr) * &inv;
161    let hl = (el + pl) / rl;
162    let hr = (er + pr) / rr;
163    let hh = (&sql * &hl + &sqr * &hr) * &inv;
164    let ch2 = c(G1) * (&hh - c(0.5) * (&uh * &uh + &vh * &vh));
165    let ch = ch2.maximum(c(1e-30)).sqrt();
166    let unh = &uh * nx + &vh * nr;
167
168    let sl = (unl - cl).minimum(&unh - &ch);
169    let sr = (unr + cr).maximum(&unh + &ch);
170    let sl_un = &sl - unl;
171    let sr_un = &sr - unr;
172    let sm = (pr - pl + rl * unl * &sl_un - rr * unr * &sr_un) / (rl * &sl_un - rr * &sr_un);
173    [sl, sr, sm]
174}

176/// The HLLC flux from the wave speeds `[S_L, S_R, S_M]` of [`hllc_speeds`].
177fn hllc_flux(l: &Side, r: &Side, nx: &A, nr: &A, speeds: &[A; 3]) -> Rec {
178    let [sl, sr, sm] = speeds;
179    let sl_un = sl - &l.un;
180    let sr_un = sr - &r.un;
181
182    // `F_K + S_K (U*_K - U_K)`, `U*_K = fac [1, u + (S_M - u_n) n_x, v + (S_M - u_n) n_r, E / rho + (S_M - u_n) (S_M + p / (rho (S_K - u_n)))]`
183    let star = |k: &Side, s: &A, s_un: &A| -> Rec {
184        let [r, u, v, p] = k.w;
185        let (f, e) = (&k.f, &k.e);
186        let fac = r * s_un / (s - sm);
187        let sm_un = sm - &k.un;
188        [
189            &f[0] + s * (&fac - r),
190            &f[1] + s * (&fac * (u + &sm_un * nx) - r * u),
191            &f[2] + s * (&fac * (v + &sm_un * nr) - r * v),
192            &f[3] + s * (&fac * (e / r + &sm_un * (sm + p / (r * s_un))) - e),
193        ]
194    };
195    let star_l = star(l, sl, &sl_un);
196    let star_r = star(r, sr, &sr_un);
197
198    let left = sl.ge(c(0.0));
199    let right = sr.le(c(0.0));
200    let contact_left = sm.ge(c(0.0));
201    rec(|k| select(&left, &l.f[k], select(&right, &r.f[k], select(&contact_left, &star_l[k], &star_r[k]))))
202}

204/// Rusanov flux `F = (F_L + F_R) / 2 - s_max (U_R - U_L) / 2`, `s_max = max(|u_nL| + c_L, |u_nR| + c_R)`.
205fn rusanov(l: &Side, r: &Side) -> Rec {
206    let [rl, ul, vl, _] = l.w;
207    let [rr, ur, vr, _] = r.w;
208    let smax = (l.un.abs() + &l.c).maximum(r.un.abs() + &r.c);
209    let dq = [rr - rl, rr * ur - rl * ul, rr * vr - rl * vl, &r.e - &l.e];
210    rec(|k| c(0.5) * (&l.f[k] + &r.f[k]) - c(0.5) * &smax * &dq[k])
211}

213/// Weight of the Rusanov flux: `s = |p_R - p_L| / min(p_L, p_R)`, `t = clamp((s - lo) / (hi - lo), 0, 1)`, `theta = t^2 (3 - 2 t)`.
214fn shock_weight(pl: &A, pr: &A) -> A {
215    let s = (pr - pl).abs() / pl.minimum(pr);
216    let t = ((s - c(SHOCK_SENSOR_LO)) / c(SHOCK_SENSOR_HI - SHOCK_SENSOR_LO)).maximum(c(0.0)).minimum(c(1.0));
217    &t * &t * (c(3.0) - c(2.0) * &t)
218}

220/// Low-Mach scaling of the face-normal velocity jump (Thornber et al. 2008): `z = clamp(max(M_L, M_R), 0.05, 1)`,
221/// `u_n' = u_avg + z (u_n - u_avg)` on both sides, untouched for `z >= 1`.
222fn low_mach_states(l: Rec, r: Rec, nx: &A, nr: &A) -> (Rec, Rec) {
223    let unl = &l[1] * nx + &l[2] * nr;
224    let unr = &r[1] * nx + &r[2] * nr;
225    let ml = (&l[1] * &l[1] + &l[2] * &l[2]).sqrt() / (c(GAMMA) * &l[3] / &l[0]).sqrt();
226    let mr = (&r[1] * &r[1] + &r[2] * &r[2]).sqrt() / (c(GAMMA) * &r[3] / &r[0]).sqrt();
227    let z = ml.maximum(&mr).maximum(c(LOW_MACH_FLOOR)).minimum(c(1.0));
228    let active = z.lt(c(1.0));
229    let avg = c(0.5) * (&unl + &unr);
230    let dl = &avg + &z * (&unl - &avg) - &unl;
231    let dr = &avg + &z * (&unr - &avg) - &unr;
232    let fix = |w: Rec, d: &A| -> Rec {
233        let [rho, u, v, p] = w;
234        [rho, select(&active, &u + d * nx, &u), select(&active, &v + d * nr, &v), p]
235    };
236    (fix(l, &dl), fix(r, &dr))
237}

239/// Complete face kernel: MUSCL + (low-Mach) + `(1 - theta) HLLC + theta Rusanov`.
240///
241/// `split = false` is one expression (what the equations say): Luma fuses it into one kernel of ~115 operations.
242/// `split = true` evaluates the reconstructed states and the shock weight first (`cache`), the flux reads them from memory.
243fn face_flux_lm(m2: &Rec, m1: &Rec, p0: &Rec, p1: &Rec, lim: Limiter, nx: &A, nr: &A, low_mach: bool, split: bool) -> Rec {
244    let (l, r) = reconstruct(m2, m1, p0, p1, lim);
245    let (l, r) = if low_mach { low_mach_states(l, r, nx, nr) } else { (l, r) };
246    let t = shock_weight(&m1[3], &p0[3]);
247    let (l, r, t) = if split { (l, r, t).cache() } else { (l, r, t) };
248    let (sl, sr) = (Side::new(&l, nx, nr), Side::new(&r, nx, nr));
249    let hllc = hllc_flux(&sl, &sr, nx, nr, &hllc_speeds(&sl, &sr, nx, nr));
250    let rus = rusanov(&sl, &sr);
251    rec(|k| (c(1.0) - &t) * &hllc[k] + &t * &rus[k])
252}

258/// Characteristic far field. With the outward normal `n`, `u_n,i` and `c_i` of the interior state `i` and `e` the free stream:
259/// `R_out = u_n,i + 2 c_i / (g - 1)`, `R_in = u_n,e - 2 c_e / (g - 1)`, `u_n,b = (R_out + R_in) / 2`,
260/// `c_b = (g - 1) (R_out - R_in) / 4`; entropy and tangential velocity come from the interior for `u_n,b >= 0`, else from the
261/// free stream (state `ref`): `rho_b = rho_ref (c_b / c_ref)^(2 / (g - 1))`, `p_b = rho_b c_b^2 / g`,
262/// `v_b = v_ref + (u_n,b - u_n,ref) n`. `M_n = u_n,i / c_i <= -1`: free stream, `>= 1`: interior.
263fn characteristic_ghost(fs: &[f64; 4], interior: &Rec, nx: &A, nr: &A) -> Rec {
264    let ci = (c(GAMMA) * &interior[3] / &interior[0]).sqrt();
265    let uni = &interior[1] * nx + &interior[2] * nr;
266    let m = &uni / &ci;
267    let ce = (GAMMA * fs[3] / fs[0]).sqrt();
268    let une = c(fs[1]) * nx + c(fs[2]) * nr;
269    let r_out = &uni + c(2.0) * &ci / c(G1);
270    let r_in = &une - c(2.0 * ce / G1);
271    let un_b = c(0.5) * (&r_out + &r_in);
272    let c_b = (c(0.25 * G1) * (&r_out - &r_in)).maximum(c(1e-8));
273    let from_interior = un_b.ge(c(0.0));
274    let rf = [
275        select(&from_interior, &interior[0], c(fs[0])),
276        select(&from_interior, &interior[1], c(fs[1])),
277        select(&from_interior, &interior[2], c(fs[2])),
278    ];
279    let un_ref = select(&from_interior, &uni, &une);
280    let c_ref = select(&from_interior, &ci, c(ce));
281    let q = &c_b / &c_ref;
282    let q2 = &q * &q;
283    let rho = &rf[0] * (&q2 * &q2 * &q);
284    let dn = &un_b - &un_ref;
285    let riemann = [rho.clone(), &rf[1] + &dn * nx, &rf[2] + &dn * nr, &rho * &c_b * &c_b / c(GAMMA)];
286    let super_in = m.le(c(-1.0));
287    let super_out = m.ge(c(1.0));
288    rec(|k| select(&super_in, c(fs[k]), select(&super_out, &interior[k], &riemann[k])))
289}

291/// Fixed static-pressure outflow: `p = p_back` where `M_n < 1`, interior state for `M_n >= 1`.
292fn pressure_outflow_ghost(p_back: f64, interior: &Rec, nx: &A, nr: &A) -> Rec {
293    let ci = (c(GAMMA) * &interior[3] / &interior[0]).sqrt();
294    let m = (&interior[1] * nx + &interior[2] * nr) / &ci;
295    [interior[0].clone(), interior[1].clone(), interior[2].clone(), select(&m.ge(c(1.0)), &interior[3], c(p_back))]
296}

359impl LazySolver {
360    /// Uploads the geometry (rounded from the grid's f64 metrics to `S`) and starts from the free stream.
361    pub fn new(ctx: &Context, grid: &Grid, cfg: &Config) -> Result<LazySolver> {
362        let (ni, nj) = (grid.ni, grid.nj);
363        let face = |f: &[crate::grid::Face], shape: [usize; 2], g: &dyn Fn(&crate::grid::Face) -> f64| {
364            let v: Vec<f64> = f.iter().map(g).collect();
365            up(ctx, &v, shape)
366        };
367        let boundary = |side: usize| -> Result<Boundary> {
368            // faces of the boundary, and the shape of the two ghost layers
369            let (faces, shape, outward): (Vec<crate::grid::Face>, [usize; 2], f64) = match side {
370                0 => ((0..nj).map(|j| grid.iface[j * (ni + 1)]).collect(), [2, nj], -1.0),
371                1 => ((0..nj).map(|j| grid.iface[j * (ni + 1) + ni]).collect(), [2, nj], 1.0),
372                2 => ((0..ni).map(|i| grid.jface[i]).collect(), [ni, 2], -1.0),
373                _ => ((0..ni).map(|i| grid.jface[nj * ni + i]).collect(), [ni, 2], 1.0),
374            };
375            // the layers are laid out `[layer, cell]` for i boundaries and `[cell, layer]` for j boundaries
376            let layered = |g: &dyn Fn(&crate::grid::Face) -> f64| -> Result<A> {
377                let v: Vec<f64> = if side < 2 { faces.iter().flat_map(|f| [g(f), g(f)]).collect() } else { [0, 1].iter().flat_map(|_| faces.iter().map(g)).collect() };
378                up(ctx, &v, shape)
379            };
380            let fixed = fixed_ghost(ctx, cfg, side, shape)?;
381            Ok(Boundary { nx: layered(&|f| f.nx)?, nr: layered(&|f| f.nr)?, onx: layered(&|f| outward * f.nx)?, onr: layered(&|f| outward * f.nr)?, fixed })
382        };
383        let geo = Geo {
384            inx: face(&grid.iface, [ni + 1, nj], &|f| f.nx)?,
385            inr: face(&grid.iface, [ni + 1, nj], &|f| f.nr)?,
386            ilen: face(&grid.iface, [ni + 1, nj], &|f| f.len)?,
387            iw: face(&grid.iface, [ni + 1, nj], &|f| f.w())?,
388            jnx: face(&grid.jface, [ni, nj + 1], &|f| f.nx)?,
389            jnr: face(&grid.jface, [ni, nj + 1], &|f| f.nr)?,
390            jlen: face(&grid.jface, [ni, nj + 1], &|f| f.len)?,
391            jw: face(&grid.jface, [ni, nj + 1], &|f| f.w())?,
392            inv_v: up(ctx, &grid.vol.iter().map(|v| 1.0 / v).collect::<Vec<_>>(), [ni, nj])?,
393            area: up(ctx, &grid.area, [ni, nj])?,
394            bnd: [boundary(0)?, boundary(1)?, boundary(2)?, boundary(3)?],
395        };
396        let mut s = LazySolver { ctx: ctx.clone(), ni, nj, cfg: cfg.clone(), low_mach: false, staging: DEFAULT_STAGING, geo, state: None, iter: 0, block: None, nozzle: None, n_active: ni * nj, scheme: Scheme::Explicit, lg: LGeo::new(ctx, grid)?, lb: None, visc: None };
397        let q = crate::gas::prim_to_cons(&cfg.freestream);
398        s.set_state(&vec![q; ni * nj])?;
399        Ok(s)
400    }

418    /// Declares the solid block (`None`: no block). Same range checks as `Solver::set_block`. The block's cells stay at their
419    /// state (residual and time step zero) and are left out of the residual norm.
420    pub fn set_block(&mut self, block: Option<Block>) -> Result<()> {
421        if let Some(b) = block {
422            assert!(
423                b.i_end >= 1
424                    && b.i_end + 2 <= self.ni
425                    && b.j_end >= 2
426                    && b.j_end + 2 <= self.nj
427                    && b.j_end > b.j_sting
428                    && (b.j_sting == 0 || b.j_sting >= 2),
429                "block out of range"
430            );
431        }
432        assert!(self.visc.is_none(), "declare the block before the viscous terms (`set_viscous`)");
433        self.block = block;
434        self.nozzle = None;
435        self.lb = block.map(|b| LBlockGeo::new(&self.ctx, self.ni, self.nj, b, None)).transpose()?;
436        self.n_active = match block {
437            Some(b) => self.ni * self.nj - b.i_end * b.j_end - (self.ni - b.i_end) * b.j_sting,
438            None => self.ni * self.nj,
439        };
440        Ok(())
441    }

443    /// Declares the nozzle exit on the base face (`None`: all base rows are wall; `Solver::set_nozzle`): rows `0..j_end` of the face
444    /// `i_end` are a fixed inflow state. Declare the block first and the viscous terms after; `set_block` clears the nozzle.
445    pub fn set_nozzle(&mut self, nozzle: Option<Nozzle>) -> Result<()> {
446        assert!(self.visc.is_none(), "declare the nozzle before the viscous terms (`set_viscous`)");
447        if let Some(n) = nozzle {
448            let b = self.block.expect("a nozzle needs a block");
449            assert!(b.j_sting == 0 && n.j_end >= 2 && n.j_end + 2 <= b.j_end && n.j_end % 2 == 0, "nozzle out of range");
450        }
451        self.nozzle = nozzle;
452        self.lb = self.block.map(|b| LBlockGeo::new(&self.ctx, self.ni, self.nj, b, nozzle)).transpose()?;
453        Ok(())
454    }

466    /// Zeros over the cells `i x j` of every field of `r`.
467    fn zero_region(&self, r: &mut Rec, i: std::ops::Range<usize>, j: std::ops::Range<usize>) {
468        if i.is_empty() || j.is_empty() {
469            return;
470        }
471        for a in r.iter_mut() {
472            a.slice_mut(0, i.clone()).slice(1, j.clone()).assign(c(0.0)).expect("zero region assign: the region lies inside the record, shapes fixed by set_block");
473        }

476    /// Solid cells of the two layers below a body or sting surface take the reflection of the fluid cells above it about the
477    /// j-face of the surface (`fill_block_ghosts`): the layers `col_start(i) - 2`, `col_start(i) - 1` get the cells
478    /// `col_start(i) + 1`, `col_start(i)`. Written after the domain ghosts, as on the CPU.
479    fn assign_block_ghosts(&self, b: Block, p: &Rec, w: &mut Rec) {
480        let g = &self.geo;
481        for (i0, i1, j0) in [(0, b.i_end, b.j_end), (b.i_end, self.ni, b.j_sting)] {
482            if j0 == 0 {
483                continue;
484            }
485            let dup = |a: A| concat(1, [&a, &a]);
486            let nx = dup(g.jnx.slice(0, i0..i1).slice(1, j0..j0 + 1));
487            let nr = dup(g.jnr.slice(0, i0..i1).slice(1, j0..j0 + 1));
488            let src = rec(|k| p[k].slice(0, i0..i1).slice(1, j0..).pad([(0, 0), (2, 0)], PadMode::Mirror).slice(1, 0..2));
489            w.slice_mut(0, i0 + 2..i1 + 2).slice(1, j0..j0 + 2).assign(self.block_wall_ghost(&src, &nx, &nr, i0)).expect("block ghost assign: shapes fixed by set_block");
490        }
491    }

493    /// The face fluxes next to the solid block, written into the i / j flux records `fi`, `fj`: the solid faces are zero, and
494    /// the base-wall faces `i_end`, `i_end + 1` (rows `j_sting..j_end`) see the solid stencil cells as the reflection of the
495    /// fluid cells about the base face:
496    ///
497    /// ```text
498    /// face i_end:     (m2, m1, p0, p1) = (R c1, R c0, c0, c1)
499    /// face i_end + 1: (m2, m1, p0, p1) = (R c0,  c0, c1, c2)         c_k the cell i_end + k, R the reflection
500    /// ```
501    fn block_fluxes(&self, b: Block, w: &Rec, fi: &mut Rec, fj: &mut Rec, lim: Limiter) {
502        let ni = self.ni;
503        let g = &self.geo;
504        // base wall faces: both faces use the normal of the base face for the reflection, their own for the flux
505        let rows = |a: &A, i0: usize, i1: usize| a.slice(0, i0..i1).slice(1, b.j_sting..b.j_end);
506        let (bx, br) = (rows(&g.inx, b.i_end, b.i_end + 1), rows(&g.inr, b.i_end, b.i_end + 1));
507        let (fx, fr) = (rows(&g.inx, b.i_end, b.i_end + 2), rows(&g.inr, b.i_end, b.i_end + 2));
508        let cells = |k: usize| rec(|f| w[f].slice(0, b.i_end + 2 + k..b.i_end + 3 + k).slice(1, b.j_sting + 2..b.j_end + 2));
509        let (c0, c1, c2) = (cells(0), cells(1), cells(2));
510        let join = |a: &Rec, bb: &Rec| rec(|f| concat(0, [&a[f], &bb[f]]));
511        let (rc0, rc1) = (self.base_wall_ghost(&c0, &bx, &br), self.base_wall_ghost(&c1, &bx, &br));
512        let (m2, m1) = (join(&rc1, &rc0), join(&rc0, &c0));
513        // nozzle rows: the exit state replaces the ghosts: face `i_end` sees (jet, jet), face `i_end + 1` sees (jet, c0)
514        let (m2, m1) = match (&self.nozzle, self.lb.as_ref().and_then(|lb| lb.jet_rows.as_ref())) {
515            (Some(n), Some([both, face0])) => {
516                let jw = n.state;
517                let m2 = rec(|k| select(both, c(jw[k]), &m2[k]));
518                let m1 = join(&rec(|k| select(face0, c(jw[k]), &rc0[k])), &c0);
519                (m2, m1)
520            }
521            _ => (m2, m1),
522        };
523        let (p0, p1) = (join(&c0, &c1), join(&c1, &c2));
524        let base = face_flux_lm(&m2, &m1, &p0, &p1, lim, &fx, &fr, self.low_mach, self.staging == Staging::Split);
525        // solid faces
526        self.zero_region(fi, 0..ni + 1, 0..b.j_sting);
527        self.zero_region(fi, 0..b.i_end, b.j_sting..b.j_end);
528        fi.slice_mut(0, b.i_end..b.i_end + 2).slice(1, b.j_sting..b.j_end).assign(&base).expect("base wall flux assign: shapes fixed by set_block");
529        self.zero_region(fj, 0..b.i_end, 0..b.j_end);
530        self.zero_region(fj, b.i_end..ni, 0..b.j_sting);
531    }

550    /// Writes the two ghost layers of boundary `side` (`[imin, imax, jmin, jmax]`) into the padded primitives `w` with a
551    /// region assign. `p` is the unpadded primitive state: the first interior cell `edge` and the cell `mirror` that a
552    /// wall reflects (`pad` with `Edge` / `Mirror` read both). `Bc::Extrapolate` keeps the `Edge` padding of `w`. Layers
553    /// are outermost first, as the cells they stand for. `w` is a lazy expression, so nothing is launched: `w` becomes the
554    /// padded state with the ghost expressions in place, and fuses into whatever reads it.
555    fn assign_ghosts(&self, side: usize, p: &Rec, w: &mut Rec) {
556        let (ni, nj) = (self.ni, self.nj);
557        let view = |a: &A, mode: PadMode| -> A {
558            match side {
559                0 => a.pad([(2, 0)], mode).slice(0, 0..2),
560                1 => a.pad([(0, 2)], mode).slice(0, ni..ni + 2),
561                2 => a.pad([(0, 0), (2, 0)], mode).slice(1, 0..2),
562                _ => a.pad([(0, 0), (0, 2)], mode).slice(1, nj..nj + 2),
563            }
564        };
565        let mut region = match side {
566            0 => w.slice_mut(0, ..2).slice(1, 2..nj + 2),
567            1 => w.slice_mut(0, ni + 2..).slice(1, 2..nj + 2),
568            2 => w.slice_mut(1, ..2).slice(0, 2..ni + 2),
569            _ => w.slice_mut(1, nj + 2..).slice(0, 2..ni + 2),
570        };
571        let b = &self.geo.bnd[side];
572        let fs = self.cfg.freestream;
573        let ghost = match self.cfg.bc[side] {
574            Bc::Freestream => b.fixed.clone().expect("a free-stream boundary carries its fixed ghost, set in the constructor"),
575            Bc::Extrapolate => return,
576            Bc::SlipWall => self.slip_wall_ghost(side, &rec(|k| view(&p[k], PadMode::Mirror)), &b.nx, &b.nr),
577            Bc::Characteristic => characteristic_ghost(&fs, &rec(|k| view(&p[k], PadMode::Edge)), &b.onx, &b.onr),
578            Bc::PressureOutflow => pressure_outflow_ghost(fs[3], &rec(|k| view(&p[k], PadMode::Edge)), &b.onx, &b.onr),
579        };
580        region.assign(ghost).expect("ghost layer assign: ghost shapes fixed by the boundary");
581    }

584    ///
585    /// ```text
586    /// dU/dt = -(1/V) sum_faces (len r_mid) F.n  +  (0, 0, p A / V, 0)        dt = cfl A / (1/2 sum_faces (|u.n| + c) len)
587    /// ```
588    pub fn eval_residual(&self, u: &Rec, lim: Limiter, compute_dt: bool) -> (Rec, Option<A>) {
589        let st = self.eval_stage(u, self.state_mu_t().as_ref(), lim, compute_dt);
590        (st.res, st.dt)
591    }

593    /// [`LazySolver::eval_residual`] with the eddy viscosity `mu_t` of an SST run (`None`: zero) and everything the turbulence update needs
594    /// (the viscous quantities, the primitives and the mass flow of the faces).
595    fn eval_stage(&self, u: &Rec, mu_t: Option<&A>, lim: Limiter, compute_dt: bool) -> Stage {
596        let (p, w) = self.padded(u);

600    /// Steps 2 to 5 of `eval_residual` for the primitives `p` and the padded primitives `w`, with the viscous terms of a viscous run.
601    fn stage_from(&self, p: Rec, w: Rec, mu_t: Option<&A>, lim: Limiter, compute_dt: bool) -> Stage {
602        let (fi, fj) = self.face_fluxes(&w, lim);
603        let vf = self.viscous_flow(&p, &w, mu_t);
604        let (res, dt) = self.cell_residual(&p, &fi, &fj, vf.as_ref(), compute_dt);
605        Stage { res, dt, mass: [fi[0].clone(), fj[0].clone()], p, w, vf }
606    }

608    /// Steps 1 of `eval_residual`: the primitives `p = cons_to_prim(u)` and the padded primitives `w` with the domain ghosts,
609    /// then the block ghosts, in place (the `Staging` cache applied).
610    fn padded(&self, u: &Rec) -> (Rec, Rec) {
611        // 1. primitives, padded by two cells on every side; the ghost layers of each boundary are one region assign
612        let p = cons_to_prim(u.clone());
613        let mut w = p.pad([(2, 2), (2, 2)], PadMode::Edge);
614        for side in 0..4 {
615            self.assign_ghosts(side, &p, &mut w);
616        }
617        if let Some(b) = self.block {
618            self.assign_block_ghosts(b, &p, &mut w);
619        }
620        // `Lines` and up: evaluate the padded record once. Without it every reader recomputes the primitives and all four ghost
621        // expressions at each stencil cell (200x100 f32: 67-91 launches per RK3 step without it, 26-30 with it; README "Speed").
622        let w = if self.staging == Staging::Fusion { w } else { w.cache() };
623        (p, w)
624    }

626    /// Steps 2 and 3 of `eval_residual`: the face fluxes of the padded primitives `w`.
627    fn face_fluxes(&self, w: &Rec, lim: Limiter) -> (Rec, Rec) {
628        let (ni, nj) = (self.ni, self.nj);
629        let g = &self.geo;
630
631        // 2. the cells along each grid line with their ghosts: i lines `[ni + 4, nj]`, j lines `[ni, nj + 4]`
632        let wi = w.slice(1, 2..nj + 2);
633        let wj = w.slice(0, 2..ni + 2);
634
635        // 3. face fluxes: face `f` sees the cells `f - 2 .. f + 1` of its line
636        let split = self.staging == Staging::Split;
637        let mut fi = face_flux_lm(&wi.slice(0, 0..ni + 1), &wi.slice(0, 1..ni + 2), &wi.slice(0, 2..ni + 3), &wi.slice(0, 3..ni + 4), lim, &g.inx, &g.inr, self.low_mach, split);
638        let mut fj = face_flux_lm(&wj.slice(1, 0..nj + 1), &wj.slice(1, 1..nj + 2), &wj.slice(1, 2..nj + 3), &wj.slice(1, 3..nj + 4), lim, &g.jnx, &g.jnr, self.low_mach, split);
639        if let Some(b) = self.block {
640            self.block_fluxes(b, w, &mut fi, &mut fj, lim);
641        }
642        // `Fluxes` and up: evaluate the face fluxes. The residual and the stage update then fuse into the next stage's padded
643        // record: three kernels per stage, 14 launches per step instead of 26-30 (the flux kernel read through four stencil
644        // views is cut at the 1024-operation fusion limit otherwise; README "Speed").
645        let (fi, fj) = if matches!(self.staging, Staging::Fluxes | Staging::Split) { (fi.cache(), fj.cache()) } else { (fi, fj) };
646        (fi, fj)
647    }

649    /// Steps 4 and 5 of `eval_residual`: the cell residual and the time step from the face fluxes `fi`, `fj` and the primitives `p`. A viscous
650    /// run (`vf`) uses the face flux `F - Fv` and the radial-momentum source `(p - t_tt) A / V`, and adds the diffusive limit to the time step.
651    fn cell_residual(&self, p: &Rec, fi: &Rec, fj: &Rec, vf: Option<&VFlow>, compute_dt: bool) -> (Rec, Option<A>) {
652        let (ni, nj) = (self.ni, self.nj);
653        let g = &self.geo;
654        let (fi, fj) = (self.total_flux(fi, vf.map(|v| &v.fvi)), self.total_flux(fj, vf.map(|v| &v.fvj)));
655
656        // 4. cell residual
657        let (wr, wl) = (g.iw.slice(0, 1..ni + 1), g.iw.slice(0, 0..ni));
658        let (wt, wb) = (g.jw.slice(1, 1..nj + 1), g.jw.slice(1, 0..nj));
659        let mut res = rec(|k| {
660            let (gr, gl) = (fi[k].slice(0, 1..ni + 1), fi[k].slice(0, 0..ni));
661            let (gt, gb) = (fj[k].slice(1, 1..nj + 1), fj[k].slice(1, 0..nj));
662            -(&gr * &wr - &gl * &wl + &gt * &wt - &gb * &wb) * &g.inv_v
663        });
664        res[2] = &res[2] + self.hoop_source(p, vf); // axisymmetric pressure source p A / V
665
666        // 5. local time step
667        let mut dt = compute_dt.then(|| {
668            let cs = (c(GAMMA) * &p[3] / &p[0]).sqrt();
669            let lam = |nx: &A, nr: &A, len: &A| ((&p[1] * nx + &p[2] * nr).abs() + &cs) * len;
670            let (fl_nx, fl_nr, fl_len) = (g.inx.slice(0, 0..ni), g.inr.slice(0, 0..ni), g.ilen.slice(0, 0..ni));
671            let (fr_nx, fr_nr, fr_len) = (g.inx.slice(0, 1..ni + 1), g.inr.slice(0, 1..ni + 1), g.ilen.slice(0, 1..ni + 1));
672            let (fb_nx, fb_nr, fb_len) = (g.jnx.slice(1, 0..nj), g.jnr.slice(1, 0..nj), g.jlen.slice(1, 0..nj));
673            let (ft_nx, ft_nr, ft_len) = (g.jnx.slice(1, 1..nj + 1), g.jnr.slice(1, 1..nj + 1), g.jlen.slice(1, 1..nj + 1));
674            let sum = lam(&fl_nx, &fl_nr, &fl_len) + lam(&fr_nx, &fr_nr, &fr_len) + lam(&fb_nx, &fb_nr, &fb_len) + lam(&ft_nx, &ft_nr, &ft_len);
675            let half = c(0.5) * &sum;
676            c(self.cfg.cfl) * &g.area / vf.map_or(half.clone(), |v| {
677                let (xi, xj) = self.diffusion_rates(v, &p[0]);
678                &half + xi + xj
679            })
680        });
681        // solid cells: residual and time step zero (region assigns, so nothing computed inside the solid reaches the state)
682        if let Some(b) = self.block {
683            self.zero_region(&mut res, 0..b.i_end, 0..b.j_end);
684            self.zero_region(&mut res, b.i_end..ni, 0..b.j_sting);
685            if let Some(dt) = &mut dt {
686                if b.j_sting > 0 {
687                    dt.slice_mut(0, b.i_end..ni).slice(1, 0..b.j_sting).assign(c(0.0)).expect("dt zero: block rows inside the grid");
688                }
689                dt.slice_mut(0, 0..b.i_end).slice(1, 0..b.j_end).assign(c(0.0)).expect("dt zero: block rows inside the grid");
690            }
691        }
692        (res, dt)
693    }

695    /// One SSP-RK3 step with local time stepping; returns the new state and the density residual of the first stage
696    /// `sqrt(sum (d rho / dt)^2 / N)`.
697    ///
698    /// ```text
699    /// u1 = u0 + dt L(u0),   u2 = 3/4 u0 + 1/4 (u1 + dt L(u1)),   u3 = 1/3 u0 + 2/3 (u2 + dt L(u2))
700    /// ```
701    fn rk3(&self, u0: &Rec, mu_t: Option<&A>, lim: Limiter) -> (Rec, A, Stage) {
702        let st0 = self.eval_stage(u0, mu_t, lim, true);
703        let (r0, dt) = (st0.res, st0.dt.expect("the residual stage was called with compute_dt = true"));
704        let rn = self.residual_norm(&r0[0]);
705        // stage 1: u1 = u0 + dt L(u0)
706        let u1 = rec(|k| &u0[k] + &dt * &r0[k]);
707        let r1 = self.eval_stage(&u1, mu_t, lim, false).res;
708        // stage 2: u2 = 3/4 u0 + 1/4 (u1 + dt L(u1))
709        let u2 = rec(|k| c(0.25) * (&u1[k] + &dt * &r1[k]) + c(0.75) * &u0[k]);
710        let st2 = self.eval_stage(&u2, mu_t, lim, false);
711        let r2 = &st2.res;
712        // stage 3: u3 = 1/3 u0 + 2/3 (u2 + dt L(u2))
713        let u3 = rec(|k| c(2.0 / 3.0) * (&u2[k] + &dt * &r2[k]) + c(1.0 / 3.0) * &u0[k]);
714        (u3, rn, Stage { dt: Some(dt), ..st2 })
715    }

723    }

725    /// Number of iterations from `self.iter` that use the same limiter (the traced step is specific to the limiter).
726    fn phase_len(&self, left: usize) -> usize {
727        if self.iter < self.cfg.first_order_iters {
728            left.min(self.cfg.first_order_iters - self.iter)
729        } else {
730            left
731        }
732    }
733
734    /// `n` iterations as one `iterate_until_all` loop per limiter phase (the step is traced once and its kernels replayed,
735    /// the density residual recorded on the GPU every iteration, the host waits once per `sync` iterations); returns the
736    /// residual history.
737    pub fn run(&mut self, n: usize, sync: usize) -> Result<Vec<f64>> {
738        let mut history = Vec::with_capacity(n);
739        let mut left = n;
740        while left > 0 {
741            let seg = self.phase_len(left);
742            let lim = self.limiter(self.iter);
743            let init = self.state.take().ok_or_else(no_state)?;
744            let run = luma::iterate_until_all(init, records(luma::Until::new(seg).check_every(sync)), |s, _| {
745                self.step(s, lim)
746            })?;
747            history.extend(read_history(&run)?);
748            self.state = Some(run.state);
749            self.iter += seg;
750            left -= seg;
751        }
752        Ok(history)
753    }
754
755    /// `n` iterations as one `iterate_all` replay per limiter phase (no residual history).
756    pub fn run_replay(&mut self, n: usize) -> Result<()> {
757        let mut left = n;
758        while left > 0 {
759            let seg = self.phase_len(left);
760            let lim = self.limiter(self.iter);
761            let init = self.state.take().ok_or_else(no_state)?;
762            self.state = Some(luma::iterate_all(init.each_ref(), seg, |s| {
763                self.step(s, lim)
764            }));
765            self.iter += seg;
766            left -= seg;
767        }
768        Ok(())
769    }
770
771    /// `n` iterations calling `rk3` every iteration and evaluating the new state with `eval_all` (no replay). Returns the
772    /// evaluated residual scalars (read them with [`Residuals::read`]).
773    pub fn run_eval(&mut self, n: usize) -> Result<Residuals> {
774        let mut history = Vec::with_capacity(n);
775        let mut s = self.state.take().ok_or_else(no_state)?;
776        for _ in 0..n {
777            s = self.step(&s, self.limiter(self.iter));
778            luma::eval_all(&s.iter().collect::<Vec<_>>())?;
779            history.push(s[4].clone());
780            self.iter += 1;
781        }
782        self.state = Some(s);
783        Ok(Residuals(history))
784    }
785
786    /// Waits for the GPU.
787    pub fn sync(&self) -> Result<()> {
788        self.ctx.sync()
789    }
790
791    /// The residual `dU/dt` and time step of the current state, on the host (`[res; 4]` of `ni * nj`, `dt`).
792    pub fn residual_f64(&self, lim: Limiter) -> Result<(Vec<[f64; 4]>, Vec<f64>)> {
793        let u = unpack(self.state.as_ref().ok_or_else(no_state)?);
794        let (r, dt) = self.eval_residual(&u, lim, true);
795        let r: Vec<Vec<f64>> = r.iter().map(to_f64).collect::<Result<_>>()?;
796        let dt = to_f64(&dt.ok_or_else(|| luma::LumaError::Unsupported("residual stage returned no time step".into()))?)?;
797        Ok(((0..self.ni * self.nj).map(|c| [r[0][c], r[1][c], r[2][c], r[3][c]]).collect(), dt))
798    }
799
800    /// GPU time per kernel of one residual evaluation (profile) and where fusion cut the expressions.
801    pub fn profile_residual(&self) -> Result<luma::Profile> {
802        let u = unpack(self.state.as_ref().ok_or_else(no_state)?);
803        let lim = self.cfg.limiter;
804        // warm up (compile) first, so that the profile shows launches only
805        let _ = self.eval_residual(&u, lim, true);
806        self.ctx.sync()?;
807        let (out, profile) = self.ctx.profile(|| -> Result<()> {
808            let (r, dt) = self.eval_residual(&u, lim, true);
809            luma::eval_all(&r.iter().chain(dt.iter()).collect::<Vec<_>>())?;
810            self.ctx.sync()
811        })?;
812        out?;
813        Ok(profile)
814    }
815}

817/// The density residual scalars of an `eval` loop, one per iteration, as evaluated GPU arrays.
818pub struct Residuals(Vec<A>);
819
820impl Residuals {
821    /// Reads the residual history (waits for the GPU).
822    pub fn read(&self) -> Result<Vec<f64>> {
823        self.0.iter().map(|a| Ok(to_f64(a)?[0])).collect()
824    }
825}
826
827impl LazySolver {
828    /// GPU time per kernel of one full SSP-RK3 step (three residual evaluations and the stage updates) as the `eval` loop
829    /// launches it, and where fusion cut the expressions.
830    pub fn profile_step(&mut self) -> Result<luma::Profile> {
831        self.run_eval(1)?;
832        self.ctx.sync()?;
833        let ctx = self.ctx.clone();
834        let (out, profile) = ctx.profile(|| -> Result<()> {
835            self.run_eval(1)?;
836            self.ctx.sync()
837        })?;
838        out?;
839        Ok(profile)
840    }
841
842    /// GPU time per kernel of `n` iterations of the `iterate_all` replay (the profile adds up all `n`), after a warm-up call
843    /// that traces and compiles it.
844    pub fn profile_replay(&mut self, n: usize) -> Result<luma::Profile> {
845        self.run_replay(n)?;
846        self.ctx.sync()?;
847        let ctx = self.ctx.clone();
848        let (out, profile) = ctx.profile(|| -> Result<()> {
849            self.run_replay(n)?;
850            self.ctx.sync()
851        })?;
852        out?;
853        Ok(profile)
854    }
855
856    /// Host seconds to build the lazy expression of one SSP-RK3 step from the current state, `n` times, without evaluating
857    /// anything (the part of an `eval` loop iteration that is Rust code building nodes).
858    pub fn trace_seconds(&self, n: usize) -> f64 {
859        let u = unpack(self.state.as_ref().expect("trace_seconds needs a state: GpuSolver::from_host sets it"));
860        let lim = self.cfg.limiter;
861        let t0 = std::time::Instant::now();
862        for _ in 0..n {
863            std::hint::black_box(self.rk3(&u, self.state_mu_t().as_ref(), lim));
864        }
865        t0.elapsed().as_secs_f64()
866    }
867}
