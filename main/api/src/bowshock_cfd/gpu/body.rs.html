1// The explicit solver in array ops. This file is `include!`d twice by `gpu/mod.rs`, once per precision, into a module that
2// defines `S` (the element type: `f32` or `Df32`), `c(f64) -> S` (a constant rounded from f64), `up(..)` (host f64 data to
3// an `Array<S>`), `to_f64(..)` (an `Array<S>` to host f64 data) and the residual bookkeeping of the loops (`records`,
4// `read_history`, `rn_host`).
5//
6// Every function mirrors the CPU routine of the same name in the CPU modules of this crate expression by expression; the array code is
7// the equation, the doc comment states it.

16/// Array of one field (`rho`, `u`, ...) over a block of cells or faces.
17pub type A = Array<S>;
18/// Four fields: primitive `[rho, u, v, p]`, conservative `[rho, rho u, rho v, E]` or a flux. A record of arrays
19/// (`luma::Fields`): `slice`, `pad`, `assign` and `cache` apply to the four fields at once.
20pub type Rec = [A; 4];
21/// The loop state: the conserved record, the density residual of the step that produced it (one element), the CFL of the next
22/// iteration (one element; the ramp of the implicit scheme is loop state, the explicit scheme passes it through) and the turbulence
23/// state of a viscous run: `k`, `omega`, the eddy viscosity (`[ni, nj]` with the SST model, one dummy element otherwise) and the
24/// density residual norm of `k` of the last update (one element).
25pub type State = [A; 10];
26
27fn rec(f: impl FnMut(usize) -> A) -> Rec {
28    std::array::from_fn(f)
29}

31/// The conserved record of a loop state.
32fn unpack(s: &State) -> Rec {
33    rec(|k| s[k].clone())
34}

36fn pack(u: Rec, rn: A, cfl: A, t: [A; 4]) -> State {
37    let [u0, u1, u2, u3] = u;
38    let [k, om, mu_t, trn] = t;
39    [u0, u1, u2, u3, rn, cfl, k, om, mu_t, trn]
40}

42/// The loop state of host data: the conserved state `u` (cell-major, `[rho, rho u, rho v, E]`), a zero residual and the CFL `cfl`.
43fn upload_state(ctx: &Context, u: &[[f64; 4]], shape: [usize; 2], cfl: f64) -> Result<State> {
44    let comp = |k: usize| up(ctx, &u.iter().map(|q| q[k]).collect::<Vec<_>>(), shape);
45    let dummy = || up(ctx, &[0.0], 1);
46    Ok([comp(0)?, comp(1)?, comp(2)?, comp(3)?, up(ctx, &[0.0], 1)?, up(ctx, &[cfl], 1)?, dummy()?, dummy()?, dummy()?, dummy()?])
47}

49/// The four conserved components stacked `[4, ..]` (component fastest) and read back with one sync.
50fn read_stacked(s: &State, ri: std::ops::Range<usize>, rj: std::ops::Range<usize>) -> Result<Vec<f64>> {
51    let part = |k: usize| s[k].slice(0, ri.clone()).slice(1, rj.clone());
52    to_f64(&luma::stack(0, [&part(0), &part(1), &part(2), &part(3)]))
53}
54
55fn download_state(s: &State) -> Result<Vec<[f64; 4]>> {
56    let (ni, nj) = (s[0].shape().dims()[0], s[0].shape().dims()[1]);
57    let v = read_stacked(s, 0..ni, 0..nj)?;
58    Ok(v.as_chunks::<4>().0.to_vec())
59}
60
61/// Reads the cells `ri x rj` of the state into `u` (`j * ni + i`) with one sync.
62fn read_rect_into(s: &State, ni: usize, ri: std::ops::Range<usize>, rj: std::ops::Range<usize>, u: &mut [[f64; 4]]) -> Result<()> {
63    if ri.is_empty() || rj.is_empty() {
64        return Ok(());
65    }
66    let v = read_stacked(s, ri.clone(), rj.clone())?;
67    let w = ri.len();
68    for (n, &q) in v.as_chunks::<4>().0.iter().enumerate() {
69        u[(rj.start + n / w) * ni + ri.start + n % w] = q;
70    }
71    Ok(())
72}

80/// `rho = q0, u = q1 / rho, v = q2 / rho, p = (g - 1) (E - rho (u^2 + v^2) / 2)`.
81pub fn cons_to_prim(q: Rec) -> Rec {
82    let u = &q[1] / &q[0];
83    let v = &q[2] / &q[0];
84    let p = c(G1) * (&q[3] - c(0.5) * &q[0] * (&u * &u + &v * &v));
85    let [rho, ..] = q;
86    [rho, u, v, p]
87}

89/// `F.n = [rho u_n, rho u u_n + p n_x, rho v u_n + p n_r, (E + p) u_n]` with `u_n = u n_x + v n_r`.
90pub fn physical_flux(w: &Rec, nx: &A, nr: &A) -> Rec {
91    let [rho, u, v, p] = w;
92    let un = u * nx + v * nr;
93    let e = p / c(G1) + c(0.5) * rho * (u * u + v * v);
94    flux_from(w, &un, &e, nx, nr)
95}
96
97/// [`physical_flux`] of a state whose normal velocity `u_n` and total energy `E` are known.
98fn flux_from(w: &Rec, un: &A, e: &A, nx: &A, nr: &A) -> Rec {
99    let [rho, u, v, p] = w;
100    [rho * un, rho * u * un + p * nx, rho * v * un + p * nr, (e + p) * un]
101}

103/// One side of a face with the quantities both Riemann solvers need, built once: normal velocity `u_n = u n_x + v n_r`,
104/// total energy `E = p / (g - 1) + rho (u^2 + v^2) / 2`, sound speed `c = sqrt(g p / rho)` and the flux `F.n`.
105struct Side<'a> {
106    w: &'a Rec,
107    un: A,
108    e: A,
109    c: A,
110    f: Rec,
111}
112
113impl<'a> Side<'a> {
114    fn new(w: &'a Rec, nx: &A, nr: &A) -> Side<'a> {
115        let [rho, u, v, p] = w;
116        let un = u * nx + v * nr;
117        let e = p / c(G1) + c(0.5) * rho * (u * u + v * v);
118        let f = flux_from(w, &un, &e, nx, nr);
119        Side { w, un, e, c: (c(GAMMA) * p / rho).sqrt(), f }
120    }
121}

123/// `w` reflected about a wall of unit normal `(n_x, n_r)`: `v - 2 (v.n) n`.
124fn reflect(w: &Rec, nx: &A, nr: &A) -> Rec {
125    let un = &w[1] * nx + &w[2] * nr;
126    [w[0].clone(), &w[1] - c(2.0) * &un * nx, &w[2] - c(2.0) * &un * nr, w[3].clone()]
127}

133/// Limited slope `phi(a, b)`, `a = q_i - q_{i-1}`, `b = q_{i+1} - q_i`; zero unless `a b > 0`.
134/// minmod: the smaller of `a`, `b` in modulus. van Albada: `(a (b^2 + e) + b (a^2 + e)) / (a^2 + b^2 + 2 e)`, `e = 1e-14`.
135fn limited_slope(a: &A, b: &A, lim: Limiter) -> A {
136    let same_sign = (a * b).gt(c(0.0));
137    match lim {
138        Limiter::FirstOrder => unreachable!("first order reconstruction has no slopes"),
139        Limiter::Minmod => select(&same_sign, select(&a.abs().lt(b.abs()), a, b), c(0.0)),
140        Limiter::VanAlbada => {
141            let eps = 1e-14;
142            let slope = (a * (b * b + c(eps)) + b * (a * a + c(eps))) / (a * a + b * b + c(2.0 * eps));
143            select(&same_sign, slope, c(0.0))
144        }
145        Limiter::Venkatakrishnan => {
146            let eps = 1e-14;
147            let slope = c(2.0) * b * (a * a + a * b + c(eps)) / (c(2.0) * a * a + a * b + b * b + c(2.0 * eps));
148            select(&same_sign, slope, c(0.0))
149        }
150    }
151}

153/// MUSCL reconstruction at the face between the cells `m1 | p0` (`m2`, `p1` the outer neighbours):
154/// `l = m1 + phi(m1 - m2, p0 - m1) / 2`, `r = p0 - phi(p0 - m1, p1 - p0) / 2`;
155/// a face whose reconstructed density or pressure is not positive falls back to the cell value.
156fn reconstruct(m2: &Rec, m1: &Rec, p0: &Rec, p1: &Rec, lim: Limiter) -> (Rec, Rec) {
157    if lim == Limiter::FirstOrder {
158        return (m1.clone(), p0.clone());
159    }
160    let l = rec(|k| &m1[k] + c(0.5) * limited_slope(&(&m1[k] - &m2[k]), &(&p0[k] - &m1[k]), lim));
161    let r = rec(|k| &p0[k] - c(0.5) * limited_slope(&(&p0[k] - &m1[k]), &(&p1[k] - &p0[k]), lim));
162    let ok_l = l[0].gt(c(0.0)) & l[3].gt(c(0.0));
163    let ok_r = r[0].gt(c(0.0)) & r[3].gt(c(0.0));
164    (rec(|k| select(&ok_l, &l[k], &m1[k])), rec(|k| select(&ok_r, &r[k], &p0[k])))
165}

167/// HLLC flux for the 2-D Euler equations projected on the unit normal `(n_x, n_r)` (Toro ch. 10), wave speeds
168/// `S_L = min(u_nL - c_L, u~_n - c~)`, `S_R = max(u_nR + c_R, u~_n + c~)` from Roe averages (`sqrt(rho)` weights):
169/// `F = F_L` for `S_L >= 0`, `F_R` for `S_R <= 0`, else `F_K + S_K (U*_K - U_K)` of the side `K` of the contact speed `S_M`.
170fn hllc_speeds(l: &Side, r: &Side, nx: &A, nr: &A) -> [A; 3] {
171    let [rl, ul, vl, pl] = l.w;
172    let [rr, ur, vr, pr] = r.w;
173    let (unl, unr, cl, cr, el, er) = (&l.un, &r.un, &l.c, &r.c, &l.e, &r.e);
174
175    // Roe averages
176    let sql = rl.sqrt();
177    let sqr = rr.sqrt();
178    let inv = c(1.0) / (&sql + &sqr);
179    let uh = (&sql * ul + &sqr * ur) * &inv;
180    let vh = (&sql * vl + &sqr * vr) * &inv;
181    let hl = (el + pl) / rl;
182    let hr = (er + pr) / rr;
183    let hh = (&sql * &hl + &sqr * &hr) * &inv;
184    let ch2 = c(G1) * (&hh - c(0.5) * (&uh * &uh + &vh * &vh));
185    let ch = ch2.maximum(c(1e-30)).sqrt();
186    let unh = &uh * nx + &vh * nr;
187
188    let sl = (unl - cl).minimum(&unh - &ch);
189    let sr = (unr + cr).maximum(&unh + &ch);
190    let sl_un = &sl - unl;
191    let sr_un = &sr - unr;
192    let sm = (pr - pl + rl * unl * &sl_un - rr * unr * &sr_un) / (rl * &sl_un - rr * &sr_un);
193    [sl, sr, sm]
194}

196/// The HLLC flux from the wave speeds `[S_L, S_R, S_M]` of [`hllc_speeds`].
197fn hllc_flux(l: &Side, r: &Side, nx: &A, nr: &A, speeds: &[A; 3]) -> Rec {
198    let [sl, sr, sm] = speeds;
199    let sl_un = sl - &l.un;
200    let sr_un = sr - &r.un;
201
202    // `F_K + S_K (U*_K - U_K)`, `U*_K = fac [1, u + (S_M - u_n) n_x, v + (S_M - u_n) n_r, E / rho + (S_M - u_n) (S_M + p / (rho (S_K - u_n)))]`
203    let star = |k: &Side, s: &A, s_un: &A| -> Rec {
204        let [r, u, v, p] = k.w;
205        let (f, e) = (&k.f, &k.e);
206        let fac = r * s_un / (s - sm);
207        let sm_un = sm - &k.un;
208        [
209            &f[0] + s * (&fac - r),
210            &f[1] + s * (&fac * (u + &sm_un * nx) - r * u),
211            &f[2] + s * (&fac * (v + &sm_un * nr) - r * v),
212            &f[3] + s * (&fac * (e / r + &sm_un * (sm + p / (r * s_un))) - e),
213        ]
214    };
215    let star_l = star(l, sl, &sl_un);
216    let star_r = star(r, sr, &sr_un);
217
218    let left = sl.ge(c(0.0));
219    let right = sr.le(c(0.0));
220    let contact_left = sm.ge(c(0.0));
221    rec(|k| select(&left, &l.f[k], select(&right, &r.f[k], select(&contact_left, &star_l[k], &star_r[k]))))
222}

224/// Rusanov flux `F = (F_L + F_R) / 2 - s_max (U_R - U_L) / 2`, `s_max = max(|u_nL| + c_L, |u_nR| + c_R)`.
225fn rusanov(l: &Side, r: &Side) -> Rec {
226    let [rl, ul, vl, _] = l.w;
227    let [rr, ur, vr, _] = r.w;
228    let smax = (l.un.abs() + &l.c).maximum(r.un.abs() + &r.c);
229    let dq = [rr - rl, rr * ur - rl * ul, rr * vr - rl * vl, &r.e - &l.e];
230    rec(|k| c(0.5) * (&l.f[k] + &r.f[k]) - c(0.5) * &smax * &dq[k])
231}

233/// Weight of the Rusanov flux: `s = |p_R - p_L| / min(p_L, p_R)`, `t = clamp((s - lo) / (hi - lo), 0, 1)`, `theta = t^2 (3 - 2 t)`.
234fn shock_weight(pl: &A, pr: &A) -> A {
235    let s = (pr - pl).abs() / pl.minimum(pr);
236    let t = ((s - c(SHOCK_SENSOR_LO)) / c(SHOCK_SENSOR_HI - SHOCK_SENSOR_LO)).maximum(c(0.0)).minimum(c(1.0));
237    &t * &t * (c(3.0) - c(2.0) * &t)
238}

240/// Low-Mach scaling of the face-normal velocity jump (Thornber et al. 2008): `z = clamp(max(M_L, M_R), 0.05, 1)`,
241/// `u_n' = u_avg + z (u_n - u_avg)` on both sides, untouched for `z >= 1`.
242fn low_mach_states(l: Rec, r: Rec, nx: &A, nr: &A) -> (Rec, Rec) {
243    let unl = &l[1] * nx + &l[2] * nr;
244    let unr = &r[1] * nx + &r[2] * nr;
245    let ml = (&l[1] * &l[1] + &l[2] * &l[2]).sqrt() / (c(GAMMA) * &l[3] / &l[0]).sqrt();
246    let mr = (&r[1] * &r[1] + &r[2] * &r[2]).sqrt() / (c(GAMMA) * &r[3] / &r[0]).sqrt();
247    let z = ml.maximum(&mr).maximum(c(LOW_MACH_FLOOR)).minimum(c(1.0));
248    let active = z.lt(c(1.0));
249    let avg = c(0.5) * (&unl + &unr);
250    let dl = &avg + &z * (&unl - &avg) - &unl;
251    let dr = &avg + &z * (&unr - &avg) - &unr;
252    let fix = |w: Rec, d: &A| -> Rec {
253        let [rho, u, v, p] = w;
254        [rho, select(&active, &u + d * nx, &u), select(&active, &v + d * nr, &v), p]
255    };
256    (fix(l, &dl), fix(r, &dr))
257}

259/// Complete face kernel: MUSCL + (low-Mach) + `(1 - theta) HLLC + theta Rusanov`.
260///
261/// `split = false` is one expression (what the equations say): Luma fuses it into one kernel of ~115 operations.
262/// `split = true` evaluates the reconstructed states and the shock weight first (`cache`), the flux reads them from memory.
263fn face_flux_lm(m2: &Rec, m1: &Rec, p0: &Rec, p1: &Rec, lim: Limiter, nx: &A, nr: &A, low_mach: bool, split: bool) -> Rec {
264    let (l, r) = reconstruct(m2, m1, p0, p1, lim);
265    let (l, r) = if low_mach { low_mach_states(l, r, nx, nr) } else { (l, r) };
266    let t = shock_weight(&m1[3], &p0[3]);
267    let (l, r, t) = if split { (l, r, t).cache() } else { (l, r, t) };
268    let (sl, sr) = (Side::new(&l, nx, nr), Side::new(&r, nx, nr));
269    let hllc = hllc_flux(&sl, &sr, nx, nr, &hllc_speeds(&sl, &sr, nx, nr));
270    let rus = rusanov(&sl, &sr);
271    rec(|k| (c(1.0) - &t) * &hllc[k] + &t * &rus[k])
272}

278/// Characteristic far field. With the outward normal `n`, `u_n,i` and `c_i` of the interior state `i` and `e` the free stream:
279/// `R_out = u_n,i + 2 c_i / (g - 1)`, `R_in = u_n,e - 2 c_e / (g - 1)`, `u_n,b = (R_out + R_in) / 2`,
280/// `c_b = (g - 1) (R_out - R_in) / 4`; entropy and tangential velocity come from the interior for `u_n,b >= 0`, else from the
281/// free stream (state `ref`): `rho_b = rho_ref (c_b / c_ref)^(2 / (g - 1))`, `p_b = rho_b c_b^2 / g`,
282/// `v_b = v_ref + (u_n,b - u_n,ref) n`. `M_n = u_n,i / c_i <= -1`: free stream, `>= 1`: interior.
283fn characteristic_ghost(fs: &[f64; 4], interior: &Rec, nx: &A, nr: &A) -> Rec {
284    let ci = (c(GAMMA) * &interior[3] / &interior[0]).sqrt();
285    let uni = &interior[1] * nx + &interior[2] * nr;
286    let m = &uni / &ci;
287    let ce = (GAMMA * fs[3] / fs[0]).sqrt();
288    let une = c(fs[1]) * nx + c(fs[2]) * nr;
289    let r_out = &uni + c(2.0) * &ci / c(G1);
290    let r_in = &une - c(2.0 * ce / G1);
291    let un_b = c(0.5) * (&r_out + &r_in);
292    let c_b = (c(0.25 * G1) * (&r_out - &r_in)).maximum(c(1e-8));
293    let from_interior = un_b.ge(c(0.0));
294    let rf = [
295        select(&from_interior, &interior[0], c(fs[0])),
296        select(&from_interior, &interior[1], c(fs[1])),
297        select(&from_interior, &interior[2], c(fs[2])),
298    ];
299    let un_ref = select(&from_interior, &uni, &une);
300    let c_ref = select(&from_interior, &ci, c(ce));
301    let q = &c_b / &c_ref;
302    let q2 = &q * &q;
303    let rho = &rf[0] * (&q2 * &q2 * &q);
304    let dn = &un_b - &un_ref;
305    let riemann = [rho.clone(), &rf[1] + &dn * nx, &rf[2] + &dn * nr, &rho * &c_b * &c_b / c(GAMMA)];
306    let super_in = m.le(c(-1.0));
307    let super_out = m.ge(c(1.0));
308    rec(|k| select(&super_in, c(fs[k]), select(&super_out, &interior[k], &riemann[k])))
309}

311/// Fixed static-pressure outflow: `p = p_back` where `M_n < 1`, interior state for `M_n >= 1`.
312fn pressure_outflow_ghost(p_back: f64, interior: &Rec, nx: &A, nr: &A) -> Rec {
313    let ci = (c(GAMMA) * &interior[3] / &interior[0]).sqrt();
314    let m = (&interior[1] * nx + &interior[2] * nr) / &ci;
315    [interior[0].clone(), interior[1].clone(), interior[2].clone(), select(&m.ge(c(1.0)), &interior[3], c(p_back))]
316}

383    /// Uploads the geometry (rounded from the grid's f64 metrics to `S`) and starts from the free stream.
384    pub fn new(ctx: &Context, grid: &Grid, cfg: &Config) -> Result<LazySolver> {
385        let (ni, nj) = (grid.ni, grid.nj);
386        let face = |f: &[crate::grid::Face], shape: [usize; 2], g: &dyn Fn(&crate::grid::Face) -> f64| {
387            let v: Vec<f64> = f.iter().map(g).collect();
388            up(ctx, &v, shape)
389        };
390        let boundary = |side: usize| -> Result<Boundary> {
391            // faces of the boundary, and the shape of the two ghost layers
392            let (faces, shape, outward): (Vec<crate::grid::Face>, [usize; 2], f64) = match side {
393                0 => ((0..nj).map(|j| grid.iface[j * (ni + 1)]).collect(), [2, nj], -1.0),
394                1 => ((0..nj).map(|j| grid.iface[j * (ni + 1) + ni]).collect(), [2, nj], 1.0),
395                2 => ((0..ni).map(|i| grid.jface[i]).collect(), [ni, 2], -1.0),
396                _ => ((0..ni).map(|i| grid.jface[nj * ni + i]).collect(), [ni, 2], 1.0),
397            };
398            // the layers are laid out `[layer, cell]` for i boundaries and `[cell, layer]` for j boundaries
399            let layered = |g: &dyn Fn(&crate::grid::Face) -> f64| -> Result<A> {
400                let v: Vec<f64> = if side < 2 { faces.iter().flat_map(|f| [g(f), g(f)]).collect() } else { [0, 1].iter().flat_map(|_| faces.iter().map(g)).collect() };
401                up(ctx, &v, shape)
402            };
403            let fixed = fixed_ghost(ctx, cfg, side, shape)?;
404            Ok(Boundary { nx: layered(&|f| f.nx)?, nr: layered(&|f| f.nr)?, onx: layered(&|f| outward * f.nx)?, onr: layered(&|f| outward * f.nr)?, fixed })
405        };
406        let geo = Geo {
407            inx: face(&grid.iface, [ni + 1, nj], &|f| f.nx)?,
408            inr: face(&grid.iface, [ni + 1, nj], &|f| f.nr)?,
409            ilen: face(&grid.iface, [ni + 1, nj], &|f| f.len)?,
410            iw: face(&grid.iface, [ni + 1, nj], &|f| f.w())?,
411            jnx: face(&grid.jface, [ni, nj + 1], &|f| f.nx)?,
412            jnr: face(&grid.jface, [ni, nj + 1], &|f| f.nr)?,
413            jlen: face(&grid.jface, [ni, nj + 1], &|f| f.len)?,
414            jw: face(&grid.jface, [ni, nj + 1], &|f| f.w())?,
415            inv_v: up(ctx, &grid.vol.iter().map(|v| 1.0 / v).collect::<Vec<_>>(), [ni, nj])?,
416            area: up(ctx, &grid.area, [ni, nj])?,
417            bnd: [boundary(0)?, boundary(1)?, boundary(2)?, boundary(3)?],
418        };
419        let mut s = LazySolver { ctx: ctx.clone(), ni, nj, cfg: cfg.clone(), low_mach: false, staging: DEFAULT_STAGING, geo, state: None, iter: 0, block: None, nozzle: None, n_active: ni * nj, scheme: Scheme::Explicit, lg: LGeo::new(ctx, grid)?, lb: None, visc: None, compiled: [None, None] };
420        let q = crate::gas::prim_to_cons(&cfg.freestream);
421        s.set_state(&vec![q; ni * nj])?;
422        Ok(s)
423    }

425    /// A lazy solver in the state of a host solver: grid, config, low-Mach flag, block, conserved state, iteration count and scheme.
426    pub fn from_host(ctx: &Context, s: &crate::solver::Solver) -> Result<LazySolver> {
427        let mut l = LazySolver::new(ctx, &s.grid, &s.cfg)?;
428        l.set_low_mach(s.low_mach);
429        l.set_block(s.block())?;
430        l.set_nozzle(s.nozzle())?;
431        l.set_state(&s.u)?;
432        l.iter = s.iter;
433        l.set_scheme(s.scheme())?;
434        l.set_viscous(s.viscous(), &s.grid)?;
435        if s.viscous().is_some_and(|v| v.turbulence == crate::solver::Turbulence::Sst) {
436            l.set_turb(&s.turb, &s.mu_t)?;
437        }
438        Ok(l)
439    }

441    /// Declares the solid block (`None`: no block). Same range checks as `Solver::set_block`. The block's cells stay at their
442    /// state (residual and time step zero) and are left out of the residual norm.
443    pub fn set_block(&mut self, block: Option<Block>) -> Result<()> {
444        if let Some(b) = block {
445            assert!(
446                b.i_end >= 1
447                    && b.i_end + 2 <= self.ni
448                    && b.j_end >= 2
449                    && b.j_end + 2 <= self.nj
450                    && b.j_end > b.j_sting
451                    && (b.j_sting == 0 || b.j_sting >= 2),
452                "block out of range"
453            );
454        }
455        assert!(self.visc.is_none(), "declare the block before the viscous terms (`set_viscous`)");
456        self.compiled = [None, None];
457        self.block = block;
458        self.nozzle = None;
459        self.lb = block.map(|b| LBlockGeo::new(&self.ctx, self.ni, self.nj, b, None)).transpose()?;
460        self.n_active = match block {
461            Some(b) => self.ni * self.nj - b.i_end * b.j_end - (self.ni - b.i_end) * b.j_sting,
462            None => self.ni * self.nj,
463        };
464        Ok(())
465    }

467    /// Declares the nozzle exit on the base face (`None`: all base rows are wall; `Solver::set_nozzle`): rows `0..j_end` of the face
468    /// `i_end` are a fixed inflow state. Declare the block first and the viscous terms after; `set_block` clears the nozzle.
469    pub fn set_nozzle(&mut self, nozzle: Option<Nozzle>) -> Result<()> {
470        assert!(self.visc.is_none(), "declare the nozzle before the viscous terms (`set_viscous`)");
471        if let Some(n) = nozzle {
472            let b = self.block.expect("a nozzle needs a block");
473            assert!(b.j_sting == 0 && n.j_end >= 2 && n.j_end + 2 <= b.j_end && n.j_end % 2 == 0, "nozzle out of range");
474        }
475        self.compiled = [None, None];
476        self.nozzle = nozzle;
477        self.lb = self.block.map(|b| LBlockGeo::new(&self.ctx, self.ni, self.nj, b, nozzle)).transpose()?;
478        Ok(())
479    }

486    /// `sqrt(sum (d rho / dt)^2 / N)`, `N` the number of fluid cells.
487    fn residual_norm(&self, r0: &A) -> A {
488        ((r0 * r0).sum() / c(self.n_active as f64)).sqrt()

491    /// Zeros over the cells `i x j` of every field of `r`.
492    fn zero_region(&self, r: &mut Rec, i: std::ops::Range<usize>, j: std::ops::Range<usize>) {
493        if i.is_empty() || j.is_empty() {
494            return;
495        }
496        for a in r.iter_mut() {
497            a.slice_mut(0, i.clone()).slice(1, j.clone()).assign(c(0.0)).expect("zero region assign: the region lies inside the record, shapes fixed by set_block");
498        }
499    }

501    /// Solid cells of the two layers below a body or sting surface take the reflection of the fluid cells above it about the
502    /// j-face of the surface (`fill_block_ghosts`): the layers `col_start(i) - 2`, `col_start(i) - 1` get the cells
503    /// `col_start(i) + 1`, `col_start(i)`. Written after the domain ghosts, as on the CPU.
504    fn assign_block_ghosts(&self, b: Block, p: &Rec, w: &mut Rec) {
505        let g = &self.geo;
506        for (i0, i1, j0) in [(0, b.i_end, b.j_end), (b.i_end, self.ni, b.j_sting)] {
507            if j0 == 0 {
508                continue;
509            }
510            let dup = |a: A| concat(1, [&a, &a]);
511            let nx = dup(g.jnx.slice(0, i0..i1).slice(1, j0..j0 + 1));
512            let nr = dup(g.jnr.slice(0, i0..i1).slice(1, j0..j0 + 1));
513            let src = rec(|k| p[k].slice(0, i0..i1).slice(1, j0..).pad([(0, 0), (2, 0)], PadMode::Mirror).slice(1, 0..2));
514            w.slice_mut(0, i0 + 2..i1 + 2).slice(1, j0..j0 + 2).assign(self.block_wall_ghost(&src, &nx, &nr, i0)).expect("block ghost assign: shapes fixed by set_block");
515        }
516    }

518    /// The face fluxes next to the solid block, written into the i / j flux records `fi`, `fj`: the solid faces are zero, and
519    /// the base-wall faces `i_end`, `i_end + 1` (rows `j_sting..j_end`) see the solid stencil cells as the reflection of the
520    /// fluid cells about the base face:
521    ///
522    /// ```text
523    /// face i_end:     (m2, m1, p0, p1) = (R c1, R c0, c0, c1)
524    /// face i_end + 1: (m2, m1, p0, p1) = (R c0,  c0, c1, c2)         c_k the cell i_end + k, R the reflection
525    /// ```
526    fn block_fluxes(&self, b: Block, w: &Rec, fi: &mut Rec, fj: &mut Rec, lim: Limiter) {
527        let ni = self.ni;
528        let g = &self.geo;
529        // base wall faces: both faces use the normal of the base face for the reflection, their own for the flux
530        let rows = |a: &A, i0: usize, i1: usize| a.slice(0, i0..i1).slice(1, b.j_sting..b.j_end);
531        let (bx, br) = (rows(&g.inx, b.i_end, b.i_end + 1), rows(&g.inr, b.i_end, b.i_end + 1));
532        let (fx, fr) = (rows(&g.inx, b.i_end, b.i_end + 2), rows(&g.inr, b.i_end, b.i_end + 2));
533        let cells = |k: usize| rec(|f| w[f].slice(0, b.i_end + 2 + k..b.i_end + 3 + k).slice(1, b.j_sting + 2..b.j_end + 2));
534        let (c0, c1, c2) = (cells(0), cells(1), cells(2));
535        let join = |a: &Rec, bb: &Rec| rec(|f| concat(0, [&a[f], &bb[f]]));
536        let (rc0, rc1) = (self.base_wall_ghost(&c0, &bx, &br), self.base_wall_ghost(&c1, &bx, &br));
537        let (m2, m1) = (join(&rc1, &rc0), join(&rc0, &c0));
538        // nozzle rows: the exit state replaces the ghosts: face `i_end` sees (jet, jet), face `i_end + 1` sees (jet, c0)
539        let (m2, m1) = match (&self.nozzle, self.lb.as_ref().and_then(|lb| lb.jet_rows.as_ref())) {
540            (Some(n), Some([both, face0])) => {
541                let jw = n.state;
542                let m2 = rec(|k| select(both, c(jw[k]), &m2[k]));
543                let m1 = join(&rec(|k| select(face0, c(jw[k]), &rc0[k])), &c0);
544                (m2, m1)
545            }
546            _ => (m2, m1),
547        };
548        let (p0, p1) = (join(&c0, &c1), join(&c1, &c2));
549        let base = face_flux_lm(&m2, &m1, &p0, &p1, lim, &fx, &fr, self.low_mach, self.staging == Staging::Split);
550        // solid faces
551        self.zero_region(fi, 0..ni + 1, 0..b.j_sting);
552        self.zero_region(fi, 0..b.i_end, b.j_sting..b.j_end);
553        fi.slice_mut(0, b.i_end..b.i_end + 2).slice(1, b.j_sting..b.j_end).assign(&base).expect("base wall flux assign: shapes fixed by set_block");
554        self.zero_region(fj, 0..b.i_end, 0..b.j_end);
555        self.zero_region(fj, b.i_end..ni, 0..b.j_sting);
556    }

576    /// Writes the two ghost layers of boundary `side` (`[imin, imax, jmin, jmax]`) into the padded primitives `w` with a
577    /// region assign. `p` is the unpadded primitive state: the first interior cell `edge` and the cell `mirror` that a
578    /// wall reflects (`pad` with `Edge` / `Mirror` read both). `Bc::Extrapolate` keeps the `Edge` padding of `w`. Layers
579    /// are outermost first, as the cells they stand for. `w` is a lazy expression, so nothing is launched: `w` becomes the
580    /// padded state with the ghost expressions in place, and fuses into whatever reads it.
581    fn assign_ghosts(&self, side: usize, p: &Rec, w: &mut Rec) {
582        let (ni, nj) = (self.ni, self.nj);
583        let view = |a: &A, mode: PadMode| -> A {
584            match side {
585                0 => a.pad([(2, 0)], mode).slice(0, 0..2),
586                1 => a.pad([(0, 2)], mode).slice(0, ni..ni + 2),
587                2 => a.pad([(0, 0), (2, 0)], mode).slice(1, 0..2),
588                _ => a.pad([(0, 0), (0, 2)], mode).slice(1, nj..nj + 2),
589            }
590        };
591        let mut region = match side {
592            0 => w.slice_mut(0, ..2).slice(1, 2..nj + 2),
593            1 => w.slice_mut(0, ni + 2..).slice(1, 2..nj + 2),
594            2 => w.slice_mut(1, ..2).slice(0, 2..ni + 2),
595            _ => w.slice_mut(1, nj + 2..).slice(0, 2..ni + 2),
596        };
597        let b = &self.geo.bnd[side];
598        let fs = self.cfg.freestream;
599        let ghost = match self.cfg.bc[side] {
600            Bc::Freestream => b.fixed.clone().expect("a free-stream boundary carries its fixed ghost, set in the constructor"),
601            Bc::Extrapolate => return,
602            Bc::SlipWall => self.slip_wall_ghost(side, &rec(|k| view(&p[k], PadMode::Mirror)), &b.nx, &b.nr),
603            Bc::Characteristic => characteristic_ghost(&fs, &rec(|k| view(&p[k], PadMode::Edge)), &b.onx, &b.onr),
604            Bc::PressureOutflow => pressure_outflow_ghost(fs[3], &rec(|k| view(&p[k], PadMode::Edge)), &b.onx, &b.onr),
605        };
606        region.assign(ghost).expect("ghost layer assign: ghost shapes fixed by the boundary");
607    }

609    /// `res = dU/dt` of the conserved state `u` (`[ni, nj]` fields) and, if asked, the local time step.
610    ///
611    /// ```text
612    /// dU/dt = -(1/V) sum_faces (len r_mid) F.n  +  (0, 0, p A / V, 0)        dt = cfl A / (1/2 sum_faces (|u.n| + c) len)
613    /// ```
614    pub fn eval_residual(&self, u: &Rec, lim: Limiter, compute_dt: bool) -> (Rec, Option<A>) {
615        let st = self.eval_stage(u, self.state_mu_t().as_ref(), lim, compute_dt);
616        (st.res, st.dt)
617    }

619    /// [`LazySolver::eval_residual`] with the eddy viscosity `mu_t` of an SST run (`None`: zero) and everything the turbulence update needs
620    /// (the viscous quantities, the primitives and the mass flow of the faces).
621    fn eval_stage(&self, u: &Rec, mu_t: Option<&A>, lim: Limiter, compute_dt: bool) -> Stage {
622        let (p, w) = self.padded(u);
623        self.stage_from(p, w, mu_t, lim, compute_dt)
624    }

626    /// Steps 2 to 5 of `eval_residual` for the primitives `p` and the padded primitives `w`, with the viscous terms of a viscous run.
627    fn stage_from(&self, p: Rec, w: Rec, mu_t: Option<&A>, lim: Limiter, compute_dt: bool) -> Stage {
628        let (fi, fj) = self.face_fluxes(&w, lim);
629        let vf = self.viscous_flow(&p, &w, mu_t);
630        let (res, dt) = self.cell_residual(&p, &fi, &fj, vf.as_ref(), compute_dt);
631        Stage { res, dt, mass: [fi[0].clone(), fj[0].clone()], p, w, vf }
632    }

634    /// Steps 1 of `eval_residual`: the primitives `p = cons_to_prim(u)` and the padded primitives `w` with the domain ghosts,
635    /// then the block ghosts, in place (the `Staging` cache applied).
636    fn padded(&self, u: &Rec) -> (Rec, Rec) {
637        // 1. primitives, padded by two cells on every side; the ghost layers of each boundary are one region assign
638        let p = cons_to_prim(u.clone());
639        let mut w = p.pad([(2, 2), (2, 2)], PadMode::Edge);
640        for side in 0..4 {
641            self.assign_ghosts(side, &p, &mut w);
642        }
643        if let Some(b) = self.block {
644            self.assign_block_ghosts(b, &p, &mut w);
645        }
646        // `Lines` and up: evaluate the padded record once. Without it every reader recomputes the primitives and all four ghost
647        // expressions at each stencil cell (200x100 f32: 67-91 launches per RK3 step without it, 26-30 with it; README "Speed").
648        let w = if self.staging == Staging::Fusion { w } else { w.cache() };
649        (p, w)
650    }

652    /// Steps 2 and 3 of `eval_residual`: the face fluxes of the padded primitives `w`.
653    fn face_fluxes(&self, w: &Rec, lim: Limiter) -> (Rec, Rec) {
654        let (ni, nj) = (self.ni, self.nj);
655        let g = &self.geo;
656
657        // 2. the cells along each grid line with their ghosts: i lines `[ni + 4, nj]`, j lines `[ni, nj + 4]`
658        let wi = w.slice(1, 2..nj + 2);
659        let wj = w.slice(0, 2..ni + 2);
660
661        // 3. face fluxes: face `f` sees the cells `f - 2 .. f + 1` of its line
662        let split = self.staging == Staging::Split;
663        let mut fi = face_flux_lm(&wi.slice(0, 0..ni + 1), &wi.slice(0, 1..ni + 2), &wi.slice(0, 2..ni + 3), &wi.slice(0, 3..ni + 4), lim, &g.inx, &g.inr, self.low_mach, split);
664        let mut fj = face_flux_lm(&wj.slice(1, 0..nj + 1), &wj.slice(1, 1..nj + 2), &wj.slice(1, 2..nj + 3), &wj.slice(1, 3..nj + 4), lim, &g.jnx, &g.jnr, self.low_mach, split);
665        if let Some(b) = self.block {
666            self.block_fluxes(b, w, &mut fi, &mut fj, lim);
667        }
668        // `Fluxes` and up: evaluate the face fluxes. The residual and the stage update then fuse into the next stage's padded
669        // record: three kernels per stage, 14 launches per step instead of 26-30 (the flux kernel read through four stencil
670        // views is cut at the 1024-operation fusion limit otherwise; README "Speed").
671        let (fi, fj) = if matches!(self.staging, Staging::Fluxes | Staging::Split) { (fi.cache(), fj.cache()) } else { (fi, fj) };
672        (fi, fj)
673    }

675    /// Steps 4 and 5 of `eval_residual`: the cell residual and the time step from the face fluxes `fi`, `fj` and the primitives `p`. A viscous
676    /// run (`vf`) uses the face flux `F - Fv` and the radial-momentum source `(p - t_tt) A / V`, and adds the diffusive limit to the time step.
677    fn cell_residual(&self, p: &Rec, fi: &Rec, fj: &Rec, vf: Option<&VFlow>, compute_dt: bool) -> (Rec, Option<A>) {
678        let (ni, nj) = (self.ni, self.nj);
679        let g = &self.geo;
680        let (fi, fj) = (self.total_flux(fi, vf.map(|v| &v.fvi)), self.total_flux(fj, vf.map(|v| &v.fvj)));
681
682        // 4. cell residual
683        let (wr, wl) = (g.iw.slice(0, 1..ni + 1), g.iw.slice(0, 0..ni));
684        let (wt, wb) = (g.jw.slice(1, 1..nj + 1), g.jw.slice(1, 0..nj));
685        let mut res = rec(|k| {
686            let (gr, gl) = (fi[k].slice(0, 1..ni + 1), fi[k].slice(0, 0..ni));
687            let (gt, gb) = (fj[k].slice(1, 1..nj + 1), fj[k].slice(1, 0..nj));
688            -(&gr * &wr - &gl * &wl + &gt * &wt - &gb * &wb) * &g.inv_v
689        });
690        res[2] = &res[2] + self.hoop_source(p, vf); // axisymmetric pressure source p A / V
691
692        // 5. local time step
693        let mut dt = compute_dt.then(|| {
694            let cs = (c(GAMMA) * &p[3] / &p[0]).sqrt();
695            let lam = |nx: &A, nr: &A, len: &A| ((&p[1] * nx + &p[2] * nr).abs() + &cs) * len;
696            let (fl_nx, fl_nr, fl_len) = (g.inx.slice(0, 0..ni), g.inr.slice(0, 0..ni), g.ilen.slice(0, 0..ni));
697            let (fr_nx, fr_nr, fr_len) = (g.inx.slice(0, 1..ni + 1), g.inr.slice(0, 1..ni + 1), g.ilen.slice(0, 1..ni + 1));
698            let (fb_nx, fb_nr, fb_len) = (g.jnx.slice(1, 0..nj), g.jnr.slice(1, 0..nj), g.jlen.slice(1, 0..nj));
699            let (ft_nx, ft_nr, ft_len) = (g.jnx.slice(1, 1..nj + 1), g.jnr.slice(1, 1..nj + 1), g.jlen.slice(1, 1..nj + 1));
700            let sum = lam(&fl_nx, &fl_nr, &fl_len) + lam(&fr_nx, &fr_nr, &fr_len) + lam(&fb_nx, &fb_nr, &fb_len) + lam(&ft_nx, &ft_nr, &ft_len);
701            let half = c(0.5) * &sum;
702            c(self.cfg.cfl) * &g.area / vf.map_or(half.clone(), |v| {
703                let (xi, xj) = self.diffusion_rates(v, &p[0]);
704                &half + xi + xj
705            })
706        });
707        // solid cells: residual and time step zero (region assigns, so nothing computed inside the solid reaches the state)
708        if let Some(b) = self.block {
709            self.zero_region(&mut res, 0..b.i_end, 0..b.j_end);
710            self.zero_region(&mut res, b.i_end..ni, 0..b.j_sting);
711            if let Some(dt) = &mut dt {
712                if b.j_sting > 0 {
713                    dt.slice_mut(0, b.i_end..ni).slice(1, 0..b.j_sting).assign(c(0.0)).expect("dt zero: block rows inside the grid");
714                }
715                dt.slice_mut(0, 0..b.i_end).slice(1, 0..b.j_end).assign(c(0.0)).expect("dt zero: block rows inside the grid");
716            }
717        }
718        (res, dt)
719    }

721    /// One SSP-RK3 step with local time stepping; returns the new state and the density residual of the first stage
722    /// `sqrt(sum (d rho / dt)^2 / N)`.
723    ///
724    /// ```text
725    /// u1 = u0 + dt L(u0),   u2 = 3/4 u0 + 1/4 (u1 + dt L(u1)),   u3 = 1/3 u0 + 2/3 (u2 + dt L(u2))
726    /// ```
727    fn rk3(&self, u0: &Rec, mu_t: Option<&A>, lim: Limiter) -> (Rec, A, Stage) {
728        let st0 = self.eval_stage(u0, mu_t, lim, true);
729        let (r0, dt) = (st0.res, st0.dt.expect("the residual stage was called with compute_dt = true"));
730        let rn = self.residual_norm(&r0[0]);
731        // stage 1: u1 = u0 + dt L(u0)
732        let u1 = rec(|k| &u0[k] + &dt * &r0[k]);
733        let r1 = self.eval_stage(&u1, mu_t, lim, false).res;
734        // stage 2: u2 = 3/4 u0 + 1/4 (u1 + dt L(u1))
735        let u2 = rec(|k| c(0.25) * (&u1[k] + &dt * &r1[k]) + c(0.75) * &u0[k]);
736        let st2 = self.eval_stage(&u2, mu_t, lim, false);
737        let r2 = &st2.res;
738        // stage 3: u3 = 1/3 u0 + 2/3 (u2 + dt L(u2))
739        let u3 = rec(|k| c(2.0 / 3.0) * (&u2[k] + &dt * &r2[k]) + c(1.0 / 3.0) * &u0[k]);
740        (u3, rn, Stage { dt: Some(dt), ..st2 })
741    }
742
743    fn limiter(&self, iter: usize) -> Limiter {
744        if iter < self.cfg.first_order_iters {
745            Limiter::FirstOrder
746        } else {
747            self.cfg.limiter
748        }
749    }
750
751    /// Number of iterations from `self.iter` that use the same limiter (the traced step is specific to the limiter).
752    fn phase_len(&self, left: usize) -> usize {
753        if self.iter < self.cfg.first_order_iters {
754            left.min(self.cfg.first_order_iters - self.iter)
755        } else {
756            left
757        }
758    }
759
760    /// `n` iterations as one `iterate_until_all` loop per limiter phase (the step is traced once and its kernels replayed,
761    /// the density residual recorded on the GPU every iteration, the host waits once per `sync` iterations); returns the
762    /// residual history.
763    pub fn run(&mut self, n: usize, sync: usize) -> Result<Vec<f64>> {
764        let mut history = Vec::with_capacity(n);
765        let mut left = n;
766        while left > 0 {
767            let seg = self.phase_len(left);
768            let lim = self.limiter(self.iter);
769            let slot = usize::from(lim != Limiter::FirstOrder);
770            let init = self.state.take().ok_or_else(no_state)?;
771            let hit = matches!(&self.compiled[slot], Some((st, m, ce, _)) if *st == self.staging && seg <= *m && *ce == sync);
772            if !hit {
773                let max_s = sync.max(seg);
774                self.compiled[slot] = match luma::compile_until_all(&init, records(luma::Until::new(max_s).check_every(sync)), |s, _| self.step(s, lim)) {
775                    Ok(cu) => Some((self.staging, max_s, sync, cu)),
776                    Err(e) if matches!(e.root(), luma::LumaError::Unsupported(_) | luma::LumaError::Chunked(_)) => None,
777                    Err(e) => return Err(e),
778                };
779            }
780            let run = match &mut self.compiled[slot] {
781                Some((.., cu)) => cu.run_steps(init, seg)?,
782                None => luma::iterate_until_all(init, records(luma::Until::new(seg).check_every(sync)), |s, _| self.step(s, lim))?,
783            };
784            history.extend(read_history(&run)?);
785            self.state = Some(run.state);
786            self.iter += seg;
787            left -= seg;
788        }
789        Ok(history)
790    }
791
792    /// Reads only the boundary stencil cells of `wall_distribution`, `friction` and `base_state` into `u` (`j * ni + i`): the first two rows
793    /// of the body wall (and the two columns behind the base face with a block).
794    pub fn read_boundary_into(&self, u: &mut [[f64; 4]]) -> Result<()> {
795        let s = self.state.as_ref().ok_or_else(no_state)?;
796        let (ni, nj) = (self.ni, self.nj);
797        match self.block {
798            None => read_rect_into(s, ni, 0..ni, 0..2.min(nj), u),
799            Some(b) => {
800                read_rect_into(s, ni, 0..b.i_end, b.j_end..(b.j_end + 2).min(nj), u)?;
801                read_rect_into(s, ni, b.i_end..(b.i_end + 2).min(ni), b.j_sting..b.j_end, u)?;
802                if b.j_sting > 0 {
803                    read_rect_into(s, ni, b.i_end..ni, b.j_sting..(b.j_sting + 1).min(nj), u)?;
804                }
805                Ok(())
806            }
807        }
808    }
809
810    /// `n` iterations as one `iterate_all` replay per limiter phase (no residual history).
811    pub fn run_replay(&mut self, n: usize) -> Result<()> {
812        let mut left = n;
813        while left > 0 {
814            let seg = self.phase_len(left);
815            let lim = self.limiter(self.iter);
816            let init = self.state.take().ok_or_else(no_state)?;
817            self.state = Some(luma::iterate_all(init.each_ref(), seg, |s| {
818                self.step(s, lim)
819            }));
820            self.iter += seg;
821            left -= seg;
822        }
823        Ok(())
824    }
825
826    /// `n` iterations calling `rk3` every iteration and evaluating the new state with `eval_all` (no replay). Returns the
827    /// evaluated residual scalars (read them with [`Residuals::read`]).
828    pub fn run_eval(&mut self, n: usize) -> Result<Residuals> {
829        let mut history = Vec::with_capacity(n);
830        let mut s = self.state.take().ok_or_else(no_state)?;
831        for _ in 0..n {
832            s = self.step(&s, self.limiter(self.iter));
833            luma::eval_all(&s.iter().collect::<Vec<_>>())?;
834            history.push(s[4].clone());
835            self.iter += 1;
836        }
837        self.state = Some(s);
838        Ok(Residuals(history))
839    }
840
841    /// Waits for the GPU.
842    pub fn sync(&self) -> Result<()> {
843        self.ctx.sync()
844    }
845
846    /// The residual `dU/dt` and time step of the current state, on the host (`[res; 4]` of `ni * nj`, `dt`).
847    pub fn residual_f64(&self, lim: Limiter) -> Result<(Vec<[f64; 4]>, Vec<f64>)> {
848        let u = unpack(self.state.as_ref().ok_or_else(no_state)?);
849        let (r, dt) = self.eval_residual(&u, lim, true);
850        let r: Vec<Vec<f64>> = r.iter().map(to_f64).collect::<Result<_>>()?;
851        let dt = to_f64(&dt.ok_or_else(|| luma::LumaError::Unsupported("residual stage returned no time step".into()))?)?;
852        Ok(((0..self.ni * self.nj).map(|c| [r[0][c], r[1][c], r[2][c], r[3][c]]).collect(), dt))
853    }

855    /// GPU time per kernel of one residual evaluation (profile) and where fusion cut the expressions.
856    pub fn profile_residual(&self) -> Result<luma::Profile> {
857        let u = unpack(self.state.as_ref().ok_or_else(no_state)?);
858        let lim = self.cfg.limiter;
859        // warm up (compile) first, so that the profile shows launches only
860        let _ = self.eval_residual(&u, lim, true);
861        self.ctx.sync()?;
862        let (out, profile) = self.ctx.profile(|| -> Result<()> {
863            let (r, dt) = self.eval_residual(&u, lim, true);
864            luma::eval_all(&r.iter().chain(dt.iter()).collect::<Vec<_>>())?;
865            self.ctx.sync()
866        })?;
867        out?;
868        Ok(profile)
869    }

872/// The density residual scalars of an `eval` loop, one per iteration, as evaluated GPU arrays.
873pub struct Residuals(Vec<A>);
874
875impl Residuals {
876    /// Reads the residual history (waits for the GPU).
877    pub fn read(&self) -> Result<Vec<f64>> {
878        self.0.iter().map(|a| Ok(to_f64(a)?[0])).collect()
879    }
880}

882impl LazySolver {
883    /// GPU time per kernel of one full SSP-RK3 step (three residual evaluations and the stage updates) as the `eval` loop
884    /// launches it, and where fusion cut the expressions.
885    pub fn profile_step(&mut self) -> Result<luma::Profile> {
886        self.run_eval(1)?;
887        self.ctx.sync()?;
888        let ctx = self.ctx.clone();
889        let (out, profile) = ctx.profile(|| -> Result<()> {
890            self.run_eval(1)?;
891            self.ctx.sync()
892        })?;
893        out?;
894        Ok(profile)
895    }
896
897    /// GPU time per kernel of `n` iterations of the `iterate_all` replay (the profile adds up all `n`), after a warm-up call
898    /// that traces and compiles it.
899    pub fn profile_replay(&mut self, n: usize) -> Result<luma::Profile> {
900        self.run_replay(n)?;
901        self.ctx.sync()?;
902        let ctx = self.ctx.clone();
903        let (out, profile) = ctx.profile(|| -> Result<()> {
904            self.run_replay(n)?;
905            self.ctx.sync()
906        })?;
907        out?;
908        Ok(profile)
909    }
910
911    /// Host seconds to build the lazy expression of one SSP-RK3 step from the current state, `n` times, without evaluating
912    /// anything (the part of an `eval` loop iteration that is Rust code building nodes).
913    pub fn trace_seconds(&self, n: usize) -> f64 {
914        let u = unpack(self.state.as_ref().expect("trace_seconds needs a state: GpuSolver::from_host sets it"));
915        let lim = self.cfg.limiter;
916        let t0 = std::time::Instant::now();
917        for _ in 0..n {
918            std::hint::black_box(self.rk3(&u, self.state_mu_t().as_ref(), lim));
919        }
920        t0.elapsed().as_secs_f64()
921    }
922}
