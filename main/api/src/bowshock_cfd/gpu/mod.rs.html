1//! The GPU solver (feature `gpu`): the axisymmetric Euler solver of the CPU modules (MUSCL + HLLC / Rusanov shock blend + optional low-Mach
2//! correction, characteristic and pressure-outflow far fields, SSP-RK3 with local time stepping or the line-implicit scheme, optional
3//! solid body block) written in plain Luma array ops: no custom kernels. Luma's lazy arrays fuse the expressions; `iterate_until_all`
4//! traces one step and replays its kernels, recording the density residual on the GPU.
5//!
6//! The solver is in `body.rs` (explicit scheme) and `implicit.rs` (line-implicit scheme), each function the mirror of the CPU routine of the
7//! same name in the CPU modules of this crate, included once per precision:
8//!
9//! * [`f32s`]: `Array<f32>` everywhere ([`Precision::F32`]).
10//! * [`df32s`]: the state, the residual and the time step in double-float `Array<Df32>` ([`Precision::Extended`]); the
11//!   state is a `Df32` record through the loops.
12//!
13//! The line-implicit scheme's left-hand side is f32 in both. The viscous terms of the compressible Navier-Stokes equations (`viscous.rs`) and the
14//! Menter SST k-omega model (`turbulence.rs`) follow, for both schemes and for grids with a solid block; their left-hand sides (the `2 lam I` of
15//! the implicit operator, the line solves of the two turbulence scalars) are f32 like the rest. [`GpuSolver`] is the precision-independent handle
16//! the other crates use. See the README.
17//!
18//! The 3D Euler solver of `solver3.rs` is in `body3.rs` (SSP-RK3) and `implicit3.rs` (three-factor line-implicit scheme), [`f32s::LazySolver3`] /
19//! [`df32s::LazySolver3`], behind [`GpuSolver3`].

29/// Errors of the GPU solver: a Luma failure or a configuration the GPU solver does not implement.
30#[derive(Debug, Clone)]
31pub enum GpuError {
32    /// Failure reported by Luma (adapter, device, kernel build, buffer transfer).
33    Luma(LumaError),
34    /// The requested scheme / block / viscous combination is not implemented on the GPU.
35    Unsupported(&'static str),
36}
37
38impl From<LumaError> for GpuError {
39    fn from(e: LumaError) -> GpuError {
40        GpuError::Luma(e)
41    }
42}

44impl std::fmt::Display for GpuError {
45    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
46        match self {
47            GpuError::Luma(e) => write!(f, "{e}"),
48            GpuError::Unsupported(m) => f.write_str(m),
49        }
50    }
51}

58/// Precision of the state, the residual and the time step.
59#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
60pub enum Precision {
61    /// `Array<f32>` everywhere ([`f32s`]); the density residual stalls about 4-4.6 orders below its peak at f32 round-off.
62    F32,
63    /// Double-float (`Array<Df32>`, ~48 bit) state, residual and time step ([`df32s`]): converges like the f64 CPU solver.
64    #[default]
65    Extended,
66}

68/// Iterations of one `iterate_until_all` submission inside [`GpuSolver::run`]: the host waits once per `RUN_SYNC` iterations.

71/// How much of the order of evaluation is left to Luma's fusion engine (`LazySolver::staging`, see README).
72#[derive(Clone, Copy, Debug, PartialEq, Eq)]
73pub enum Staging {
74    /// No `cache()` anywhere: the whole step is one lazy expression and Luma's fusion decides where to cut.
75    Fusion,
76    /// The padded primitive record (state with its ghost layers) is evaluated once per residual evaluation.
77    Lines,
78    /// Additionally the face fluxes are evaluated: three kernels per Runge-Kutta stage (padded state, i fluxes, j fluxes),
79    /// the stage update and the residual fuse into the padded-state kernel of the next stage.
80    Fluxes,
81    /// Additionally the reconstructed states and the shock weight are evaluated before the Riemann solvers.
82    Split,
83}

85/// Single precision.
86pub mod f32s {
87    pub type S = f32;
88
89    fn c(x: f64) -> S {
90        x as f32
91    }
92
93    /// Host data (row-major, `i` fastest) as an array of `shape`.
94    fn up(ctx: &Context, data: &[f64], shape: impl Into<luma::Shape>) -> Result<A> {
95        let v: Vec<f32> = data.iter().map(|&x| x as f32).collect();
96        Array::from_slice_on(ctx, &v, shape)
97    }
98
99    /// Measured fastest (README, speed table): `Fluxes` is 14 launches per step, `Fusion` (no `cache()`) 67-91.
100    const DEFAULT_STAGING: Staging = Staging::Fluxes;
101
102    fn to_f64(a: &A) -> Result<Vec<f64>> {
103        Ok(a.to_vec()?.into_iter().map(f64::from).collect())
104    }
105
106    /// The density residual of every step, recorded on the GPU.
107    fn records(until: luma::Until<'static, State>) -> luma::Until<'static, State> {
108        until.record("residual", |s: &State| s[4].clone())
109    }
110
111    fn read_history(run: &luma::Run<State>) -> Result<Vec<f64>> {
112        let h = run.history("residual").ok_or_else(|| luma::LumaError::Unsupported("the run recorded no residual history".into()))?;
113        Ok(h.iter().map(|&r| f64::from(r)).collect())
114    }
115
116    /// The density residual of every step of a 3D run, recorded on the GPU (`State3` holds it at index 5).
117    fn records3(until: luma::Until<'static, State3>) -> luma::Until<'static, State3> {
118        until.record("residual", |s: &State3| s[5].clone())
119    }
120
121    fn read_history3(run: &luma::Run<State3>) -> Result<Vec<f64>> {
122        let h = run.history("residual").ok_or_else(|| luma::LumaError::Unsupported("the run recorded no residual history".into()))?;
123        Ok(h.iter().map(|&r| f64::from(r)).collect())
124    }
125
126    /// The left-hand side of the implicit scheme runs in f32: the f32 view of `a` (here `a` itself) and back.
127    fn lhs(a: &A) -> Array<f32> {
128        a.clone()
129    }
130
131    fn from_lhs(a: &Array<f32>) -> A {
132        a.clone()
133    }

144/// Double-float state, residual and time step.
145pub mod df32s {
146    use luma::Df32;
147
148    pub type S = Df32;
149
150    fn c(x: f64) -> S {
151        Df32::from(x)
152    }
153
154    fn up(ctx: &Context, data: &[f64], shape: impl Into<luma::Shape>) -> Result<A> {
155        let v: Vec<Df32> = data.iter().map(|&x| Df32::from(x)).collect();
156        Array::from_slice_on(ctx, &v, shape)
157    }
158
159    /// Measured fastest (README, speed table): `Fluxes` is 14 launches per step, `Fusion` (no `cache()`) 67-91.
160    const DEFAULT_STAGING: Staging = Staging::Fluxes;
161
162    fn to_f64(a: &A) -> Result<Vec<f64>> {
163        Ok(a.to_vec()?.into_iter().map(|d| d.to_f64()).collect())
164    }
165
166    /// The density residual of every step, recorded on the GPU as two `f32` values: the history of `iterate_until_all` is
167    /// `f32`, so the double-float residual `hi + lo` is recorded as its high word and its low word.
168    fn records(until: luma::Until<'static, State>) -> luma::Until<'static, State> {
169        until.record("residual", |s: &State| s[4].to_f32()).record("residual_lo", |s: &State| (&s[4] - s[4].to_f32().to_df32()).to_f32())
170    }
171
172    fn read_history(run: &luma::Run<State>) -> Result<Vec<f64>> {
173        let missing = || luma::LumaError::Unsupported("the run recorded no residual history".into());
174        let (hi, lo) = (run.history("residual").ok_or_else(missing)?, run.history("residual_lo").ok_or_else(missing)?);
175        Ok(hi.iter().zip(lo).map(|(&h, &l)| f64::from(h) + f64::from(l)).collect())
176    }
177
178    /// The 3D counterpart of `records`: the double-float residual (`State3` index 5) as its high and low word.
179    fn records3(until: luma::Until<'static, State3>) -> luma::Until<'static, State3> {
180        until.record("residual", |s: &State3| s[5].to_f32()).record("residual_lo", |s: &State3| (&s[5] - s[5].to_f32().to_df32()).to_f32())
181    }
182
183    fn read_history3(run: &luma::Run<State3>) -> Result<Vec<f64>> {
184        let missing = || luma::LumaError::Unsupported("the run recorded no residual history".into());
185        let (hi, lo) = (run.history("residual").ok_or_else(missing)?, run.history("residual_lo").ok_or_else(missing)?);
186        Ok(hi.iter().zip(lo).map(|(&h, &l)| f64::from(h) + f64::from(l)).collect())
187    }
188
189    /// The left-hand side of the implicit scheme runs in f32: the f32 view of `a` and back.
190    fn lhs(a: &A) -> Array<f32> {
191        a.to_f32()
192    }
193
194    fn from_lhs(a: &Array<f32>) -> A {
195        a.to_df32()
196    }

212/// Runs `$e` on the solver `$s` of whichever precision `$self` holds.
213macro_rules! on {
214    ($self:expr, $s:ident => $e:expr) => {
215        match &$self.inner {
216            Inner::F32($s) => $e,
217            Inner::Ext($s) => $e,
218        }
219    };
220}

222macro_rules! on_mut {
223    ($self:expr, $s:ident => $e:expr) => {
224        match &mut $self.inner {
225            Inner::F32($s) => $e,
226            Inner::Ext($s) => $e,
227        }
228    };

242    /// Uploads the geometry (rounded from the grid's f64 metrics), starts from the free stream.
243    pub fn new(ctx: &Context, grid: &Grid, cfg: &Config, prec: Precision) -> Result<GpuSolver> {
244        let inner = match prec {
245            Precision::F32 => Inner::F32(f32s::LazySolver::new(ctx, grid, cfg)?),
246            Precision::Extended => Inner::Ext(df32s::LazySolver::new(ctx, grid, cfg)?),
247        };
248        Ok(GpuSolver { inner, ctx: ctx.clone(), prec, ni: grid.ni, nj: grid.nj })
249    }

251    /// The solver of a host `crate::solver::Solver`: its grid and config, low-Mach flag, solid block, scheme (`Scheme::Explicit` or
252    /// `Scheme::Implicit`), conserved state, iteration count and, for a viscous host, the viscous parameters with the turbulence state and the
253    /// wall distance (the SST state is copied from `host.turb` / `host.mu_t`).
254    pub fn from_host(ctx: &Context, host: &Solver, prec: Precision) -> Result<GpuSolver> {
255        if !matches!(host.scheme(), Scheme::Explicit | Scheme::Implicit(_)) {
256            return Err(GpuError::Unsupported("only Scheme::Explicit and Scheme::Implicit are implemented on the GPU"));
257        }
258        let inner = match prec {
259            Precision::F32 => Inner::F32(f32s::LazySolver::from_host(ctx, host)?),
260            Precision::Extended => Inner::Ext(df32s::LazySolver::from_host(ctx, host)?),
261        };
262        Ok(GpuSolver { inner, ctx: ctx.clone(), prec, ni: host.grid.ni, nj: host.grid.nj })
263    }

277    /// Replaces the configuration: free stream, CFL, limiter, `first_order_iters` and the boundary conditions may change between runs
278    /// (a Mach sweep) without touching the state. The next [`GpuSolver::run`] uses it (the fixed ghost states are uploaded again).
279    pub fn set_config(&mut self, cfg: &Config) -> Result<()> {

282        }

349    /// Switches the viscous terms on (`Some`, with the SST model if `v.turbulence` says so) or off (`None`, the Euler equations). `grid` is the grid

354        }
355        on_mut!(self, s => s.set_viscous(v, grid)?);
356        Ok(())

505    pub fn set_state(&mut self, u: &[Cons3]) -> Result<()> {
506        on3_mut!(self, s => s.set_state(u)?);
507        Ok(())
508    }

510    /// Resets to the free stream of the current configuration at iteration 0, the CFL ramp restarted.
511    pub fn reset(&mut self) -> Result<()> {
512        let fs = crate::gas3::prim_to_cons3(&self.config().freestream);
513        self.set_state(&vec![fs; self.ni * self.nj * self.nk])
514    }
