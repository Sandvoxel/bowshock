1//! The GPU solver (feature `gpu`): the axisymmetric Euler solver of the CPU modules (MUSCL + HLLC / Rusanov shock blend + optional low-Mach
2//! correction, characteristic and pressure-outflow far fields, SSP-RK3 with local time stepping or the line-implicit scheme, optional
3//! solid body block) written in plain Luma array ops: no custom kernels. Luma's lazy arrays fuse the expressions; `iterate_until_all`
4//! traces one step and replays its kernels, recording the density residual on the GPU.
5//!
6//! The solver is in `body.rs` (explicit scheme) and `implicit.rs` (line-implicit scheme), each function the mirror of the CPU routine of the
7//! same name in the CPU modules of this crate, included once per precision:
8//!
9//! * [`f32s`]: `Array<f32>` everywhere ([`Precision::F32`]).
10//! * [`df32s`]: the state, the residual and the time step in double-float `Array<Df32>` ([`Precision::Extended`]); the
11//!   state is a `Df32` record through the loops.
12//!
13//! The line-implicit scheme's left-hand side is f32 in both. The viscous terms of the compressible Navier-Stokes equations (`viscous.rs`) and the
14//! Menter SST k-omega model (`turbulence.rs`) follow, for both schemes and for grids with a solid block; their left-hand sides (the `2 lam I` of
15//! the implicit operator, the line solves of the two turbulence scalars) are f32 like the rest. [`GpuSolver`] is the precision-independent handle
16//! the other crates use. See the README.
17

23/// Errors of the GPU solver: a Luma failure or a configuration the GPU solver does not implement.
24#[derive(Debug, Clone)]
25pub enum GpuError {
26    /// Failure reported by Luma (adapter, device, kernel build, buffer transfer).
27    Luma(LumaError),
28    /// The requested scheme / block / viscous combination is not implemented on the GPU.
29    Unsupported(&'static str),
30}
31
32impl From<LumaError> for GpuError {
33    fn from(e: LumaError) -> GpuError {
34        GpuError::Luma(e)
35    }
36}
37
38impl std::fmt::Display for GpuError {
39    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
40        match self {
41            GpuError::Luma(e) => write!(f, "{e}"),
42            GpuError::Unsupported(m) => f.write_str(m),
43        }
44    }
45}

52/// Precision of the state, the residual and the time step.
53#[derive(Clone, Copy, Debug, PartialEq, Eq, Default)]
54pub enum Precision {
55    /// `Array<f32>` everywhere ([`f32s`]); the density residual stalls about 4-4.6 orders below its peak at f32 round-off.
56    F32,
57    /// Double-float (`Array<Df32>`, ~48 bit) state, residual and time step ([`df32s`]): converges like the f64 CPU solver.
58    #[default]
59    Extended,
60}
61
62/// Iterations of one `iterate_until_all` submission inside [`GpuSolver::run`]: the host waits once per `RUN_SYNC` iterations.
63const RUN_SYNC: usize = 50;
64
65/// How much of the order of evaluation is left to Luma's fusion engine (`LazySolver::staging`, see README).
66#[derive(Clone, Copy, Debug, PartialEq, Eq)]
67pub enum Staging {
68    /// No `cache()` anywhere: the whole step is one lazy expression and Luma's fusion decides where to cut.
69    Fusion,
70    /// The padded primitive record (state with its ghost layers) is evaluated once per residual evaluation.
71    Lines,
72    /// Additionally the face fluxes are evaluated: three kernels per Runge-Kutta stage (padded state, i fluxes, j fluxes),
73    /// the stage update and the residual fuse into the padded-state kernel of the next stage.
74    Fluxes,
75    /// Additionally the reconstructed states and the shock weight are evaluated before the Riemann solvers.
76    Split,
77}

79/// Single precision.
80pub mod f32s {
81    pub type S = f32;
82
83    fn c(x: f64) -> S {
84        x as f32
85    }
86
87    /// Host data (row-major, `i` fastest) as an array of `shape`.
88    fn up(ctx: &Context, data: &[f64], shape: impl Into<luma::Shape>) -> Result<A> {
89        let v: Vec<f32> = data.iter().map(|&x| x as f32).collect();
90        Array::from_slice_on(ctx, &v, shape)
91    }
92
93    /// Measured fastest (README, speed table): `Fluxes` is 14 launches per step, `Fusion` (no `cache()`) 67-91.
94    const DEFAULT_STAGING: Staging = Staging::Fluxes;
95
96    fn to_f64(a: &A) -> Result<Vec<f64>> {
97        Ok(a.to_vec()?.into_iter().map(f64::from).collect())
98    }
99
100    /// The density residual of every step, recorded on the GPU.
101    fn records(until: luma::Until<'static, State>) -> luma::Until<'static, State> {
102        until.record("residual", |s: &State| s[4].clone())
103    }
104
105    fn read_history(run: &luma::Run<State>) -> Result<Vec<f64>> {
106        let h = run.history("residual").ok_or_else(|| luma::LumaError::Unsupported("the run recorded no residual history".into()))?;
107        Ok(h.iter().map(|&r| f64::from(r)).collect())
108    }
109
110    /// The left-hand side of the implicit scheme runs in f32: the f32 view of `a` (here `a` itself) and back.
111    fn lhs(a: &A) -> Array<f32> {
112        a.clone()
113    }
114
115    fn from_lhs(a: &Array<f32>) -> A {
116        a.clone()
117    }
118
119    include!("body.rs");
120    include!("implicit.rs");
121    include!("viscous.rs");
122    include!("turbulence.rs");
123    include!("control.rs");
124}

126/// Double-float state, residual and time step.
127pub mod df32s {
128    use luma::Df32;
129
130    pub type S = Df32;
131
132    fn c(x: f64) -> S {
133        Df32::from(x)
134    }
135
136    fn up(ctx: &Context, data: &[f64], shape: impl Into<luma::Shape>) -> Result<A> {
137        let v: Vec<Df32> = data.iter().map(|&x| Df32::from(x)).collect();
138        Array::from_slice_on(ctx, &v, shape)
139    }
140
141    /// Measured fastest (README, speed table): `Fluxes` is 14 launches per step, `Fusion` (no `cache()`) 67-91.
142    const DEFAULT_STAGING: Staging = Staging::Fluxes;
143
144    fn to_f64(a: &A) -> Result<Vec<f64>> {
145        Ok(a.to_vec()?.into_iter().map(|d| d.to_f64()).collect())
146    }
147
148    /// The density residual of every step, recorded on the GPU as two `f32` values: the history of `iterate_until_all` is
149    /// `f32`, so the double-float residual `hi + lo` is recorded as its high word and its low word.
150    fn records(until: luma::Until<'static, State>) -> luma::Until<'static, State> {
151        until.record("residual", |s: &State| s[4].to_f32()).record("residual_lo", |s: &State| (&s[4] - s[4].to_f32().to_df32()).to_f32())
152    }
153
154    fn read_history(run: &luma::Run<State>) -> Result<Vec<f64>> {
155        let missing = || luma::LumaError::Unsupported("the run recorded no residual history".into());
156        let (hi, lo) = (run.history("residual").ok_or_else(missing)?, run.history("residual_lo").ok_or_else(missing)?);
157        Ok(hi.iter().zip(lo).map(|(&h, &l)| f64::from(h) + f64::from(l)).collect())
158    }
159
160    /// The left-hand side of the implicit scheme runs in f32: the f32 view of `a` and back.
161    fn lhs(a: &A) -> Array<f32> {
162        a.to_f32()
163    }
164
165    fn from_lhs(a: &Array<f32>) -> A {
166        a.to_df32()
167    }
168
169    include!("body.rs");
170    include!("implicit.rs");
171    include!("viscous.rs");
172    include!("turbulence.rs");
173    include!("control.rs");
174}

176enum Inner {
177    F32(f32s::LazySolver),
178    Ext(df32s::LazySolver),
179}

181/// Runs `$e` on the solver `$s` of whichever precision `$self` holds.
182macro_rules! on {
183    ($self:expr, $s:ident => $e:expr) => {
184        match &$self.inner {
185            Inner::F32($s) => $e,
186            Inner::Ext($s) => $e,
187        }
188    };
189}
190
191macro_rules! on_mut {
192    ($self:expr, $s:ident => $e:expr) => {
193        match &mut $self.inner {
194            Inner::F32($s) => $e,
195            Inner::Ext($s) => $e,
196        }
197    };
198}

200/// GPU solver of one grid: [`f32s::LazySolver`] or [`df32s::LazySolver`] behind one type. Mirrors `crate::solver::Solver` for the
201/// supported options (explicit SSP-RK3 or line-implicit scheme, solid block, low-Mach correction, viscous terms with the SST model).
202pub struct GpuSolver {
203    inner: Inner,
204    ctx: Context,
205    prec: Precision,
206    pub ni: usize,
207    pub nj: usize,
208}
209
210impl GpuSolver {
211    /// Uploads the geometry (rounded from the grid's f64 metrics), starts from the free stream.
212    pub fn new(ctx: &Context, grid: &Grid, cfg: &Config, prec: Precision) -> Result<GpuSolver> {
213        let inner = match prec {
214            Precision::F32 => Inner::F32(f32s::LazySolver::new(ctx, grid, cfg)?),
215            Precision::Extended => Inner::Ext(df32s::LazySolver::new(ctx, grid, cfg)?),
216        };
217        Ok(GpuSolver { inner, ctx: ctx.clone(), prec, ni: grid.ni, nj: grid.nj })
218    }
219
220    /// The solver of a host `crate::solver::Solver`: its grid and config, low-Mach flag, solid block, scheme (`Scheme::Explicit` or
221    /// `Scheme::Implicit`), conserved state, iteration count and, for a viscous host, the viscous parameters with the turbulence state and the
222    /// wall distance (the SST state is copied from `host.turb` / `host.mu_t`).
223    pub fn from_host(ctx: &Context, host: &Solver, prec: Precision) -> Result<GpuSolver> {
224        if !matches!(host.scheme(), Scheme::Explicit | Scheme::Implicit(_)) {
225            return Err(GpuError::Unsupported("only Scheme::Explicit and Scheme::Implicit are implemented on the GPU"));
226        }
227        let inner = match prec {
228            Precision::F32 => Inner::F32(f32s::LazySolver::from_host(ctx, host)?),
229            Precision::Extended => Inner::Ext(df32s::LazySolver::from_host(ctx, host)?),
230        };
231        Ok(GpuSolver { inner, ctx: ctx.clone(), prec, ni: host.grid.ni, nj: host.grid.nj })
232    }

246    /// Replaces the configuration: free stream, CFL, limiter, `first_order_iters` and the boundary conditions may change between runs
247    /// (a Mach sweep) without touching the state. The next [`GpuSolver::run`] uses it (the fixed ghost states are uploaded again).
248    pub fn set_config(&mut self, cfg: &Config) -> Result<()> {
249        if self.viscous().is_some() && cfg.bc != self.config().bc {
250            return Err(GpuError::Unsupported("the boundary conditions of a viscous run (no-slip walls, wall distance) are fixed by set_viscous"));
251        }
252        on_mut!(self, s => s.set_config(cfg)?);
253        Ok(())
254    }

259    }

262        on_mut!(self, s => s.iter = iter)
263    }

270    /// Declares the solid body block of a base / wake grid (`crate::solver::Solver::set_block`).
271    pub fn set_block(&mut self, block: Option<Block>) -> Result<()> {
272        if self.viscous().is_some() {
273            return Err(GpuError::Unsupported("the solid block must be declared before the viscous terms"));
274        }
275        on_mut!(self, s => s.set_block(block)?);
276        Ok(())
277    }

283    /// Declares the nozzle exit on the base face (`crate::solver::Solver::set_nozzle`): rows `0..j_end` of the face `i_end` are a fixed
284    /// inflow state. Must follow [`GpuSolver::set_block`] and precede [`GpuSolver::set_viscous`].
285    pub fn set_nozzle(&mut self, nozzle: Option<Nozzle>) -> Result<()> {
286        if self.viscous().is_some() || (nozzle.is_some() && self.block().is_none()) {
287            return Err(GpuError::Unsupported("set_nozzle must follow set_block and precede set_viscous"));
288        }
289        on_mut!(self, s => s.set_nozzle(nozzle)?);
290        Ok(())
291    }

298    /// Selects the convergence scheme: `Scheme::Explicit` or `Scheme::Implicit` (the CFL ramp restarts); multigrid and residual

302            return Err(GpuError::Unsupported("only Scheme::Explicit and Scheme::Implicit are implemented on the GPU"));
303        }

305        Ok(())
306    }

318    /// Switches the viscous terms on (`Some`, with the SST model if `v.turbulence` says so) or off (`None`, the Euler equations). `grid` is the grid
319    /// of the solver; declare the solid block first. The turbulence state restarts at the free stream, the mean flow is not touched.
320    pub fn set_viscous(&mut self, v: Option<Viscous>, grid: &Grid) -> Result<()> {
321        if (grid.ni, grid.nj) != (self.ni, self.nj) {
322            return Err(GpuError::Unsupported("set_viscous: the grid is not the grid of the solver"));
323        }
324        on_mut!(self, s => s.set_viscous(v, grid)?);
325        Ok(())
326    }

328    /// The viscous-flow parameters, `None` for the Euler equations.
329    pub fn viscous(&self) -> Option<Viscous> {
330        on!(self, s => s.viscous())
331    }

333    /// `([k, omega], mu_t)` of every cell of an SST run (`Solver::turb`, `Solver::mu_t`), as f64.
334    pub fn turb_state(&self) -> Result<(Vec<[f64; 2]>, Vec<f64>)> {
335        if !self.viscous().is_some_and(|v| v.turbulence == crate::solver::Turbulence::Sst) {
336            return Err(GpuError::Unsupported("turb_state needs the SST model"));
337        }
338        Ok(on!(self, s => s.turb_state_f64()?))
339    }

344    }

352    /// Resets to the free stream of the current configuration at iteration 0, the implicit CFL ramp restarted.
353    pub fn reset(&mut self) -> Result<()> {
354        let fs = crate::gas::prim_to_cons(&self.config().freestream);
355        self.set_state(&vec![fs; self.ni * self.nj])
356    }

358    /// Conserved state, `ni * nj`, same layout as `crate::solver::Solver::u`.
359    pub fn state_f64(&self) -> Result<Vec<[f64; 4]>> {
360        Ok(on!(self, s => s.state_f64()?))
361    }

363    /// Runs `n` iterations and returns the density residual of every one (L2 of `d rho / dt` over `sqrt(N)` of the fluid cells, as in
364    /// the CPU solver). The residual is recorded on the GPU, the host waits once per `RUN_SYNC` iterations and at the end.
365    pub fn run(&mut self, n: usize) -> Result<Vec<f64>> {
366        Ok(on_mut!(self, s => s.run(n, RUN_SYNC.min(n.max(1)))?))
367    }

369    /// Conserved state of the two first cells behind the base face, `[(i_end, j), (i_end + 1, j)]`, for every row `j_sting..j_end` (the
370    /// extrapolation stencil of `bowshock::engine::run::base_state`). Reads back only these rows.
371    pub fn base_cells(&self) -> Result<Vec<[[f64; 4]; 2]>> {
372        if self.block().is_none() {
373            return Err(GpuError::Unsupported("base_cells needs a solid block"));
374        }
375        Ok(on!(self, s => s.base_cells()?))
376    }
