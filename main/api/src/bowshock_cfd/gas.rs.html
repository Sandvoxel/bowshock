1//! Ideal-gas helpers. State layouts:
2//! * conservative `Cons = [rho, rho*u, rho*v, E]`, `E = p/(g-1) + rho (u^2+v^2)/2`
3//! * primitive    `Prim = [rho, u, v, p]`
4//!   with `u` the axial (x) and `v` the radial (r) velocity.
5
6pub const GAMMA: f64 = 1.4;
7
8pub type Cons = [f64; 4];
9pub type Prim = [f64; 4];
10
11#[inline]
12pub fn prim_to_cons(w: &Prim) -> Cons {
13    let [rho, u, v, p] = *w;
14    [rho, rho * u, rho * v, p / (GAMMA - 1.0) + 0.5 * rho * (u * u + v * v)]
15}
16
17#[inline]
18pub fn cons_to_prim(q: &Cons) -> Prim {
19    let rho = q[0];
20    let u = q[1] / rho;
21    let v = q[2] / rho;
22    let p = (GAMMA - 1.0) * (q[3] - 0.5 * rho * (u * u + v * v));
23    [rho, u, v, p]
24}
25
26#[inline]
27pub fn sound_speed(w: &Prim) -> f64 {
28    (GAMMA * w[3] / w[0]).sqrt()
29}
30
31/// Physical Euler flux projected on the unit normal `(nx, nr)`: `F.n`.
32#[inline]
33pub fn physical_flux(w: &Prim, nx: f64, nr: f64) -> Cons {
34    let [rho, u, v, p] = *w;
35    let un = u * nx + v * nr;
36    let e = p / (GAMMA - 1.0) + 0.5 * rho * (u * u + v * v);
37    [rho * un, rho * u * un + p * nx, rho * v * un + p * nr, (e + p) * un]
38}
39
40/// Free stream in non-dimensional units: rho=1, c=1 (so p=1/gamma), u=Mach, v=0.
41pub fn freestream(mach: f64) -> Prim {
42    [1.0, mach, 0.0, 1.0 / GAMMA]
43}
44
45/// Reflect the velocity about a wall with unit normal `(nx, nr)`.
46#[inline]
47pub fn reflect(w: &Prim, nx: f64, nr: f64) -> Prim {
48    let un = w[1] * nx + w[2] * nr;
49    [w[0], w[1] - 2.0 * un * nx, w[2] - 2.0 * un * nr, w[3]]
50}
