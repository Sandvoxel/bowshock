1//! 3D ideal-gas helpers, the Cartesian counterpart of [`crate::gas`]. State layouts:
2//! * conservative `Cons3 = [rho, rho*u, rho*v, rho*w, E]`, `E = p/(g-1) + rho (u^2+v^2+w^2)/2`
3//! * primitive    `Prim3 = [rho, u, v, w, p]`
4//!
5//! Velocities are Cartesian: `u` axial (x), `v` lateral (y), `w` vertical (z). Incidence `alpha > 0` means the air moves
6//! towards +z relative to the body, so the windward side is -z and a positive normal force points to +z.
7
8pub use crate::gas::GAMMA;
9
10pub type Cons3 = [f64; 5];
11pub type Prim3 = [f64; 5];
12
13#[inline]
14pub fn prim_to_cons3(w: &Prim3) -> Cons3 {
15    let [rho, u, v, ww, p] = *w;
16    [rho, rho * u, rho * v, rho * ww, p / (GAMMA - 1.0) + 0.5 * rho * (u * u + v * v + ww * ww)]
17}
18
19#[inline]
20pub fn cons_to_prim3(q: &Cons3) -> Prim3 {
21    let rho = q[0];
22    let (u, v, w) = (q[1] / rho, q[2] / rho, q[3] / rho);
23    let p = (GAMMA - 1.0) * (q[4] - 0.5 * rho * (u * u + v * v + w * w));
24    [rho, u, v, w, p]
25}
26
27#[inline]
28pub fn sound_speed3(w: &Prim3) -> f64 {
29    (GAMMA * w[4] / w[0]).sqrt()
30}
31
32/// Physical Euler flux projected on the unit normal `n`: `F.n`.
33#[inline]
34pub fn physical_flux3(w: &Prim3, n: [f64; 3]) -> Cons3 {
35    let [rho, u, v, ww, p] = *w;
36    let un = u * n[0] + v * n[1] + ww * n[2];
37    let e = p / (GAMMA - 1.0) + 0.5 * rho * (u * u + v * v + ww * ww);
38    [rho * un, rho * u * un + p * n[0], rho * v * un + p * n[1], rho * ww * un + p * n[2], (e + p) * un]
39}
40
41/// Free stream in non-dimensional units (rho=1, c=1, p=1/gamma) at incidence `alpha` \[rad\]: velocity `M (cos a, 0, sin a)`.
42pub fn freestream3(mach: f64, alpha: f64) -> Prim3 {
43    [1.0, mach * alpha.cos(), 0.0, mach * alpha.sin(), 1.0 / GAMMA]
44}
45
46/// Reflect the velocity about a wall with unit normal `n`.
47#[inline]
48pub fn reflect3(w: &Prim3, n: [f64; 3]) -> Prim3 {
49    let un = w[1] * n[0] + w[2] * n[1] + w[3] * n[2];
50    [w[0], w[1] - 2.0 * un * n[0], w[2] - 2.0 * un * n[1], w[3] - 2.0 * un * n[2], w[4]]
51}

53#[cfg(test)]
54mod tests {
55    use super::*;
56    use crate::gas;
57
58    #[test]
59    fn round_trip() {
60        let w = [1.3, 0.7, -0.2, 0.4, 0.9];
61        let back = cons_to_prim3(&prim_to_cons3(&w));
62        for k in 0..5 {
63            assert!((w[k] - back[k]).abs() < 1e-14);
64        }
65    }
66
67    #[test]
68    fn flux_matches_2d_in_xz_plane() {
69        let (nx, nr) = (0.6, 0.8);
70        let w3 = [1.2, 0.9, 0.0, -0.35, 0.7];
71        let f3 = physical_flux3(&w3, [nx, 0.0, nr]);
72        let f2 = gas::physical_flux(&[w3[0], w3[1], w3[3], w3[4]], nx, nr);
73        assert_eq!(f3[2], 0.0);
74        for (a, b) in [(f3[0], f2[0]), (f3[1], f2[1]), (f3[3], f2[2]), (f3[4], f2[3])] {
75            assert!((a - b).abs() < 1e-14);
76        }
77        let r3 = reflect3(&w3, [nx, 0.0, nr]);
78        let r2 = gas::reflect(&[w3[0], w3[1], w3[3], w3[4]], nx, nr);
79        assert!((r3[1] - r2[1]).abs() < 1e-14 && (r3[3] - r2[2]).abs() < 1e-14);
80        let fs = freestream3(2.0, 0.1);
81        assert!((fs[1] - 2.0 * 0.1f64.cos()).abs() < 1e-15 && (fs[3] - 2.0 * 0.1f64.sin()).abs() < 1e-15);
82    }
