1//! 3D face kernels (Cartesian), ported one-to-one from [`crate::flux`]: MUSCL reconstruction, HLLC, Rusanov, the
2//! shock-sensor blend and the low-Mach correction, all projected on a unit normal `n: [f64; 3]`.
3//! The limiters, `limited_slope`, `shock_weight` and the sensor / low-Mach constants are shared with 2D.
4
5use crate::flux::{limited_slope, shock_weight, Limiter, LOW_MACH_FLOOR};
6use crate::gas3::{physical_flux3, Cons3, Prim3, GAMMA};

8/// Cylindrical frame about the x axis at a face centroid `c`: `[sin, cos]` of its azimuth, `[c_y / r, c_z / r]` with
9/// `r = hypot(c_y, c_z)`; `[0, 1]` on the axis.
10#[inline]
11pub fn face_rot(c: [f64; 3]) -> [f64; 2] {
12    let r = c[1].hypot(c[2]);
13    if r < 1e-300 {
14        [0.0, 1.0]
15    } else {
16        [c[1] / r, c[2] / r]
17    }
18}

20/// MUSCL reconstruction of the primitive variables at the face between cells `m1` (left) and `p0` (right);
21/// `m2` and `p1` are the outer neighbours. The lateral velocity is limited in the face's cylindrical frame about the x
22/// axis (`rot = face_rot(centroid)`: radial `v s + w co`, tangential `v co - w s`), so axisymmetric flow stays exactly
23/// axisymmetric; the reconstructed states are transformed back. Falls back to first order for a side whose
24/// reconstructed density or pressure is not positive.
25#[inline]
26pub fn reconstruct3(m2: &Prim3, m1: &Prim3, p0: &Prim3, p1: &Prim3, lim: Limiter, rot: [f64; 2]) -> (Prim3, Prim3) {
27    let [s, co] = rot;
28    let fwd = |w: &Prim3| [w[0], w[1], w[2] * s + w[3] * co, w[2] * co - w[3] * s, w[4]];
29    let (a, b, c, d) = (fwd(m2), fwd(m1), fwd(p0), fwd(p1));
30    let mut l = b;
31    let mut r = c;
32    for k in 0..5 {
33        let phi_l = limited_slope(b[k] - a[k], c[k] - b[k], lim);
34        let phi_r = limited_slope(c[k] - b[k], d[k] - c[k], lim);
35        l[k] = b[k] + 0.5 * phi_l;
36        r[k] = c[k] - 0.5 * phi_r;
37    }
38    if l[0] <= 0.0 || l[4] <= 0.0 {
39        l = b;
40    }
41    if r[0] <= 0.0 || r[4] <= 0.0 {
42        r = c;
43    }
44    let back = |w: &Prim3, orig: &Prim3, fallback: bool| {
45        if fallback {
46            *orig
47        } else {
48            [w[0], w[1], w[2] * s + w[3] * co, w[2] * co - w[3] * s, w[4]]
49        }
50    };
51    (back(&l, m1, l == b), back(&r, p0, r == c))
52}

54/// HLLC flux for the 3D Euler equations projected on the unit normal `n` (see [`crate::flux::hllc`]). The contact
55/// wave carries both tangential velocity components passively. Returns `F.n` per unit face area.
56#[inline]
57pub fn hllc3(wl: &Prim3, wr: &Prim3, n: [f64; 3]) -> Cons3 {
58    let [rl, ul, vl, wl_z, pl] = *wl;
59    let [rr, ur, vr, wr_z, pr] = *wr;
60    let unl = ul * n[0] + vl * n[1] + wl_z * n[2];
61    let unr = ur * n[0] + vr * n[1] + wr_z * n[2];
62    let cl = (GAMMA * pl / rl).sqrt();
63    let cr = (GAMMA * pr / rr).sqrt();
64    let el = pl / (GAMMA - 1.0) + 0.5 * rl * (ul * ul + vl * vl + wl_z * wl_z);
65    let er = pr / (GAMMA - 1.0) + 0.5 * rr * (ur * ur + vr * vr + wr_z * wr_z);
66
67    // Roe averages
68    let sql = rl.sqrt();
69    let sqr = rr.sqrt();
70    let inv = 1.0 / (sql + sqr);
71    let uh = (sql * ul + sqr * ur) * inv;
72    let vh = (sql * vl + sqr * vr) * inv;
73    let wh = (sql * wl_z + sqr * wr_z) * inv;
74    let hl = (el + pl) / rl;
75    let hr = (er + pr) / rr;
76    let hh = (sql * hl + sqr * hr) * inv;
77    let ch2 = (GAMMA - 1.0) * (hh - 0.5 * (uh * uh + vh * vh + wh * wh));
78    let ch = ch2.max(1e-300).sqrt();
79    let unh = uh * n[0] + vh * n[1] + wh * n[2];
80
81    let sl = (unl - cl).min(unh - ch);
82    let sr = (unr + cr).max(unh + ch);
83    if sl >= 0.0 {
84        return physical_flux3(wl, n);
85    }
86    if sr <= 0.0 {
87        return physical_flux3(wr, n);
88    }
89
90    let sm = (pr - pl + rl * unl * (sl - unl) - rr * unr * (sr - unr)) / (rl * (sl - unl) - rr * (sr - unr));
91
92    if sm >= 0.0 {
93        let f = physical_flux3(wl, n);
94        let fac = rl * (sl - unl) / (sl - sm);
95        let d = sm - unl;
96        let star = [fac, fac * (ul + d * n[0]), fac * (vl + d * n[1]), fac * (wl_z + d * n[2]), fac * (el / rl + d * (sm + pl / (rl * (sl - unl))))];
97        let q = [rl, rl * ul, rl * vl, rl * wl_z, el];
98        let mut out = [0.0; 5];
99        for k in 0..5 {
100            out[k] = f[k] + sl * (star[k] - q[k]);
101        }
102        out
103    } else {
104        let f = physical_flux3(wr, n);
105        let fac = rr * (sr - unr) / (sr - sm);
106        let d = sm - unr;
107        let star = [fac, fac * (ur + d * n[0]), fac * (vr + d * n[1]), fac * (wr_z + d * n[2]), fac * (er / rr + d * (sm + pr / (rr * (sr - unr))))];
108        let q = [rr, rr * ur, rr * vr, rr * wr_z, er];
109        let mut out = [0.0; 5];
110        for k in 0..5 {
111            out[k] = f[k] + sr * (star[k] - q[k]);
112        }
113        out
114    }
115}

117/// Rusanov (local Lax-Friedrichs) flux projected on `n`: `0.5 (F_L + F_R) - 0.5 smax (U_R - U_L)` with
118/// `smax = max(|un_L| + c_L, |un_R| + c_R)`.
119#[inline]
120pub fn rusanov3(wl: &Prim3, wr: &Prim3, n: [f64; 3]) -> Cons3 {
121    let [rl, ul, vl, zl, pl] = *wl;
122    let [rr, ur, vr, zr, pr] = *wr;
123    let unl = ul * n[0] + vl * n[1] + zl * n[2];
124    let unr = ur * n[0] + vr * n[1] + zr * n[2];
125    let smax = (unl.abs() + (GAMMA * pl / rl).sqrt()).max(unr.abs() + (GAMMA * pr / rr).sqrt());
126    let fl = physical_flux3(wl, n);
127    let fr = physical_flux3(wr, n);
128    let ql = [rl, rl * ul, rl * vl, rl * zl, pl / (GAMMA - 1.0) + 0.5 * rl * (ul * ul + vl * vl + zl * zl)];
129    let qr = [rr, rr * ur, rr * vr, rr * zr, pr / (GAMMA - 1.0) + 0.5 * rr * (ur * ur + vr * vr + zr * zr)];
130    let mut out = [0.0; 5];
131    for k in 0..5 {
132        out[k] = 0.5 * (fl[k] + fr[k]) - 0.5 * smax * (qr[k] - ql[k]);
133    }
134    out
135}

137/// Complete face kernel: MUSCL reconstruction + HLLC, blended towards Rusanov inside shocks
138/// (see [`crate::flux::face_flux`]).
139#[inline]
140pub fn face_flux3(m2: &Prim3, m1: &Prim3, p0: &Prim3, p1: &Prim3, lim: Limiter, n: [f64; 3], rot: [f64; 2]) -> Cons3 {
141    let (l, r) = reconstruct3(m2, m1, p0, p1, lim, rot);
142    blend3(&l, &r, m1[4], p0[4], n)
143}
144
145#[inline]
146fn blend3(l: &Prim3, r: &Prim3, pl: f64, pr: f64, n: [f64; 3]) -> Cons3 {
147    let c = hllc3(l, r, n);
148    let t = shock_weight(pl, pr);
149    if t == 0.0 {
150        return c;
151    }
152    let f = rusanov3(l, r, n);
153    let mut out = c;
154    for k in 0..5 {
155        out[k] = (1.0 - t) * c[k] + t * f[k];
156    }
157    out
158}

160/// Low-Mach correction of the reconstructed face states: scales only the jump of the face-normal velocity by
161/// `z = clamp(max(M_L, M_R), LOW_MACH_FLOOR, 1)` (see [`crate::flux::low_mach_states`]).
162#[inline]
163pub fn low_mach_states3(l: &Prim3, r: &Prim3, n: [f64; 3]) -> (Prim3, Prim3) {
164    let unl = l[1] * n[0] + l[2] * n[1] + l[3] * n[2];
165    let unr = r[1] * n[0] + r[2] * n[1] + r[3] * n[2];
166    let ml = (l[1] * l[1] + l[2] * l[2] + l[3] * l[3]).sqrt() / (GAMMA * l[4] / l[0]).sqrt();
167    let mr = (r[1] * r[1] + r[2] * r[2] + r[3] * r[3]).sqrt() / (GAMMA * r[4] / r[0]).sqrt();
168    let z = ml.max(mr).clamp(LOW_MACH_FLOOR, 1.0);
169    if z >= 1.0 {
170        return (*l, *r);
171    }
172    let avg = 0.5 * (unl + unr);
173    let dl = avg + z * (unl - avg) - unl;
174    let dr = avg + z * (unr - avg) - unr;
175    ([l[0], l[1] + dl * n[0], l[2] + dl * n[1], l[3] + dl * n[2], l[4]], [r[0], r[1] + dr * n[0], r[2] + dr * n[1], r[3] + dr * n[2], r[4]])
176}

178/// [`face_flux3`] with the optional low-Mach correction; `low_mach = false` is exactly [`face_flux3`].
179#[inline]
180pub fn face_flux_lm3(m2: &Prim3, m1: &Prim3, p0: &Prim3, p1: &Prim3, lim: Limiter, n: [f64; 3], rot: [f64; 2], low_mach: bool) -> Cons3 {
181    if !low_mach {
182        return face_flux3(m2, m1, p0, p1, lim, n, rot);
183    }
184    let (l, r) = reconstruct3(m2, m1, p0, p1, lim, rot);
185    let (l, r) = low_mach_states3(&l, &r, n);
186    blend3(&l, &r, m1[4], p0[4], n)
187}

194    pub(crate) struct Rng(u64);
195    impl Rng {
196        pub(crate) fn new(seed: u64) -> Self {
197            Rng(seed)

199        pub(crate) fn next(&mut self) -> f64 {
200            self.0 = self.0.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
201            ((self.0 >> 11) as f64) / ((1u64 << 53) as f64)
202        }
203        pub(crate) fn range(&mut self, a: f64, b: f64) -> f64 {
204            a + (b - a) * self.next()
205        }

208    pub(crate) type Rot = [[f64; 3]; 3];

210    pub(crate) fn random_rotation(rng: &mut Rng) -> Rot {
211        let mut a = [rng.range(-1.0, 1.0), rng.range(-1.0, 1.0), rng.range(-1.0, 1.0)];
212        let l = (a[0] * a[0] + a[1] * a[1] + a[2] * a[2]).sqrt();
213        for x in &mut a {
214            *x /= l;
215        }
216        let th = rng.range(0.0, std::f64::consts::TAU);
217        let (s, c) = th.sin_cos();
218        let k = [[0.0, -a[2], a[1]], [a[2], 0.0, -a[0]], [-a[1], a[0], 0.0]];
219        let mut r = [[0.0; 3]; 3];
220        for i in 0..3 {
221            for j in 0..3 {
222                let kk: f64 = (0..3).map(|m| k[i][m] * k[m][j]).sum();
223                r[i][j] = if i == j { 1.0 } else { 0.0 } + s * k[i][j] + (1.0 - c) * kk;
224            }
225        }
226        r
227    }

229    pub(crate) fn rot(r: &Rot, v: [f64; 3]) -> [f64; 3] {
230        [r[0][0] * v[0] + r[0][1] * v[1] + r[0][2] * v[2], r[1][0] * v[0] + r[1][1] * v[1] + r[1][2] * v[2], r[2][0] * v[0] + r[2][1] * v[1] + r[2][2] * v[2]]
231    }

233    /// Rotate the velocity (indices 1..4) of a primitive or conservative vector.
234    pub(crate) fn rot5(r: &Rot, q: &[f64; 5]) -> [f64; 5] {
235        let v = rot(r, [q[1], q[2], q[3]]);
236        [q[0], v[0], v[1], v[2], q[4]]
237    }

239    fn rand_state(rng: &mut Rng, planar: bool, vmax: f64) -> Prim3 {
240        let v = if planar { 0.0 } else { rng.range(-vmax, vmax) };
241        [rng.range(0.5, 2.0), rng.range(-vmax, vmax), v, rng.range(-vmax, vmax), rng.range(0.5, 2.0)]
242    }

244    fn to2(w: &Prim3) -> [f64; 4] {
245        [w[0], w[1], w[3], w[4]]
246    }

248    fn close4(f3: &Cons3, f2: &[f64; 4]) {
249        assert_eq!(f3[2], 0.0, "y-momentum flux must vanish");
250        for (a, b) in [(f3[0], f2[0]), (f3[1], f2[1]), (f3[3], f2[2]), (f3[4], f2[3])] {
251            assert!((a - b).abs() < 1e-14 * (1.0 + b.abs()), "{a} vs {b}");
252        }
253    }

255    fn rand_normal_xz(rng: &mut Rng) -> (f64, f64) {
256        let t = rng.range(0.0, std::f64::consts::TAU);
257        (t.cos(), t.sin())
258    }

309        for it in 0..2000 {
310            let vmax = if it % 2 == 0 { 3.0 } else { 0.1 };
311            let th = rng.range(0.0, std::f64::consts::TAU);

319            // in-plane state (u, w_r) -> (u, w_r sin, w_r cos)
320            let sr: Vec<Prim3> = s.iter().map(|w| [w[0], w[1], w[3] * st, w[3] * ct, w[4]]).collect();
