5#[derive(Clone, Copy, Debug, PartialEq, Eq)]
6pub enum Limiter {
7    /// Piecewise-constant (first order).
8    FirstOrder,
9    Minmod,
10    VanAlbada,
11}
12
13/// Limited cell slope `phi(a, b)` with `a = q_i - q_{i-1}`, `b = q_{i+1} - q_i`.
14/// Face values are `q_i +- phi/2`.
15#[inline]
16pub fn limited_slope(a: f64, b: f64, lim: Limiter) -> f64 {
17    match lim {
18        Limiter::FirstOrder => 0.0,
19        Limiter::Minmod => {
20            if a * b <= 0.0 {
21                0.0
22            } else if a.abs() < b.abs() {
23                a
24            } else {
25                b
26            }
27        }
28        Limiter::VanAlbada => {
29            // smooth van Albada: (a(b^2+e) + b(a^2+e)) / (a^2+b^2+2e); == a when a == b
30            const EPS: f64 = 1e-14;
31            if a * b <= 0.0 {
32                0.0
33            } else {
34                (a * (b * b + EPS) + b * (a * a + EPS)) / (a * a + b * b + 2.0 * EPS)
35            }
36        }
37    }
38}

40/// MUSCL reconstruction of the primitive variables at the face between cells
41/// `m1` (left) and `p0` (right); `m2` and `p1` are the outer neighbours.
42/// Falls back to first order for a face whose reconstructed density or pressure is not positive.
43#[inline]
44pub fn reconstruct(m2: &Prim, m1: &Prim, p0: &Prim, p1: &Prim, lim: Limiter) -> (Prim, Prim) {
45    let mut l = *m1;
46    let mut r = *p0;
47    for k in 0..4 {
48        let phi_l = limited_slope(m1[k] - m2[k], p0[k] - m1[k], lim);
49        let phi_r = limited_slope(p0[k] - m1[k], p1[k] - p0[k], lim);
50        l[k] = m1[k] + 0.5 * phi_l;
51        r[k] = p0[k] - 0.5 * phi_r;
52    }
53    if l[0] <= 0.0 || l[3] <= 0.0 {
54        l = *m1;
55    }
56    if r[0] <= 0.0 || r[3] <= 0.0 {
57        r = *p0;
58    }
59    (l, r)
60}

62/// HLLC flux (Toro, *Riemann Solvers and Numerical Methods for Fluid Dynamics*, ch. 10)
63/// for the 2D Euler equations, projected on the unit normal `(nx, nr)`.
64/// Wave speeds: Einfeldt/Roe-averaged estimates `SL = min(unL-cL, u~-c~)`, `SR = max(unR+cR, u~+c~)`.
65/// Returns the flux per unit face length `F.n` (no geometric weighting).
66#[inline]
67pub fn hllc(wl: &Prim, wr: &Prim, nx: f64, nr: f64) -> Cons {
68    let [rl, ul, vl, pl] = *wl;
69    let [rr, ur, vr, pr] = *wr;
70    let unl = ul * nx + vl * nr;
71    let unr = ur * nx + vr * nr;
72    let cl = (GAMMA * pl / rl).sqrt();
73    let cr = (GAMMA * pr / rr).sqrt();
74    let el = pl / (GAMMA - 1.0) + 0.5 * rl * (ul * ul + vl * vl);
75    let er = pr / (GAMMA - 1.0) + 0.5 * rr * (ur * ur + vr * vr);
76
77    // Roe averages
78    let sql = rl.sqrt();
79    let sqr = rr.sqrt();
80    let inv = 1.0 / (sql + sqr);
81    let uh = (sql * ul + sqr * ur) * inv;
82    let vh = (sql * vl + sqr * vr) * inv;
83    let hl = (el + pl) / rl;
84    let hr = (er + pr) / rr;
85    let hh = (sql * hl + sqr * hr) * inv;
86    let ch2 = (GAMMA - 1.0) * (hh - 0.5 * (uh * uh + vh * vh));
87    let ch = ch2.max(1e-300).sqrt();
88    let unh = uh * nx + vh * nr;
89
90    let sl = (unl - cl).min(unh - ch);
91    let sr = (unr + cr).max(unh + ch);
92    if sl >= 0.0 {
93        return physical_flux(wl, nx, nr);
94    }
95    if sr <= 0.0 {
96        return physical_flux(wr, nx, nr);
97    }
98
99    let sm = (pr - pl + rl * unl * (sl - unl) - rr * unr * (sr - unr)) / (rl * (sl - unl) - rr * (sr - unr));
100
101    if sm >= 0.0 {
102        let f = physical_flux(wl, nx, nr);
103        let fac = rl * (sl - unl) / (sl - sm);
104        let ul_star = [fac, fac * (ul + (sm - unl) * nx), fac * (vl + (sm - unl) * nr), fac * (el / rl + (sm - unl) * (sm + pl / (rl * (sl - unl))))];
105        let q = [rl, rl * ul, rl * vl, el];
106        [f[0] + sl * (ul_star[0] - q[0]), f[1] + sl * (ul_star[1] - q[1]), f[2] + sl * (ul_star[2] - q[2]), f[3] + sl * (ul_star[3] - q[3])]
107    } else {
108        let f = physical_flux(wr, nx, nr);
109        let fac = rr * (sr - unr) / (sr - sm);
110        let ur_star = [fac, fac * (ur + (sm - unr) * nx), fac * (vr + (sm - unr) * nr), fac * (er / rr + (sm - unr) * (sm + pr / (rr * (sr - unr))))];
111        let q = [rr, rr * ur, rr * vr, er];
112        [f[0] + sr * (ur_star[0] - q[0]), f[1] + sr * (ur_star[1] - q[1]), f[2] + sr * (ur_star[2] - q[2]), f[3] + sr * (ur_star[3] - q[3])]
113    }
114}

116/// Rusanov (local Lax-Friedrichs) flux, projected on `(nx, nr)`: `0.5 (F_L + F_R) - 0.5 smax (U_R - U_L)`
117/// with `smax = max(|un_L| + c_L, |un_R| + c_R)`. Dissipates every wave family (incl. contact/shear) at the
118/// maximum rate.
119#[inline]
120pub fn rusanov(wl: &Prim, wr: &Prim, nx: f64, nr: f64) -> Cons {
121    let [rl, ul, vl, pl] = *wl;
122    let [rr, ur, vr, pr] = *wr;
123    let unl = ul * nx + vl * nr;
124    let unr = ur * nx + vr * nr;
125    let smax = (unl.abs() + (GAMMA * pl / rl).sqrt()).max(unr.abs() + (GAMMA * pr / rr).sqrt());
126    let fl = physical_flux(wl, nx, nr);
127    let fr = physical_flux(wr, nx, nr);
128    let ql = [rl, rl * ul, rl * vl, pl / (GAMMA - 1.0) + 0.5 * rl * (ul * ul + vl * vl)];
129    let qr = [rr, rr * ur, rr * vr, pr / (GAMMA - 1.0) + 0.5 * rr * (ur * ur + vr * vr)];
130    let mut out = [0.0; 4];
131    for k in 0..4 {
132        out[k] = 0.5 * (fl[k] + fr[k]) - 0.5 * smax * (qr[k] - ql[k]);
133    }
134    out
135}

137/// Shock-sensor lower/upper threshold on the relative cell-to-cell pressure jump
138/// `|p_R - p_L| / min(p_R, p_L)` (see [`shock_weight`]).
139pub const SHOCK_SENSOR_LO: f64 = 0.002;
140pub const SHOCK_SENSOR_HI: f64 = 0.02;
141
142/// Weight `theta in [0, 1]` of the Rusanov flux in [`face_flux`]: smoothstep of the relative pressure jump
143/// between the two cells adjacent to the face (Quirk-type shock sensor). `0` in smooth flow (pure HLLC),
144/// `1` inside a captured shock (pure Rusanov); C1-continuous so the steady state stays a smooth fixed point.
145#[inline]
146pub fn shock_weight(pl: f64, pr: f64) -> f64 {
147    let s = (pr - pl).abs() / pl.min(pr);
148    let t = ((s - SHOCK_SENSOR_LO) / (SHOCK_SENSOR_HI - SHOCK_SENSOR_LO)).clamp(0.0, 1.0);
149    t * t * (3.0 - 2.0 * t)
150}

152/// Complete face kernel: MUSCL reconstruction + HLLC, blended towards Rusanov inside shocks.
153/// `m2, m1 | p0, p1` are the four primitive cell states around the face (`m1`/`p0` adjacent to it).
154///
155/// Why the blend: HLLC resolves contact/shear waves with zero dissipation; combined with second-order
156/// reconstruction a captured oblique shock that is not aligned with the grid is a (convectively) unstable
157/// discrete structure: perturbations in the shock cells grow with distance downstream and saturate as a
158/// limit cycle (README, "Convergence stall"). Rusanov at shock faces removes it; away from shocks the
159/// accurate HLLC flux is used unchanged.
160#[inline]
161pub fn face_flux(m2: &Prim, m1: &Prim, p0: &Prim, p1: &Prim, lim: Limiter, nx: f64, nr: f64) -> Cons {
162    let (l, r) = reconstruct(m2, m1, p0, p1, lim);
163    let c = hllc(&l, &r, nx, nr);
164    let t = shock_weight(m1[3], p0[3]);
165    if t == 0.0 {
166        return c;
167    }
168    let f = rusanov(&l, &r, nx, nr);
169    let mut out = c;
170    for k in 0..4 {
171        out[k] = (1.0 - t) * c[k] + t * f[k];
172    }
173    out
174}

176/// Lower bound of the low-Mach scaling factor `z` in [`low_mach_states`].
177pub const LOW_MACH_FLOOR: f64 = 0.05;
178
179/// Low-Mach correction of the reconstructed face states (Thornber et al., "An improved reconstruction method for
180/// compressible flows with low Mach number features", J. Comput. Phys. 227 (2008) 4873): the jump of the face-normal
181/// velocity between the left and right state is scaled by `z = min(max(M_L, M_R), 1)` (`M = |V|/c` of the reconstructed
182/// states, floored at [`LOW_MACH_FLOOR`]):
183///
184/// ```text
185/// u_n,L' = u_n,avg + z (u_n,L - u_n,avg),     u_n,R' = u_n,avg + z (u_n,R - u_n,avg)
186/// ```
187///
188/// Without it the dissipation of an upwind Riemann solver contains a pressure term `~ rho c Delta u_n` that is `O(1/M)`
189/// larger than the dynamic pressure, so the pressure field degrades as `M -> 0` (and its error does not vanish with
190/// the mesh at fixed `M`). Only the normal velocity is touched (tangential velocity, density and pressure are unchanged),
191/// so a uniform state, a stationary contact and any state with `u_n,L = u_n,R` are unchanged; `z = 1` for `M >= 1`, i.e.
192/// the correction is inactive in supersonic regions. `p`, `rho` and the tangential velocity are not modified.
193#[inline]
194pub fn low_mach_states(l: &Prim, r: &Prim, nx: f64, nr: f64) -> (Prim, Prim) {
195    let unl = l[1] * nx + l[2] * nr;
196    let unr = r[1] * nx + r[2] * nr;
197    let ml = (l[1] * l[1] + l[2] * l[2]).sqrt() / (GAMMA * l[3] / l[0]).sqrt();
198    let mr = (r[1] * r[1] + r[2] * r[2]).sqrt() / (GAMMA * r[3] / r[0]).sqrt();
199    let z = ml.max(mr).clamp(LOW_MACH_FLOOR, 1.0);
200    if z >= 1.0 {
201        return (*l, *r);
202    }
203    let avg = 0.5 * (unl + unr);
204    let dl = avg + z * (unl - avg) - unl;
205    let dr = avg + z * (unr - avg) - unr;
206    ([l[0], l[1] + dl * nx, l[2] + dl * nr, l[3]], [r[0], r[1] + dr * nx, r[2] + dr * nr, r[3]])
207}

209/// [`face_flux`] with the optional low-Mach correction [`low_mach_states`] applied to the reconstructed states before the
210/// Riemann solvers; `low_mach = false` is exactly [`face_flux`].
211#[inline]
212pub fn face_flux_lm(m2: &Prim, m1: &Prim, p0: &Prim, p1: &Prim, lim: Limiter, nx: f64, nr: f64, low_mach: bool) -> Cons {
213    if !low_mach {
214        return face_flux(m2, m1, p0, p1, lim, nx, nr);
215    }
216    let (l, r) = reconstruct(m2, m1, p0, p1, lim);
217    let (l, r) = low_mach_states(&l, &r, nx, nr);
218    let c = hllc(&l, &r, nx, nr);
219    let t = shock_weight(m1[3], p0[3]);
220    if t == 0.0 {
221        return c;
222    }
223    let f = rusanov(&l, &r, nx, nr);
224    let mut out = c;
225    for k in 0..4 {
226        out[k] = (1.0 - t) * c[k] + t * f[k];
227    }
228    out
229}

235    #[test]
236    fn hllc_consistency() {
237        // F(U,U) must equal the physical flux for any state and normal direction.
238        let states: [Prim; 5] = [[1.0, 2.0, 0.0, 1.0 / GAMMA], [0.3, -1.5, 0.7, 0.2], [2.5, 0.0, 0.0, 3.0], [1.2, 0.3, -2.4, 0.9], [0.9, 5.0, 3.0, 1.3]];
239        for w in &states {
240            for k in 0..16 {
241                let a = 0.4 * k as f64;
242                let (nr, nx) = a.sin_cos();
243                let f = hllc(w, w, nx, nr);
244                let g = physical_flux(w, nx, nr);
245                for c in 0..4 {
246                    assert!((f[c] - g[c]).abs() <= 1e-13 * (1.0 + g[c].abs()), "{w:?} {a} {c}");
247                }
248            }
249        }
250    }

271    #[test]
272    fn rusanov_consistency_and_shock_blend() {
273        let w: Prim = [0.9, 5.0, 3.0, 1.3];
274        for k in 0..8 {
275            let (nr, nx) = (0.8 * k as f64).sin_cos();
276            let (f, g) = (rusanov(&w, &w, nx, nr), physical_flux(&w, nx, nr));
277            for c in 0..4 {
278                assert!((f[c] - g[c]).abs() <= 1e-13 * (1.0 + g[c].abs()));
279            }
280        }
281        // smooth flow: pure HLLC; strong jump: pure Rusanov; monotone in between
282        assert_eq!(shock_weight(1.0, 1.0 + 0.5 * SHOCK_SENSOR_LO), 0.0);
283        assert_eq!(shock_weight(1.0, 1.5), 1.0);
284        assert!(shock_weight(1.0, 1.0 + 0.5 * (SHOCK_SENSOR_LO + SHOCK_SENSOR_HI)) > 0.4);
285        let (a, b): (Prim, Prim) = ([1.0, 2.0, 0.0, 0.7], [1.5, 1.7, 0.2, 1.0]);
286        let f = face_flux(&a, &a, &b, &b, Limiter::FirstOrder, 1.0, 0.0);
287        let r = rusanov(&a, &b, 1.0, 0.0);
288        for c in 0..4 {
289            assert!((f[c] - r[c]).abs() < 1e-14);
290        }

293    #[test]
294    fn low_mach_flux_is_consistent_and_reduces_to_the_standard_flux() {
295        // F(U, U) = F(U) with the correction on (the normal-velocity jump is zero)
296        let w: Prim = [1.1, 0.04, -0.02, 0.8];
297        for k in 0..8 {
298            let (nr, nx) = (0.8 * k as f64).sin_cos();
299            let f = face_flux_lm(&w, &w, &w, &w, Limiter::VanAlbada, nx, nr, true);
300            let g = physical_flux(&w, nx, nr);

302                assert!((f[c] - g[c]).abs() <= 1e-13 * (1.0 + g[c].abs()), "{k} {c}");
303            }
304        }
305        // low_mach = false is exactly the standard kernel
306        let (a, b, c, d): (Prim, Prim, Prim, Prim) = ([1.0, 0.1, 0.0, 0.7], [1.01, 0.12, 0.01, 0.71], [1.02, 0.15, 0.02, 0.72], [1.04, 0.2, 0.0, 0.73]);

314    #[test]
315    fn low_mach_scaling_reduces_the_normal_velocity_jump_only() {
316        let (l, r): (Prim, Prim) = ([1.0, 0.12, 0.05, 0.714], [1.0, 0.08, 0.05, 0.714]);
317        let (lm, rm) = low_mach_states(&l, &r, 1.0, 0.0);

319        // jump scaled by z = M_local, mean preserved, everything but u unchanged
320        assert!(((rm[1] - lm[1]) - m_local * (0.08 - 0.12)).abs() < 1e-14);
321        assert!((0.5 * (lm[1] + rm[1]) - 0.10).abs() < 1e-14);
322        assert_eq!((lm[0], lm[2], lm[3]), (l[0], l[2], l[3]));
323        assert_eq!((rm[0], rm[2], rm[3]), (r[0], r[2], r[3]));
324        // z is floored: a near-stagnant pair still keeps LOW_MACH_FLOOR of the jump
325        let (a, b): (Prim, Prim) = ([1.0, 1e-6, 0.0, 0.714], [1.0, -1e-6, 0.0, 0.714]);
326        let (am, bm) = low_mach_states(&a, &b, 1.0, 0.0);
327        assert!(((bm[1] - am[1]) - LOW_MACH_FLOOR * (-2e-6)).abs() < 1e-18);
