1//! 3D Euler finite-volume solver on a [`Grid3`] (CPU, f64, `rayon`): MUSCL + HLLC/Rusanov face fluxes of [`crate::flux3`], SSP-RK3
2//! with local time stepping (`Scheme3::Explicit`) or the three-factor line-implicit scheme of [`crate::implicit3`].
3//!
4//! The data flow is the 2D one of [`crate::solver`] with one more direction and no source term:
5//! 1. `cons -> prim` per cell into the padded array `w` (2 ghost layers on every side of all three axes, [`pidx3`]);
6//! 2. ghost fill (no corner ghosts: all stencils are 1D);
7//! 3. flux per i / j / k face (`face_flux_lm3`; the lateral velocity is limited in the face's cylindrical frame, [`face_rot`]);
8//! 4. cell residual `R = -(1/V) sum_f S_f F_f`.
9//!
10//! Boundaries (see [`Solver3::new`] for the grid conventions): `i = 0`, `i = ni`, `j = nj` characteristic far field;
11//! `j = 0, i < axis_cells` the symmetry axis (ghost = the cell across the axis, found through the pitch-plane symmetry
12//! `theta -> pi - theta` and a sign flip of `v`); `j = 0, i >= axis_cells` slip wall; `k = 0`, `k = nk` symmetry planes.
13//! Cells inside a [`crate::grid3::SolidBox`] (fins) are never updated and act as slip walls on their fluid neighbours.
14

23/// Ghost layers per side.
24pub(crate) const NG: usize = 2;
25
26/// Index of cell `(i, j, k)` (ghost cells have negative / out-of-range indices) in the padded primitive array.
27#[inline]
28pub(crate) fn pidx3(ni: usize, nj: usize, _nk: usize, i: isize, j: isize, k: isize) -> usize {
29    let ng = NG as isize;
30    ((k + ng) as usize * (nj + 2 * NG) + (j + ng) as usize) * (ni + 2 * NG) + (i + ng) as usize
31}

82impl Solver3 {
83    /// Solver on `grid` with the free stream everywhere.
84    pub fn new(grid: Grid3, cfg: Config3) -> Solver3 {
85        let (ni, nj, nk) = (grid.ni, grid.nj, grid.nk);
86        let nc = ni * nj * nk;
87        let mut solid = vec![false; nc];
88        for b in &grid.solid {
89            for k in b.k.clone() {
90                for j in b.j.clone() {
91                    for i in b.i.clone() {
92                        solid[(k * nj + j) * ni + i] = true;
93                    }
94                }
95            }
96        }
97        let n_active = solid.iter().filter(|s| !**s).count();
98        // cell across the axis: the one whose centroid azimuth is closest to pi - theta
99        let theta = |i: usize, k: usize| {
100            let c = grid.xc[(k * nj) * ni + i];
101            c[1].atan2(c[2])
102        };
103        let mut axis_k = vec![0; grid.axis_cells.min(ni) * nk];
104        for i in 0..grid.axis_cells.min(ni) {
105            for k in 0..nk {
106                let target = std::f64::consts::PI - theta(i, k);
107                let mut best = (f64::INFINITY, 0);
108                for kk in 0..nk {
109                    let d = (theta(i, kk) - target).abs();
110                    if d < best.0 {
111                        best = (d, kk);
112                    }
113                }
114                axis_k[i * nk + k] = best.1;
115            }
116        }
117        let rot = |f: &[Face3]| f.iter().map(|f| face_rot(f.c)).collect::<Vec<_>>();
118        let q = prim_to_cons3(&cfg.freestream);
119        Solver3 {
120            u: vec![q; nc],
121            w: vec![cfg.freestream; (ni + 2 * NG) * (nj + 2 * NG) * (nk + 2 * NG)],
122            res: vec![[0.0; 5]; nc],
123            dt: vec![0.0; nc],
124            iter: 0,
125            fi: vec![[0.0; 5]; grid.iface.len()],
126            fj: vec![[0.0; 5]; grid.jface.len()],
127            fk: vec![[0.0; 5]; grid.kface.len()],
128            u0: vec![[0.0; 5]; nc],
129            solid,
130            n_active,
131            ramp_k: 0,
132            axis_k,
133            rot_i: rot(&grid.iface),
134            rot_j: rot(&grid.jface),
135            rot_k: rot(&grid.kface),
136            grid,
137            cfg,
138        }
139    }

141    /// Restart the CFL ramp of the implicit scheme (call at the start of a run / after a change of the free stream).

143        self.ramp_k = 0;

146    /// Primitive state of interior cell `(i, j, k)` as of the last residual evaluation.
147    pub fn prim(&self, i: usize, j: usize, k: usize) -> Prim3 {
148        self.w[pidx3(self.grid.ni, self.grid.nj, self.grid.nk, i as isize, j as isize, k as isize)]
149    }

151    /// L2 norm of `d rho / dt` normalised by sqrt(N) (N = number of fluid cells).
152    pub fn density_residual(&self) -> f64 {
153        let s: f64 = self.res.par_iter().map(|r| r[0] * r[0]).sum();
154        (s / self.n_active as f64).sqrt()
155    }

157    /// One iteration with the configured [`Scheme3`]; returns the density residual at the start of the iteration.
158    pub fn step(&mut self) -> f64 {
159        let rn = match self.cfg.scheme {
160            Scheme3::Explicit => self.rk3_step(),
161            Scheme3::Implicit(p) => crate::implicit3::step(self, &p),
162        };
163        self.iter += 1;
164        rn
165    }

167    fn ghost_char(fs: &Prim3, edge: &Prim3, n: [f64; 3], outward: f64) -> Prim3 {
168        characteristic_ghost3(fs, edge, [outward * n[0], outward * n[1], outward * n[2]]).0
169    }

171    fn fill_ghosts(&mut self) {
172        let (ni, nj, nk) = (self.grid.ni, self.grid.nj, self.grid.nk);
173        let fs = self.cfg.freestream;
174        let (w, g, axis_k) = (&mut self.w, &self.grid, &self.axis_k);
175        let id = |i: isize, j: isize, k: isize| pidx3(ni, nj, nk, i, j, k);
176        let (nii, njj, nkk) = (ni as isize, nj as isize, nk as isize);
177        for k in 0..nkk {
178            for j in 0..njj {
179                let base = (k as usize * nj + j as usize) * (ni + 1);
180                let (fl, fr) = (&g.iface[base], &g.iface[base + ni]);
181                let gl = Self::ghost_char(&fs, &w[id(0, j, k)], fl.n, -1.0);
182                let gr = Self::ghost_char(&fs, &w[id(nii - 1, j, k)], fr.n, 1.0);
183                for m in 1..=NG as isize {
184                    w[id(-m, j, k)] = gl;
185                    w[id(nii + m - 1, j, k)] = gr;
186                }
187            }
188            for i in 0..nii {
189                let iu = i as usize;
190                if iu < g.axis_cells {
191                    let kp = axis_k[iu * nk + k as usize] as isize;
192                    for m in 1..=NG as isize {
193                        let mut s = w[id(i, m - 1, kp)];
194                        s[2] = -s[2];
195                        w[id(i, -m, k)] = s;
196                    }
197                } else {
198                    let f = &g.jface[(k as usize * (nj + 1)) * ni + iu];
199                    for m in 1..=NG as isize {
200                        w[id(i, -m, k)] = reflect3(&w[id(i, m - 1, k)], f.n);
201                    }
202                }
203                let f = &g.jface[(k as usize * (nj + 1) + nj) * ni + iu];
204                let gh = Self::ghost_char(&fs, &w[id(i, njj - 1, k)], f.n, 1.0);
205                for m in 1..=NG as isize {
206                    w[id(i, njj + m - 1, k)] = gh;
207                }
208            }
209        }
210        for j in 0..njj {
211            for i in 0..nii {
212                let (fl, fh) = (&g.kface[j as usize * ni + i as usize], &g.kface[(nk * nj + j as usize) * ni + i as usize]);
213                for m in 1..=NG as isize {
214                    w[id(i, j, -m)] = reflect3(&w[id(i, j, m - 1)], fl.n);
215                    w[id(i, j, nkk + m - 1)] = reflect3(&w[id(i, j, nkk - m)], fh.n);
216                }
217            }
218        }
219    }

221    /// Evaluate `res = dU/dt` for the current `self.u` (also refreshes `w` and the face fluxes).
222    pub fn eval_residual(&mut self) {
223        let (ni, nj, nk) = (self.grid.ni, self.grid.nj, self.grid.nk);
224        let lim = if self.iter < self.cfg.first_order_iters { Limiter::FirstOrder } else { self.cfg.limiter };
225        let low_mach = self.cfg.low_mach;
226        {
227            let (nw, plane) = (ni + 2 * NG, (ni + 2 * NG) * (nj + 2 * NG));
228            let u = &self.u;
229            self.w.par_chunks_mut(plane).enumerate().filter(|(kk, _)| *kk >= NG && *kk < nk + NG).for_each(|(kk, pl)| {
230                let k = kk - NG;
231                for j in 0..nj {
232                    for i in 0..ni {
233                        pl[(j + NG) * nw + i + NG] = cons_to_prim3(&u[(k * nj + j) * ni + i]);
234                    }
235                }
236            });
237        }
238        self.fill_ghosts();
239        let Solver3 { grid, w, fi, fj, fk, res, solid, rot_i, rot_j, rot_k, .. } = self;
240        let has_solid = !grid.solid.is_empty();
241        let fam = Family { ni, nj, nk, w, solid, has_solid, lim, low_mach };
242        fam.fluxes(0, &grid.iface, rot_i, fi);
243        fam.fluxes(1, &grid.jface, rot_j, fj);
244        fam.fluxes(2, &grid.kface, rot_k, fk);
245        let (grid, fi, fj, fk) = (&*grid, &*fi, &*fj, &*fk);
246        let solid: &[bool] = solid;
247        res.par_chunks_mut(ni * nj).enumerate().for_each(|(k, pl)| {
248            for j in 0..nj {
249                for i in 0..ni {
250                    let c = (k * nj + j) * ni + i;
251                    if solid[c] {
252                        pl[j * ni + i] = [0.0; 5];
253                        continue;
254                    }
255                    let (il, jl) = ((k * nj + j) * (ni + 1) + i, (k * (nj + 1) + j) * ni + i);
256                    let (ir, jr, kr) = (il + 1, jl + ni, c + ni * nj);
257                    let inv_v = 1.0 / grid.vol[c];
258                    let mut out = [0.0; 5];
259                    for q in 0..5 {
260                        out[q] = -((fi[ir][q] * grid.iface[ir].area - fi[il][q] * grid.iface[il].area)
261                            + (fj[jr][q] * grid.jface[jr].area - fj[jl][q] * grid.jface[jl].area)
262                            + (fk[kr][q] * grid.kface[kr].area - fk[c][q] * grid.kface[c].area))
263                            * inv_v;
264                    }
265                    pl[j * ni + i] = out;
266                }
267            }
268        });
269    }

271    /// Local pseudo-time step into `self.dt` from the primitives of the last residual evaluation:
272    /// `CFL V / (L_i + L_j + L_k)` (`DtMode::Sum`) or `CFL V / (3 min L_d)`; `L_d = (lam(lo) + lam(hi)) / 2`,
273    /// `lam(f) = (|u.n| + c) area`. Zero in solid cells.
274    pub(crate) fn compute_dt(&mut self, cfl: f64, mode: DtMode) {
275        let (ni, nj, nk) = (self.grid.ni, self.grid.nj, self.grid.nk);
276        let (grid, w, solid) = (&self.grid, &self.w, &self.solid);
277        self.dt.par_chunks_mut(ni * nj).enumerate().for_each(|(k, pl)| {
278            for j in 0..nj {
279                for i in 0..ni {
280                    let c = (k * nj + j) * ni + i;
281                    if solid[c] {
282                        pl[j * ni + i] = 0.0;
283                        continue;
284                    }
285                    let wc = &w[pidx3(ni, nj, nk, i as isize, j as isize, k as isize)];
286                    let cs = sound_speed3(wc);
287                    let lam = |f: &Face3| ((wc[1] * f.n[0] + wc[2] * f.n[1] + wc[3] * f.n[2]).abs() + cs) * f.area;
288                    let il = (k * nj + j) * (ni + 1) + i;
289                    let jl = (k * (nj + 1) + j) * ni + i;
290                    let li = 0.5 * (lam(&grid.iface[il]) + lam(&grid.iface[il + 1]));
291                    let lj = 0.5 * (lam(&grid.jface[jl]) + lam(&grid.jface[jl + ni]));
292                    let lk = 0.5 * (lam(&grid.kface[c]) + lam(&grid.kface[c + ni * nj]));
293                    pl[j * ni + i] = match mode {
294                        DtMode::Sum => cfl * grid.vol[c] / (li + lj + lk),
295                        DtMode::MinDirection => cfl * grid.vol[c] / (3.0 * li.min(lj).min(lk)),
296                    };
297                }
298            }
299        });
300    }

302    /// One SSP-RK3 (Shu-Osher) step with local time stepping; returns the density residual of the first stage.
303    fn rk3_step(&mut self) -> f64 {
304        self.eval_residual();
305        let rn = self.density_residual();
306        self.compute_dt(self.cfg.cfl, DtMode::Sum);
307        self.u0.copy_from_slice(&self.u);
308        self.stage(1.0, 0.0);
309        self.eval_residual();
310        self.stage(0.25, 0.75);
311        self.eval_residual();
312        self.stage(2.0 / 3.0, 1.0 / 3.0);
313        rn
314    }
315
316    /// `u = a (u + dt res) + b u0` (fluid cells).
317    fn stage(&mut self, a: f64, b: f64) {
318        let (res, dt, u0, solid) = (&self.res, &self.dt, &self.u0, &self.solid);
319        self.u.par_iter_mut().enumerate().for_each(|(c, q)| {
320            if solid[c] {
321                return;
322            }
323            for k in 0..5 {
324                q[k] = a * (q[k] + dt[c] * res[c][k]) + b * u0[c][k];
325            }
326        });
327    }

330/// Everything the face-flux kernel of one direction reads.
331struct Family<'a> {
332    ni: usize,
333    nj: usize,
334    nk: usize,
335    w: &'a [Prim3],
336    solid: &'a [bool],
337    has_solid: bool,
338    lim: Limiter,
339    low_mach: bool,
340}
341
342impl Family<'_> {
343    #[inline]
344    fn is_solid(&self, c: [isize; 3]) -> bool {
345        self.has_solid
346            && c[0] >= 0
347            && c[1] >= 0
348            && c[2] >= 0
349            && (c[0] as usize) < self.ni
350            && (c[1] as usize) < self.nj
351            && (c[2] as usize) < self.nk
352            && self.solid[(c[2] as usize * self.nj + c[1] as usize) * self.ni + c[0] as usize]
353    }
354
355    #[inline]
356    fn at(&self, c: [isize; 3]) -> Prim3 {
357        self.w[pidx3(self.ni, self.nj, self.nk, c[0], c[1], c[2])]
358    }
359
360    /// Fluxes of the faces of direction `dir` (`faces` / `rots` / `out` share the layout of `Grid3::{iface, jface, kface}`).
361    ///
362    /// Solid cells: a face between two solid cells carries no flux; between a fluid cell `a` and a solid cell the right
363    /// (solid-side) states are `reflect3(a)` and `reflect3(a-1)` (`a-1`: the next fluid cell behind `a`, `a` itself if that
364    /// one is solid too); a fluid face whose outer stencil cell is solid takes the reflection of the adjacent fluid cell.
365    fn fluxes(&self, dir: usize, faces: &[Face3], rots: &[[f64; 2]], out: &mut [Cons3]) {
366        let nfi = self.ni + (dir == 0) as usize;
367        let nfj = self.nj + (dir == 1) as usize;
368        let plane = nfi * nfj;
369        let e: [isize; 3] = match dir {
370            0 => [1, 0, 0],
371            1 => [0, 1, 0],
372            _ => [0, 0, 1],
373        };
374        let fstride = [1, nfi, plane][dir];
375        out.par_chunks_mut(plane).enumerate().for_each(|(k, po)| {
376            for j in 0..nfj {
377                for i in 0..nfi {
378                    let fidx = k * plane + j * nfi + i;
379                    let f = &faces[fidx];
380                    if f.area == 0.0 {
381                        po[j * nfi + i] = [0.0; 5];
382                        continue;
383                    }
384                    let p0 = [i as isize, j as isize, k as isize];
385                    let off = |s: isize| [p0[0] + s * e[0], p0[1] + s * e[1], p0[2] + s * e[2]];
386                    let (cm2, cm1, cp1) = (off(-2), off(-1), off(1));
387                    let (mut wm2, mut wm1, mut wp0, mut wp1) = (self.at(cm2), self.at(cm1), self.at(p0), self.at(cp1));
388                    if self.has_solid {
389                        let (sm1, sp0) = (self.is_solid(cm1), self.is_solid(p0));
390                        if sm1 && sp0 {
391                            po[j * nfi + i] = [0.0; 5];
392                            continue;
393                        }
394                        if sm1 {
395                            wm1 = reflect3(&wp0, f.n);
396                            wm2 = if self.is_solid(cp1) { wm1 } else { reflect3(&wp1, f.n) };
397                        } else if sp0 {
398                            wp0 = reflect3(&wm1, f.n);
399                            wp1 = if self.is_solid(cm2) { wp0 } else { reflect3(&wm2, f.n) };
400                        } else {
401                            if self.is_solid(cm2) {
402                                wm2 = reflect3(&wm1, faces[fidx - fstride].n);
403                            }
404                            if self.is_solid(cp1) {
405                                wp1 = reflect3(&wp0, faces[fidx + fstride].n);
406                            }
407                        }
408                    }
409                    po[j * nfi + i] = face_flux_lm3(&wm2, &wm1, &wp0, &wp1, self.lim, f.n, rots[fidx], self.low_mach);
410                }
411            }
412        });
413    }

428    fn max_du(s: &Solver3, q0: &Cons3) -> f64 {

486    #[test]
487    fn axisymmetric_flow_stays_axisymmetric() {
488        let g2 = cone2d(32, 16);
489        let nk = 8;
490        let g3 = cone3(&g2, nk);
491        let fs = freestream3(2.0, 0.0);
492        // The scheme is rotation-equivariant, so azimuthal velocity stays at roundoff for Minmod and first order. The smooth
493        // van Albada limiter (shared with 2D, `EPS = 1e-14`) is discontinuous at `b = 0` for `|a| ~ 1e-9`: a roundoff sign flip
494        // of `b` changes the slope by up to `a / 2`, which seeds ~1e-11 azimuthal velocity in the first step; it does not grow.
495        let ip = ImplicitParams { cfl_max: 20.0, ..ImplicitParams::DEFAULT };
496        // The implicit scheme (two defect-correction passes) amplifies roundoff in the azimuthal direction by ~1e5 in the
497        // far-field corner before it saturates at ~5e-11 (1e-13 with `sub_iters = 1`); the converged solution is unaffected.
498        for (scheme, lim, tol, tol_p) in [
499            (Scheme3::Explicit, Limiter::Minmod, 1e-12, 1e-10),
500            (Scheme3::Implicit(ip), Limiter::Minmod, 1e-9, 1e-10),
501            (Scheme3::Explicit, Limiter::VanAlbada, 1e-9, 1e-8),
502        ] {
503            let mut s = Solver3::new(g3.clone(), cfg(fs, scheme, lim));
504            for _ in 0..50 {
505                s.step();
506            }
507            let (mut az, mut dp) = (0.0f64, 0.0f64);
508            for k in 0..nk {
509                for j in 0..g3.nj {
510                    for i in 0..g3.ni {
511                        let c = g3.cell(i, j, k);
512                        let w = cons_to_prim3(&s.u[c]);
513                        let xc = g3.xc[c];
514                        let th = xc[1].atan2(xc[2]);
515                        az = az.max((w[2] * th.cos() - w[3] * th.sin()).abs());
516                        let w0 = cons_to_prim3(&s.u[g3.cell(i, j, 0)]);
517                        dp = dp.max((w[4] - w0[4]).abs());
518                    }
519                }
520            }
521            println!("symmetry {scheme:?} {lim:?}: max azimuthal velocity {az:e}, max pressure spread over k {dp:e}");
522            assert!(az <= tol && dp <= tol_p, "{az:e} {dp:e}");
523        }
524    }

526    /// Wall `Cp` (first fluid cell) of the 3D solver at `k = 0` and of the 2D solver, converged; max over wall cells of the difference.
527    fn cone_error(nk: usize) -> f64 {
528        use crate::gas::{cons_to_prim, freestream};
529        use crate::solver::{Bc, Config, Scheme, Solver};
530        let g2 = cone2d(40, 20);
531        let g3 = cone3(&g2, nk);
532        let mach = 2.0;
533        let ip = ImplicitParams { cfl_max: 20.0, ..ImplicitParams::DEFAULT };
534        let mut s3 = Solver3::new(g3, Config3 { first_order_iters: 50, ..cfg(freestream3(mach, 0.0), Scheme3::Implicit(ip), Limiter::VanAlbada) });
535        let mut s2 = Solver::new(
536            cone2d(40, 20),
537            Config {
538                cfl: 0.5,
539                limiter: Limiter::VanAlbada,
540                freestream: freestream(mach),
541                bc: [Bc::Characteristic, Bc::Characteristic, Bc::SlipWall, Bc::Characteristic],
542                first_order_iters: 50,
543            },
544        );
545        s2.set_scheme(Scheme::Implicit(ip));
546        let (mut r0, mut r) = (0.0, f64::INFINITY);
547        for it in 0..3000 {
548            r = s2.step();
549            if it == 20 {
550                r0 = r;
551            }
552            if it > 20 && r < 1e-6 * r0 {
553                break;
554            }
555        }
556        println!("2D residual {r:e} (r0 {r0:e}) after {} iterations", s2.iter);
557        let (mut r0, mut r) = (0.0, f64::INFINITY);
558        let t0 = std::time::Instant::now();
559        for it in 0..3000 {
560            r = s3.step();
561            if it == 20 {
562                r0 = r;
563            }
564            if it > 20 && r < 1e-6 * r0 {
565                break;
566            }
567        }
568        let us = t0.elapsed().as_secs_f64() * 1e6 / (s3.iter as f64 * s3.u.len() as f64);
569        println!("3D nk={nk}: {us:.2} us per cell per implicit iteration ({} cells, {} threads)", s3.u.len(), rayon::current_num_threads());
570        println!("3D nk={nk} residual {r:e} (r0 {r0:e}) after {} iterations", s3.iter);
571        let (pinf, q) = (1.0 / crate::gas::GAMMA, 0.5 * mach * mach);
572        let mut e = 0.0f64;
573        for i in n_up(&g2)..g2.ni {
574            let cp2 = (cons_to_prim(&s2.u[i])[3] - pinf) / q;
575            let cp3 = (cons_to_prim3(&s3.u[s3.grid.cell(i, 0, 0)])[4] - pinf) / q;
576            e = e.max((cp3 - cp2).abs());
577        }
578        e
579    }

589    /// Rectangular (aligned) revolved grid with one solid box.
590    pub(crate) fn fin_grid() -> Grid3 {
591        let (ni, nj, nk) = (10, 8, 8);
592        let (mut x, mut r) = (vec![0.0; (ni + 1) * (nj + 1)], vec![0.0; (ni + 1) * (nj + 1)]);
593        for j in 0..=nj {
594            for i in 0..=ni {
595                x[j * (ni + 1) + i] = 2.0 * i as f64 / ni as f64;
596                r[j * (ni + 1) + i] = j as f64 / nj as f64;
597            }
598        }
599        let g2 = Grid::try_from_nodes(ni, nj, x, r).unwrap();
600        let theta: Vec<f64> = (0..=nk).map(|k| std::f64::consts::PI * k as f64 / nk as f64).collect();
601        let mut g = revolve(&g2, nk, &theta).unwrap();
602        g.axis_cells = ni;
603        g.solid = vec![SolidBox { i: 3..5, j: 0..3, k: 3..4 }];
604        g
605    }

607    #[test]
608    fn fin_wall_is_impermeable_and_the_run_converges() {
609        let g = fin_grid();
610        let fs = freestream3(2.0, 0.0);
611        let mut s = Solver3::new(g.clone(), cfg(fs, Scheme3::Implicit(ImplicitParams { cfl_max: 20.0, ..ImplicitParams::DEFAULT }), Limiter::VanAlbada));
612        s.cfg.first_order_iters = 50;
613        let (mut rmax, mut rn) = (0.0f64, 0.0);
614        for _ in 0..200 {
615            rn = s.step();
616            rmax = rmax.max(rn);
617        }
618        let q0 = prim_to_cons3(&fs);
619        assert!(s.u.iter().flatten().all(|v| v.is_finite()));
620        println!("fin wall: residual max {rmax:e} -> {rn:e} (drop {:.2} orders)", (rmax / rn).log10());
621        assert!(rn <= 1e-2 * rmax, "{rmax:e} -> {rn:e}");
622        // solid cells untouched
623        for k in 3..4 {
624            for j in 0..3 {
625                for i in 3..5 {
626                    assert_eq!(s.u[g.cell(i, j, k)], q0);
627                }
628            }
629        }
630        s.eval_residual();
631        let scale = fs[0] * fs[1];
632        let mut worst = 0.0f64;
633        let mut nwall = 0;
634        for k in 0..g.nk {
635            for j in 0..g.nj {
636                for i in 0..=g.ni {
637                    let a = i > 0 && s.solid[g.cell(i - 1, j, k)];
638                    let b = i < g.ni && s.solid[g.cell(i, j, k)];
639                    if a != b {
640                        worst = worst.max(s.fi[(k * g.nj + j) * (g.ni + 1) + i][0].abs());
641                        nwall += 1;
642                    }
643                }
644            }
645            for j in 0..=g.nj {
646                for i in 0..g.ni {
647                    let a = j > 0 && s.solid[g.cell(i, j - 1, k)];
648                    let b = j < g.nj && s.solid[g.cell(i, j, k)];
649                    if a != b {
650                        worst = worst.max(s.fj[(k * (g.nj + 1) + j) * g.ni + i][0].abs());
651                        nwall += 1;
652                    }
653                }
654            }
655        }
656        for k in 0..=g.nk {
657            for j in 0..g.nj {
658                for i in 0..g.ni {
659                    let a = k > 0 && s.solid[g.cell(i, j, k - 1)];
660                    let b = k < g.nk && s.solid[g.cell(i, j, k)];
661                    if a != b {
662                        worst = worst.max(s.fk[(k * g.nj + j) * g.ni + i][0].abs());
663                        nwall += 1;
664                    }
665                }
666            }
667        }
668        println!("fin wall: {nwall} wall faces, max |F[0]| = {worst:e} (rho u = {scale})");
669        assert!(nwall > 0 && worst <= 1e-14 * scale, "{worst:e}");
670    }
