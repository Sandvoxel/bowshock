1//! Menter SST k-omega turbulence model (2003 version) for the viscous solver (`Turbulence::Sst`).
2//!
3//! The model transports the specific turbulence kinetic energy `k` and the specific dissipation rate `omega` of every cell
4//! (`Solver::turb`, `[k, omega]`) with a loosely coupled pseudo-time step at the end of every `Solver::step`; the eddy
5//! viscosity `mu_t` it produces (`Solver::mu_t`) enters the mean-flow equations through the viscous terms of `viscous.rs`.
6//! Units as in `viscous.rs` (`rho_inf = c_inf = 1`, lengths in nose diameters, `nu = mu / rho`).
7//!
8//! ```text
9//!   constants: a1 = 0.31, beta* = 0.09;  set 1: sigma_k 0.85, sigma_w 0.5, beta 0.075, gamma 5/9;
10//!              set 2: sigma_k 1.0, sigma_w 0.856, beta 0.0828, gamma 0.44;    phi = F1 phi_1 + (1 - F1) phi_2
11//!   free stream: k_inf = 9e-9,  omega_inf = 1e-6 re_c                         (mu_t / mu = 0.009)
12//!   d = distance of the cell centroid to the nearest no-slip wall face
13//!   S^2 = 2 (u_x^2 + v_r^2 + (v/r)^2) + (u_r + v_x)^2 - 2/3 div^2,   S = sqrt(max(S^2, 0))     (cell gradients)
14//!   X   = 2 rho sigma_w2 (grad k . grad omega) / omega,                CD = max(X, 1e-20)
15//!   F1  = tanh(a^4),  a = min( max( sqrt(k) / (beta* omega d), 500 nu / (d^2 omega) ), 4 rho sigma_w2 k / (CD d^2) )
16//!   F2  = tanh(b^2),  b = max( 2 sqrt(k) / (beta* omega d), 500 nu / (d^2 omega) )
17//!   mu_t = min( rho a1 k / max(a1 omega, S F2), 1e5 / re_c )
18//!   P    = min( mu_t S^2, 10 beta* rho k omega )
19//!   rho dk/dt     = (-conv_k + diff_k) / V + P - beta* rho omega k
20//!   rho domega/dt = (-conv_w + diff_w) / V + gamma rho P / max(mu_t, 1e-30) - beta rho omega^2 + (1 - F1) X
21//!   conv = sum over the four faces of (outward mass flow) (phi_upwind - phi_c), inflow faces only; the mass flow of a face is
22//!          `F[0] w_f` of the stored inviscid face flux
23//!   diff = sum over the four faces of the outward (mu_f + sigma_f mu_t,f) (g_f . n) w_f, `g_f` the corrected face gradient of
24//!          `viscous.rs` (a compact normal derivative with the averaged tangential part), `sigma_f` the mean of the blended
25//!          `sigma` of the two sides
26//! ```
27//!
28//! Boundary states of the two scalars (the ghost of a face, mirrored in the face centre like the mean-flow ghosts):
29//! `Bc::Freestream | Characteristic | PressureOutflow` take `(k_inf, omega_inf)`; `Extrapolate`, slip walls and the axis the
30//! cell values; a no-slip wall `k_ghost = -k_c`, `omega_ghost = 2 omega_w - omega_c`, `omega_w = 60 nu_c / (0.075 d_c^2)`
31//! (the face values are then `k = 0`, `omega = omega_w`). With a wall roughness `ks > 0` (`Viscous::roughness`, equivalent sand grain in
32//! nose diameters) the wall cells take the Aupoix (2015) "Colebrook" rough-wall values of [`aupoix_wall`]: `k_ghost = 2 k_w - k_c`,
33//! `omega_ghost = 2 omega_w - omega_c`, and the face eddy viscosity of a no-slip face is the wall value `mu_t,w = rho k_w / omega_w`
34//! instead of 0 (`Solver::rough_wall`, read by `viscous::face_mu`: the mean-flow wall shear is `(mu + mu_t,w) u_c / d_c`). The rough state is for
35//! the body surface (`Walls::rough_jface`; the base wall and the sting keep the smooth state) and its roughness is phased in over the first
36//! [`ROUGH_RAMP_CELLS`] wall cells ([`rough_ramp`]); `ks = 0` takes none of these paths.
37//!
38//! The update is a sequence of data-parallel kernels (`update`): cell gradients of `k`, `omega`; the blending functions; the
39//! face coefficients; the per-cell residual and the coefficients of the two linear systems; the line solves; the clipped
40//! update with the new eddy viscosity. Per equation, with `D_k = beta* omega`, `D_w = 2 beta omega`, `M-` / `M+` the mass flow
41//! in the positive direction through the lower / upper face of a line direction and `G = (mu_f + sigma_f mu_t,f) w_f / |d|`:
42//!
43//! ```text
44//!   diag0 = rho V / dt + rho V D
45//!   direction dir:  lo = -(max(M-, 0) + G-),  up = -(max(-M+, 0) + G+),  dd = max(M-, 0) + max(-M+, 0) + G- + G+
46//!   i lines:  (diag0 + dd_i) x_c + lo_i x_{c-1} + up_i x_{c+1} = V R_c                    R_c = rho dphi/dt
47//!   j lines:  (1 + dd_j / diag0) dphi_c + (lo_j / diag0) dphi_{c-1} + (up_j / diag0) dphi_{c+1} = x_c
48//!   phi_new = max(phi + dphi, 0.1 phi), then k >= 1e-3 k_inf, omega >= 1e-3 omega_inf;  then mu_t is recomputed
49//! ```
50//!
51//! A neighbour that is a ghost or a solid cell drops its off-diagonal (the diagonal part stays). The two scalars are solved
52//! together with the block-tridiagonal line solver of `implicit.rs` (4x4 blocks that are diagonal: entries 0 and 5 hold the
53//! `k` and `omega` coefficients, entries 10 and 15 are `1` on the diagonal block and `0` elsewhere).

82/// Distance of every cell centroid to the nearest no-slip wall face segment of the viscous run `v` with boundary conditions `bc` and
83/// body block `blk` and nozzle exit `nozzle` (jet faces are not walls; `f64::INFINITY` without a no-slip face and in solid cells): what `Solver::set_viscous` computes for the
84/// turbulence model, for ports that keep their own copy of the field.
85pub fn wall_distances(grid: &Grid, v: Viscous, bc: &[Bc; 4], blk: Option<Block>, nozzle: Option<Nozzle>) -> Vec<f64> {
86    wall_distance(grid, &Walls::new(Some(v), bc, blk, nozzle))
87}

89/// Distance of every cell centroid to the nearest no-slip wall face segment (`f64::INFINITY` without a no-slip face and in
90/// solid cells); brute force over the wall faces.
91pub(crate) fn wall_distance(grid: &Grid, walls: &Walls) -> Vec<f64> {
92    let (ni, nj) = (grid.ni, grid.nj);
93    let mut segs: Vec<[f64; 4]> = Vec::new();
94    for j in 0..=nj {
95        for i in 0..ni {
96            if walls.noslip_jface(i, j) {
97                let (a, b) = (grid.node(i, j), grid.node(i + 1, j));
98                segs.push([a.0, a.1, b.0, b.1]);
99            }
100        }
101    }
102    for j in 0..nj {
103        for i in 0..=ni {
104            if walls.noslip_iface(i, j) {
105                let (a, b) = (grid.node(i, j), grid.node(i, j + 1));
106                segs.push([a.0, a.1, b.0, b.1]);
107            }
108        }
109    }
110    let mut d = vec![f64::INFINITY; ni * nj];
111    d.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
112        for (i, out) in row.iter_mut().enumerate() {
113            if walls.blk.is_some_and(|b| b.solid(i, j)) {
114                continue;
115            }
116            let (px, pr) = (grid.xc[j * ni + i], grid.rc[j * ni + i]);
117            for s in &segs {
118                let (ex, er) = (s[2] - s[0], s[3] - s[1]);
119                let t = (((px - s[0]) * ex + (pr - s[1]) * er) / (ex * ex + er * er)).clamp(0.0, 1.0);
120                *out = out.min((px - (s[0] + t * ex)).hypot(pr - (s[1] + t * er)));
121            }
122        }
123    });
124    d
125}

127/// Initialise the turbulence state of `s` (free stream), its wall distance and the eddy viscosity `rho k / omega`.
128pub(crate) fn init(s: &mut Solver, re_c: f64) {
129    let n = s.grid.ni * s.grid.nj;
130    s.turb = vec![[K_INF, omega_inf(re_c)]; n];
131    s.dwall = wall_distance(&s.grid, &s.walls());
132    let (ni, blk) = (s.grid.ni, s.block);
133    for c in 0..n {
134        if !blk.is_some_and(|b| b.solid(c % ni, c / ni)) {
135            s.mu_t[c] = s.u[c][0] * K_INF / omega_inf(re_c);
136        }
137    }
138}

140/// Where a ghost state of a face comes from.
141#[derive(Clone, Copy, PartialEq)]
142enum Kind {
143    /// The values of the adjacent cell (extrapolation, slip wall, axis).
144    Cell,
145    /// The free-stream values.
146    Inf,
147    /// The fixed `[k, omega]` of a nozzle exit.
148    Fixed([f64; 2]),
149    /// A no-slip wall.
150    Wall,
151}
152
153fn bc_kind(bc: Bc) -> Kind {
154    match bc {
155        Bc::Freestream | Bc::Characteristic | Bc::PressureOutflow => Kind::Inf,
156        Bc::Extrapolate | Bc::SlipWall => Kind::Cell,
157    }
158}

160/// Read-only data of one update.
161struct Ctx<'a> {
162    /// `[k_w, omega_w, mu_t,w]` of the wall cells of a rough run (`Solver::rough_wall`); empty for smooth walls.
163    rough: &'a [[f64; 3]],
164    grid: &'a Grid,
165    w: &'a [Prim],
166    walls: Walls,
167    bc: [Bc; 4],
168    mu: &'a [f64],
169    mu_t: &'a [f64],
170    turb: &'a [[f64; 2]],
171    dwall: &'a [f64],
172    inf: [f64; 2],
173}
174
175impl Ctx<'_> {
176    #[inline]
177    fn solid(&self, i: usize, j: usize) -> bool {
178        self.walls.blk.is_some_and(|b| b.solid(i, j))
179    }
180
181    #[inline]
182    fn rho(&self, c: usize) -> f64 {
183        let ni = self.grid.ni;
184        self.w[pidx(ni, (c % ni) as isize, (c / ni) as isize)][0]
185    }
186
187    /// Sides of i-face `(i, j)` and the origin of its ghost state; `None` inside the solid block.
188    fn iface(&self, i: usize, j: usize) -> Option<(FaceSides, Kind)> {
189        let sides = iface_sides(self.grid, self.w, &self.walls, i, j)?;
190        let kind = if i == 0 {
191            bc_kind(self.bc[0])
192        } else if i == self.grid.ni {
193            bc_kind(self.bc[1])
194        } else if let (true, Some(n)) = (sides.l.ghost && sides.l.fixed.is_some(), self.walls.nozzle) {
195            Kind::Fixed(n.turb)
196        } else if sides.l.ghost && sides.noslip {
197            Kind::Wall
198        } else {
199            Kind::Cell
200        };
201        Some((sides, kind))
202    }
203
204    /// Sides of j-face `(i, j)` and the origin of its ghost state; `None` inside the solid block.
205    fn jface(&self, i: usize, j: usize) -> Option<(FaceSides, Kind)> {
206        let sides = jface_sides(self.grid, self.w, &self.walls, i, j)?;
207        let kind = if sides.l.ghost {
208            if sides.noslip {
209                Kind::Wall
210            } else if j == 0 {
211                bc_kind(self.bc[2])
212            } else {
213                Kind::Cell
214            }
215        } else if sides.r.ghost {
216            bc_kind(self.bc[3])
217        } else {
218            Kind::Cell
219        };
220        Some((sides, kind))
221    }
222
223    /// `[k, omega]` of one side of a face.
224    fn kw(&self, s: &Side, kind: Kind) -> [f64; 2] {
225        let c = s.cell;
226        if !s.ghost {
227            return self.turb[c];
228        }
229        match kind {
230            Kind::Cell => self.turb[c],
231            Kind::Inf => self.inf,
232            Kind::Fixed(v) => v,
233            Kind::Wall => {
234                let [k, om] = self.turb[c];
235                if self.rough.is_empty() {
236                    let nu = self.mu[c] / self.rho(c);
237                    let d = self.dwall[c];
238                    [-k, 2.0 * (60.0 * nu / (BETA1 * d * d)) - om]
239                } else {
240                    let r = &self.rough[c];
241                    [2.0 * r[0] - k, 2.0 * r[1] - om]
242                }
243            }
244        }
245    }
246}

248/// `S^2` of a cell from its gradient `(u_x, u_r, v_x, v_r, ..)`, radial velocity and radius.
249#[inline]
250fn strain2(g: &[f64; 6], v: f64, r: f64) -> f64 {
251    let vr = v / r;
252    let div = g[0] + g[3] + vr;
253    let shear = g[1] + g[2];
254    2.0 * (g[0] * g[0] + g[3] * g[3] + vr * vr) + shear * shear - 2.0 / 3.0 * div * div
255}

257/// Argument pieces `(sqrt(k) / (beta* omega d), 500 nu / (d^2 omega))` of the blending functions.
258#[inline]
259fn blend_args(nu: f64, k: f64, om: f64, d: f64) -> (f64, f64) {
260    (k.sqrt() / (BETA_STAR * om * d), 500.0 * nu / (d * d * om))
261}

263/// `F2` of a cell.
264#[inline]
265fn f2_of(nu: f64, k: f64, om: f64, d: f64) -> f64 {
266    let (t1, t2) = blend_args(nu, k, om, d);
267    let b = (2.0 * t1).max(t2);
268    (b * b).tanh()
269}

271/// Fixed-point iterations of `aupoix_wall` (and its GPU twin). The map `u_tau^2 -> (nu + nu_t,w) U / d` contracts by at most about 0.65 per iteration, from
272/// a start that is up to a factor `1 + nu_t,w / nu` (10 and more at a leading edge) below the fixed point: 4 iterations leave that error unresolved and the
273/// plate of G2r did not converge (density residual stuck at 0.5-1), 5 or more converge; 16 leave under 1e-3.
274pub const AUPOIX_ITERATIONS: usize = 16;
275
276/// Upper bound of `nu_t,w / nu`: the cap `1e5 / re_c` of the eddy viscosity of the cells (`eddy_viscosity`) in units of the free-stream `nu`. Where the
277/// smooth `omega_w` bound is active (`d` large against the roughness) `nu_t,w` grows with `u_tau^2` and the fixed point has no solution without it.
278pub const NU_T_WALL_MAX: f64 = 1e5;
279
280/// Number of wall cells over which the roughness is phased in from the wall start (`Viscous::wall_i_start`): the roughness of the first wall cells is
281/// `ks rough_ramp(i)`. A roughness wall state that starts as a step feeds turbulence (`k_w` with the low rough `omega_w`) into the slip cells ahead of the
282/// wall start, where it spreads up their column (`mu_t / mu` 15-45 against 3 smooth) and the column oscillates with period 2 at transonic speed (M 0.9-1.3,
283/// ks / d 6e-4 to 1.2e-3: the stall at 80 000 iterations); a ramp of 2 and 3 cells still oscillates, 4 and 16 do not.
284pub const ROUGH_RAMP_CELLS: usize = 8;
285
286/// Fraction of the roughness `ks` that wall cell `i` carries: `min(1, (i + 1 - wall_i_start) / ROUGH_RAMP_CELLS)`.
287#[inline]
288pub fn rough_ramp(i: usize, wall_i_start: usize) -> f64 {
289    ((i + 1).saturating_sub(wall_i_start) as f64 / ROUGH_RAMP_CELLS as f64).min(1.0)
290}

292/// Aupoix (2015) "Colebrook" rough-wall state of a no-slip wall cell with kinematic viscosity `nu`, speed `speed`, wall distance `d`
293/// and equivalent sand-grain roughness `ks > 0`: `[k_w, omega_w, nu_t,w]` with `nu_t,w = k_w / omega_w`.
294///
295/// ```text
296///   ks+      = u_tau ks / nu                                                    (floored at 1e-30)
297///   k_w+     = max(0, tanh( (log10(ks+ / 30) + 1 - tanh(ks+ / 125)) tanh(ks+ / 125) ) / sqrt(beta*))
298///   omega_w+ = (300 / ks+^2) / tanh(15 / (4 ks+)) + (191 / ks+) (1 - exp(-ks+ / 250))
299///   k_w = k_w+ u_tau^2,   omega_w = min(omega_w+ u_tau^2 / nu, 60 nu / (beta1 d^2))
300///   u_tau^2 = (nu + nu_t,w) speed / d      (the discrete wall shear (mu + mu_t,w) U / d), `AUPOIX_ITERATIONS` fixed-point iterations from nu speed / d
301/// ```
302///
303/// `omega_w+ u_tau^2 / nu` is evaluated as `300 nu / ks^2 / tanh(..) + 191 u_tau / ks (1 - exp(..))`, which stays finite at rest.
304pub fn aupoix_wall(nu: f64, speed: f64, d: f64, ks: f64) -> [f64; 3] {
305    let smooth = 60.0 * nu / (BETA1 * d * d);
306    let at = |u2: f64| -> [f64; 3] {
307        let u_tau = u2.sqrt();
308        let ks_plus = (u_tau * ks / nu).max(1e-30);
309        let t = (ks_plus / 125.0).tanh();
310        let k_plus = (((ks_plus / 30.0).log10() + 1.0 - t) * t).tanh().max(0.0) / BETA_STAR.sqrt();
311        let om_rough = 300.0 * nu / (ks * ks) / (15.0 / (4.0 * ks_plus)).tanh() + 191.0 * u_tau / ks * (1.0 - (-ks_plus / 250.0).exp());
312        let (k_w, om_w) = (k_plus * u2, om_rough.min(smooth));
313        [k_w, om_w, (k_w / om_w).min(NU_T_WALL_MAX * nu)]
314    };
315    let mut u2 = nu * speed / d;
316    for _ in 0..AUPOIX_ITERATIONS {
317        u2 = (nu + at(u2)[2]) * speed / d;
318    }
319    at(u2)
320}

322/// Eddy viscosity `min(rho a1 k / max(a1 omega, S F2), 1e5 / re_c)`.
323#[inline]
324fn eddy_viscosity(rho: f64, k: f64, om: f64, s: f64, f2: f64, re_c: f64) -> f64 {
325    (rho * A1 * k / (A1 * om).max(s * f2)).min(1e5 / re_c)
326}

336/// Kernel 1: Green-Gauss gradient `(k_x, k_r, omega_x, omega_r)` of every fluid cell, `phi_f = (phi_c + phi_neighbour) / 2`.
337fn scalar_gradients(cx: &Ctx) -> Vec<[f64; 4]> {
338    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
339    let mut out = vec![[0.0; 4]; ni * nj];
340    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
341        for (i, g_out) in row.iter_mut().enumerate() {
342            if cx.solid(i, j) {
343                continue;
344            }
345            let c = j * ni + i;
346            let pc = cx.turb[c];
347            let (sl, kl) = cx.iface(i, j).expect("fluid cell face");
348            let (sr, kr) = cx.iface(i + 1, j).expect("fluid cell face");
349            let (sb, kb) = cx.jface(i, j).expect("fluid cell face");
350            let (st, kt) = cx.jface(i, j + 1).expect("fluid cell face");
351            let faces = [
352                (&cx.grid.iface[j * (ni + 1) + i + 1], 1.0, cx.kw(&sr.r, kr)),
353                (&cx.grid.iface[j * (ni + 1) + i], -1.0, cx.kw(&sl.l, kl)),
354                (&cx.grid.jface[(j + 1) * ni + i], 1.0, cx.kw(&st.r, kt)),
355                (&cx.grid.jface[j * ni + i], -1.0, cx.kw(&sb.l, kb)),
356            ];
357            let mut g = [0.0; 4];
358            for (f, sgn, nb) in faces {
359                for m in 0..2 {
360                    let pf = 0.5 * (pc[m] + nb[m]);
361                    g[2 * m] += sgn * pf * f.nx * f.len;
362                    g[2 * m + 1] += sgn * pf * f.nr * f.len;
363                }
364            }
365            let inv_a = 1.0 / cx.grid.area[c];
366            for x in g.iter_mut() {
367                *x *= inv_a;
368            }
369            *g_out = g;
370        }
371    });
372    out
373}

375/// Kernel 2: `F1` and the cross-diffusion `X` of every fluid cell.
376fn blends(cx: &Ctx, tg: &[[f64; 4]]) -> Vec<Blend> {
377    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
378    let mut out = vec![Blend::default(); ni * nj];
379    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
380        for (i, b) in row.iter_mut().enumerate() {
381            if cx.solid(i, j) {
382                continue;
383            }
384            let c = j * ni + i;
385            let rho = cx.rho(c);
386            let nu = cx.mu[c] / rho;
387            let [k, om] = cx.turb[c];
388            let d = cx.dwall[c];
389            let g = &tg[c];
390            let x = 2.0 * rho * SIGMA_W2 * (g[0] * g[2] + g[1] * g[3]) / om;
391            let cd = x.max(1e-20);
392            let (t1, t2) = blend_args(nu, k, om, d);
393            let t3 = 4.0 * rho * SIGMA_W2 * k / (cd * d * d);
394            let a = t1.max(t2).min(t3);
395            *b = Blend { f1: a.powi(4).tanh(), x };
396        }
397    });
398    out
399}

414/// Kernel 3 (one face): see the module documentation; `m` is the mass flow of the face, `(xf, rf)` its centre.
415fn tface(cx: &Ctx, tg: &[[f64; 4]], blend: &[Blend], sides: &FaceSides, kind: Kind, f: &Face, xf: f64, rf: f64, m: f64) -> TFace {
416    let (kl, kr) = (cx.kw(&sides.l, kind), cx.kw(&sides.r, kind));
417    if f.rm == 0.0 {
418        return TFace { m, g: [0.0; 2], flux: [0.0; 2], l: kl, r: kr };
419    }
420    let w = f.w();
421    let (dd, ex, er) = stencil(cx.grid, sides, xf, rf);
422    let (gl, gr) = (&tg[sides.l.cell], &tg[sides.r.cell]);
423    let (mu_f, mut_f) = face_mu(cx.mu, cx.mu_t, cx.rough, sides);
424    let (f1l, f1r) = (blend[sides.l.cell].f1, blend[sides.r.cell].f1);
425    let mut out = TFace { m, g: [0.0; 2], flux: [0.0; 2], l: kl, r: kr };
426    for e in 0..2 {
427        let (s1, s2) = if e == 0 { (SIGMA_K1, SIGMA_K2) } else { (SIGMA_W1, SIGMA_W2) };
428        let sigma = 0.5 * (f1l * s1 + (1.0 - f1l) * s2 + f1r * s1 + (1.0 - f1r) * s2);
429        let coef = mu_f + sigma * mut_f;
430        let (gbx, gbr) = (0.5 * (gl[2 * e] + gr[2 * e]), 0.5 * (gl[2 * e + 1] + gr[2 * e + 1]));
431        let corr = (kr[e] - kl[e]) / dd - (gbx * ex + gbr * er);
432        let gn = (gbx + corr * ex) * f.nx + (gbr + corr * er) * f.nr;
433        out.flux[e] = coef * gn * w;
434        out.g[e] = coef * w / dd;
435    }
436    out
437}

439/// Kernel 3: scalar face data of every i-face and j-face.
440fn face_pass(cx: &Ctx, tg: &[[f64; 4]], blend: &[Blend], fi: &[Cons], fj: &[Cons]) -> (Vec<TFace>, Vec<TFace>) {
441    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
442    let mut ti = vec![TFace::default(); (ni + 1) * nj];
443    ti.par_chunks_mut(ni + 1).enumerate().for_each(|(j, row)| {
444        for (i, out) in row.iter_mut().enumerate() {
445            if let Some((sides, kind)) = cx.iface(i, j) {
446                let f = &cx.grid.iface[j * (ni + 1) + i];
447                let (xf, rf) = iface_centre(cx.grid, i, j);
448                *out = tface(cx, tg, blend, &sides, kind, f, xf, rf, fi[j * (ni + 1) + i][0] * f.w());
449            }
450        }
451    });
452    let mut tj = vec![TFace::default(); ni * (nj + 1)];
453    tj.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
454        for (i, out) in row.iter_mut().enumerate() {
455            if let Some((sides, kind)) = cx.jface(i, j) {
456                let f = &cx.grid.jface[j * ni + i];
457                let (xf, rf) = jface_centre(cx.grid, i, j);
458                *out = tface(cx, tg, blend, &sides, kind, f, xf, rf, fj[j * ni + i][0] * f.w());
459            }
460        }
461    });
462    (ti, tj)
463}

465/// Residual and line coefficients of one cell, per scalar.
466#[derive(Clone, Copy, Default)]
467struct CellTerms {
468    /// `R = rho dphi/dt`.
469    r: [f64; 2],
470    d0: [f64; 2],
471    /// `[direction][scalar]`, direction 0 = i, 1 = j.
472    lo: [[f64; 2]; 2],
473    dd: [[f64; 2]; 2],
474    up: [[f64; 2]; 2],
475}
476
477/// Kernel 4: residual `R`, `diag0` and the line coefficients of every fluid cell.
478fn cell_terms(cx: &Ctx, grad: &[[f64; 6]], blend: &[Blend], dt: &[f64], ti: &[TFace], tj: &[TFace]) -> Vec<CellTerms> {
479    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
480    let blk = cx.walls.blk;
481    let mut out = vec![CellTerms::default(); ni * nj];
482    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
483        for (i, o) in row.iter_mut().enumerate() {
484            if cx.solid(i, j) {
485                continue;
486            }
487            let c = j * ni + i;
488            let vol = cx.grid.vol[c];
489            let rho = cx.rho(c);
490            let [k, om] = cx.turb[c];
491            let phi = [k, om];
492            let b = blend[c];
493            let wc = &cx.w[pidx(ni, i as isize, j as isize)];
494            let s2 = strain2(&grad[c], wc[2], cx.grid.rc[c]);
495            let mut_c = cx.mu_t[c];
496            let prod = (mut_c * s2).min(10.0 * BETA_STAR * rho * k * om);
497            let beta = b.f1 * BETA1 + (1.0 - b.f1) * BETA2;
498            let gamma = b.f1 * GAMMA1 + (1.0 - b.f1) * GAMMA2;
499            let sources = [prod - BETA_STAR * rho * om * k, gamma * rho * prod / mut_c.max(1e-30) - beta * rho * om * om + (1.0 - b.f1) * b.x];
500            let dest = [BETA_STAR * om, 2.0 * beta * om];
501            let faces = [(&ti[j * (ni + 1) + i], &ti[j * (ni + 1) + i + 1]), (&tj[j * ni + i], &tj[(j + 1) * ni + i])];
502            let ends = [(i == blk.map_or(0, |b| b.row_start(j).unwrap_or(ni)), i == ni - 1), (j == blk.map_or(0, |b| b.col_start(i)), j == nj - 1)];
503            let mut rate = [0.0; 2];
504            for (dir, (lo_f, up_f)) in faces.iter().enumerate() {
505                let (a_lo, a_up) = (lo_f.m.max(0.0), (-up_f.m).max(0.0));
506                for e in 0..2 {
507                    // -conv + diff of this direction
508                    rate[e] += a_lo * (lo_f.l[e] - phi[e]) + a_up * (up_f.r[e] - phi[e]) + up_f.flux[e] - lo_f.flux[e];
509                    o.lo[dir][e] = if ends[dir].0 { 0.0 } else { -(a_lo + lo_f.g[e]) };
510                    o.up[dir][e] = if ends[dir].1 { 0.0 } else { -(a_up + up_f.g[e]) };
511                    o.dd[dir][e] = a_lo + a_up + lo_f.g[e] + up_f.g[e];
512                }
513            }
514            for e in 0..2 {
515                o.r[e] = rate[e] / vol + sources[e];
516                o.d0[e] = rho * vol / dt[c] + rho * vol * dest[e];
517            }
518        }
519    });
520    out
521}

525    let mut m = ZERO;
526    m[0] = a;
527    m[5] = b;
528    m[10] = 1.0;
529    m[15] = 1.0;
530    m
531}

534fn offdiag(a: f64, b: f64) -> M4 {

537    m[5] = b;

541/// One turbulence update at the end of a `Solver::step`, from the primitives, face fluxes, gradients and time step of the
542/// residual evaluation of this iteration (see the module documentation): residual and coefficients, the i- and j-line solves,
543/// the clipped update of `k`, `omega` and the new eddy viscosity.
544pub(crate) fn update(s: &mut Solver) {
545    let v = s.viscous.expect("SST needs the viscous terms");
546    let (ni, nj) = (s.grid.ni, s.grid.nj);
547    let n = ni * nj;
548    let inf = [K_INF, omega_inf(v.re_c)];
549    let (turb, mu_t, rnorm) = {
550        let cx =
551            Ctx { rough: &s.rough_wall, grid: &s.grid, w: &s.w, walls: s.walls(), bc: s.cfg.bc, mu: &s.mu, mu_t: &s.mu_t, turb: &s.turb, dwall: &s.dwall, inf };
552        let tg = scalar_gradients(&cx);
553        let blend = blends(&cx, &tg);
554        let (ti, tj) = face_pass(&cx, &tg, &blend, &s.fi, &s.fj);
555        let cells = cell_terms(&cx, &s.grad, &blend, &s.dt, &ti, &tj);
556
557        // coefficients of the two linear systems
558        let mut fac_i = Factor { lo: vec![ZERO; n], d: vec![EYE; n], up: vec![ZERO; n] };
559        let mut fac_j = Factor { lo: vec![ZERO; n], d: vec![EYE; n], up: vec![ZERO; n] };
560        let mut x: Vec<Cons> = vec![[0.0; 4]; n];
561        for c in 0..n {
562            if cx.solid(c % ni, c / ni) {
563                continue;
564            }
565            let t = &cells[c];
566            let vol = s.grid.vol[c];
567            x[c] = [vol * t.r[0], vol * t.r[1], 0.0, 0.0];
568            fac_i.lo[c] = offdiag(t.lo[0][0], t.lo[0][1]);
569            fac_i.up[c] = offdiag(t.up[0][0], t.up[0][1]);
570            fac_i.d[c] = diag(t.d0[0] + t.dd[0][0], t.d0[1] + t.dd[0][1]);
571            fac_j.lo[c] = offdiag(t.lo[1][0] / t.d0[0], t.lo[1][1] / t.d0[1]);
572            fac_j.up[c] = offdiag(t.up[1][0] / t.d0[0], t.up[1][1] / t.d0[1]);
573            fac_j.d[c] = diag(1.0 + t.dd[1][0] / t.d0[0], 1.0 + t.dd[1][1] / t.d0[1]);
574        }
575        solve_lines(s, &fac_i, true, &mut x);
576        solve_lines(s, &fac_j, false, &mut x);
577
578        // clipped update and eddy viscosity
579        let mut turb = vec![[0.0; 2]; n];
580        let mut mu_t = vec![0.0; n];
581        turb.par_chunks_mut(ni).zip(mu_t.par_chunks_mut(ni)).enumerate().for_each(|(j, (trow, mrow))| {
582            for i in 0..ni {
583                let c = j * ni + i;
584                if cx.solid(i, j) {
585                    continue;
586                }
587                let [k0, w0] = s.turb[c];
588                let k = (k0 + x[c][0]).max(0.1 * k0).max(1e-3 * inf[0]);
589                let om = (w0 + x[c][1]).max(0.1 * w0).max(1e-3 * inf[1]);
590                let rho = cx.rho(c);
591                let nu = cx.mu[c] / rho;
592                let g = &s.grad[c];
593                let strain = strain2(g, cx.w[pidx(ni, i as isize, j as isize)][2], cx.grid.rc[c]).max(0.0).sqrt();
594                trow[i] = [k, om];
595                mrow[i] = eddy_viscosity(rho, k, om, strain, f2_of(nu, k, om, cx.dwall[c]), v.re_c);
596            }
597        });
598        let sum: f64 = cells.par_iter().map(|t| t.r[0] * t.r[0]).sum();
599        (turb, mu_t, (sum / s.n_active as f64).sqrt())
600    };
601    s.turb = turb;
602    s.mu_t = mu_t;
603    s.turb_rnorm = rnorm;
604}

793    /// Colebrook formulas at the fixed point `u_tau^2 = (nu + nu_t,w) U / d`.
794    #[test]
795    fn aupoix_wall_limits() {
796        let (nu, d, speed) = (1e-6, 1e-4, 1.0);
797        let smooth = 60.0 * nu / (BETA1 * d * d);
798        let [k_w, om_w, nut] = aupoix_wall(nu, speed, d, 1e-9);
799        assert_eq!((k_w, om_w, nut), (0.0, smooth, 0.0), "ks+ = 1e-4 must give the smooth wall");
800        // at rest the values stay finite
801        assert!(aupoix_wall(nu, 0.0, d, 1e-3).iter().all(|x| x.is_finite()));
802        // the roughness whose fixed point is ks+ = 100
803        let ks_plus_of = |ks: f64| {
804            let nt = aupoix_wall(nu, speed, d, ks)[2];
805            ((nu + nt) * speed / d).sqrt() * ks / nu
806        };
807        let (mut lo, mut hi) = (1e-4f64, 1e-2f64);
808        for _ in 0..60 {
809            let mid = (lo * hi).sqrt();
810            if ks_plus_of(mid) < 100.0 {
811                lo = mid;
812            } else {
813                hi = mid;
814            }
815        }
816        let ks = (lo * hi).sqrt();
817        let [k_w, om_w, nut] = aupoix_wall(nu, speed, d, ks);
818        let u2 = (nu + nut) * speed / d;
819        let ks_plus = u2.sqrt() * ks / nu;
820        assert!((ks_plus / 100.0 - 1.0).abs() < 1e-6, "ks+ = {ks_plus}");
821        let t = (ks_plus / 125.0).tanh();
822        let k_plus = (((ks_plus / 30.0).log10() + 1.0 - t) * t).tanh().max(0.0) / BETA_STAR.sqrt();
823        let om_plus = 300.0 / (ks_plus * ks_plus) / (15.0 / (4.0 * ks_plus)).tanh() + 191.0 / ks_plus * (1.0 - (-ks_plus / 250.0).exp());
824        assert!((k_w / u2 / k_plus - 1.0).abs() < 2e-2, "k_w+ = {} against {k_plus}", k_w / u2);
825        assert!((om_w * nu / u2 / om_plus - 1.0).abs() < 2e-2, "omega_w+ = {} against {om_plus}", om_w * nu / u2);
826        assert!((nut - k_w / om_w).abs() <= 1e-15 * nut && nut > 0.0 && om_w < smooth, "nu_t,w = {nut}");
827    }
