1//! Menter SST k-omega turbulence model (2003 version) for the viscous solver (`Turbulence::Sst`).
2//!
3//! The model transports the specific turbulence kinetic energy `k` and the specific dissipation rate `omega` of every cell
4//! (`Solver::turb`, `[k, omega]`) with a loosely coupled pseudo-time step at the end of every `Solver::step`; the eddy
5//! viscosity `mu_t` it produces (`Solver::mu_t`) enters the mean-flow equations through the viscous terms of `viscous.rs`.
6//! Units as in `viscous.rs` (`rho_inf = c_inf = 1`, lengths in nose diameters, `nu = mu / rho`).
7//!
8//! ```text
9//!   constants: a1 = 0.31, beta* = 0.09;  set 1: sigma_k 0.85, sigma_w 0.5, beta 0.075, gamma 5/9;
10//!              set 2: sigma_k 1.0, sigma_w 0.856, beta 0.0828, gamma 0.44;    phi = F1 phi_1 + (1 - F1) phi_2
11//!   free stream: k_inf = 9e-9,  omega_inf = 1e-6 re_c                         (mu_t / mu = 0.009)
12//!   d = distance of the cell centroid to the nearest no-slip wall face
13//!   S^2 = 2 (u_x^2 + v_r^2 + (v/r)^2) + (u_r + v_x)^2 - 2/3 div^2,   S = sqrt(max(S^2, 0))     (cell gradients)
14//!   X   = 2 rho sigma_w2 (grad k . grad omega) / omega,                CD = max(X, 1e-20)
15//!   F1  = tanh(a^4),  a = min( max( sqrt(k) / (beta* omega d), 500 nu / (d^2 omega) ), 4 rho sigma_w2 k / (CD d^2) )
16//!   F2  = tanh(b^2),  b = max( 2 sqrt(k) / (beta* omega d), 500 nu / (d^2 omega) )
17//!   mu_t = min( rho a1 k / max(a1 omega, S F2), 1e5 / re_c )
18//!   P    = min( mu_t S^2, 10 beta* rho k omega )
19//!   rho dk/dt     = (-conv_k + diff_k) / V + P - beta* rho omega k
20//!   rho domega/dt = (-conv_w + diff_w) / V + gamma rho P / max(mu_t, 1e-30) - beta rho omega^2 + (1 - F1) X
21//!   conv = sum over the four faces of (outward mass flow) (phi_upwind - phi_c), inflow faces only; the mass flow of a face is
22//!          `F[0] w_f` of the stored inviscid face flux
23//!   diff = sum over the four faces of the outward (mu_f + sigma_f mu_t,f) (g_f . n) w_f, `g_f` the corrected face gradient of
24//!          `viscous.rs` (a compact normal derivative with the averaged tangential part), `sigma_f` the mean of the blended
25//!          `sigma` of the two sides
26//! ```
27//!
28//! Boundary states of the two scalars (the ghost of a face, mirrored in the face centre like the mean-flow ghosts):
29//! `Bc::Freestream | Characteristic | PressureOutflow` take `(k_inf, omega_inf)`; `Extrapolate`, slip walls and the axis the
30//! cell values; a no-slip wall `k_ghost = -k_c`, `omega_ghost = 2 omega_w - omega_c`, `omega_w = 60 nu_c / (0.075 d_c^2)`
31//! (the face values are then `k = 0`, `omega = omega_w`).
32//!
33//! The update is a sequence of data-parallel kernels (`update`): cell gradients of `k`, `omega`; the blending functions; the
34//! face coefficients; the per-cell residual and the coefficients of the two linear systems; the line solves; the clipped
35//! update with the new eddy viscosity. Per equation, with `D_k = beta* omega`, `D_w = 2 beta omega`, `M-` / `M+` the mass flow
36//! in the positive direction through the lower / upper face of a line direction and `G = (mu_f + sigma_f mu_t,f) w_f / |d|`:
37//!
38//! ```text
39//!   diag0 = rho V / dt + rho V D
40//!   direction dir:  lo = -(max(M-, 0) + G-),  up = -(max(-M+, 0) + G+),  dd = max(M-, 0) + max(-M+, 0) + G- + G+
41//!   i lines:  (diag0 + dd_i) x_c + lo_i x_{c-1} + up_i x_{c+1} = V R_c                    R_c = rho dphi/dt
42//!   j lines:  (1 + dd_j / diag0) dphi_c + (lo_j / diag0) dphi_{c-1} + (up_j / diag0) dphi_{c+1} = x_c
43//!   phi_new = max(phi + dphi, 0.1 phi), then k >= 1e-3 k_inf, omega >= 1e-3 omega_inf;  then mu_t is recomputed
44//! ```
45//!
46//! A neighbour that is a ghost or a solid cell drops its off-diagonal (the diagonal part stays). The two scalars are solved
47//! together with the block-tridiagonal line solver of `implicit.rs` (4x4 blocks that are diagonal: entries 0 and 5 hold the
48//! `k` and `omega` coefficients, entries 10 and 15 are `1` on the diagonal block and `0` elsewhere).

57/// Free-stream specific turbulence kinetic energy.
58pub const K_INF: f64 = 9e-9;
59
60/// Free-stream specific dissipation rate `1e-6 re_c`.
61pub fn omega_inf(re_c: f64) -> f64 {
62    1e-6 * re_c
63}

65/// Model constants of the SST model (see the module documentation); public so that GPU ports can take them from here.
66pub const A1: f64 = 0.31;
67pub const BETA_STAR: f64 = 0.09;
68pub const SIGMA_K1: f64 = 0.85;
69pub const SIGMA_K2: f64 = 1.0;
70pub const SIGMA_W1: f64 = 0.5;
71pub const SIGMA_W2: f64 = 0.856;
72pub const BETA1: f64 = 0.075;
73pub const BETA2: f64 = 0.0828;
74pub const GAMMA1: f64 = 5.0 / 9.0;

77/// Distance of every cell centroid to the nearest no-slip wall face segment of the viscous run `v` with boundary conditions `bc` and
78/// body block `blk` and nozzle exit `nozzle` (jet faces are not walls; `f64::INFINITY` without a no-slip face and in solid cells): what `Solver::set_viscous` computes for the
79/// turbulence model, for ports that keep their own copy of the field.
80pub fn wall_distances(grid: &Grid, v: Viscous, bc: &[Bc; 4], blk: Option<Block>, nozzle: Option<Nozzle>) -> Vec<f64> {
81    wall_distance(grid, &Walls::new(Some(v), bc, blk, nozzle))
82}

84/// Distance of every cell centroid to the nearest no-slip wall face segment (`f64::INFINITY` without a no-slip face and in
85/// solid cells); brute force over the wall faces.
86pub(crate) fn wall_distance(grid: &Grid, walls: &Walls) -> Vec<f64> {
87    let (ni, nj) = (grid.ni, grid.nj);
88    let mut segs: Vec<[f64; 4]> = Vec::new();
89    for j in 0..=nj {
90        for i in 0..ni {
91            if walls.noslip_jface(i, j) {
92                let (a, b) = (grid.node(i, j), grid.node(i + 1, j));
93                segs.push([a.0, a.1, b.0, b.1]);
94            }
95        }
96    }
97    for j in 0..nj {
98        for i in 0..=ni {
99            if walls.noslip_iface(i, j) {
100                let (a, b) = (grid.node(i, j), grid.node(i, j + 1));
101                segs.push([a.0, a.1, b.0, b.1]);
102            }
103        }
104    }
105    let mut d = vec![f64::INFINITY; ni * nj];
106    d.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
107        for (i, out) in row.iter_mut().enumerate() {
108            if walls.blk.is_some_and(|b| b.solid(i, j)) {
109                continue;
110            }
111            let (px, pr) = (grid.xc[j * ni + i], grid.rc[j * ni + i]);
112            for s in &segs {
113                let (ex, er) = (s[2] - s[0], s[3] - s[1]);
114                let t = (((px - s[0]) * ex + (pr - s[1]) * er) / (ex * ex + er * er)).clamp(0.0, 1.0);
115                *out = out.min((px - (s[0] + t * ex)).hypot(pr - (s[1] + t * er)));
116            }
117        }
118    });
119    d
120}

122/// Initialise the turbulence state of `s` (free stream), its wall distance and the eddy viscosity `rho k / omega`.
123pub(crate) fn init(s: &mut Solver, re_c: f64) {
124    let n = s.grid.ni * s.grid.nj;
125    s.turb = vec![[K_INF, omega_inf(re_c)]; n];
126    s.dwall = wall_distance(&s.grid, &s.walls());
127    let (ni, blk) = (s.grid.ni, s.block);
128    for c in 0..n {
129        if !blk.is_some_and(|b| b.solid(c % ni, c / ni)) {
130            s.mu_t[c] = s.u[c][0] * K_INF / omega_inf(re_c);
131        }
132    }
133}

135/// Where a ghost state of a face comes from.
136#[derive(Clone, Copy, PartialEq)]
137enum Kind {
138    /// The values of the adjacent cell (extrapolation, slip wall, axis).
139    Cell,
140    /// The free-stream values.
141    Inf,
142    /// The fixed `[k, omega]` of a nozzle exit.
143    Fixed([f64; 2]),
144    /// A no-slip wall.
145    Wall,
146}
147
148fn bc_kind(bc: Bc) -> Kind {
149    match bc {
150        Bc::Freestream | Bc::Characteristic | Bc::PressureOutflow => Kind::Inf,
151        Bc::Extrapolate | Bc::SlipWall => Kind::Cell,
152    }
153}

155/// Read-only data of one update.
156struct Ctx<'a> {
157    grid: &'a Grid,
158    w: &'a [Prim],
159    walls: Walls,
160    bc: [Bc; 4],
161    mu: &'a [f64],
162    mu_t: &'a [f64],
163    turb: &'a [[f64; 2]],
164    dwall: &'a [f64],
165    inf: [f64; 2],
166}
167
168impl Ctx<'_> {
169    #[inline]
170    fn solid(&self, i: usize, j: usize) -> bool {
171        self.walls.blk.is_some_and(|b| b.solid(i, j))
172    }
173
174    #[inline]
175    fn rho(&self, c: usize) -> f64 {
176        let ni = self.grid.ni;
177        self.w[pidx(ni, (c % ni) as isize, (c / ni) as isize)][0]
178    }
179
180    /// Sides of i-face `(i, j)` and the origin of its ghost state; `None` inside the solid block.
181    fn iface(&self, i: usize, j: usize) -> Option<(FaceSides, Kind)> {
182        let sides = iface_sides(self.grid, self.w, &self.walls, i, j)?;
183        let kind = if i == 0 {
184            bc_kind(self.bc[0])
185        } else if i == self.grid.ni {
186            bc_kind(self.bc[1])
187        } else if let (true, Some(n)) = (sides.l.ghost && sides.l.fixed.is_some(), self.walls.nozzle) {
188            Kind::Fixed(n.turb)
189        } else if sides.l.ghost && sides.noslip {
190            Kind::Wall
191        } else {
192            Kind::Cell
193        };
194        Some((sides, kind))
195    }
196
197    /// Sides of j-face `(i, j)` and the origin of its ghost state; `None` inside the solid block.
198    fn jface(&self, i: usize, j: usize) -> Option<(FaceSides, Kind)> {
199        let sides = jface_sides(self.grid, self.w, &self.walls, i, j)?;
200        let kind = if sides.l.ghost {
201            if sides.noslip {
202                Kind::Wall
203            } else if j == 0 {
204                bc_kind(self.bc[2])
205            } else {
206                Kind::Cell
207            }
208        } else if sides.r.ghost {
209            bc_kind(self.bc[3])
210        } else {
211            Kind::Cell
212        };
213        Some((sides, kind))
214    }
215
216    /// `[k, omega]` of one side of a face.
217    fn kw(&self, s: &Side, kind: Kind) -> [f64; 2] {
218        let c = s.cell;
219        if !s.ghost {
220            return self.turb[c];
221        }
222        match kind {
223            Kind::Cell => self.turb[c],
224            Kind::Inf => self.inf,
225            Kind::Fixed(v) => v,
226            Kind::Wall => {
227                let [k, om] = self.turb[c];
228                let nu = self.mu[c] / self.rho(c);
229                let d = self.dwall[c];
230                [-k, 2.0 * (60.0 * nu / (BETA1 * d * d)) - om]
231            }
232        }
233    }
234}

236/// `S^2` of a cell from its gradient `(u_x, u_r, v_x, v_r, ..)`, radial velocity and radius.
237#[inline]
238fn strain2(g: &[f64; 6], v: f64, r: f64) -> f64 {
239    let vr = v / r;
240    let div = g[0] + g[3] + vr;
241    let shear = g[1] + g[2];
242    2.0 * (g[0] * g[0] + g[3] * g[3] + vr * vr) + shear * shear - 2.0 / 3.0 * div * div
243}

245/// Argument pieces `(sqrt(k) / (beta* omega d), 500 nu / (d^2 omega))` of the blending functions.
246#[inline]
247fn blend_args(nu: f64, k: f64, om: f64, d: f64) -> (f64, f64) {
248    (k.sqrt() / (BETA_STAR * om * d), 500.0 * nu / (d * d * om))
249}
250
251/// `F2` of a cell.
252#[inline]
253fn f2_of(nu: f64, k: f64, om: f64, d: f64) -> f64 {
254    let (t1, t2) = blend_args(nu, k, om, d);
255    let b = (2.0 * t1).max(t2);
256    (b * b).tanh()
257}

259/// Eddy viscosity `min(rho a1 k / max(a1 omega, S F2), 1e5 / re_c)`.
260#[inline]
261fn eddy_viscosity(rho: f64, k: f64, om: f64, s: f64, f2: f64, re_c: f64) -> f64 {
262    (rho * A1 * k / (A1 * om).max(s * f2)).min(1e5 / re_c)
263}

273/// Kernel 1: Green-Gauss gradient `(k_x, k_r, omega_x, omega_r)` of every fluid cell, `phi_f = (phi_c + phi_neighbour) / 2`.
274fn scalar_gradients(cx: &Ctx) -> Vec<[f64; 4]> {
275    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
276    let mut out = vec![[0.0; 4]; ni * nj];
277    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
278        for (i, g_out) in row.iter_mut().enumerate() {
279            if cx.solid(i, j) {
280                continue;
281            }
282            let c = j * ni + i;
283            let pc = cx.turb[c];
284            let (sl, kl) = cx.iface(i, j).expect("fluid cell face");
285            let (sr, kr) = cx.iface(i + 1, j).expect("fluid cell face");
286            let (sb, kb) = cx.jface(i, j).expect("fluid cell face");
287            let (st, kt) = cx.jface(i, j + 1).expect("fluid cell face");
288            let faces = [
289                (&cx.grid.iface[j * (ni + 1) + i + 1], 1.0, cx.kw(&sr.r, kr)),
290                (&cx.grid.iface[j * (ni + 1) + i], -1.0, cx.kw(&sl.l, kl)),
291                (&cx.grid.jface[(j + 1) * ni + i], 1.0, cx.kw(&st.r, kt)),
292                (&cx.grid.jface[j * ni + i], -1.0, cx.kw(&sb.l, kb)),
293            ];
294            let mut g = [0.0; 4];
295            for (f, sgn, nb) in faces {
296                for m in 0..2 {
297                    let pf = 0.5 * (pc[m] + nb[m]);
298                    g[2 * m] += sgn * pf * f.nx * f.len;
299                    g[2 * m + 1] += sgn * pf * f.nr * f.len;
300                }
301            }
302            let inv_a = 1.0 / cx.grid.area[c];
303            for x in g.iter_mut() {
304                *x *= inv_a;
305            }
306            *g_out = g;
307        }
308    });
309    out
310}

312/// Kernel 2: `F1` and the cross-diffusion `X` of every fluid cell.
313fn blends(cx: &Ctx, tg: &[[f64; 4]]) -> Vec<Blend> {
314    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
315    let mut out = vec![Blend::default(); ni * nj];
316    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
317        for (i, b) in row.iter_mut().enumerate() {
318            if cx.solid(i, j) {
319                continue;
320            }
321            let c = j * ni + i;
322            let rho = cx.rho(c);
323            let nu = cx.mu[c] / rho;
324            let [k, om] = cx.turb[c];
325            let d = cx.dwall[c];
326            let g = &tg[c];
327            let x = 2.0 * rho * SIGMA_W2 * (g[0] * g[2] + g[1] * g[3]) / om;
328            let cd = x.max(1e-20);
329            let (t1, t2) = blend_args(nu, k, om, d);
330            let t3 = 4.0 * rho * SIGMA_W2 * k / (cd * d * d);
331            let a = t1.max(t2).min(t3);
332            *b = Blend { f1: a.powi(4).tanh(), x };
333        }
334    });
335    out
336}

338/// Scalar diffusion and convection data of one face (`l` / `r`: the two sides, lower / upper index).
339#[derive(Clone, Copy, Default)]
340struct TFace {
341    /// Mass flow `F[0] w_f` in the positive direction.
342    m: f64,
343    /// Diffusive conductance `G = (mu_f + sigma_f mu_t,f) w_f / |d|` per scalar.
344    g: [f64; 2],
345    /// Diffusive flux `(mu_f + sigma_f mu_t,f) (g_f . n) w_f` per scalar in the positive direction.
346    flux: [f64; 2],
347    l: [f64; 2],
348    r: [f64; 2],
349}

351/// Kernel 3 (one face): see the module documentation; `m` is the mass flow of the face, `(xf, rf)` its centre.
352fn tface(cx: &Ctx, tg: &[[f64; 4]], blend: &[Blend], sides: &FaceSides, kind: Kind, f: &Face, xf: f64, rf: f64, m: f64) -> TFace {
353    let (kl, kr) = (cx.kw(&sides.l, kind), cx.kw(&sides.r, kind));
354    if f.rm == 0.0 {
355        return TFace { m, g: [0.0; 2], flux: [0.0; 2], l: kl, r: kr };
356    }
357    let w = f.w();
358    let (dd, ex, er) = stencil(cx.grid, sides, xf, rf);
359    let (gl, gr) = (&tg[sides.l.cell], &tg[sides.r.cell]);
360    let (mu_f, mut_f) = face_mu(cx.mu, cx.mu_t, sides);
361    let (f1l, f1r) = (blend[sides.l.cell].f1, blend[sides.r.cell].f1);
362    let mut out = TFace { m, g: [0.0; 2], flux: [0.0; 2], l: kl, r: kr };
363    for e in 0..2 {
364        let (s1, s2) = if e == 0 { (SIGMA_K1, SIGMA_K2) } else { (SIGMA_W1, SIGMA_W2) };
365        let sigma = 0.5 * (f1l * s1 + (1.0 - f1l) * s2 + f1r * s1 + (1.0 - f1r) * s2);
366        let coef = mu_f + sigma * mut_f;
367        let (gbx, gbr) = (0.5 * (gl[2 * e] + gr[2 * e]), 0.5 * (gl[2 * e + 1] + gr[2 * e + 1]));
368        let corr = (kr[e] - kl[e]) / dd - (gbx * ex + gbr * er);
369        let gn = (gbx + corr * ex) * f.nx + (gbr + corr * er) * f.nr;
370        out.flux[e] = coef * gn * w;
371        out.g[e] = coef * w / dd;
372    }
373    out
374}

376/// Kernel 3: scalar face data of every i-face and j-face.
377fn face_pass(cx: &Ctx, tg: &[[f64; 4]], blend: &[Blend], fi: &[Cons], fj: &[Cons]) -> (Vec<TFace>, Vec<TFace>) {
378    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
379    let mut ti = vec![TFace::default(); (ni + 1) * nj];
380    ti.par_chunks_mut(ni + 1).enumerate().for_each(|(j, row)| {
381        for (i, out) in row.iter_mut().enumerate() {
382            if let Some((sides, kind)) = cx.iface(i, j) {
383                let f = &cx.grid.iface[j * (ni + 1) + i];
384                let (xf, rf) = iface_centre(cx.grid, i, j);
385                *out = tface(cx, tg, blend, &sides, kind, f, xf, rf, fi[j * (ni + 1) + i][0] * f.w());
386            }
387        }
388    });
389    let mut tj = vec![TFace::default(); ni * (nj + 1)];
390    tj.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
391        for (i, out) in row.iter_mut().enumerate() {
392            if let Some((sides, kind)) = cx.jface(i, j) {
393                let f = &cx.grid.jface[j * ni + i];
394                let (xf, rf) = jface_centre(cx.grid, i, j);
395                *out = tface(cx, tg, blend, &sides, kind, f, xf, rf, fj[j * ni + i][0] * f.w());
396            }
397        }
398    });
399    (ti, tj)
400}

414/// Kernel 4: residual `R`, `diag0` and the line coefficients of every fluid cell.
415fn cell_terms(cx: &Ctx, grad: &[[f64; 6]], blend: &[Blend], dt: &[f64], ti: &[TFace], tj: &[TFace]) -> Vec<CellTerms> {
416    let (ni, nj) = (cx.grid.ni, cx.grid.nj);
417    let blk = cx.walls.blk;
418    let mut out = vec![CellTerms::default(); ni * nj];
419    out.par_chunks_mut(ni).enumerate().for_each(|(j, row)| {
420        for (i, o) in row.iter_mut().enumerate() {
421            if cx.solid(i, j) {
422                continue;
423            }
424            let c = j * ni + i;
425            let vol = cx.grid.vol[c];
426            let rho = cx.rho(c);
427            let [k, om] = cx.turb[c];
428            let phi = [k, om];
429            let b = blend[c];
430            let wc = &cx.w[pidx(ni, i as isize, j as isize)];
431            let s2 = strain2(&grad[c], wc[2], cx.grid.rc[c]);
432            let mut_c = cx.mu_t[c];
433            let prod = (mut_c * s2).min(10.0 * BETA_STAR * rho * k * om);
434            let beta = b.f1 * BETA1 + (1.0 - b.f1) * BETA2;
435            let gamma = b.f1 * GAMMA1 + (1.0 - b.f1) * GAMMA2;
436            let sources = [prod - BETA_STAR * rho * om * k, gamma * rho * prod / mut_c.max(1e-30) - beta * rho * om * om + (1.0 - b.f1) * b.x];
437            let dest = [BETA_STAR * om, 2.0 * beta * om];
438            let faces = [(&ti[j * (ni + 1) + i], &ti[j * (ni + 1) + i + 1]), (&tj[j * ni + i], &tj[(j + 1) * ni + i])];
439            let ends = [(i == blk.map_or(0, |b| b.row_start(j).unwrap_or(ni)), i == ni - 1), (j == blk.map_or(0, |b| b.col_start(i)), j == nj - 1)];
440            let mut rate = [0.0; 2];
441            for (dir, (lo_f, up_f)) in faces.iter().enumerate() {
442                let (a_lo, a_up) = (lo_f.m.max(0.0), (-up_f.m).max(0.0));
443                for e in 0..2 {
444                    // -conv + diff of this direction
445                    rate[e] += a_lo * (lo_f.l[e] - phi[e]) + a_up * (up_f.r[e] - phi[e]) + up_f.flux[e] - lo_f.flux[e];
446                    o.lo[dir][e] = if ends[dir].0 { 0.0 } else { -(a_lo + lo_f.g[e]) };
447                    o.up[dir][e] = if ends[dir].1 { 0.0 } else { -(a_up + up_f.g[e]) };
448                    o.dd[dir][e] = a_lo + a_up + lo_f.g[e] + up_f.g[e];
449                }
450            }
451            for e in 0..2 {
452                o.r[e] = rate[e] / vol + sources[e];
453                o.d0[e] = rho * vol / dt[c] + rho * vol * dest[e];
454            }
455        }
456    });
457    out
458}

478/// One turbulence update at the end of a `Solver::step`, from the primitives, face fluxes, gradients and time step of the
479/// residual evaluation of this iteration (see the module documentation): residual and coefficients, the i- and j-line solves,
480/// the clipped update of `k`, `omega` and the new eddy viscosity.
481pub(crate) fn update(s: &mut Solver) {
482    let v = s.viscous.expect("SST needs the viscous terms");
483    let (ni, nj) = (s.grid.ni, s.grid.nj);
484    let n = ni * nj;
485    let inf = [K_INF, omega_inf(v.re_c)];
486    let (turb, mu_t, rnorm) = {
487        let cx = Ctx { grid: &s.grid, w: &s.w, walls: s.walls(), bc: s.cfg.bc, mu: &s.mu, mu_t: &s.mu_t, turb: &s.turb, dwall: &s.dwall, inf };
488        let tg = scalar_gradients(&cx);
489        let blend = blends(&cx, &tg);
490        let (ti, tj) = face_pass(&cx, &tg, &blend, &s.fi, &s.fj);
491        let cells = cell_terms(&cx, &s.grad, &blend, &s.dt, &ti, &tj);
492
493        // coefficients of the two linear systems
494        let mut fac_i = Factor { lo: vec![ZERO; n], d: vec![EYE; n], up: vec![ZERO; n] };
495        let mut fac_j = Factor { lo: vec![ZERO; n], d: vec![EYE; n], up: vec![ZERO; n] };
496        let mut x: Vec<Cons> = vec![[0.0; 4]; n];
497        for c in 0..n {
498            if cx.solid(c % ni, c / ni) {
499                continue;
500            }
501            let t = &cells[c];
502            let vol = s.grid.vol[c];
503            x[c] = [vol * t.r[0], vol * t.r[1], 0.0, 0.0];
504            fac_i.lo[c] = offdiag(t.lo[0][0], t.lo[0][1]);
505            fac_i.up[c] = offdiag(t.up[0][0], t.up[0][1]);
506            fac_i.d[c] = diag(t.d0[0] + t.dd[0][0], t.d0[1] + t.dd[0][1]);
507            fac_j.lo[c] = offdiag(t.lo[1][0] / t.d0[0], t.lo[1][1] / t.d0[1]);
508            fac_j.up[c] = offdiag(t.up[1][0] / t.d0[0], t.up[1][1] / t.d0[1]);
509            fac_j.d[c] = diag(1.0 + t.dd[1][0] / t.d0[0], 1.0 + t.dd[1][1] / t.d0[1]);
510        }
511        solve_lines(s, &fac_i, true, &mut x);
512        solve_lines(s, &fac_j, false, &mut x);
513
514        // clipped update and eddy viscosity
515        let mut turb = vec![[0.0; 2]; n];
516        let mut mu_t = vec![0.0; n];
517        turb.par_chunks_mut(ni).zip(mu_t.par_chunks_mut(ni)).enumerate().for_each(|(j, (trow, mrow))| {
518            for i in 0..ni {
519                let c = j * ni + i;
520                if cx.solid(i, j) {
521                    continue;
522                }
523                let [k0, w0] = s.turb[c];
524                let k = (k0 + x[c][0]).max(0.1 * k0).max(1e-3 * inf[0]);
525                let om = (w0 + x[c][1]).max(0.1 * w0).max(1e-3 * inf[1]);
526                let rho = cx.rho(c);
527                let nu = cx.mu[c] / rho;
528                let g = &s.grad[c];
529                let strain = strain2(g, cx.w[pidx(ni, i as isize, j as isize)][2], cx.grid.rc[c]).max(0.0).sqrt();
530                trow[i] = [k, om];
531                mrow[i] = eddy_viscosity(rho, k, om, strain, f2_of(nu, k, om, cx.dwall[c]), v.re_c);
532            }
533        });
534        let sum: f64 = cells.par_iter().map(|t| t.r[0] * t.r[0]).sum();
535        (turb, mu_t, (sum / s.n_active as f64).sqrt())
536    };
537    s.turb = turb;
538    s.mu_t = mu_t;
539    s.turb_rnorm = rnorm;
540}
