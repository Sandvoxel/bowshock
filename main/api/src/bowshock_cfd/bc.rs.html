1//! Characteristic (Riemann-invariant) far-field and static-pressure outflow ghost states.
2//!
3//! All functions take the primitive state of the first interior cell and the **outward** unit normal
4//! `(nx, nr)` of the boundary face and return the ghost (boundary) state; both ghost layers are
5//! filled with it. They are pure functions of their inputs so they port 1:1 to WGSL.
6//!
7//! # Characteristic far field (`characteristic_ghost`)
8//!
9//! With the outward normal `n`, `u_n = v.n`, `c` the sound speed and `g = gamma`, the one-dimensional
10//! (normal to the boundary) Riemann invariants are
11//!
12//! ```text
13//!   R_out = u_n,i + 2 c_i/(g-1)     carried out of the domain by the u_n + c wave  (from the interior state)
14//!   R_in  = u_n,e - 2 c_e/(g-1)     carried into the domain by the u_n - c wave    (from the free stream)
15//!   u_n,b = (R_out + R_in)/2,   c_b = (g-1)/4 (R_out - R_in)
16//! ```
17//!
18//! The entropy `s = p/rho^g` and the tangential velocity are convected with the flow: taken from the interior
19//! for outflow (`u_n,b >= 0`) and from the free stream for inflow. Then `rho_b = (c_b^2/(g s))^(1/(g-1))`,
20//! `p_b = rho_b c_b^2/g`, `v_b = v_ref + (u_n,b - u_n,ref) n`.
21//!
22//! Which branch is used is decided per face by the **interior normal Mach number** `M_n = u_n,i/c_i`
23//! ([`Regime`]):
24//!
25//! | `M_n` | regime | ghost |
26//! |---|---|---|
27//! | `<= -1` | supersonic inflow | free stream |
28//! | `(-1, 0)` | subsonic inflow | Riemann (inflow/outflow split of entropy + tangential velocity by the sign of `u_n,b`) |
29//! | `[0, 1)` | subsonic outflow | Riemann (same) |
30//! | `>= 1` | supersonic outflow | interior (zero gradient) |
31//!
32//! If the interior state is the free stream, every branch returns the free stream (to round-off), so a uniform
33//! flow produces no spurious waves.
34
35use crate::gas::{sound_speed, Prim, GAMMA};
36
37/// Flow regime of a far-field face, selected by the interior normal Mach number.
38#[derive(Clone, Copy, Debug, PartialEq, Eq)]
39pub enum Regime {
40    SupersonicInflow,
41    SubsonicInflow,
42    SubsonicOutflow,
43    SupersonicOutflow,
44}
45
46/// Normal Mach number `v.n / c` of `w` for the outward unit normal `(nx, nr)`.
47#[inline]
48pub fn normal_mach(w: &Prim, nx: f64, nr: f64) -> f64 {
49    (w[1] * nx + w[2] * nr) / sound_speed(w)
50}
51
52/// Regime of a face from the interior state and the outward normal.
53#[inline]
54pub fn regime(interior: &Prim, nx: f64, nr: f64) -> Regime {
55    let m = normal_mach(interior, nx, nr);
56    if m <= -1.0 {
57        Regime::SupersonicInflow
58    } else if m < 0.0 {
59        Regime::SubsonicInflow
60    } else if m < 1.0 {
61        Regime::SubsonicOutflow
62    } else {
63        Regime::SupersonicOutflow
64    }
65}

67/// Ghost state of the characteristic far-field BC. `fs` is the free stream, `interior` the first interior cell,
68/// `(nx, nr)` the outward unit normal. Returns the regime used as well.
69pub fn characteristic_ghost(fs: &Prim, interior: &Prim, nx: f64, nr: f64) -> (Prim, Regime) {
70    let reg = regime(interior, nx, nr);
71    let g1 = GAMMA - 1.0;
72    let ghost = match reg {
73        Regime::SupersonicInflow => *fs,
74        Regime::SupersonicOutflow => *interior,
75        Regime::SubsonicInflow | Regime::SubsonicOutflow => {
76            let (ci, ce) = (sound_speed(interior), sound_speed(fs));
77            let uni = interior[1] * nx + interior[2] * nr;
78            let une = fs[1] * nx + fs[2] * nr;
79            let r_out = uni + 2.0 * ci / g1;
80            let r_in = une - 2.0 * ce / g1;
81            let un_b = 0.5 * (r_out + r_in);
82            let c_b = (0.25 * g1 * (r_out - r_in)).max(1e-8);
83            // reference state carrying entropy and tangential velocity
84            let (rf, un_ref) = if un_b >= 0.0 { (interior, uni) } else { (fs, une) };
85            let s = rf[3] / rf[0].powf(GAMMA);
86            let rho = (c_b * c_b / (GAMMA * s)).powf(1.0 / g1);
87            let p = rho * c_b * c_b / GAMMA;
88            let dn = un_b - un_ref;
89            [rho, rf[1] + dn * nx, rf[2] + dn * nr, p]
90        }
91    };
92    (ghost, reg)
93}

95/// Fixed static-pressure outflow: subsonic faces (`M_n < 1`) take `rho, u, v` from the interior and
96/// `p = p_back`; supersonic faces extrapolate.
97pub fn pressure_outflow_ghost(p_back: f64, interior: &Prim, nx: f64, nr: f64) -> Prim {
98    if normal_mach(interior, nx, nr) >= 1.0 {
99        *interior
100    } else {
101        [interior[0], interior[1], interior[2], p_back]
102    }
103}

110    #[test]
111    fn regime_selected_by_normal_mach() {
112        let c = 1.0f64; // these states have rho = 1, p = 1/g, hence c = 1
113        let mk = |u: f64, v: f64| [1.0, u, v, 1.0 / GAMMA];
114        // normal pointing in +x
115        assert_eq!(regime(&mk(-1.5 * c, 0.0), 1.0, 0.0), Regime::SupersonicInflow);
116        assert_eq!(regime(&mk(-c, 0.0), 1.0, 0.0), Regime::SupersonicInflow);
117        assert_eq!(regime(&mk(-0.5 * c, 0.0), 1.0, 0.0), Regime::SubsonicInflow);
118        assert_eq!(regime(&mk(0.5 * c, 0.0), 1.0, 0.0), Regime::SubsonicOutflow);
119        assert_eq!(regime(&mk(0.0, 0.0), 1.0, 0.0), Regime::SubsonicOutflow);
120        assert_eq!(regime(&mk(1.0 * c, 0.0), 1.0, 0.0), Regime::SupersonicOutflow);
121        assert_eq!(regime(&mk(2.0 * c, 0.0), 1.0, 0.0), Regime::SupersonicOutflow);
122        // same state, normal -x: flips inflow <-> outflow
123        assert_eq!(regime(&mk(0.5 * c, 0.0), -1.0, 0.0), Regime::SubsonicInflow);
124        assert_eq!(regime(&mk(2.0 * c, 0.0), -1.0, 0.0), Regime::SupersonicInflow);
125        // radial normal: axial flow is tangential -> subsonic whatever the Mach number
126        assert_eq!(regime(&mk(3.0 * c, 0.0), 0.0, 1.0), Regime::SubsonicOutflow);
127        // oblique normal: M_n = M cos(60 deg)
128        let (nx, nr) = (0.5, 3f64.sqrt() / 2.0);
129        assert_eq!(regime(&mk(1.9, 0.0), nx, nr), Regime::SubsonicOutflow);
130        assert_eq!(regime(&mk(2.1, 0.0), nx, nr), Regime::SupersonicOutflow);
131    }

133    #[test]
134    fn ghost_follows_regime() {
135        let fs = freestream(0.9);
136        let s = |w: &Prim| w[3] / w[0].powf(GAMMA);
137        // supersonic inflow: free stream regardless of the interior; supersonic outflow: interior
138        let int = [1.3, 2.0, 0.1, 1.2];
139        let (g, r) = characteristic_ghost(&fs, &int, -1.0, 0.0);
140        assert_eq!(r, Regime::SupersonicInflow);
141        assert_eq!(g, fs);
142        let (g, r) = characteristic_ghost(&fs, &int, 1.0, 0.0);
143        assert_eq!(r, Regime::SupersonicOutflow);
144        assert_eq!(g, int);
145        // subsonic outflow (+x): entropy and tangential velocity from the interior, Riemann invariants reproduced
146        let int = [1.1, 0.4, 0.2, 0.8];
147        let (g, r) = characteristic_ghost(&fs, &int, 1.0, 0.0);
148        assert_eq!(r, Regime::SubsonicOutflow);
149        assert!((g[2] - int[2]).abs() < 1e-14, "tangential velocity must come from the interior");
150        assert!((s(&g) - s(&int)).abs() < 1e-13, "entropy must come from the interior");
151        let g1 = GAMMA - 1.0;
152        let (ci, ce, cb) = (sound_speed(&int), sound_speed(&fs), sound_speed(&g));
153        assert!((g[1] + 2.0 * cb / g1 - (int[1] + 2.0 * ci / g1)).abs() < 1e-12);
154        assert!((g[1] - 2.0 * cb / g1 - (fs[1] - 2.0 * ce / g1)).abs() < 1e-12);
155        // subsonic inflow (outward normal -x): entropy and tangential velocity from the free stream
156        let (g, r) = characteristic_ghost(&fs, &int, -1.0, 0.0);
157        assert_eq!(r, Regime::SubsonicInflow);
158        assert!((g[2] - fs[2]).abs() < 1e-14, "tangential velocity must come from the free stream");
159        assert!((s(&g) - s(&fs)).abs() < 1e-13, "entropy must come from the free stream");
160    }

162    #[test]
163    fn freestream_is_a_fixed_point_for_any_normal() {
164        for mach in [0.3, 0.5, 0.9, 0.98, 1.0, 1.1, 2.0] {
165            let fs = freestream(mach);
166            for k in 0..16 {
167                let a = std::f64::consts::TAU * k as f64 / 16.0;
168                let (g, _) = characteristic_ghost(&fs, &fs, a.cos(), a.sin());
169                for q in 0..4 {
170                    assert!((g[q] - fs[q]).abs() < 1e-13, "M={mach} a={a}: {g:?} vs {fs:?}");
171                }
172            }
173        }
174    }
